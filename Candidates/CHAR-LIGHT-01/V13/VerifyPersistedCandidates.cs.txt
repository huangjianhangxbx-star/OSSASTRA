/* 定向读回四候选及原始Profile/Main不变证据；无场景/资产保存。 */
using System;using System.IO;using System.Linq;using System.Security.Cryptography;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Newtonsoft.Json.Linq;using Ossastra.CharacterLighting;using Funplay.Editor.Tools.Scripting;
public sealed class VerifyCharacterSunlightCandidates:IFunplayCommand{
 static string Hash(string p){using(var h=SHA256.Create())using(var s=File.OpenRead(p))return BitConverter.ToString(h.ComputeHash(s)).Replace("-","");}
 public void Execute(ExecutionContext ctx){var r=JObject.Parse(File.ReadAllText("artifacts/CHAR-LIGHT-01/v1.3/runtime-baseline-read-v2.json"));var baseline=JObject.Parse(File.ReadAllText((string)r["data"]["returnValue"]["output"]));var saved=JArray.Parse(File.ReadAllText("artifacts/CHAR-LIGHT-01/v1.3/saved-candidates.json"));var rows=new JArray();foreach(var c in saved){string path=(string)c["path"];var asset=AssetDatabase.LoadAssetAtPath<CharacterLightingProfileSO>(path);if(asset==null)throw new InvalidOperationException("Missing exact saved candidate "+path);var actor=baseline["actors"].Single(v=>(string)v["name"]==(string)c["actor"]);var expected=JObject.Parse((string)actor["runtimeJson"]);if((string)c["mode"]=="B"){expected["knee"]=.7f;expected["ceiling"]=1f;}var actual=JObject.Parse(JsonUtility.ToJson(asset.settings));bool same=JToken.DeepEquals(expected,actual);if(!same)throw new InvalidOperationException("Candidate full field difference "+path);rows.Add(JObject.FromObject(new{path,guid=AssetDatabase.AssetPathToGUID(path),same,sha=Hash(path)}));}
 var original=baseline["actors"].Select(a=>new{path=(string)a["profilePath"],expected=(string)a["profileFileSha"],actual=Hash((string)a["profilePath"])}).ToArray();bool sourceSame=original.All(a=>a.expected==a.actual);var route=JObject.Parse(File.ReadAllText("artifacts/CHAR-LIGHT-01/v1.3/route-active.json"));var scene=SceneManager.GetActiveScene();bool sceneSame=Hash(scene.path)==(string)route["sceneDiskSha"];if(!sourceSame||!sceneSame)throw new InvalidOperationException("Original author asset/scene disk changed");string output=Path.GetFullPath("artifacts/CHAR-LIGHT-01/v1.3/persistence-"+DateTime.UtcNow.ToString("yyyyMMdd-HHmmss-fffffff")+".json");File.WriteAllText(output,JObject.FromObject(new{playing=Application.isPlaying,rows,original,sourceSame,sceneSame,sceneDirty=scene.isDirty,sceneSaved=false}).ToString());ctx.ReturnValue=new{output,sourceSame,sceneSame,sceneDirty=scene.isDirty,candidates=rows.Count};}}
