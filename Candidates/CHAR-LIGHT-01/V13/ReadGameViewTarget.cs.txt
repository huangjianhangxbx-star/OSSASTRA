using System;using System.IO;using System.Linq;using System.Reflection;using UnityEngine;using UnityEditor;using Funplay.Editor.Tools.Scripting;
public sealed class ReadCharacterGameViewTarget:IFunplayCommand{
 public void Execute(ExecutionContext ctx){var rows=Resources.FindObjectsOfTypeAll<EditorWindow>().Where(w=>w.GetType().FullName=="UnityEditor.GameView").Select(w=>{var f=w.GetType().GetField("m_RenderTexture",BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic);var rt=f==null?null:f.GetValue(w) as RenderTexture;return new{type=w.GetType().FullName,target=rt==null?null:new{name=rt.name,rt.width,rt.height,format=rt.graphicsFormat.ToString(),rt.antiAliasing,rt.sRGB}};}).ToArray();string output=Path.GetFullPath("artifacts/CHAR-LIGHT-01/v1.3/game-view-target.json");File.WriteAllText(output,Newtonsoft.Json.JsonConvert.SerializeObject(rows,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=new{output,rows};}}
