using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using Ossastra.Development.CharacterLighting;using Funplay.Editor.Tools.Scripting;
public sealed class ShowSunlightPanel:IFunplayCommand{public void Execute(ExecutionContext ctx){var p=SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<CharacterLightingPanel>(true)).Single();p.environmentPanel.SetExpanded(false);p.SetExpanded(true);ctx.ReturnValue=new{p.Expanded,p.enabled,p.isActiveAndEnabled,p.Status,surfaceActive=p.Surface.gameObject.activeInHierarchy,bodyActive=p.Surface.Find("Body").gameObject.activeInHierarchy,rigs=p.receivers.Select(r=>new{r.name,r.Settings.knee,r.Settings.ceiling}).ToArray()};}}
