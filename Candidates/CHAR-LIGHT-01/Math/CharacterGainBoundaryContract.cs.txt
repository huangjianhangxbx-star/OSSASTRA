/* 功能：正式Spine片元的合计高亮边界：8灯叠加后保护、接收者关闭、信息角色、黑色和alpha0加色附件排除。
 * 边界：增益软肩限制光照乘数，不宣称任意HDR原图的最终RGB分量都小于固定阈值；本合同使用LDR原图。
 * 生命周期：不创建场景对象；所有借用全局和GPU对象在finally归还。
 */
using System;using System.Linq;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public class CharacterGainBoundaryContract:IFunplayCommand
{
 static float Diff(Color a,Color b)=>Mathf.Max(Mathf.Abs(a.r-b.r),Mathf.Abs(a.g-b.g),Mathf.Abs(a.b-b.b));
 static Color Draw(Material m,Mesh mesh,RenderTexture rt,Texture2D read,Vector4 controls)
 {
  Shader.SetGlobalVector("_CharacterLightControls",controls);var cmd=new CommandBuffer{name="Character gain finite light-sum and exclusion contract"};
  try{cmd.SetRenderTarget(rt);cmd.SetViewport(new Rect(0,0,16,16));cmd.ClearRenderTarget(false,true,Color.clear);cmd.SetViewProjectionMatrices(Matrix4x4.identity,Matrix4x4.identity);cmd.DrawMesh(mesh,Matrix4x4.identity,m,0,1);Graphics.ExecuteCommandBuffer(cmd);}finally{cmd.Release();}
  RenderTexture.active=rt;read.ReadPixels(new Rect(0,0,16,16),0,0);read.Apply();return read.GetPixel(8,8);
 }
 static void Fill(Texture2D texture,Color color){texture.SetPixels(Enumerable.Repeat(color,4).ToArray());texture.Apply();}
 public void Execute(ExecutionContext ctx)
 {
  if(Application.dataPath!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new Exception("Project guard");var shader=Shader.Find("Hidden/Ossastra/Rooms/SightSpine");if(shader==null||!shader.isSupported)throw new Exception("Actual Spine shader unavailable");
  var fs=new[]{"_EnvironmentLightingEnabled","_EnvironmentIndoorAmbientScale","_EnvironmentWetness","_RoomSightStaticEnvironment","_EnvironmentFullSurface","_EnvironmentSurfaceRole","_EnvironmentResponseOverride","_EnvironmentReceiveShadow","_EnvironmentMinimumOverride"};
  var ns=new[]{"_EnvironmentIndoorCount","_EnvironmentLocalLightCount","_EnvironmentShelterCount","_RoomSightOwnerCount","_RoomSightOwnerExterior","_RoomSightSeedCount","_RoomSightBuildingCount","_RoomSightDomainCount","_RoomSightBlockerCount","_RoomSightWindowCount","_RoomSightCameraBlockerCount","_RoomSightExteriorView","_RoomSightPassMode"};
  var vs=new[]{"_EnvironmentStylizedSkyTint","_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient","_EnvironmentLightResponse","_EnvironmentCloudParameters","_EnvironmentCloudMotion","_MainLightPosition","_MainLightColor","_RoomSightTargetSize","_CharacterLightControls","_CharacterLightShoulder"};
  var oldF=fs.ToDictionary(n=>n,n=>Shader.GetGlobalFloat(n));var oldN=ns.ToDictionary(n=>n,n=>Shader.GetGlobalInteger(n));var oldV=vs.ToDictionary(n=>n,n=>Shader.GetGlobalVector(n));var oldPositions=Shader.GetGlobalVectorArray("_EnvironmentLocalLightPositions");var oldColors=Shader.GetGlobalVectorArray("_EnvironmentLocalLightColors");var inverse=Shader.GetGlobalMatrix("_RoomSightInverseViewProjection");var previous=RenderTexture.active;
  Material material=null;Texture2D texture=null,read=null;Mesh mesh=null;RenderTexture rt=null;
  try
  {
   foreach(var n in fs)Shader.SetGlobalFloat(n,0);foreach(var n in ns)Shader.SetGlobalInteger(n,0);Shader.SetGlobalVector("_EnvironmentStylizedSkyTint",new Vector4(1,1,1,0));Shader.SetGlobalVector("_CharacterLightShoulder",new Vector4(1.2f,2,0,0));Shader.SetGlobalInteger("_RoomSightExteriorView",1);Shader.SetGlobalFloat("_EnvironmentFullSurface",1);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",Matrix4x4.identity);Shader.SetGlobalVector("_RoomSightTargetSize",new Vector4(16,16,1f/16,1f/16));
   Shader.SetGlobalFloat("_EnvironmentLightingEnabled",1);Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);Shader.SetGlobalFloat("_EnvironmentResponseOverride",1);Shader.SetGlobalVector("_EnvironmentLightResponse",new Vector4(1,1,1,0));Shader.SetGlobalVector("_MainLightPosition",new Vector4(0,1,0,0));Shader.SetGlobalVector("_MainLightColor",Vector4.zero);Shader.SetGlobalVector("_EnvironmentCloudParameters",Vector4.zero);Shader.SetGlobalVector("_EnvironmentCloudMotion",Vector4.zero);foreach(var n in new[]{"_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient"})Shader.SetGlobalVector(n,new Vector4(.4f,.4f,.4f,0));
   var positions=new Vector4[8];var colors=new Vector4[8];for(int i=0;i<8;i++){positions[i]=new Vector4(0,0,2,10);colors[i]=new Vector4(16,16,16,0);}Shader.SetGlobalVectorArray("_EnvironmentLocalLightPositions",positions);Shader.SetGlobalVectorArray("_EnvironmentLocalLightColors",colors);Shader.SetGlobalInteger("_EnvironmentLocalLightCount",8);
   material=new Material(shader){hideFlags=HideFlags.HideAndDontSave};material.DisableKeyword("_STRAIGHT_ALPHA_INPUT");material.SetFloat("_StraightAlphaInput",0);texture=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true);material.SetTexture("_MainTex",texture);read=new Texture2D(16,16,TextureFormat.RGBAFloat,false,true);rt=new RenderTexture(16,16,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);if(!rt.Create())throw new Exception("Private target failed");mesh=new Mesh();mesh.vertices=new[]{new Vector3(-1,-1,.5f),new Vector3(3,-1,.5f),new Vector3(-1,3,.5f)};mesh.triangles=new[]{0,1,2};mesh.uv=new[]{Vector2.zero,Vector2.right*2,Vector2.up*2};mesh.colors=new[]{Color.white,Color.white,Color.white};mesh.RecalculateBounds();
   var off=Vector4.zero;var on=new Vector4(1,0,1,.35f);Fill(texture,new Color(.2f,.4f,.6f,1));var sumOff=Draw(material,mesh,rt,read,off);var sumOn=Draw(material,mesh,rt,read,on);bool fixture=sumOff.r>2;bool summedBudget=sumOn.r<=.401f&&sumOn.g<=.801f&&sumOn.b<=1.201f&&sumOn.r>.35f;var receiverOff=Draw(material,mesh,rt,read,new Vector4(0,1,1,2));bool moduleExcluded=Diff(receiverOff,sumOff)<.001f;
   Shader.SetGlobalFloat("_EnvironmentSurfaceRole",-1);var infoOff=Draw(material,mesh,rt,read,off);var infoOn=Draw(material,mesh,rt,read,on);bool infoExcluded=Diff(infoOff,infoOn)<.001f&&Mathf.Abs(infoOn.r-.2f)<.003f;Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);
   Fill(texture,new Color(0,0,0,.5f));var black=Draw(material,mesh,rt,read,on);bool blackDoesNotGlow=black.r==0&&black.g==0&&black.b==0&&Mathf.Abs(black.a-.5f)<.0001f;
   Fill(texture,new Color(.1f,.2f,.3f,0));var additiveOff=Draw(material,mesh,rt,read,off);var additiveOn=Draw(material,mesh,rt,read,on);bool additiveFixture=additiveOff.r>1;bool additiveExcluded=Diff(additiveOff,additiveOn)<.001f&&additiveOn.a==0;
   bool shaderErrors=ShaderUtil.GetShaderMessages(shader).Any(m=>m.severity.ToString()=="Error");bool pass=fixture&&summedBudget&&moduleExcluded&&infoExcluded&&blackDoesNotGlow&&additiveFixture&&additiveExcluded&&!shaderErrors;ctx.ReturnValue=JsonConvert.SerializeObject(new{pass,fixture,summedBudget,moduleExcluded,infoExcluded,blackDoesNotGlow,additiveFixture,additiveExcluded,shaderErrors,sumOff=sumOff.ToString(),sumOn=sumOn.ToString(),additiveOff=additiveOff.ToString(),additiveOn=additiveOn.ToString()});
  }
  finally
  {
   foreach(var p in oldF)Shader.SetGlobalFloat(p.Key,p.Value);foreach(var p in oldN)Shader.SetGlobalInteger(p.Key,p.Value);foreach(var p in oldV)Shader.SetGlobalVector(p.Key,p.Value);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",inverse);Shader.SetGlobalVectorArray("_EnvironmentLocalLightPositions",oldPositions!=null&&oldPositions.Length>0?oldPositions:new Vector4[8]);Shader.SetGlobalVectorArray("_EnvironmentLocalLightColors",oldColors!=null&&oldColors.Length>0?oldColors:new Vector4[8]);RenderTexture.active=previous;if(rt!=null)rt.Release();foreach(var o in new UnityEngine.Object[]{material,texture,read,mesh,rt})if(o!=null)UnityEngine.Object.DestroyImmediate(o);
  }
 }
}
