/* 功能：在正式WorldTile片元中检查只给环境贡献染色，局部灯不受重复染色，保留旧暗部与许可/alpha分支。
 * 边界：不修改生产资产；没有功能时应报告颜色合同失败，而非fixture/编译失败。
 * 生命周期：私有GPU对象finally释放，所有借用Shader全局与RT活动目标完整恢复。
 */
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public class CharacterAmbientTintContract:IFunplayCommand
{
 static readonly Vector4 Off=new Vector4(1,1,1,0),Warm=new Vector4(1,.45f,.25f,1),Cool=new Vector4(.25f,.45f,1,1);
 static float Luma(Color c)=>c.r*.2126f+c.g*.7152f+c.b*.0722f;
 static float Diff(Color a,Color b)=>Mathf.Max(Mathf.Abs(a.r-b.r),Mathf.Abs(a.g-b.g),Mathf.Abs(a.b-b.b));
 static bool Same(Color a,Color b)=>Diff(a,b)<.001f&&Mathf.Abs(a.a-b.a)<.0001f;
 static void Ambient(float x){foreach(string n in new[]{"_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient"})Shader.SetGlobalVector(n,new Vector4(x,x,x,0));}
 static Color Draw(Material material,Mesh mesh,RenderTexture rt,Texture2D read,Vector4 tint)
 {
  Shader.SetGlobalVector("_EnvironmentStylizedSkyTint",tint);var cmd=new CommandBuffer{name="Character ambient contribution GPU contract"};
  try{cmd.SetRenderTarget(rt);cmd.SetViewport(new Rect(0,0,16,16));cmd.ClearRenderTarget(false,true,Color.clear);cmd.SetViewProjectionMatrices(Matrix4x4.identity,Matrix4x4.identity);cmd.DrawMesh(mesh,Matrix4x4.identity,material,0,0);Graphics.ExecuteCommandBuffer(cmd);}finally{cmd.Release();}
  RenderTexture.active=rt;read.ReadPixels(new Rect(0,0,16,16),0,0);read.Apply();return read.GetPixel(8,8);
 }
 public void Execute(ExecutionContext ctx)
 {
  if(Application.dataPath!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new Exception("Project guard");
  var shader=Shader.Find("Hidden/Ossastra/Rooms/SightWorldTile");if(shader==null||!shader.isSupported)throw new Exception("WorldTile missing/unsupported");
  var fs=new[]{"_EnvironmentLightingEnabled","_EnvironmentIndoorAmbientScale","_EnvironmentWetness","_RoomSightStaticEnvironment","_EnvironmentFullSurface","_EnvironmentSurfaceRole","_EnvironmentResponseOverride","_EnvironmentReceiveShadow","_EnvironmentMinimumOverride"};
  var ns=new[]{"_EnvironmentIndoorCount","_EnvironmentLocalLightCount","_EnvironmentShelterCount","_RoomSightOwnerCount","_RoomSightOwnerExterior","_RoomSightSeedCount","_RoomSightBuildingCount","_RoomSightDomainCount","_RoomSightBlockerCount","_RoomSightWindowCount","_RoomSightCameraBlockerCount","_RoomSightExteriorView","_RoomSightPassMode"};
  var vs=new[]{"_EnvironmentStylizedSkyTint","_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient","_EnvironmentLightResponse","_EnvironmentCloudParameters","_EnvironmentCloudMotion","_MainLightPosition","_MainLightColor","_RoomSightTargetSize","_CharacterLightControls","_CharacterLightShoulder"};
  var f=fs.ToDictionary(n=>n,n=>Shader.GetGlobalFloat(n));var ints=ns.ToDictionary(n=>n,n=>Shader.GetGlobalInteger(n));var v=vs.ToDictionary(n=>n,n=>Shader.GetGlobalVector(n));var inverse=Shader.GetGlobalMatrix("_RoomSightInverseViewProjection");var oldMatrices=Shader.GetGlobalMatrixArray("_EnvironmentIndoorMatrices");var oldExtents=Shader.GetGlobalVectorArray("_EnvironmentIndoorHalfExtents");var oldPositions=Shader.GetGlobalVectorArray("_EnvironmentLocalLightPositions");var oldColors=Shader.GetGlobalVectorArray("_EnvironmentLocalLightColors");var previous=RenderTexture.active;
  Material mat=null;Texture2D texture=null,read=null;Mesh mesh=null;RenderTexture rt=null;
  try
  {
   foreach(var n in fs)Shader.SetGlobalFloat(n,0);foreach(var n in ns)Shader.SetGlobalInteger(n,0);Shader.SetGlobalVector("_CharacterLightControls",new Vector4(1,0,0,.35f));
   Shader.SetGlobalInteger("_RoomSightExteriorView",1);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",Matrix4x4.identity);Shader.SetGlobalVector("_RoomSightTargetSize",new Vector4(16,16,1f/16,1f/16));
   mat=new Material(shader){hideFlags=HideFlags.HideAndDontSave};texture=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true);texture.SetPixels(Enumerable.Repeat(new Color(.5f,.5f,.5f,1),4).ToArray());texture.Apply();read=new Texture2D(16,16,TextureFormat.RGBAFloat,false,true);rt=new RenderTexture(16,16,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);if(!rt.Create())throw new Exception("GPU target creation failed");
   mat.SetTexture("_MainTex",Texture2D.whiteTexture);mat.SetTexture("_SecondTex",texture);mat.SetColor("_Color",Color.white);mat.SetVector("_GroundBounds",new Vector4(-1000,-1000,1000,1000));mat.SetFloat("_BaseCoverage",1);mat.SetFloat("_DebugMask",0);mat.SetFloat("_WorldPeriod",1);
   mesh=new Mesh();mesh.vertices=new[]{new Vector3(-1,-1,.5f),new Vector3(3,-1,.5f),new Vector3(-1,3,.5f)};mesh.triangles=new[]{0,1,2};mesh.uv=new[]{Vector2.zero,Vector2.right*2,Vector2.up*2};mesh.colors=new[]{Color.white,Color.white,Color.white};mesh.RecalculateBounds();
   var matrices=new Matrix4x4[32];matrices[0]=Matrix4x4.identity;var extents=new Vector4[32];extents[0]=new Vector4(10,10,10,0);Shader.SetGlobalMatrixArray("_EnvironmentIndoorMatrices",matrices);Shader.SetGlobalVectorArray("_EnvironmentIndoorHalfExtents",extents);
   var positions=new Vector4[8];positions[0]=new Vector4(0,0,2,10);var colors=new Vector4[8];colors[0]=new Vector4(.1f,.3f,.8f,0);Shader.SetGlobalVectorArray("_EnvironmentLocalLightPositions",positions);Shader.SetGlobalVectorArray("_EnvironmentLocalLightColors",colors);
   Shader.SetGlobalFloat("_EnvironmentLightingEnabled",1);Shader.SetGlobalFloat("_EnvironmentFullSurface",1);Shader.SetGlobalFloat("_EnvironmentResponseOverride",1);Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);Shader.SetGlobalFloat("_EnvironmentIndoorAmbientScale",.22f);Shader.SetGlobalVector("_EnvironmentLightResponse",new Vector4(1,1,1,0));Shader.SetGlobalVector("_EnvironmentCloudParameters",Vector4.zero);Shader.SetGlobalVector("_EnvironmentCloudMotion",Vector4.zero);Shader.SetGlobalVector("_MainLightPosition",new Vector4(0,1,0,0));Shader.SetGlobalVector("_MainLightColor",Vector4.zero);
   Ambient(.4f);var baseline=Draw(mat,mesh,rt,read,Off);Ambient(.8f);var bright=Draw(mat,mesh,rt,read,Off);Ambient(.4f);var warm=Draw(mat,mesh,rt,read,Warm);var cool=Draw(mat,mesh,rt,read,Cool);var zero=Draw(mat,mesh,rt,read,new Vector4(.25f,1,.25f,0));
   bool validFixture=Mathf.Abs(baseline.r-.2f)<.003f&&Mathf.Abs(bright.r-.4f)<.003f;
   bool tintWarm=Diff(baseline,warm)>.02f&&warm.r>warm.g&&warm.g>warm.b,tintCool=Diff(baseline,cool)>.02f&&cool.b>cool.g&&cool.g>cool.r,ambientLuminance=Mathf.Abs(Luma(warm)-Luma(baseline))<.001f&&Mathf.Abs(Luma(cool)-Luma(baseline))<.001f;
   Ambient(0);Shader.SetGlobalInteger("_EnvironmentLocalLightCount",1);var localOff=Draw(mat,mesh,rt,read,Off);var localWarm=Draw(mat,mesh,rt,read,Warm);bool localFixture=Luma(localOff)>.01f;bool localUnchanged=Same(localOff,localWarm);
   Ambient(.4f);var mixedOff=Draw(mat,mesh,rt,read,Off);var mixedWarm=Draw(mat,mesh,rt,read,Warm);bool localNotRetinted=Diff(mixedWarm-mixedOff,warm-baseline)<.0015f;
   Shader.SetGlobalInteger("_EnvironmentLocalLightCount",0);Shader.SetGlobalInteger("_EnvironmentIndoorCount",1);var indoorOff=Draw(mat,mesh,rt,read,Off);var indoorWarm=Draw(mat,mesh,rt,read,Warm);bool indoorUnchanged=Same(indoorOff,indoorWarm)&&indoorOff.r<baseline.r*.5f;Shader.SetGlobalInteger("_EnvironmentIndoorCount",0);
   Shader.SetGlobalFloat("_EnvironmentSurfaceRole",-1);var infoOff=Draw(mat,mesh,rt,read,Off);var infoWarm=Draw(mat,mesh,rt,read,Warm);bool informationUnchanged=Same(infoOff,infoWarm);Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);Shader.SetGlobalVector("_EnvironmentLightResponse",new Vector4(1,1,.7f,.12f));Ambient(.02f);var floorOff=Draw(mat,mesh,rt,read,Off);var floorWarm=Draw(mat,mesh,rt,read,Warm);bool oldNightFloor=Mathf.Abs(floorOff.r-.192f)<.003f&&Mathf.Abs(floorOff.g-.192f)<.003f&&Mathf.Abs(floorOff.b-.192f)<.003f&&Mathf.Abs(Luma(floorWarm)-Luma(floorOff))<.001f;
   Shader.SetGlobalFloat("_EnvironmentLightingEnabled",0);var disabledOff=Draw(mat,mesh,rt,read,Off);var disabledWarm=Draw(mat,mesh,rt,read,Warm);bool disabledUnchanged=Same(disabledOff,disabledWarm)&&Mathf.Abs(disabledOff.r-.5f)<.003f;
   var all=new[]{baseline,bright,warm,cool,zero,localOff,localWarm,mixedOff,mixedWarm,indoorOff,indoorWarm,infoOff,infoWarm,floorOff,floorWarm,disabledOff,disabledWarm};bool alpha=all.All(c=>Mathf.Abs(c.a-1)<.0001f),finite=all.All(c=>!float.IsNaN(c.r)&&!float.IsInfinity(c.r)&&!float.IsNaN(c.g)&&!float.IsInfinity(c.g)&&!float.IsNaN(c.b)&&!float.IsInfinity(c.b));bool shaderErrors=ShaderUtil.GetShaderMessages(shader).Any(m=>m.severity.ToString()=="Error");
   bool pass=validFixture&&localFixture&&!shaderErrors&&tintWarm&&tintCool&&ambientLuminance&&Same(zero,baseline)&&localUnchanged&&localNotRetinted&&indoorUnchanged&&informationUnchanged&&oldNightFloor&&disabledUnchanged&&alpha&&finite;
   ctx.ReturnValue=JsonConvert.SerializeObject(new{pass,validFixture,localFixture,shaderErrors,tintWarm,tintCool,ambientLuminance,strengthZero=Same(zero,baseline),localUnchanged,localNotRetinted,indoorUnchanged,informationUnchanged,oldNightFloor,disabledUnchanged,alpha,finite,baseline=baseline.ToString(),warm=warm.ToString(),cool=cool.ToString(),local=localOff.ToString(),mixedOff=mixedOff.ToString(),mixedWarm=mixedWarm.ToString(),floorOff=floorOff.ToString(),floorWarm=floorWarm.ToString()});
  }
  finally
  {
   foreach(var p in f)Shader.SetGlobalFloat(p.Key,p.Value);foreach(var p in ints)Shader.SetGlobalInteger(p.Key,p.Value);foreach(var p in v)Shader.SetGlobalVector(p.Key,p.Value);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",inverse);Shader.SetGlobalMatrixArray("_EnvironmentIndoorMatrices",oldMatrices!=null&&oldMatrices.Length>0?oldMatrices:new Matrix4x4[32]);Shader.SetGlobalVectorArray("_EnvironmentIndoorHalfExtents",oldExtents!=null&&oldExtents.Length>0?oldExtents:new Vector4[32]);Shader.SetGlobalVectorArray("_EnvironmentLocalLightPositions",oldPositions!=null&&oldPositions.Length>0?oldPositions:new Vector4[8]);Shader.SetGlobalVectorArray("_EnvironmentLocalLightColors",oldColors!=null&&oldColors.Length>0?oldColors:new Vector4[8]);RenderTexture.active=previous;if(rt!=null)rt.Release();foreach(var o in new UnityEngine.Object[]{mat,texture,read,mesh,rt})if(o!=null)UnityEngine.Object.DestroyImmediate(o);
  }
 }
}
