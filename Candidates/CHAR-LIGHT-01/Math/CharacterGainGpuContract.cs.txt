/* 功能：使用正式Spine及DepthPlane片元检查角色合计光照增益软肩，不依赖Bloom或虚构数学副本。
 * 边界：只测试当前两个真实shader家族与两输入alpha路径；私有RT不授予任何主场显示权限。
 * 生命周期：借用全局完整恢复，所有私有材质/纹理/mesh/RT及命令缓冲finally释放。
 */
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public class CharacterGainGpuContract:IFunplayCommand
{
 static float Luma(Color c)=>c.r*.2126f+c.g*.7152f+c.b*.0722f;
 static float Diff(Color a,Color b)=>Mathf.Max(Mathf.Abs(a.r-b.r),Mathf.Abs(a.g-b.g),Mathf.Abs(a.b-b.b));
 static void Ambient(float x){foreach(var n in new[]{"_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient"})Shader.SetGlobalVector(n,new Vector4(x,x,x,0));}
 static void Fill(Texture2D t,float alpha,bool pma){var c=new Color(.2f,.4f,.6f,alpha);if(pma){c.r*=alpha;c.g*=alpha;c.b*=alpha;}t.SetPixels(Enumerable.Repeat(c,4).ToArray());t.Apply();}
 static Color Draw(Material m,Mesh mesh,RenderTexture rt,Texture2D read,Vector4 controls)
 {
  Shader.SetGlobalVector("_CharacterLightControls",controls);var cmd=new CommandBuffer{name="Character gain protection actual shader contract"};
  try{cmd.SetRenderTarget(rt);cmd.SetViewport(new Rect(0,0,16,16));cmd.ClearRenderTarget(false,true,Color.clear);cmd.SetViewProjectionMatrices(Matrix4x4.identity,Matrix4x4.identity);cmd.DrawMesh(mesh,Matrix4x4.identity,m,0,1);Graphics.ExecuteCommandBuffer(cmd);}finally{cmd.Release();}
  RenderTexture.active=rt;read.ReadPixels(new Rect(0,0,16,16),0,0);read.Apply();return read.GetPixel(8,8);
 }
 public void Execute(ExecutionContext ctx)
 {
  if(Application.dataPath!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new Exception("Project guard");
  var shaders=new[]{Shader.Find("Hidden/Ossastra/Rooms/SightSpine"),Shader.Find("Hidden/Ossastra/Rooms/SightDepthPlane")};if(shaders.Any(s=>s==null||!s.isSupported))throw new Exception("Actual character shaders missing/unsupported");
  var fs=new[]{"_EnvironmentLightingEnabled","_EnvironmentIndoorAmbientScale","_EnvironmentWetness","_RoomSightStaticEnvironment","_EnvironmentFullSurface","_EnvironmentSurfaceRole","_EnvironmentResponseOverride","_EnvironmentReceiveShadow","_EnvironmentMinimumOverride"};
  var ns=new[]{"_EnvironmentIndoorCount","_EnvironmentLocalLightCount","_EnvironmentShelterCount","_RoomSightOwnerCount","_RoomSightOwnerExterior","_RoomSightSeedCount","_RoomSightBuildingCount","_RoomSightDomainCount","_RoomSightBlockerCount","_RoomSightWindowCount","_RoomSightCameraBlockerCount","_RoomSightExteriorView","_RoomSightPassMode"};
  var vs=new[]{"_EnvironmentStylizedSkyTint","_EnvironmentSkyAmbient","_EnvironmentHorizonAmbient","_EnvironmentGroundAmbient","_EnvironmentLightResponse","_EnvironmentCloudParameters","_EnvironmentCloudMotion","_MainLightPosition","_MainLightColor","_RoomSightTargetSize","_CharacterLightControls","_CharacterLightShoulder"};
  var oldF=fs.ToDictionary(n=>n,n=>Shader.GetGlobalFloat(n));var oldN=ns.ToDictionary(n=>n,n=>Shader.GetGlobalInteger(n));var oldV=vs.ToDictionary(n=>n,n=>Shader.GetGlobalVector(n));var inverse=Shader.GetGlobalMatrix("_RoomSightInverseViewProjection");var previous=RenderTexture.active;
  Texture2D texture=null,read=null;Mesh mesh=null;RenderTexture rt=null;var materials=new List<Material>();var reports=new List<object>();
  try
  {
   foreach(var n in fs)Shader.SetGlobalFloat(n,0);foreach(var n in ns)Shader.SetGlobalInteger(n,0);Shader.SetGlobalVector("_EnvironmentStylizedSkyTint",new Vector4(1,1,1,0));Shader.SetGlobalVector("_CharacterLightShoulder",new Vector4(1.2f,2,0,0));
   Shader.SetGlobalInteger("_RoomSightExteriorView",1);Shader.SetGlobalFloat("_EnvironmentFullSurface",1);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",Matrix4x4.identity);Shader.SetGlobalVector("_RoomSightTargetSize",new Vector4(16,16,1f/16,1f/16));
   Shader.SetGlobalFloat("_EnvironmentLightingEnabled",1);Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);Shader.SetGlobalFloat("_EnvironmentResponseOverride",1);Shader.SetGlobalVector("_EnvironmentLightResponse",new Vector4(1,1,1,0));Shader.SetGlobalVector("_MainLightPosition",new Vector4(0,1,0,0));Shader.SetGlobalVector("_MainLightColor",Vector4.zero);Shader.SetGlobalVector("_EnvironmentCloudParameters",Vector4.zero);Shader.SetGlobalVector("_EnvironmentCloudMotion",Vector4.zero);
   texture=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true);read=new Texture2D(16,16,TextureFormat.RGBAFloat,false,true);rt=new RenderTexture(16,16,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);if(!rt.Create())throw new Exception("Private float GPU target failed");
   mesh=new Mesh();mesh.vertices=new[]{new Vector3(-1,-1,.5f),new Vector3(3,-1,.5f),new Vector3(-1,3,.5f)};mesh.triangles=new[]{0,1,2};mesh.uv=new[]{Vector2.zero,Vector2.right*2,Vector2.up*2};mesh.colors=new[]{Color.white,Color.white,Color.white};mesh.RecalculateBounds();
   var off=Vector4.zero;var on=new Vector4(1,0,1,.35f);bool all=true;
   foreach(var shader in shaders)
   {
    var mat=new Material(shader){hideFlags=HideFlags.HideAndDontSave};materials.Add(mat);mat.SetTexture("_MainTex",texture);if(mat.HasProperty("_Color"))mat.SetColor("_Color",Color.white);if(mat.HasProperty("_Cutoff"))mat.SetFloat("_Cutoff",.001f);
    bool spine=shader.name.EndsWith("SightSpine");bool modePass=true;
    for(int mode=0;mode<2;mode++)
    {
     bool pma=mode==1;if(spine){mat.SetFloat("_StraightAlphaInput",pma?0:1);if(pma)mat.DisableKeyword("_STRAIGHT_ALPHA_INPUT");else mat.EnableKeyword("_STRAIGHT_ALPHA_INPUT");}else mat.SetFloat("_PmaInput",pma?1:0);
     Fill(texture,1,pma);Ambient(.4f);var normalOff=Draw(mat,mesh,rt,read,off);var normalOn=Draw(mat,mesh,rt,read,on);Ambient(16);var highOff=Draw(mat,mesh,rt,read,off);var highOn=Draw(mat,mesh,rt,read,on);var noStrength=Draw(mat,mesh,rt,read,new Vector4(1,0,0,.35f));
     bool fixture=Mathf.Abs(normalOff.r-.08f)<.004f&&highOff.r>3;bool normalSame=Diff(normalOff,normalOn)<.001f;bool limits=highOn.r<.41f&&highOn.g<.81f&&highOn.b<1.21f&&highOn.r>normalOn.r+.1f;bool zero=Diff(highOff,noStrength)<.001f;bool hue=Mathf.Abs(highOn.r*2-highOn.g)<.0015f&&Mathf.Abs(highOn.r*3-highOn.b)<.002f;
     float[] values={1,2,4,16};var curve=new float[4];for(int i=0;i<values.Length;i++){Ambient(values[i]);curve[i]=Luma(Draw(mat,mesh,rt,read,on));}bool monotonic=curve[1]>curve[0]&&curve[2]>curve[1]&&curve[3]>=curve[2]-.0001f;bool softShoulder=(curve[1]-curve[0])>(curve[2]-curve[1])/2&&(curve[2]-curve[1])/2>(curve[3]-curve[2])/12;
     bool alphaSame=true,pmaSame=true;foreach(float alpha in new[]{.5f,.125f}){Fill(texture,alpha,pma);Ambient(16);var edgeOff=Draw(mat,mesh,rt,read,off);var edgeOn=Draw(mat,mesh,rt,read,on);alphaSame&=Mathf.Abs(edgeOff.a-edgeOn.a)<.0001f;pmaSame&=Diff(edgeOn/alpha,highOn)<.003f;}
     Shader.SetGlobalFloat("_EnvironmentSurfaceRole",1);Fill(texture,1,pma);Ambient(16);var decorOff=Draw(mat,mesh,rt,read,off);var decorOn=Draw(mat,mesh,rt,read,on);bool decorationSame=Diff(decorOff,decorOn)<.001f;Shader.SetGlobalFloat("_EnvironmentSurfaceRole",2);
     Shader.SetGlobalFloat("_EnvironmentLightingEnabled",0);var disabledOff=Draw(mat,mesh,rt,read,off);var disabledOn=Draw(mat,mesh,rt,read,on);bool disabledSame=Diff(disabledOff,disabledOn)<.001f;Shader.SetGlobalFloat("_EnvironmentLightingEnabled",1);
     bool finite=new[]{normalOff,normalOn,highOff,highOn}.All(c=>!float.IsNaN(c.r)&&!float.IsInfinity(c.r)&&!float.IsNaN(c.g)&&!float.IsInfinity(c.g)&&!float.IsNaN(c.b)&&!float.IsInfinity(c.b));bool pass=fixture&&normalSame&&limits&&zero&&hue&&monotonic&&softShoulder&&alphaSame&&pmaSame&&decorationSame&&disabledSame&&finite;modePass&=pass;
     reports.Add(new{shader=shader.name,pma,pass,fixture,normalSame,limits,zero,hue,monotonic,softShoulder,alphaSame,pmaSame,decorationSame,disabledSame,finite,normalOff=normalOff.ToString(),normalOn=normalOn.ToString(),highOff=highOff.ToString(),highOn=highOn.ToString(),curve});
    }
    all&=modePass;
   }
   bool shaderErrors=shaders.Any(s=>ShaderUtil.GetShaderMessages(s).Any(m=>m.severity.ToString()=="Error"));ctx.ReturnValue=JsonConvert.SerializeObject(new{pass=all&&!shaderErrors,shaderErrors,reports});
  }
  finally
  {
   foreach(var p in oldF)Shader.SetGlobalFloat(p.Key,p.Value);foreach(var p in oldN)Shader.SetGlobalInteger(p.Key,p.Value);foreach(var p in oldV)Shader.SetGlobalVector(p.Key,p.Value);Shader.SetGlobalMatrix("_RoomSightInverseViewProjection",inverse);RenderTexture.active=previous;if(rt!=null)rt.Release();foreach(var m in materials)if(m!=null)UnityEngine.Object.DestroyImmediate(m);foreach(var o in new UnityEngine.Object[]{texture,read,mesh,rt})if(o!=null)UnityEngine.Object.DestroyImmediate(o);
  }
 }
}
