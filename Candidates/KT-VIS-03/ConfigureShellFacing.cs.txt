// Specific imported open-shell model adaptation; full 3D models are not auto-classified by size/name.
using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
public sealed class Vis03ConfigureShellFacing:IFunplayCommand
{
 public void Execute(ExecutionContext ctx){var s=SceneManager.GetActiveScene();if(Application.isPlaying||EditorApplication.isCompiling||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/CRAFT-01/SC_CraftingIntegration.unity"||SceneManager.sceneCount!=1||PrefabStageUtility.GetCurrentPrefabStage()!=null)throw new Exception("Wrong Edit source");const string meshPath="Assets/OSSASTRA/Art/WorldPropsReferenceLocal/KT_Stone3D.asset";var mesh=AssetDatabase.LoadAssetAtPath<Mesh>(meshPath);if(mesh==null)throw new Exception("Missing audited mesh");var world=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Ossastra.WorldProps.WorldPropsRoot>(true)).Single();var props=world.props.Where(p=>new[]{"mesh-a","decor-a","platform-a"}.Contains(p.instanceId)).ToArray();if(props.Length!=3)throw new Exception("Exact shell instance set mismatch");foreach(var p in props){var body=p.visualRoot.GetComponentsInChildren<MeshFilter>(true).Single();if(body.sharedMesh!=mesh||body.transform.parent!=p.visualRoot||body.GetComponent<Renderer>().sharedMaterial.GetFloat("_Cull")!=2)throw new Exception("Source mesh/material mismatch");ctx.RegisterObjectModification(p);ctx.RegisterObjectModification(p.visualRoot);ctx.RegisterObjectModification(body.transform);p.faceCamera=true;p.visualRoot.rotation=Quaternion.Euler(0,world.gameplayCamera.transform.eulerAngles.y,0);body.transform.localRotation=Quaternion.Euler(0,90,0);EditorUtility.SetDirty(p);}
 const string path="Assets/OSSASTRA/WorldProps/Prefabs/WorldProp_3D_Template.prefab";var root=PrefabUtility.LoadPrefabContents(path);try{var p=root.GetComponent<Ossastra.WorldProps.WorldPropRoot>();if(p==null||p.visualRoot==null)throw new Exception("Missing explicit prefab view");var body=p.visualRoot.GetComponentsInChildren<MeshFilter>(true).Single();if(body.sharedMesh!=mesh)throw new Exception("Prefab not audited shell");ctx.RegisterObjectModification(p);ctx.RegisterObjectModification(p.visualRoot);ctx.RegisterObjectModification(body.transform);p.faceCamera=true;p.visualRoot.localRotation=Quaternion.Euler(0,45,0);body.transform.localRotation=Quaternion.Euler(0,90,0);if(PrefabUtility.SaveAsPrefabAsset(root,path)==null)throw new Exception("Prefab save failed");}finally{PrefabUtility.UnloadPrefabContents(root);}EditorSceneManager.MarkSceneDirty(s);ctx.ReturnValue=new{specificMesh=meshPath,instances=props.Select(p=>new{p.instanceId,p.faceCamera,rootPosition=p.transform.position.ToString("R"),visualYaw=p.visualRoot.eulerAngles.y,bodyLocalYaw=p.visualRoot.GetComponentInChildren<MeshFilter>().transform.localEulerAngles.y}).ToArray(),prefabSaved=path,cullingRetained=2,sceneSaved=false};}
}
