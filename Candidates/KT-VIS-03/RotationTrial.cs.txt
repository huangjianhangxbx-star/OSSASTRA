// Tests source-authored facing as the only variable. No culling/material/geometry edit.
using System;using System.IO;using System.Linq;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public sealed class Vis03RotationTrial:IFunplayCommand
{
 public void Execute(ExecutionContext ctx){var w=UnityEngine.Object.FindObjectsOfType<Ossastra.WorldProps.WorldPropsRoot>().Single();if(!Application.isPlaying||w.commerce.profileId!="kt-vis03-01"||w.commerce.saveDirectory!="E:/WORLDCREATOR/XingHaiHuiLang/研究/可见性修复/KT-VIS-03/profiles")throw new Exception("Wrong isolated context");var renderers=w.props.Where(p=>p.instanceId=="mesh-a"||p.instanceId=="decor-a").Select(p=>p.visualRoot.GetComponentInChildren<MeshRenderer>()).ToArray();if(renderers.Length!=2)throw new Exception("Wrong exact renderers");var old=renderers.Select(r=>r.transform.rotation).ToArray();var c=w.gameplayCamera;var oldTarget=c.targetTexture;var oldActive=RenderTexture.active;var rt=new RenderTexture(c.pixelWidth,c.pixelHeight,24,RenderTextureFormat.ARGB32);var t=new Texture2D(c.pixelWidth,c.pixelHeight,TextureFormat.RGB24,false);string output="E:/WORLDCREATOR/XingHaiHuiLang/研究/可见性修复/KT-VIS-03/";try{rt.Create();c.targetTexture=rt;c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);t.Apply();File.WriteAllBytes(output+"rotation-trial-before.png",t.EncodeToPNG());foreach(var r in renderers)r.transform.rotation=Quaternion.Euler(0,135,0);c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);t.Apply();File.WriteAllBytes(output+"rotation-trial-after.png",t.EncodeToPNG());ctx.ReturnValue=new{oneVariable="Source-authored body Y135 instead of lost Y0; materials Cull2 and mesh unchanged",before=output+"rotation-trial-before.png",after=output+"rotation-trial-after.png"};}finally{for(int i=0;i<2;i++)renderers[i].transform.rotation=old[i];c.targetTexture=oldTarget;RenderTexture.active=oldActive;UnityEngine.Object.DestroyImmediate(t);rt.Release();UnityEngine.Object.DestroyImmediate(rt);}}
}
