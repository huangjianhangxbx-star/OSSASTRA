// One-variable culling ablation. Own material copies only; original source assets and slots restored.
using System;using System.IO;using System.Linq;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public sealed class Vis03CullTrial:IFunplayCommand
{
 public void Execute(ExecutionContext ctx){var w=UnityEngine.Object.FindObjectsOfType<Ossastra.WorldProps.WorldPropsRoot>().Single();if(!Application.isPlaying||w.commerce.profileId!="kt-vis03-01"||w.commerce.saveDirectory!="E:/WORLDCREATOR/XingHaiHuiLang/研究/可见性修复/KT-VIS-03/profiles")throw new Exception("Wrong isolated context");var renderers=w.props.Where(p=>p.instanceId=="mesh-a"||p.instanceId=="decor-a").Select(p=>p.visualRoot.GetComponentInChildren<MeshRenderer>()).ToArray();if(renderers.Length!=2||renderers.Any(r=>r==null||r.sharedMaterials.Length!=1||!r.sharedMaterial.HasProperty("_Cull")))throw new Exception("Wrong exact source renderers");var old=renderers.Select(r=>r.sharedMaterial).ToArray();var own=old.Select(m=>new Material(m){hideFlags=HideFlags.HideAndDontSave}).ToArray();var c=w.gameplayCamera;var oldTarget=c.targetTexture;var oldActive=RenderTexture.active;var rt=new RenderTexture(c.pixelWidth,c.pixelHeight,24,RenderTextureFormat.ARGB32);var t=new Texture2D(c.pixelWidth,c.pixelHeight,TextureFormat.RGB24,false);string output="E:/WORLDCREATOR/XingHaiHuiLang/研究/可见性修复/KT-VIS-03/";try{rt.Create();c.targetTexture=rt;c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);t.Apply();File.WriteAllBytes(output+"cull-trial-before.png",t.EncodeToPNG());for(int i=0;i<2;i++){own[i].SetFloat("_Cull",0);renderers[i].sharedMaterial=own[i];}c.Render();RenderTexture.active=rt;t.ReadPixels(new Rect(0,0,rt.width,rt.height),0,0);t.Apply();File.WriteAllBytes(output+"cull-trial-after.png",t.EncodeToPNG());ctx.ReturnValue=new{oneVariable="Only cloned material Cull2 -> Cull0; no source geometry, texture, light, pose or ground change",before=output+"cull-trial-before.png",after=output+"cull-trial-after.png"};}finally{for(int i=0;i<2;i++)renderers[i].sharedMaterial=old[i];c.targetTexture=oldTarget;RenderTexture.active=oldActive;UnityEngine.Object.DestroyImmediate(t);rt.Release();UnityEngine.Object.DestroyImmediate(rt);foreach(var m in own)UnityEngine.Object.DestroyImmediate(m);}}
}
