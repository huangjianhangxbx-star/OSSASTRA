using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;using Ossastra.SceneVisibility;
public sealed class Vis03ConfigureLocalPolicy:IFunplayCommand
{
 public void Execute(ExecutionContext ctx){var s=SceneManager.GetActiveScene();if(Application.isPlaying||EditorApplication.isCompiling||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/CRAFT-01/SC_CraftingIntegration.unity"||AssetDatabase.AssetPathToGUID(s.path)!="d694fa04240ccaf40808408e233c70ba"||SceneManager.sceneCount!=1||PrefabStageUtility.GetCurrentPrefabStage()!=null)throw new Exception("Wrong own Edit scene");var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(v=>v!=null).ToArray();var vis=all.OfType<SceneVisibilityRoot>().Single();var world=all.OfType<Ossastra.WorldProps.WorldPropsRoot>().Single();var craft=all.OfType<Ossastra.Crafting.CraftingRoot>().Single();if(world.props.Length!=17||craft.stations.Length!=3)throw new Exception("Unexpected sample counts");var groups=world.props.Select(p=>p.visibilityGroup).Concat(craft.stations.Select(p=>p.GetComponent<SceneOccluderGroup>())).ToArray();if(groups.Length!=20||groups.Any(g=>g==null||!vis.groups.Contains(g))||groups.Distinct().Count()!=20)throw new Exception("Wrong exact sample group references");foreach(var group in groups){ctx.RegisterObjectModification(group);group.occlusionPolicy=SceneOcclusionPolicy.KeepNaturalOcclusion;EditorUtility.SetDirty(group);}EditorSceneManager.MarkSceneDirty(s);
 var prefabs=new[]{"Assets/OSSASTRA/WorldProps/Prefabs/WorldProp_2D_Template.prefab","Assets/OSSASTRA/WorldProps/Prefabs/WorldProp_3D_Template.prefab","Assets/OSSASTRA/Crafting/Prefabs/Station2D.prefab","Assets/OSSASTRA/Crafting/Prefabs/Station3D.prefab"};foreach(var path in prefabs){if(AssetDatabase.LoadAssetAtPath<GameObject>(path)==null)throw new Exception("Missing explicit prefab:"+path);var root=PrefabUtility.LoadPrefabContents(path);try{var group=root.GetComponent<SceneOccluderGroup>();if(group==null)throw new Exception("Prefab has no own group:"+path);ctx.RegisterObjectModification(group);group.occlusionPolicy=SceneOcclusionPolicy.KeepNaturalOcclusion;if(PrefabUtility.SaveAsPrefabAsset(root,path)==null)throw new Exception("Prefab save failed:"+path);}finally{PrefabUtility.UnloadPrefabContents(root);}}
 ctx.ReturnValue=new{scene=s.path,normalOcclusion=groups.Select(g=>g.name).ToArray(),unchangedLarge=vis.groups.Except(groups).Select(g=>new{g.name,policy=g.occlusionPolicy.ToString()}).ToArray(),prefabsSaved=prefabs,sourceSceneSaved=false};}
}
