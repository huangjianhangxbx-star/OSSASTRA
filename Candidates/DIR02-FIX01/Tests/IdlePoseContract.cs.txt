using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using Spine;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class IdlePoseContract:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var root=UnityEngine.Object.FindObjectsOfType<C.CharacterRoot>().Single(r=>r.ActorId=="temporary-a");var body=root.GetComponentInChildren<A.CharacterAnimationController>().Skeleton;var current=body.Skeleton;var track=body.AnimationState.GetTrack(0);if(root.ActionActive||track?.Animation?.Name!="_stand")throw new Exception("Need actual post-action Idle");var fresh=new Skeleton(current.Data);fresh.ScaleX=current.ScaleX;fresh.ScaleY=current.ScaleY;var state=new Spine.AnimationState(new AnimationStateData(current.Data));state.SetAnimation(0,"_stand",true).TrackTime=track.TrackTime;var structure=body.AnimationState.GetTrack(9);if(structure!=null)state.SetAnimation(9,structure.Animation.Name,true);state.Apply(fresh);fresh.UpdateWorldTransform(Spine.Physics.Pose);var mask=new C.CharacterSlotVisibility(current,root.Definition.directionProfile.boundarySlot,Array.Empty<string>());mask.Apply(false,false);var differences=new List<object>();int count=0;for(int i=0;i<current.Bones.Count;i++){var actual=current.Bones.Items[i];var expected=fresh.Bones.Items[i];if(!mask.Contains(actual))continue;var a=actual.AppliedPose;var e=expected.AppliedPose;float error=Mathf.Abs(a.WorldX-e.WorldX)+Mathf.Abs(a.WorldY-e.WorldY)+Mathf.Abs(a.A-e.A)+Mathf.Abs(a.B-e.B)+Mathf.Abs(a.C-e.C)+Mathf.Abs(a.D-e.D);if(error>.02f){count++;if(differences.Count<14)differences.Add(new{bone=actual.Data.Name,error,actual=new[]{a.WorldX,a.WorldY,a.A,a.B,a.C,a.D},expected=new[]{e.WorldX,e.WorldY,e.A,e.B,e.C,e.D}});}}mask.Restore();ctx.ReturnValue=new{failed=count==0?0:1,changedVisibleBones=count,clip=track.Animation.Name,time=track.TrackTime,differences};}}
