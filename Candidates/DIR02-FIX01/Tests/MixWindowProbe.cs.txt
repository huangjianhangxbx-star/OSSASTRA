/* 只观察生成网格后的混合窗口；不驱动动画。列出混合时进入显示区、而当前目标完整姿态应在场外的slot。 */
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Spine;using Spine.Unity;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;

void Mesh(ISkeletonRenderer renderer){observed++;var track=body.AnimationState.GetTrack(0);if(track?.Animation==null)return;bool changed=priorFace!=direction.FaceId||priorMirror!=direction.Mirrored||priorClip!=track.Animation.Name;priorFace=direction.FaceId;priorMirror=direction.Mirrored;priorClip=track.Animation.Name;if(!changed&&track.MixingFrom==null||rows.Count>=180)return;var actual=body.Skeleton;var fresh=new Skeleton(actual.Data);fresh.ScaleX=actual.ScaleX;fresh.ScaleY=actual.ScaleY;var state=new Spine.AnimationState(new AnimationStateData(actual.Data));state.SetAnimation(0,track.Animation.Name,track.Loop).TrackTime=track.TrackTime;var structural=body.AnimationState.GetTrack(9);if(structural!=null)state.SetAnimation(9,structural.Animation.Name,true);state.Apply(fresh);fresh.UpdateWorldTransform(Spine.Physics.Pose);var expectedMask=new C.CharacterSlotVisibility(fresh,profile.boundarySlot,Array.Empty<string>());expectedMask.Apply(false,false);mask.Apply(false,false);var leaks=new List<object>();for(int i=0;i<actual.Slots.Count;i++){var slot=actual.Slots.Items[i];if(!(slot.AppliedPose.Attachment is RegionAttachment)&&!(slot.AppliedPose.Attachment is body.OnMeshAndMaterialsUpdated+=Mesh;EditorApplication.CallbackFunction finish=null;finish=()=>{if(EditorApplication.timeSinceStartup-start<55&&Application.isPlaying)return;EditorApplication.update-=finish;body.OnMeshAndMaterialsUpdated-=Mesh;mask.Restore();SessionState.SetString("FIX01_MIX_WINDOW",Newtonsoft.Json.JsonConvert.SerializeObject(new{state="completed",observed,leakFrames,rows}));};EditorApplication.update+=finish;ctx.ReturnValue="Finite 55s mesh-window probe attached";}}
