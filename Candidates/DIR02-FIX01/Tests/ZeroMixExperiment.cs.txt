using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using A=Ossastra.CharacterAnimation;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ZeroMixExperiment:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Play only");var root=UnityEngine.Object.FindObjectsOfType<C.CharacterRoot>().Single(r=>r.ActorId=="temporary-a");var config=root.GetComponentInChildren<A.CharacterAnimationConfigManager>();var oldMap=config.baseMap;var oldOverrides=config.modelOverrides;var map=UnityEngine.Object.Instantiate(oldMap);var overrides=oldOverrides.Select(o=>UnityEngine.Object.Instantiate(o)).ToList();foreach(var entry in map.plansOther)foreach(var clip in entry.plan.clips)if(clip.track==0)clip.mixDuration=0;foreach(var o in overrides)foreach(var rule in o.rulesOther)foreach(var clip in rule.replacementPlan.clips)if(clip.track==0)clip.mixDuration=0;config.baseMap=map;config.modelOverrides=overrides;double start=EditorApplication.timeSinceStartup;EditorApplication.CallbackFunction cleanup=null;cleanup=()=>{if(Application.isPlaying&&EditorApplication.timeSinceStartup-start<50)return;EditorApplication.update-=cleanup;if(config!=null){config.baseMap=oldMap;config.modelOverrides=oldOverrides;}UnityEngine.Object.DestroyImmediate(map);foreach(var o in overrides)UnityEngine.Object.DestroyImmediate(o);SessionState.SetString("FIX01_ZERO_MIX_RESTORED","true");};EditorApplication.update+=cleanup;ctx.ReturnValue=new{temporary=true,onlyBlue=true,bodyMix=0,restoreAfterSeconds=50};}}
