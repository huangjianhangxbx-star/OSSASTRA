/* 经真实mix窗口A/B后，仅小蓝两份生效SO的身体默认mix改0；保留逐clip作者配置及其他角色资产。 */
using System;using System.Linq;using UnityEngine;using UnityEditor;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class BlueDiscreteMix:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(Application.isPlaying)throw new Exception("Edit required");const string folder="Assets/OSSASTRA/Configs/Characters/BlueFiveFace/";var map=AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(folder+"BlueFiveFace_Animation_Mapping.asset");var model=AssetDatabase.LoadAssetAtPath<A.AnimationPlanOverrideRuleSetSO>(folder+"BlueFiveFace_Model_Override.asset");if(map==null||model==null)throw new Exception("Missing own Blue mapping/override");int changed=0;ctx.RegisterObjectModification(map);ctx.RegisterObjectModification(model);foreach(var p in map.plansOther.Select(e=>e.plan).Concat(model.rulesOther.Select(r=>r.replacementPlan)))foreach(var clip in p.clips)if(clip.track==0&&clip.mixDuration!=0){clip.mixDuration=0;changed++;}map.Revision++;model.Revision++;EditorUtility.SetDirty(map);EditorUtility.SetDirty(model);AssetDatabase.SaveAssetIfDirty(map);AssetDatabase.SaveAssetIfDirty(model);ctx.ReturnValue=new{changed,bodyZero=map.plansOther.SelectMany(e=>e.plan.clips).Where(c=>c.track==0).All(c=>c.mixDuration==0)&&model.rulesOther.SelectMany(r=>r.replacementPlan.clips).Where(c=>c.track==0).All(c=>c.mixDuration==0),structureZero=map.plansOther.SelectMany(e=>e.plan.clips).Where(c=>c.track>0).All(c=>c.mixDuration==0),saved=new[]{AssetDatabase.GetAssetPath(map),AssetDatabase.GetAssetPath(model)}};}}
