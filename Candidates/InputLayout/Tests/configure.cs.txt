using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using Ossastra.Characters;using Ossastra.Inventory;using Ossastra.UserInterface;using Funplay.Editor.Tools.Scripting;
public class ConfigureInputLayout:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var stage=PrefabStageUtility.GetCurrentPrefabStage();if(stage==null)throw new Exception("User prefab stage missing");var input=stage.prefabContentsRoot.GetComponent<Ossastra.Characters.CharacterInputHandler>();
string path="Assets/OSSASTRA/VC02B/PlayerInputLayout.asset";if(AssetDatabase.LoadMainAssetAtPath(path)!=null)throw new Exception("Already exists; inspect instead of replay");var p=ScriptableObject.CreateInstance<PlayerInputLayoutSO>();
InputAction Button(string name,string keyboard,string gamepad){var a=new InputAction(name,InputActionType.Button);if(keyboard!=null)a.AddBinding(keyboard);if(gamepad!=null)a.AddBinding(gamepad);return a;}
void Chord(InputAction a,string modifier,string button){a.AddCompositeBinding("ButtonWithOneModifier").With("Modifier","<Gamepad>/"+modifier).With("Button","<Gamepad>/"+button);}
p.move=input.movementAction.Clone();if(!p.move.bindings.Any(b=>b.path=="<Gamepad>/leftStick"))p.move.AddBinding("<Gamepad>/leftStick");p.pointer=input.mouseAction.Clone();p.aim=input.gamepadAimAction.Clone();if(!p.aim.bindings.Any(b=>b.path=="<Gamepad>/rightStick"))p.aim.AddBinding("<Gamepad>/rightStick");
p.rightHand=Button("RightHandAction","<Mouse>/leftButton","<Gamepad>/leftTrigger");p.leftHand=Button("LeftHandAction","<Mouse>/rightButton","<Gamepad>/rightTrigger");p.dodge=Button("Dodge","<Keyboard>/space","<Gamepad>/buttonEast");
p.skillOne=Button("SkillOne","<Keyboard>/q",null);Chord(p.skillOne,"leftShoulder","buttonWest");p.skillTwo=Button("SkillTwo","<Keyboard>/e",null);Chord(p.skillTwo,"leftShoulder","buttonNorth");p.skillThree=Button("SkillThree","<Keyboard>/r",null);Chord(p.skillThree,"leftShoulder","buttonEast");p.interact=Button("Interact","<Keyboard>/f","<Gamepad>/buttonSouth");p.heal=Button("Heal","<Keyboard>/y",null);Chord(p.heal,"leftShoulder","buttonSouth");p.switchCharacter=Button("SwitchCharacter","<Keyboard>/z",null);Chord(p.switchCharacter,"leftShoulder","rightShoulder");p.inventory=Button("Inventory","<Keyboard>/tab","<Gamepad>/start");p.menu=Button("Menu",null,"<Gamepad>/select");p.weaponMode=Button("WeaponMode",null,null);Chord(p.weaponMode,"leftTrigger","rightTrigger");
AssetDatabase.CreateAsset(p,path);Undo.RecordObject(input,"Shared player input layout");input.inputLayout=p;EditorUtility.SetDirty(input);
var scene=UnityEngine.SceneManagement.SceneManager.GetSceneByPath("Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity");var objects=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();T Find<T>()where T:Component=>objects.Select(t=>t.GetComponent<T>()).FirstOrDefault(c=>c!=null);
foreach(var i in objects.Select(t=>t.GetComponent<Ossastra.Characters.CharacterInputHandler>()).Where(i=>i!=null)){Undo.RecordObject(i,"Shared input layout");i.inputLayout=p;p.Apply(i);EditorUtility.SetDirty(i);}
var installer=Undo.AddComponent<PlayerInputLayoutInstaller>(Find<InventorySession>().gameObject);installer.layout=p;installer.control=Find<CharacterControlCoordinator>();installer.consumables=Find<PartyConsumableInput>();installer.interaction=Find<InventoryInteraction>();installer.inventory=Find<InventoryView>();EditorUtility.SetDirty(installer);AssetDatabase.SaveAssets();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=new{asset=path,scene=scene.path,installer=installer.GetInstanceID(),prefabStageNotSaved=true};}}
