using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.UserInterface;using Funplay.Editor.Tools.Scripting;
public class InputWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var installer=UnityEngine.Object.FindObjectOfType<PlayerInputLayoutInstaller>();var view=installer.inventory;var c=installer.control;if(!Application.isPlaying||c.Current==null)throw new Exception("Live B required");var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
foreach(var r in c.Characters)r.SendMessage("OnApplicationFocus",true);c.SendMessage("OnApplicationFocus",true);var pad=InputSystem.AddDevice<Gamepad>();var rows=new List<object>();int failed=0,index=0;double at=EditorApplication.timeSinceStartup;var first=c.Current;double clock=first.SimulationTime;string mode="";var steps=new List<Action>();void C(string name,bool pass){rows.Add(new{name,pass});if(!pass)failed++;}void Pad(GamepadState s)=>InputSystem.QueueStateEvent(pad,s);void Neutral()=>Pad(new GamepadState());
steps.Add(()=>{Neutral();C("both bodies share definition only",c.Characters.All(r=>r.GetComponent<Ossastra.Characters.CharacterInputHandler>().inputLayout==installer.layout)&&!ReferenceEquals(c.Characters[0].GetComponent<Ossastra.Characters.CharacterInputHandler>().dodgeAction,c.Characters[1].GetComponent<Ossastra.Characters.CharacterInputHandler>().dodgeAction));});
steps.Add(()=>Pad(new GamepadState().WithButton(GamepadButton.Start)));steps.Add(()=>{Neutral();C("Start opened actual inventory",view.IsOpen);});steps.Add(()=>Pad(new GamepadState().WithButton(GamepadButton.Start)));steps.Add(()=>{Neutral();C("Start closed inventory",!view.IsOpen);});steps.Add(()=>Pad(new GamepadState().WithButton(GamepadButton.Select)));steps.Add(()=>{Neutral();var menu=view.transform.Find("InventoryCanvas/Windows/SystemMenu");C("Select opened active actual menu",menu!=null&&menu.gameObject.activeInHierarchy);clock=first.SimulationTime;});steps.Add(()=>{C("menu does not pause world",first.SimulationTime>clock&&Time.timeScale==1);Pad(new GamepadState().WithButton(GamepadButton.Select));});steps.Add(()=>{Neutral();C("Select closed menu",!view.transform.Find("InventoryCanvas/Windows/SystemMenu").gameObject.activeInHierarchy);});steps.Add(()=>Pad(new GamepadState().WithButton(GamepadButton.LeftShoulder).WithButton(GamepadButton.RightShoulder)));steps.Add(()=>{Neutral();C("LB RB changed Current",c.Current!=first);});steps.Add(()=>{mode=c.Current.GetComponent<CharacterMainAction>().SelectedId;Pad(new GamepadState{leftTrigger=1,rightTrigger=1});});steps.Add(()=>{Neutral();C("LT RT changed actual mode",c.Current.GetComponent<CharacterMainAction>().SelectedId!=mode);});
EditorApplication.CallbackFunction tick=null;void Done(){EditorApplication.update-=tick;InputSystem.RemoveDevice(pad);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);SessionState.SetString("INPUT_LAYOUT_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state="completed",failed,rows}));}
tick=()=>{try{if(EditorApplication.timeSinceStartup-at<.3)return;at=EditorApplication.timeSinceStartup;steps[index++]();if(index==steps.Count)Done();}catch(Exception e){failed++;rows.Add(new{error=e.ToString()});Done();}};EditorApplication.update+=tick;ctx.ReturnValue="scheduled actual scene input story";}}
