using System;using System.Linq;using UnityEngine;using Ossastra.CharacterCore;using Ossastra.Characters;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class CoarseOwnerContract:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(!Application.isPlaying||UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>().Length!=1)throw new Exception("Single M01 Play required");var p=UnityEngine.Object.FindObjectOfType<Ossastra.Characters.CharacterRoot>();var targets=UnityEngine.Object.FindObjectsOfType<CombatTestTarget>();var left=targets.Single(t=>t.Health.Id=="target-b");var right=targets.Single(t=>t.Health.Id=="target-a");var before=p.transform.position;var lp=left.transform.position;var rp=right.transform.position;try{p.CancelAction("coarse-fixture");p.SetPaused(false);p.GetComponent<CharacterMovement>().Teleport(new Vector3(0,.03f,0));left.transform.position=new Vector3(-1.15f,0,0);right.transform.position=new Vector3(1.15f,0,0);left.RestoreHealth();right.RestoreHealth();Physics.SyncTransforms();var runtime=new BasicActionRuntime(p.ActorId);runtime.Request(new BasicAttackDefinition(.3,.5,.8,1.1,12),"coarse",new ControlAuthority());var releases=runtime.Advance(3,false).Where(f=>f.Release!=null).ToArray();var hit=p.GetComponent<CharacterMeleeHit>();hit.Resolve(releases[0].Release,Vector3.left);bool duplicate=false;try{hit.Resolve(releases[0].Release,Vector3.right);}catch(InvalidOperationException){duplicate=true;}ctx.Log(Newtonsoft.Json.JsonConvert.SerializeObject(new{pass=releases.Length==1&&left.Health.Current==88&&right.Health.Current==100&&duplicate,source=p.ActorId,leftHp=left.Health.Current,rightHp=right.Health.Current,duplicate}));}finally{p.GetComponent<CharacterMovement>().Teleport(before);left.transform.position=lp;right.transform.position=rp;left.RestoreHealth();right.RestoreHealth();}}}
