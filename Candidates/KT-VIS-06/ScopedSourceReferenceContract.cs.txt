using System;using System.Linq;using System.Collections.Generic;using System.IO;using UnityEngine;using UnityEngine.Rendering;using UnityEngine.SceneManagement;using UnityEditor;using Funplay.Editor.Tools.Scripting;using Ossastra.SceneVisibility;using Ossastra.SceneVisibility.Rooms.Rendering;using Newtonsoft.Json;
public class ScopedSourceReferenceContract:IFunplayCommand{
 public static object Result;static SceneVisibilityRoot root;static Renderer[] renderers;static Material[][] prior;static int frames;static double deadline;static bool running;static string error;
 static void Need(bool b,string m){if(!b)throw new Exception(m);}static bool Same(Material[] a,Material[] b){return a.Length==b.Length&&a.Select((m,i)=>ReferenceEquals(m,b[i])).All(v=>v);}
 public void Execute(ExecutionContext ctx){Need(!running,"Already running");var s=SceneManager.GetActiveScene();Need(Application.isPlaying&&s.path=="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity","Exact Lab Play");var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(v=>v!=null).ToArray();Need(all.OfType<Ossastra.Commerce.Runtime.CommerceService>().Single().profileId=="kt-vis06-natural-01","Wrong isolated profile");root=all.OfType<SceneVisibilityRoot>().Single();Need(root.roomModule.IsReady&&!root.roomModule.truthMode,"Final module required");renderers=root.coordinator.Characters.Select(a=>a.GetComponent<Ossastra.Characters.CharacterEquipmentView>()).Where(v=>v!=null&&v.PresentedRenderer!=null).Select(v=>(Renderer)v.PresentedRenderer).ToArray();Need(renderers.Length>0,"Need original equipment Sprite renderer");prior=renderers.Select(r=>r.sharedMaterials).ToArray();Need(prior.SelectMany(a=>a).All(m=>m!=null&&m.shader!=null),"Source was already broken before regression");for(int i=0;i<3;++i){root.roomModule.BeginCamera(root,root.gameplayCamera,true);try{Need(RoomMaskCameraRegistry.TryGet(root.gameplayCamera,out var f)&&f.PositiveSurfaces.Length>0,"Empty/rejected source frame");}finally{root.roomModule.EndCamera(root,root.gameplayCamera);}for(int j=0;j<renderers.Length;++j)Need(Same(renderers[j].sharedMaterials,prior[j]),"Manual End changed original Sprite source refs");}frames=0;error=null;running=true;deadline=EditorApplication.timeSinceStartup+6;RenderPipelineManager.endFrameRendering+=End;EditorApplication.update+=Tick;Result=new{status="running",manualPairs=3,sources=renderers.Length};ctx.ReturnValue=Result;}
 static void Tick(){if(running&&(!Application.isPlaying||EditorApplication.timeSinceStartup>deadline)){error="Context/deadline";Finish();}}
 static void End(ScriptableRenderContext c,Camera[] cams){if(!running||!cams.Contains(root.gameplayCamera))return;try{for(int i=0;i<renderers.Length;++i)Need(renderers[i]!=null&&Same(renderers[i].sharedMaterials,prior[i])&&renderers[i].sharedMaterials.All(m=>m!=null&&m.shader!=null),"Source refs/null changed across real Prepare");Need(RoomMaskCameraRegistry.TryGetStatistics(root.gameplayCamera,out var s)&&s.Active&&s.UnsupportedMaterialCount==0&&string.IsNullOrEmpty(s.Error),"Final drawing failed after manual scope");if(++frames>=2)Finish();}catch(Exception e){error=e.ToString();Finish();}}
 static void Finish(){running=false;RenderPipelineManager.endFrameRendering-=End;EditorApplication.update-=Tick;Result=new{passed=error==null,error,manualPairs=3,realFrames=frames,sources=renderers.Select((r,i)=>new{renderer=r==null?null:r.name,id=r==null?0:r.GetInstanceID(),allOriginalReferences= r!=null&&Same(r.sharedMaterials,prior[i]),allSourceMaterialsAlive=r!=null&&r.sharedMaterials.All(m=>m!=null&&m.shader!=null)}).ToArray(),scope="original equipment renderer refs, 3 manual pairs then 2 actual main frames; no fake replacement or save mutation"};File.WriteAllText("Candidates/KT-VIS-06/Evidence/scoped-source-reference.json",JsonConvert.SerializeObject(Result,Formatting.Indented));}
}
