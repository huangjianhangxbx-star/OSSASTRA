    static Vector3 Center(ViewPortal portal){var points=portal.GetWorldAperture();Need(points.Length>=3,"No actual aperture");Vector3 sum=Vector3.zero;foreach(var p in points)sum+=p;return sum/points.Length;}
    static Vector3 WindowGround(ViewPortal portal)
    {
        var center=Center(portal);var delta=center-camera.transform.position;
        Need(Mathf.Abs(delta.y)>.001f,"Camera/aperture ray does not reach ground");
        float amount=(.08f-camera.transform.position.y)/delta.y;
        Need(amount>1,"Ground sample is not behind actual window");
        return camera.transform.position+delta*amount;
    }
    static bool InBox(RoomMaskVolume box,Vector3 point)
    {
        var p=box.WorldToLocal.MultiplyPoint3x4(point);var e=box.HalfExtents;
        return Mathf.Abs(p.x)<=e.x+.0002f&&Mathf.Abs(p.y)<=e.y+.0002f&&Mathf.Abs(p.z)<=e.z+.0002f;
    }
    static bool Blocked(RoomMaskVolume[] boxes,Vector3 origin,Vector3 target)
    {
        foreach(var box in boxes)
        {
            Vector3 a=box.WorldToLocal.MultiplyPoint3x4(origin),delta=box.WorldToLocal.MultiplyPoint3x4(target)-a;
            float enter=0,leave=1-.02f/Mathf.Max(Vector3.Distance(origin,target),.02f);bool hit=true;
            for(int axis=0;axis<3;axis++)
            {
                float extent=box.HalfExtents[axis];
                if(Mathf.Abs(delta[axis])<.0000001f){if(Mathf.Abs(a[axis])>extent){hit=false;break;}}
                else{float first=(-extent-a[axis])/delta[axis],second=(extent-a[axis])/delta[axis];enter=Mathf.Max(enter,Mathf.Min(first,second));leave=Mathf.Min(leave,Mathf.Max(first,second));if(enter>leave){hit=false;break;}}
            }
            if(hit&&leave>=enter&&leave>=0)return true;
        }
        return false;
    }
    static bool CameraAllows(RoomMaskFrame frame,Vector3 point)
    {
        if(frame.BuildingVolumes.Any(v=>InBox(v,point)))return false;
        foreach(var domain in frame.WindowSightDomains)
        {
            if(!domain.TargetExterior||!domain.IsCameraWindow)continue;
            bool inside=true;foreach(var p in domain.Planes)if(Vector4.Dot(p,new Vector4(point.x,point.y,point.z,1))<0){inside=false;break;}
            if(inside&&!Blocked(frame.CameraOpaqueBlockers,domain.ObserverPosition,point))return true;
        }
        return false;
    }
    static Color Patch(Texture2D image,Vector3 world,out Vector3 projected)
    {
        projected=camera.WorldToScreenPoint(world);
        Need(projected.z>0&&projected.x>=3&&projected.y>=3&&projected.x<image.width-3&&projected.y<image.height-3,"Chosen actual world sample outside Game View: "+world+" => "+projected);
        int x=Mathf.RoundToInt(projected.x),y=Mathf.RoundToInt(projected.y);Color sum=Color.clear;
        for(int dy=-1;dy<=1;dy++)for(int dx=-1;dx<=1;dx++)sum+=image.GetPixel(x+dx,y+dy);
        return sum/9;
    }
    static object CaptureWindow(RoomMaskFrame frame,ViewPortal portal,string label,bool expectExterior)
    {
        Texture2D image=null;
        try
        {
            // 此调用仅发生于真实主相机 endFrame；不更换目标RT、不引入额外camera或手工重渲。
            image=ScreenCapture.CaptureScreenshotAsTexture();Need(image!=null&&image.width>0&&image.height>0,"No actual screen readback");
            Need(image.width==Screen.width&&image.height==Screen.height,"Unexpected screen capture dimensions");
            var point=WindowGround(portal);var color=Patch(image,point,out var projected);
            Need(CameraAllows(frame,point)==expectExterior,"Actual camera-to-hole exterior sample has wrong current geometric permission");
            Vector3 tangent=Vector3.Cross(Vector3.up,portal.OutwardNormal).normalized;Need(tangent.sqrMagnitude>.9f,"Expected a vertical Lab aperture");
            var outside=point+tangent*5;bool outsideCamera=CameraAllows(frame,outside),outsideEye=module.AllowsWorldPoint(outside);
            Need(!outsideCamera,"Camera cone leaked five metres to side of actual opening");
            Need(!outsideEye,"Side-of-opening exterior point unexpectedly gained avatar permission");
            Color? outsideColor=null;Vector3 outsideProjected=camera.WorldToScreenPoint(outside);
            if(outsideProjected.z>0&&outsideProjected.x>=3&&outsideProjected.y>=3&&outsideProjected.x<image.width-3&&outsideProjected.y<image.height-3)outsideColor=Patch(image,outside,out outsideProjected);
            var neighbour=new Vector3(-6,.12f,-12);Color? neighbourColor=null;Vector3 neighbourProjected=camera.WorldToScreenPoint(neighbour);
            if(neighbourProjected.z>0&&neighbourProjected.x>=3&&neighbourProjected.y>=3&&neighbourProjected.x<image.width-3&&neighbourProjected.y<image.height-3)
            {
                neighbourColor=Patch(image,neighbour,out neighbourProjected);
                if(label=="north-avatar-back")Need(Mathf.Max(neighbourColor.Value.r,Mathf.Max(neighbourColor.Value.g,neighbourColor.Value.b))<.06f,"Actual unknown-neighbour pixel is not black");
            }
            if(expectExterior)Need(Mathf.Max(color.r,Mathf.Max(color.g,color.b))>.025f,"Real window aperture remains black despite valid camera exterior ray");
            float? closedDelta=null;
            if(label=="west-open"){openWestColor=color;hasOpenWestColor=true;}
            else if(label=="west-solid-only"&&hasOpenWestColor){closedDelta=Mathf.Abs(color.r-openWestColor.r)+Mathf.Abs(color.g-openWestColor.g)+Mathf.Abs(color.b-openWestColor.b);Need(closedDelta>.02f,"Actual closed shutter pixel did not differ from open exterior pixel");}
            string path=Path.Combine(EvidenceDirectory,label+".png");File.WriteAllBytes(path,image.EncodeToPNG());
            return new{path,width=image.width,height=image.height,source="actual main endFrame ScreenCapture readback",world=point.ToString("R"),screen=projected.ToString("R"),rgb=new[]{color.r,color.g,color.b},expectExterior,cameraAllowed=CameraAllows(frame,point),eyeAllowed=module.AllowsWorldPoint(point),outsideWorld=outside.ToString("R"),outsideCamera,outsideEye,outsideScreen=outsideProjected.ToString("R"),outsideRgb=outsideColor.HasValue?new[]{outsideColor.Value.r,outsideColor.Value.g,outsideColor.Value.b}:null,neighbourScreen=neighbourProjected.ToString("R"),neighbourRgb=neighbourColor.HasValue?new[]{neighbourColor.Value.r,neighbourColor.Value.g,neighbourColor.Value.b}:null,closedDelta,scope="outside-opening colour may be solid wall; it is not required to be black"};
        }
        finally{if(image!=null)UnityEngine.Object.Destroy(image);}
    }
