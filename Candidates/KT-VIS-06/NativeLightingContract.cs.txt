    static void OnEnd(ScriptableRenderContext context,Camera current){if(Running&&current==camera)endSerial++;}
    static void OnFrameEnd(ScriptableRenderContext context,Camera[] rendered)
    {
        if(!Running||endSerial==processedSerial)return;processedSerial=endSerial;
        try
        {
            Need(Application.isPlaying&&root!=null&&module!=null&&camera!=null&&SceneManager.GetActiveScene().path==Lab,"Play context changed");
            if(EditorApplication.timeSinceStartup-stageAt<(stage==0?.8:.3)||endSerial-startSerial<2)return;
            Need(module.SuppressedRendererCount==0,"Must sample after original real camera scope cleanup");
            Need(RoomMaskCameraRegistry.TryGet(camera,out var frame)&&frame.Enabled&&frame.PositiveRenderingEnabled&&!frame.TruthMode,"No actual final frame");
            Need(RoomMaskCameraRegistry.TryGetStatistics(camera,out var stats)&&stats.Active&&string.IsNullOrEmpty(stats.Error),"Actual backend error: "+(stats==null?"missing":stats.Error));
            if(stage==0)
            {
                Need(module.ActiveRoom!=null&&module.ActiveRoom.roomId=="room-a"&&root.coordinator.Current==actor,"Original Current did not resolve actual A");
                SelectFloorPoints(frame);CheckCasterScope();onColors=Capture("original-interior-light-on");local.enabled=false;
                facts.Add(new{stage="light-on",frame=Time.frameCount,actualEnd=endSerial,samples=points.Count,pointLightIntensity=local.intensity,color=Rgba(local.color)});Advance(1);
            }
            else if(stage==1)
            {
                var off=Capture("original-interior-light-off");float max=0,sum=0;int detected=0;var rows=new List<object>();
                for(int i=0;i<off.Length;i++){float d=Luma(onColors[i])-Luma(off[i]);max=Mathf.Max(max,d);sum+=d;if(d>.01f)detected++;rows.Add(new{world=Vec(points[i]),on=Rgba(onColors[i]),off=Rgba(off[i]),onMinusOffLuma=d});}
                facts.Add(new{stage="light-difference",frame=Time.frameCount,actualEnd=endSerial,maxOnMinusOffLuma=max,meanOnMinusOffLuma=sum/off.Length,detectedAbovePointZeroOne=detected,rows});
                Need(max>.01f&&detected>=2,"Existing local light ON/OFF actual Lit-floor delta insufficient; max="+max+", detected="+detected);
                RestoreState();Need(restoreErrors.Count==0,"Initial restore failed");Advance(2);
            }
            else
            {
                var lightAfter=LightSnapshots();var rendererAfter=RendererSnapshots();var colliderAfter=ColliderSnapshots();
                bool lightsEqual=JsonConvert.SerializeObject(lightAfter)==lightsBeforeJson,renderersEqual=JsonConvert.SerializeObject(rendererAfter)==renderersBeforeJson,collidersEqual=JsonConvert.SerializeObject(colliderAfter)==collidersBeforeJson;
                facts.Add(new{stage="actual-restored-frame",frame=Time.frameCount,actualEnd=endSerial,lightsEqual,renderersEqual,collidersEqual,lightsBefore,lightAfter});
                Need(lightsEqual&&renderersEqual&&collidersEqual,"Original Light/roof/cabinet/collision primitive snapshot changed after restored real frame");
                Need(root.coordinator.Current==actor,"Original Current changed");Finish();
            }
        }
        catch(Exception e){error="stage "+stage+": "+e;Finish();}
    }
    static void Advance(int next){stage=next;stageAt=EditorApplication.timeSinceStartup;startSerial=endSerial;}
    static float Luma(Color c)=>c.r*.2126f+c.g*.7152f+c.b*.0722f;
    static void SelectFloorPoints(RoomMaskFrame frame)
    {
        var floorCollider=floor.GetComponent<Collider>();var bounds=floor.bounds;var physics=scene.GetPhysicsScene();var hits=new RaycastHit[64];var candidates=new List<Vector3>();
        for(int z=0;z<9;z++)for(int x=0;x<9;x++)
        {
            var p=new Vector3(Mathf.Lerp(bounds.min.x+.65f,bounds.max.x-.65f,x/8f),bounds.max.y+.002f,Mathf.Lerp(bounds.min.z+.65f,bounds.max.z-.65f,z/8f));
            if(actors.Any(a=>new Vector2(a.transform.position.x-p.x,a.transform.position.z-p.z).sqrMagnitude<1.3f*1.3f))continue;
            var view=camera.WorldToViewportPoint(p);if(view.z<=0||view.x<.04f||view.x>.96f||view.y<.04f||view.y>.96f)continue;
            var ray=camera.ViewportPointToRay(view);float distance=Vector3.Dot(p-ray.origin,ray.direction);int count=physics.Raycast(ray.origin,ray.direction,hits,distance+.08f,camera.cullingMask,QueryTriggerInteraction.Ignore);
            if(count==hits.Length)continue;Collider nearest=null;float nearestDistance=float.PositiveInfinity;
            for(int i=0;i<count;i++)if(hits[i].collider!=null&&Array.IndexOf(frame.LabelRaycastExclusions,hits[i].collider)<0&&hits[i].distance<nearestDistance){nearest=hits[i].collider;nearestDistance=hits[i].distance;}
            if(nearest==floorCollider)candidates.Add(p);
        }
        points.AddRange(candidates.OrderBy(p=>(p-local.transform.position).sqrMagnitude).Take(24));Need(points.Count>=4,"Need >=4 actual Lit Floor ROI, not wall/cabinet/actor pixels");
    }
    static void CheckCasterScope()
    {
        string before=JsonConvert.SerializeObject(RendererSnapshots()),physicsBefore=JsonConvert.SerializeObject(ColliderSnapshots());var cabinetMode=cabinet.shadowCastingMode;bool cabinetOff=cabinet.forceRenderingOff;bool entered=false;
        try
        {
            module.BeginCamera(root,camera,true);entered=true;
            Need(module.ActiveRoom!=null&&module.ActiveRoom.roomId=="room-a","Manual scope seed lost A");
            Need(cabinet.shadowCastingMode==cabinetMode&&cabinet.shadowCastingMode!=ShadowCastingMode.Off&&cabinet.forceRenderingOff==cabinetOff,"Known static cabinet lost original Caster");
            Need(roofRenderers.All(r=>r.shadowCastingMode==ShadowCastingMode.ShadowsOnly&&!r.forceRenderingOff&&r.enabled),"Entered building roof must keep ShadowsOnly, never Off");
            Need(JsonConvert.SerializeObject(ColliderSnapshots())==physicsBefore,"Collider changed inside manual display scope");
            facts.Add(new{stage="manual-caster-scope",cabinetMode=(int)cabinet.shadowCastingMode,roofs=roofRenderers.Select(r=>new{id=r.GetInstanceID(),mode=(int)r.shadowCastingMode,r.forceRenderingOff}).ToArray(),collidersUnchanged=true});
        }
        finally{if(entered)module.EndCamera(root,camera);}
        Need(JsonConvert.SerializeObject(RendererSnapshots())==before,"Manual End did not return original roof/cabinet source state/materials");
    }
    static Color[] Capture(string label)
    {
        Texture2D image=null;
        try
        {
            image=ScreenCapture.CaptureScreenshotAsTexture();Need(image!=null&&image.width==Screen.width&&image.height==Screen.height,"Actual main endFrame screen readback failed");
            var colors=new Color[points.Count];for(int i=0;i<points.Count;i++){var pixel=camera.WorldToScreenPoint(points[i]);Need(pixel.z>0&&pixel.x>=3&&pixel.y>=3&&pixel.x<image.width-3&&pixel.y<image.height-3,"Floor ROI left actual Game View");int x=Mathf.RoundToInt(pixel.x),y=Mathf.RoundToInt(pixel.y);Color sum=Color.clear;for(int dy=-1;dy<=1;dy++)for(int dx=-1;dx<=1;dx++)sum+=image.GetPixel(x+dx,y+dy);colors[i]=sum/9;}
            string path=Path.Combine(DirectoryPath,label+".png");File.WriteAllBytes(path,image.EncodeToPNG());facts.Add(new{stage="actual-color-capture",label,path,width=image.width,height=image.height,samples=colors.Length});return colors;
        }
        finally{if(image!=null)UnityEngine.Object.Destroy(image);}
    }
    static void Restore(Action action,string label){try{action();}catch(Exception e){restoreErrors.Add(label+": "+e.Message);}}
    static void RestoreState()
    {
        if(!acquired||restored)return;
        Restore(()=>{if(local!=null)local.enabled=oldLightEnabled;},"original local light");
        if(actors!=null)for(int i=0;i<actors.Length;i++){int index=i;Restore(()=>{var a=actors[index];if(a!=null){a.GetComponent<Ossastra.Characters.CharacterMovement>().Teleport(actorPositions[index]);var o=a.GetComponent<Ossastra.Characters.CharacterOrientation>();o.OrientTowards(actorPositions[index]+actorDirections[index]*5);o.FaceAttackDirection();o.OrientTowards(actorPositions[index]+actorAttackDirections[index]*5);}},"actor pose");Restore(()=>{if(actors[index]!=null)actors[index].SetViewInputCapture(Lease,false);},"actor input lease");}
        Restore(()=>{if(map!=null){map.followParticipant=oldFollow;map.followSmoothTime=oldSmooth;map.fixedTarget=oldFixedTarget;map.showOverlay=oldMapOverlay;for(int i=0;i<mapFields.Length;i++)mapFields[i].SetValue(map,mapValues[i]);}},"camera map/orbit");
        Restore(()=>{if(map!=null)map.SetUiCameraCapture(Lease,false);},"camera input lease");
        Restore(()=>{if(camera!=null)camera.transform.SetPositionAndRotation(oldCameraPosition,oldCameraRotation);},"camera pose");
        Restore(()=>{if(root!=null){root.showDiagnostics=oldRootDiagnostics;module.RequestImmediateMembershipRefresh();if(!oldFocus)focusCallback.Invoke(root.coordinator,new object[]{false});}if(panel!=null){panel.showPanel=oldPanel;panel.showPortalProjection=oldProjection;}},"UI/focus");
        restored=true;
    }
    static void Watchdog(){if(Running&&(!Application.isPlaying||EditorApplication.timeSinceStartup-began>9)){error=error??"Play interrupted / 9-second watchdog";Finish();}}
    static void OnPlayMode(PlayModeStateChange state){if(Running&&state==PlayModeStateChange.ExitingPlayMode){error=error??"Play exit requested";Finish();}}
    static void Finish()
    {
        RenderPipelineManager.endCameraRendering-=OnEnd;RenderPipelineManager.endFrameRendering-=OnFrameEnd;EditorApplication.update-=Watchdog;EditorApplication.playModeStateChanged-=OnPlayMode;Running=false;RestoreState();acquired=false;
        Result=new{status=error==null&&restoreErrors.Count==0&&stage==2?"passed":"failed",error,seconds=EditorApplication.timeSinceStartup-began,actualMainEnds=endSerial,stage,facts,restoreErrors,originalCurrentRestored=root!=null&&root.coordinator.Current==actor,originalOrbitReferenceRestored=map!=null&&ReferenceEquals(mapFields[0].GetValue(map),mapValues[0]),originalLightPrimitiveSnapshots=lightsBefore,scope="Existing realtime InteriorLocalLight_A ON/OFF actual Lit Floor pixels and roof/cabinet Caster state only; no full external daylight, GI, baking, reflection, SSAO, Player or hardware certification"};
        ResultJson=JsonConvert.SerializeObject(Result,Formatting.Indented);try{Directory.CreateDirectory(DirectoryPath);File.WriteAllText(Path.Combine(DirectoryPath,"native-lighting-contract.json"),ResultJson);}catch(Exception e){ResultJson+="\nEvidence write failure: "+e.Message;}
    }
