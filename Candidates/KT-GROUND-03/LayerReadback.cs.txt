using System.Linq;using UnityEngine;using Ossastra.GroundSurfaces.Editor;using Funplay.Editor.Tools.Scripting;
public sealed class KT03LayerReadback:IFunplayCommand{public void Execute(ExecutionContext ctx){ctx.ReturnValue=GroundSurfaceAuthoring.GetBindings().Select(b=>new{b.SurfaceId,layers=b.Surface.layers.Select((l,i)=>new{i,l.displayName,purpose=b.map.theme.layers[i].purpose.ToString(),locked=Ossastra.GroundResearch.Editor.GroundWorkshopAuthoring.IsLayerLocked(b.map,i),cells=l.tilemap.GetTilesBlock(l.tilemap.cellBounds).Count(t=>t!=null),recipe=b.map.layerBindings[i].recipe.displayName}).ToArray()}).ToArray();}}
