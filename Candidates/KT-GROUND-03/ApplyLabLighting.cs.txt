/* 新场自有岩壁可读性，不改原地表Shader/旧场灯光/全局管线。 */
using System;using UnityEditor;using UnityEngine;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;using Ossastra.GroundSurfaces.Editor;
public sealed class ApplyLabLighting:IFunplayCommand{public void Execute(ExecutionContext ctx){if(Application.isPlaying||SceneManager.GetActiveScene().path!=GroundSurfaceAuthoring.ScenePath)throw new InvalidOperationException("需新Lab Edit");GroundSurfaceLabBuilder.ConfigureWallFill();for(int i=0;i<2;i++){var m=AssetDatabase.LoadAssetAtPath<Material>(GroundSurfaceTechnicalModels.Root+"/Materials/Wall_"+i+".mat");if(m==null)throw new InvalidOperationException("缺自有岩壁材质");ctx.RegisterObjectModification(m);m.EnableKeyword("_EMISSION");m.SetColor("_EmissionColor",new Color(.045f,.039f,.03f));AssetDatabase.SaveAssetIfDirty(m);}GroundSurfaceAuthoring.SaveLab();ctx.ReturnValue=new{saved=true};}}
