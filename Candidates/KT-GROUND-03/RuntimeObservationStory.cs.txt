/* 真实Play中的观察生命周期；只调用已有两套观察入口，不写角色/碰撞/作者，也不暂停时间。 */
using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using UnityEditor;using Ossastra.GroundSurfaces;using Ossastra.GroundSurfaces.Editor;using M=Ossastra.MapResearch;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public sealed class KT03RuntimeObservationStory:IFunplayCommand
{
 public static object Report;static bool running;static M.MapResearchRoot map;static GroundSurfaceLabView view;static Transform[] roots;static Vector3[] positions;static Quaternion[] rotations;static int[] builds;static int phase,frame;static float clock;static double next;static readonly string folder="E:/WORLDCREATOR/XingHaiHuiLang/研究/开拓物语地表/KT-GROUND-03/02-visuals/RuntimeObservation";
 public void Execute(ExecutionContext ctx){if(running)throw new InvalidOperationException("正在观察");var s=SceneManager.GetActiveScene();if(!Application.isPlaying||s.path!=GroundSurfaceAuthoring.ScenePath)throw new InvalidOperationException("需新Lab真实Play");map=All<M.MapResearchRoot>().Single();view=All<GroundSurfaceLabView>().Single();if(map.IsStructureMode||view.IsObserving)throw new InvalidOperationException("已有观察现场，未接管");roots=All<C.CharacterRoot>().Select(c=>c.transform).Concat(All<GroundSurfaceBinding>().Select(b=>b.transform)).Concat(All<GroundSurfaceBinding>().Select(b=>b.capCollider.transform)).ToArray();positions=roots.Select(t=>t.position).ToArray();rotations=roots.Select(t=>t.rotation).ToArray();builds=All<GroundSurfaceMeshCache>().Select(c=>c.rebuildCount).ToArray();frame=Time.frameCount;clock=Time.time;phase=0;next=EditorApplication.timeSinceStartup;System.IO.Directory.CreateDirectory(folder);running=true;EditorApplication.update+=Tick;EditorApplication.playModeStateChanged+=Mode;Report=new{status="RUNNING"};ctx.ReturnValue=Report;}
 static T[] All<T>()where T:Component=>SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<T>(true)).ToArray();
 static void Tick(){if(!running||EditorApplication.timeSinceStartup<next)return;next=EditorApplication.timeSinceStartup+.5;try{switch(phase++){case 0:if(!view.SetMode(GroundSurfaceLabViewMode.BareSeam))throw new Exception(view.LastMessage);break;case 1:ScreenCapture.CaptureScreenshot(folder+"/01-bare35.png");break;case 2:if(!map.SetStructureMode(true))throw new Exception("原12度观察入口拒绝");break;case 3:if(Mathf.Abs(map.observerCamera.transform.eulerAngles.x-12)>.01)throw new Exception("原侧视角不是12度");ScreenCapture.CaptureScreenshot(folder+"/02-bare12.png");break;case 4:if(!view.SetMode(GroundSurfaceLabViewMode.Structure))throw new Exception(view.LastMessage);break;case 5:ScreenCapture.CaptureScreenshot(folder+"/03-structure12.png");break;case 6:if(!view.SetMode(GroundSurfaceLabViewMode.Exploded))throw new Exception(view.LastMessage);break;case 7:ScreenCapture.CaptureScreenshot(folder+"/04-exploded12.png");break;default:Finish(null);break;}}catch(Exception e){Finish(e.ToString());}}
 static void Mode(PlayModeStateChange p){if(p==PlayModeStateChange.ExitingPlayMode)Finish("Play提前退出");}
 static void Finish(string error){if(!running)return;running=false;EditorApplication.update-=Tick;EditorApplication.playModeStateChanged-=Mode;try{view.Restore();map.SetStructureMode(false);bool physicalSame=roots.Select((t,i)=>Vector3.Distance(t.position,positions[i])<.0001f&&Quaternion.Angle(t.rotation,rotations[i])<.01f).All(x=>x);bool cachesSame=builds.SequenceEqual(All<GroundSurfaceMeshCache>().Select(c=>c.rebuildCount));bool normal=view.Mode==GroundSurfaceLabViewMode.Final&&view.DiagnosticRendererCount==0&&!map.IsStructureMode&&map.gameplayCamera.enabled&&Mathf.Abs(map.gameplayCamera.transform.eulerAngles.x-35)<.01f;Report=new{status=error==null&&physicalSame&&cachesSame&&normal&&Time.time>clock?"PASS":"FAILED",error,physicalSame,cachesSame,normal,worldClockAdvanced=Time.time>clock,frames=Time.frameCount-frame,elapsedGameSeconds=Time.time-clock,images=folder,scope="Runtime observer APIs and actual GameView screenshots; no physical keyboard claim"};}catch(Exception e){Report=new{status="FAILED",error=error+e.ToString()};}}
}
