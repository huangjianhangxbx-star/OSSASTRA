using System.Linq;using UnityEngine;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Ossastra.GroundSurfaces.Editor;using Funplay.Editor.Tools.Scripting;
public sealed class KT03SaveLab:IFunplayCommand{public void Execute(ExecutionContext ctx){GroundSurfaceAuthoring.SaveLab();var s=SceneManager.GetActiveScene();ctx.ReturnValue=new{saved=!s.isDirty,path=s.path,bindings=GroundSurfaceAuthoring.GetBindings().Select(b=>new{b.SurfaceId,variant=b.currentModelVariant,caches=b.GetComponentsInChildren<Ossastra.GroundSurfaces.GroundSurfaceMeshCache>(true).Length,colliders=b.GetComponentsInChildren<MeshCollider>(true).Length}).ToArray()};}}
