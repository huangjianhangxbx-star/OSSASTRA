using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Ossastra.GroundSurfaces;using Ossastra.GroundSurfaces.Editor;using Funplay.Editor.Tools.Scripting;
public sealed class KT03CacheHealth:IFunplayCommand{public void Execute(ExecutionContext ctx){ctx.ReturnValue=Resources.FindObjectsOfTypeAll<GroundSurfaceBinding>().Where(b=>b.gameObject.scene==SceneManager.GetActiveScene()).Select(b=>{var cache=b.cacheRoot==null?null:b.cacheRoot.GetComponent<GroundSurfaceMeshCache>();string fp="";try{fp=GroundSurfaceModelReader.Fingerprint(b);}catch(Exception e){fp=e.ToString();}return new{b.SurfaceId,b.lastError,enabled=b.isActiveAndEnabled,cacheExists=cache!=null,fpEqual=fp==b.boundary.sourceFingerprint,fp,boundaryFp=b.boundary.sourceFingerprint,cap=b.capCollider!=null,capMeshExists=b.capCollider!=null&&b.capCollider.sharedMesh!=null,capReadable=b.capCollider!=null&&b.capCollider.sharedMesh!=null&&b.capCollider.sharedMesh.isReadable,cacheCapExists=cache!=null&&cache.colliderMesh!=null,chunks=cache?.chunks.Select(c=>new{mesh=c.filter.sharedMesh!=null,material=c.renderer.sharedMaterial!=null,persistent=c.filter.sharedMesh!=null&&EditorUtility.IsPersistent(c.filter.sharedMesh),c.layerIndex}).ToArray()};}).ToArray();}}
