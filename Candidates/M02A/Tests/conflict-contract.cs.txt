using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConflictRed:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();if(c==null||!Application.isPlaying)throw new Exception("Dual Play required");var a=c.Current;if(a==null)throw new Exception("Current missing");var input=a.GetComponent<CharacterInputHandler>();var keyboard=InputSystem.AddDevice<Keyboard>("M02ConflictKeyboard");var json=input.movementAction.SaveBindingOverridesAsJson();var go=new GameObject("ConflictProbe");go.SetActive(false);try{for(int i=0;i<input.movementAction.bindings.Count;i++){var bind=input.movementAction.bindings[i];if(bind.path!=null&&bind.path.Contains("<Keyboard>"))input.movementAction.ApplyBindingOverride(i,bind.path.Replace("<Keyboard>",keyboard.path));}InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.D));InputSystem.Update();var before=input.Read().move;var other=go.AddComponent<CharacterControlCoordinator>();var so=new SerializedObject(other);var arr=so.FindProperty("characters");arr.arraySize=2;arr.GetArrayElementAtIndex(0).objectReferenceValue=c.Characters[0];arr.GetArrayElementAtIndex(1).objectReferenceValue=c.Characters[1];so.ApplyModifiedPropertiesWithoutUndo();other.switchAction.AddBinding("<Keyboard>/tab");go.SetActive(true);ctx.Log(Newtonsoft.Json.JsonConvert.SerializeObject(new{before=before.ToString(),after=input.CurrentFrame.move.ToString(),rejected=!other.enabled,status=other.Status,pass=before.x>.9f&&input.CurrentFrame.move.x>.9f}));}finally{UnityEngine.Object.DestroyImmediate(go);input.movementAction.LoadBindingOverridesFromJson(json);InputSystem.RemoveDevice(keyboard);}}}
