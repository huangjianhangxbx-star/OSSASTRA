// 双控RED：真实共享Prefab与虚拟键盘，观察两个运动组件是否消费同一D输入。
using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class DualRed:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");var prefab=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/OSSASTRA/Prefabs/Characters/PlayerCharacter.prefab");if(prefab==null)throw new Exception("Prefab missing");var parent=new GameObject("M02RedProbe");parent.SetActive(false);var a=UnityEngine.Object.Instantiate(prefab,parent.transform);var b=UnityEngine.Object.Instantiate(prefab,parent.transform);a.transform.position=new Vector3(-3,.03f,-3);b.transform.position=new Vector3(2,.03f,-3);var keyboard=InputSystem.AddDevice<Keyboard>("M02RedKeyboard");int id=0;foreach(var g in new[]{a,b}){var so=new SerializedObject(g.GetComponent<Ossastra.Characters.CharacterRoot>());so.FindProperty("actorId").stringValue="red-"+(id++);so.ApplyModifiedPropertiesWithoutUndo();var input=g.GetComponent<CharacterInputHandler>();input.movementAction.Dispose();input.movementAction=new InputAction("Move",InputActionType.Value,keyboard.path+"/d",expectedControlType:"Vector2");input.movementAction.Dispose();input.movementAction=new InputAction("Move",InputActionType.Value);input.movementAction.AddCompositeBinding("2DVector").With("Right",keyboard.path+"/d");g.GetComponent<Ossastra.Characters.CharacterRoot>().SendMessage("OnApplicationFocus",true);}
parent.SetActive(true);var posA=a.transform.position;var posB=b.transform.position;var start=EditorApplication.timeSinceStartup;EditorApplication.CallbackFunction tick=null;bool queued=false;
tick=()=>{try{var elapsed=EditorApplication.timeSinceStartup-start;if(elapsed>.15&&!queued){InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.D));queued=true;}if(elapsed<.65)return;var da=a.transform.position.x-posA.x;var db=b.transform.position.x-posB.x;SessionState.SetString("OSSASTRA_M02_RED",Newtonsoft.Json.JsonConvert.SerializeObject(new{expected="only selected A moves",aDelta=da,bDelta=db,aInput=a.GetComponent<CharacterInputHandler>().CurrentFrame.move.ToString(),bInput=b.GetComponent<CharacterInputHandler>().CurrentFrame.move.ToString(),pass=da>.1f&&Mathf.Abs(db)<.01f}));}finally{if(EditorApplication.timeSinceStartup-start>=.65){EditorApplication.update-=tick;InputSystem.RemoveDevice(keyboard);UnityEngine.Object.DestroyImmediate(parent);}}};EditorApplication.update+=tick;ctx.Log("RED scheduled .65s");}}
