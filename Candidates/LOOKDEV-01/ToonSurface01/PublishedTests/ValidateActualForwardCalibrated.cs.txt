/* Root-only transient validation. Uses the installed production ForwardLit pass,
 * not a copy of its equations and not the MathProbe shader.
 * Known main-light ABI is a controlled fixture, not evidence for scene shadows,
 * camera culling, actual light selection, GI, Room or post-processing.
 */
using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEditor;
using Funplay.Editor.Tools.Scripting;

public sealed class ValidateActualForwardCalibrated : IFunplayCommand
{
    const int Width=192,Height=48;
    static readonly float[] Angles={.12f,.22f,.4f,.6f,.8f,.95f};
    Material material;
    Mesh mesh;
    Texture2D baseMap,white,read;
    RenderTexture target;
    CommandBuffer command;
    MaterialPropertyBlock perDraw;
    Matrix4x4 view,projection;
    static readonly Color ClearSentinel=new Color(.017f,.031f,.047f,1);
    readonly List<object> rasterizations=new List<object>();

    static float[] RGB(Color c)=>new[]{c.r,c.g,c.b};
    static float Difference(Color a,Color b)=>Mathf.Max(Mathf.Abs(a.r-b.r),Mathf.Abs(a.g-b.g),Mathf.Abs(a.b-b.b));
    static bool Finite(Color c)=>!float.IsNaN(c.r)&&!float.IsNaN(c.g)&&!float.IsNaN(c.b)&&!float.IsInfinity(c.r)&&!float.IsInfinity(c.g)&&!float.IsInfinity(c.b);

    public void Execute(ExecutionContext ctx)
    {
        if(!Application.dataPath.Replace('\\','/').Equals("E:/WORLDCREATOR/new/Project_ANKALUTE/Assets",StringComparison.OrdinalIgnoreCase))
            throw new InvalidOperationException("wrong Unity project");
        var shader=Shader.Find("Ossastra/WorldRendering/ToonSurface01");
        if(shader==null||!shader.isSupported)throw new InvalidOperationException("installed ToonSurface01 shader unavailable/unsupported");
        if(!SystemInfo.SupportsRenderTextureFormat(RenderTextureFormat.ARGBFloat))
            throw new InvalidOperationException("float RT unavailable; no faithful linear RGB readback");
        var priorKeywords=Shader.enabledGlobalKeywords.Select(k=>k.name).OrderBy(x=>x,StringComparer.Ordinal).ToArray();
        string[] vectors={"_MainLightPosition","_MainLightColor","unity_LightData","_WorldSpaceCameraPos","_EnvironmentSceneTint","_EnvironmentSceneChroma"};
        var priorVectors=vectors.ToDictionary(x=>x,Shader.GetGlobalVector,StringComparer.Ordinal);
        float priorEnvironment=Shader.GetGlobalFloat("_EnvironmentLightingEnabled");
        var priorEnvironmentFloats=new[]{"_EnvironmentGroundBaseStrength","_EnvironmentSurfaceRole","_EnvironmentWetness"}.ToDictionary(x=>x,Shader.GetGlobalFloat,StringComparer.Ordinal);
        var priorEnvironmentInts=new[]{"_EnvironmentIndoorCount","_EnvironmentShelterCount"}.ToDictionary(x=>x,Shader.GetGlobalInt,StringComparer.Ordinal);
        var priorView=Shader.GetGlobalMatrix("unity_MatrixV");
        var priorProjection=Shader.GetGlobalMatrix("unity_MatrixP");
        var priorViewProjection=Shader.GetGlobalMatrix("unity_MatrixVP");
        var priorGlProjection=Shader.GetGlobalMatrix("glstate_matrix_projection");
        var priorActive=RenderTexture.active;
        bool priorSrgb=GL.sRGBWrite,priorAsync=ShaderUtil.allowAsyncCompilation;
        Color[] albedo=null,linear=null,ramp=null;
        var inputCalibrations=new List<object>();
        Color gpuBase=Color.clear,gpuTexture=Color.clear,gpuVertex=Color.clear;
        string pipelineTag=Shader.globalRenderPipeline;
        string[] shaderErrors=Array.Empty<string>();
        var messages=new List<string>();
        var controlledNames=new HashSet<string>(new[]{"_MAIN_LIGHT_SHADOWS","_MAIN_LIGHT_SHADOWS_CASCADE","_MAIN_LIGHT_SHADOWS_SCREEN","_ADDITIONAL_LIGHTS","_ADDITIONAL_LIGHTS_VERTEX","_ADDITIONAL_LIGHT_SHADOWS","_LIGHT_LAYERS","_SCREEN_SPACE_OCCLUSION","_CLUSTERED_LIGHTING","_LIGHT_COOKIES","_WRITE_RENDERING_LAYERS","_DBUFFER_MRT1","_DBUFFER_MRT2","_DBUFFER_MRT3","LIGHTMAP_ON","DYNAMICLIGHTMAP_ON","LIGHTMAP_SHADOW_MIXING","SHADOWS_SHADOWMASK","USE_PROBE_SYSTEM","EVALUATE_SH_VERTEX","EVALUATE_SH_MIXED"},StringComparer.Ordinal);
        var disabledForFixture=priorKeywords.Where(x=>controlledNames.Contains(x)).ToArray();
        bool globalsRestored=false;
        var rows=new List<object>();int failures=0;
        try
        {
            ShaderUtil.allowAsyncCompilation=false;
            foreach(var keyword in disabledForFixture)Shader.DisableKeyword(keyword);
            // This is an explicitly controlled white directional light. No shadow
            // variants are enabled; actual shadow-map correctness is tested in Lab.
            Shader.SetGlobalVector("_MainLightPosition",new Vector4(0,0,-1,0));
            Shader.SetGlobalVector("_MainLightColor",Vector4.one);
            Shader.SetGlobalVector("unity_LightData",new Vector4(0,0,1,0));
            Shader.SetGlobalVector("_WorldSpaceCameraPos",new Vector4(0,0,-4,1));
            Shader.SetGlobalFloat("_EnvironmentLightingEnabled",0);
            Shader.SetGlobalFloat("_EnvironmentGroundBaseStrength",0);Shader.SetGlobalFloat("_EnvironmentSurfaceRole",-1);Shader.SetGlobalFloat("_EnvironmentWetness",0);
            Shader.SetGlobalInt("_EnvironmentIndoorCount",0);Shader.SetGlobalInt("_EnvironmentShelterCount",0);
            Shader.SetGlobalVector("_EnvironmentSceneTint",Vector4.zero);Shader.SetGlobalVector("_EnvironmentSceneChroma",Vector4.zero);
            GL.sRGBWrite=false;

            material=new Material(shader){name="Owned ToonSurface01 actual Forward fixture",hideFlags=HideFlags.HideAndDontSave};
            material.shaderKeywords=Array.Empty<string>();
            material.EnableKeyword("_SPECULARHIGHLIGHTS_OFF");
            material.EnableKeyword("_ENVIRONMENTREFLECTIONS_OFF");
            material.SetFloat("_Surface",0);material.SetFloat("_AlphaClip",0);
            material.SetFloat("_SrcBlend",1);material.SetFloat("_DstBlend",0);
            material.SetFloat("_SrcBlendAlpha",1);material.SetFloat("_DstBlendAlpha",0);
            material.SetFloat("_ZWrite",1);material.SetFloat("_Cull",0);material.SetFloat("_Metallic",0);
            material.SetFloat("_Smoothness",0);material.SetFloat("_VoxelVertexColors",1);
            material.SetFloat("_VoxelIndirectDiffuse",0);material.SetFloat("_VoxelEnvironmentSpecular",0);
            material.SetFloat("_VoxelDirectSpecular",0);material.SetFloat("_VoxelHighlightStrength",0);
            material.SetFloat("_VoxelBackFaceDarkening",0);material.SetFloat("_VoxelLitFaceBoost",0);
            material.SetFloat("_ToonSpecularEnabled",0);material.SetFloat("_ToonRimEnabled",0);
            material.SetVector("_BaseColor",new Vector4(.6f,.5f,.4f,1));
            material.SetVector("_ToonThresholds",new Vector4(.32f,.70f,.05f,0));
            material.SetVector("_ToonDarkColor",Vector4.one*.2f);
            material.SetVector("_ToonMidColor",Vector4.one*.6f);
            material.SetVector("_ToonLightColor",Vector4.one);
            material.SetVector("_ToonShadowColor",Vector4.one);
            baseMap=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true){name="Owned linear BaseMap",hideFlags=HideFlags.HideAndDontSave,wrapMode=TextureWrapMode.Clamp,filterMode=FilterMode.Point};
            baseMap.SetPixels(Enumerable.Repeat(new Color(.4f,.5f,.6f,1),4).ToArray());baseMap.Apply(false,false);material.SetTexture("_BaseMap",baseMap);
            perDraw=new MaterialPropertyBlock();
            // UnityPerDraw's explicit fixture value accompanies the global ABI.
            // If the actual Forward backend ignores this it will fail nonempty
            // main-light checks instead of reporting a synthetic Math result.
            perDraw.SetVector("unity_LightData",new Vector4(0,0,1,0));
            mesh=BuildMesh();
            target=new RenderTexture(Width,Height,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear){name="Owned actual Forward linear RT",hideFlags=HideFlags.HideAndDontSave};
            if(!target.Create())throw new InvalidOperationException("float target failed");
            read=new Texture2D(Width,Height,TextureFormat.RGBAFloat,false,true){hideFlags=HideFlags.HideAndDontSave};
            command=new CommandBuffer{name="ToonSurface01 production ForwardLit controlled fixture"};
            view=Matrix4x4.Scale(new Vector3(1,1,-1))*Matrix4x4.TRS(new Vector3(0,0,-4),Quaternion.identity,Vector3.one).inverse;
            projection=GL.GetGPUProjectionMatrix(Matrix4x4.Ortho(-3,3,-.75f,.75f,.1f,10),true);
            int pass=material.FindPass("ForwardLit");if(pass<0)throw new InvalidOperationException("actual ForwardLit pass missing");
            white=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true){name="Owned linear calibration white",hideFlags=HideFlags.HideAndDontSave,wrapMode=TextureWrapMode.Clamp,filterMode=FilterMode.Point};white.SetPixels(Enumerable.Repeat(Color.white,4).ToArray());white.Apply(false,false);
            // Three independent white-factor renders establish upload semantics.
            // No expected RGB is fitted to the final composite readback.
            material.SetTexture("_BaseMap",white);mesh.colors=Enumerable.Repeat(Color.white,mesh.vertexCount).ToArray();
            var baseOnly=Render(pass,1,0);gpuBase=baseOnly[0];
            material.SetVector("_BaseColor",Vector4.one);material.SetTexture("_BaseMap",baseMap);
            var textureOnly=Render(pass,1,0);gpuTexture=textureOnly[0];
            material.SetTexture("_BaseMap",white);mesh.colors=Enumerable.Repeat(new Color(.5f,.8f,.7f,1),mesh.vertexCount).ToArray();
            var vertexOnly=Render(pass,1,0);gpuVertex=vertexOnly[0];
            var authorBase=new Color(.6f,.5f,.4f,1);var baseExpected=QualitySettings.activeColorSpace==ColorSpace.Linear?authorBase.linear:authorBase;
            var mapExpected=new Color(.4f,.5f,.6f,1);var vertexExpected=new Color(.5f,.8f,.7f,1);
            bool baseInputPass=baseOnly.All(c=>Finite(c)&&Difference(c,baseExpected)<.003f);
            bool textureInputPass=textureOnly.All(c=>Finite(c)&&Difference(c,mapExpected)<.003f);
            bool vertexInputPass=vertexOnly.All(c=>Finite(c)&&Difference(c,vertexExpected)<.003f);
            if(!baseInputPass)failures++;if(!textureInputPass)failures++;if(!vertexInputPass)failures++;
            inputCalibrations.Add(new{input="BaseColor only; white image and vertex",pass=baseInputPass,author=RGB(authorBase),expected=RGB(baseExpected),actual=RGB(gpuBase)});
            inputCalibrations.Add(new{input="linear float BaseMap only; white color and vertex",pass=textureInputPass,expected=RGB(mapExpected),actual=RGB(gpuTexture)});
            inputCalibrations.Add(new{input="Mesh COLOR only; white image and base color",pass=vertexInputPass,expected=RGB(vertexExpected),actual=RGB(gpuVertex)});
            material.SetVector("_BaseColor",new Vector4(.6f,.5f,.4f,1));material.SetTexture("_BaseMap",baseMap);
            albedo=Render(pass,1,0);linear=Render(pass,3,0);ramp=Render(pass,3,1);
            // Product of separately measured factors proves the actual composition,
            // while the calibration rows independently check the conversion rule.
            var expectedAlbedo=gpuBase*gpuTexture*gpuVertex;
            var expectedMainFactor=new[]{.2f,.2f,.6f,.6f,1f,1f}.Select(x=>QualitySettings.activeColorSpace==ColorSpace.Linear?Mathf.GammaToLinearSpace(x):x).ToArray();
            for(int i=0;i<Angles.Length;i++)
            {
                bool rawPass=Finite(albedo[i])&&Difference(albedo[i],expectedAlbedo)<.003f;
                // Metallic=0 URP BRDF diffuse retains its dielectric 0.96 factor;
                // endpoints are hand-derived, not calls to the candidate equations.
                var expectedLinear=expectedAlbedo*(.96f*Angles[i]);
                var expectedRamp=expectedAlbedo*(.96f*expectedMainFactor[i]);
                bool finite=Finite(linear[i])&&Finite(ramp[i]);
                bool linearPass=finite&&Difference(linear[i],expectedLinear)<.004f;
                bool rampPass=finite&&Difference(ramp[i],expectedRamp)<.004f;
                if(!rawPass||!linearPass||!rampPass)failures++;
                rows.Add(new{panel=i,normalDotLight=Angles[i],rawPass,linearPass,rampPass,raw=RGB(albedo[i]),rawExpected=RGB(expectedAlbedo),linear=RGB(linear[i]),linearExpected=RGB(expectedLinear),ramp=RGB(ramp[i]),rampExpected=RGB(expectedRamp)});
            }
            for(int i=0;i<6;i+=2)
            {
                bool sameBand=Difference(ramp[i],ramp[i+1])<.003f;
                bool stillContinuous=Difference(linear[i],linear[i+1])>.005f;
                if(!sameBand||!stillContinuous)failures++;
                rows.Add(new{pair=i/2,sameBand,continuousOff=stillContinuous});
            }
            shaderErrors=ShaderUtil.GetShaderMessages(shader).Where(m=>m.severity.ToString()=="Error").Select(m=>m.message).ToArray();
            messages.AddRange(ShaderUtil.GetShaderMessages(shader).Select(m=>m.severity+": "+m.message));
            if(shaderErrors.Length>0)failures++;
            // Return only after finally has restored shared state.
        }
        finally
        {
            foreach(var keyword in Shader.enabledGlobalKeywords)Shader.DisableKeyword(keyword);
            foreach(var keyword in priorKeywords)Shader.EnableKeyword(keyword);
            foreach(var p in priorVectors)Shader.SetGlobalVector(p.Key,p.Value);
            Shader.SetGlobalFloat("_EnvironmentLightingEnabled",priorEnvironment);
            foreach(var p in priorEnvironmentFloats)Shader.SetGlobalFloat(p.Key,p.Value);foreach(var p in priorEnvironmentInts)Shader.SetGlobalInt(p.Key,p.Value);
            var restore=new CommandBuffer{name="Restore actual Forward fixture GPU view state"};
            try{restore.SetViewProjectionMatrices(priorView,priorProjection);restore.SetGlobalMatrix("unity_MatrixV",priorView);restore.SetGlobalMatrix("unity_MatrixP",priorProjection);restore.SetGlobalMatrix("unity_MatrixVP",priorViewProjection);restore.SetGlobalMatrix("glstate_matrix_projection",priorGlProjection);Graphics.ExecuteCommandBuffer(restore);}finally{restore.Release();}
            RenderTexture.active=priorActive;GL.sRGBWrite=priorSrgb;ShaderUtil.allowAsyncCompilation=priorAsync;
            if(command!=null)command.Release();
            if(target!=null){target.Release();UnityEngine.Object.DestroyImmediate(target);}
            if(read!=null)UnityEngine.Object.DestroyImmediate(read);
            if(mesh!=null)UnityEngine.Object.DestroyImmediate(mesh);
            if(material!=null)UnityEngine.Object.DestroyImmediate(material);
            if(baseMap!=null)UnityEngine.Object.DestroyImmediate(baseMap);if(white!=null)UnityEngine.Object.DestroyImmediate(white);
            globalsRestored=Shader.enabledGlobalKeywords.Select(k=>k.name).OrderBy(x=>x,StringComparer.Ordinal).SequenceEqual(priorKeywords)
                &&priorVectors.All(p=>Shader.GetGlobalVector(p.Key)==p.Value)
                &&Shader.GetGlobalFloat("_EnvironmentLightingEnabled")==priorEnvironment
                &&priorEnvironmentFloats.All(p=>Shader.GetGlobalFloat(p.Key)==p.Value)&&priorEnvironmentInts.All(p=>Shader.GetGlobalInt(p.Key)==p.Value)
                &&RenderTexture.active==priorActive&&GL.sRGBWrite==priorSrgb&&ShaderUtil.allowAsyncCompilation==priorAsync;
        }
        if(!globalsRestored)failures++;
        ctx.ReturnValue=new{pass=failures==0,failures,checks=rows.Count+inputCalibrations.Count,globalsRestored,actualShader="Ossastra/WorldRendering/ToonSurface01",passName="ForwardLit",path="CommandBuffer.DrawMesh actual production ForwardLit",linearTarget=true,ordinaryMeshColorEntry=failures==0,sceneCameraOrRoomCertified=false,fixtureCull="Off; source backface behavior unverified",pipelineTag,priorEnabledKeywords=priorKeywords,disabledForFixture,shaderErrors,shaderMessages=messages.ToArray(),rasterizations=rasterizations.ToArray(),projectColorSpace=QualitySettings.activeColorSpace.ToString(),previousAssumptionUndecodedRaw=new[]{.12f,.2f,.168f},inputCalibrations=inputCalibrations.ToArray(),fixtureEnvironmentDisabled=true,rows};
    }

    Mesh BuildMesh()
    {
        var vertices=new List<Vector3>();var normals=new List<Vector3>();var uv=new List<Vector2>();var colors=new List<Color>();var triangles=new List<int>();
        for(int i=0;i<Angles.Length;i++)
        {
            int k=vertices.Count;float x=-2.5f+i;
            vertices.Add(new Vector3(x-.4f,-.55f,0));vertices.Add(new Vector3(x+.4f,-.55f,0));vertices.Add(new Vector3(x+.4f,.55f,0));vertices.Add(new Vector3(x-.4f,.55f,0));
            for(int n=0;n<4;n++){normals.Add(new Vector3(Mathf.Sqrt(1-Angles[i]*Angles[i]),0,-Angles[i]));uv.Add(Vector2.one*.5f);colors.Add(new Color(.5f,.8f,.7f,1));}
            triangles.AddRange(new[]{k,k+2,k+1,k,k+3,k+2});
        }
        var result=new Mesh{name="Owned six-normal actual Forward mesh",hideFlags=HideFlags.HideAndDontSave};result.SetVertices(vertices);result.SetNormals(normals);result.SetUVs(0,uv);result.SetColors(colors);result.SetTriangles(triangles,0);result.RecalculateBounds();return result;
    }
    Color[] Render(int pass,int diagnostic,int enabled)
    {
        material.SetFloat("_VoxelDiagnostic",diagnostic);material.SetFloat("_ToonRampEnabled",enabled);
        command.Clear();command.SetRenderTarget(target);command.SetViewport(new Rect(0,0,Width,Height));command.ClearRenderTarget(false,true,ClearSentinel);command.SetViewProjectionMatrices(view,projection);command.DrawMesh(mesh,Matrix4x4.identity,material,0,pass,perDraw);Graphics.ExecuteCommandBuffer(command);
        RenderTexture.active=target;read.ReadPixels(new Rect(0,0,Width,Height),0,0,false);read.Apply(false,false);var pixels=read.GetPixels();rasterizations.Add(new{diagnostic,enabled,nonClearPixels=pixels.Count(p=>Difference(p,ClearSentinel)>.001f),magentaPixels=pixels.Count(p=>Difference(p,Color.magenta)<.001f),total=pixels.Length});var result=new Color[Angles.Length];for(int i=0;i<result.Length;i++)result[i]=read.GetPixel(i*32+16,Height/2);return result;
    }
}
