 sealed class AdapterRow{public Material Source,Adapter,Clone;public int Pass;public string Kind;}
 static List<AdapterRow> Adapters(object pass)
 {
  var rows=new List<AdapterRow>();var active=Member(Member(pass,"materialBuckets"),"Active") as IEnumerable;
  if(active!=null)foreach(var bucket in active){var source=Member(bucket,"Original") as Material;var adapter=Member(bucket,"Adapter") as Material;var clone=Member(bucket,"Clone") as Material;if(source!=null&&adapter!=null)rows.Add(new AdapterRow{Source=source,Adapter=adapter,Clone=clone,Pass=Convert.ToInt32(Member(bucket,"Pass")),Kind="color-bucket"});}
  var others=Member(pass,"materials") as IDictionary;if(others!=null)foreach(DictionaryEntry entry in others){var adapter=entry.Value as Material;var source=Member(entry.Key,"Material") as Material;if(source==null)source=Member(entry.Key,"Source") as Material;if(source!=null&&adapter!=null)rows.Add(new AdapterRow{Source=source,Adapter=adapter,Pass=adapter.FindPass("ForwardLit"),Kind=adapter.shader!=null&&adapter.shader.name==AdapterName?"color-independent":"auxiliary-truth-or-other"});}
  return rows;
 }
 static bool SameBaseInputs(Material source,Material actual)
 {
  if(source==null||actual==null)return false;
  if(source.HasProperty("_BaseColor")&&(!actual.HasProperty("_BaseColor")||source.GetColor("_BaseColor")!=actual.GetColor("_BaseColor")))return false;
  if(source.HasProperty("_BaseMap")&&(!actual.HasProperty("_BaseMap")||source.GetTexture("_BaseMap")!=actual.GetTexture("_BaseMap")||source.GetTextureScale("_BaseMap")!=actual.GetTextureScale("_BaseMap")||source.GetTextureOffset("_BaseMap")!=actual.GetTextureOffset("_BaseMap")))return false;
  float vertex=source.HasProperty("_VoxelVertexColors")?source.GetFloat("_VoxelVertexColors"):0;
  return actual.HasProperty("_VoxelVertexColors")&&Mathf.Abs(actual.GetFloat("_VoxelVertexColors")-vertex)<.000001f;
 }
 static List<string> ManifestDifferences(Material material,string category,JObject spec)
 {
  var differences=new List<string>();var fields=(JObject)spec["shared"];var categoryFields=(JObject)spec[category];var merged=(JObject)fields.DeepClone();merged.Merge(categoryFields,new Newtonsoft.Json.Linq.JsonMergeSettings{MergeArrayHandling=MergeArrayHandling.Replace});if(category=="stone")merged["_ToonRimEnabled"]=1;
  foreach(var p in merged.Properties())
  {
   if(!material.HasProperty(p.Name)){differences.Add(p.Name+": missing");continue;}
   if(p.Value is JArray a){var expected=new Vector4((float)a[0],(float)a[1],(float)a[2],(float)a[3]);Vector4 actual;if(p.Name.EndsWith("Color",StringComparison.Ordinal)){var c=material.GetColor(p.Name);actual=new Vector4(c.r,c.g,c.b,c.a);}else actual=material.GetVector(p.Name);if((actual-expected).sqrMagnitude>1e-10f)differences.Add(p.Name+": differs from setup initial coefficient");}
   else if(Mathf.Abs(material.GetFloat(p.Name)-(float)p.Value)>1e-6f)differences.Add(p.Name+": differs from setup initial coefficient");
  }
  return differences;
 }
