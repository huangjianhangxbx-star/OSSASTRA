using System;
using System.Collections.Generic;
using System.Linq;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using Funplay.Editor.Tools.Scripting;

// Real production shader passes, private finite MRT, no camera.Render or Main publish.
public sealed class StyleCoverageGpuContract : IFunplayCommand
{
 const int N=48;
 sealed class Pair { public Color32[] Item1,Item2; public Pair(Color32[] a,Color32[] b){Item1=a;Item2=b;} }
 static bool Eq(Color32 a,Color32 b)=>a.r==b.r&&a.g==b.g&&a.b==b.b&&a.a==b.a;
 static bool Near(Color32 p,int r,int g,int b,int a)=>Math.Abs(p.r-r)<=1&&Math.Abs(p.g-g)<=1&&Math.Abs(p.b-b)<=1&&Math.Abs(p.a-a)<=1;
 public void Execute(ExecutionContext ctx)
 {
  if(Application.isPlaying||EditorApplication.isCompiling||EditorApplication.isUpdating)throw new InvalidOperationException("Stable Edit Mode required");
  if(SystemInfo.supportedRenderTargetCount<2)throw new InvalidOperationException("MRT2 unavailable; unsupported must use legacy world, not fake zero GPU coverage");
  var oldActive=RenderTexture.active;var oldV=Shader.GetGlobalMatrix("unity_MatrixV");var oldP=Shader.GetGlobalMatrix("unity_MatrixP");var oldVP=Shader.GetGlobalMatrix("unity_MatrixVP");
  var rows=new List<object>();var failures=new List<string>();var owned=new List<UnityEngine.Object>();var rts=new List<RenderTexture>();
  Action<string,bool,object> check=(name,pass,detail)=>{rows.Add(new{name,pass,detail});if(!pass)failures.Add(name);};
  string exception=null;
  try
  {
   var mesh=new Mesh{name="StyleCoverage private quad",hideFlags=HideFlags.HideAndDontSave};owned.Add(mesh);
   mesh.vertices=new[]{new Vector3(-.7f,-.7f,0),new Vector3(.7f,-.7f,0),new Vector3(.7f,.7f,0),new Vector3(-.7f,.7f,0)};
   mesh.uv=new[]{Vector2.zero,Vector2.right,Vector2.one,Vector2.up};mesh.SetUVs(1,new List<Vector4>{Vector4.zero,Vector4.zero,Vector4.zero,Vector4.zero});
   mesh.colors=Enumerable.Repeat(Color.white,4).ToArray();mesh.normals=Enumerable.Repeat(Vector3.back,4).ToArray();mesh.tangents=Enumerable.Repeat(new Vector4(1,0,0,1),4).ToArray();mesh.triangles=new[]{0,2,1,0,3,2};mesh.RecalculateBounds();
   var position=new Vector3(0,0,-3);var view=Matrix4x4.Scale(new Vector3(1,1,-1))*Matrix4x4.TRS(position,Quaternion.identity,Vector3.one).inverse;
   var projection=GL.GetGPUProjectionMatrix(Matrix4x4.Ortho(-1,1,-1,1,.1f,10),true);var inverse=(projection*view).inverse;
   foreach(int requested in new[]{1,4})
   {
    var label=new RenderTexture(N,N,24,RenderTextureFormat.ARGB32,RenderTextureReadWrite.Linear){antiAliasing=requested,bindTextureMS=false,hideFlags=HideFlags.HideAndDontSave};label.Create();rts.Add(label);
    var style=new RenderTexture(N,N,0,RenderTextureFormat.ARGB32,RenderTextureReadWrite.Linear){antiAliasing=requested,bindTextureMS=false,hideFlags=HideFlags.HideAndDontSave};style.Create();rts.Add(style);
    int actual=label.antiAliasing;check("same actual MRT samples requested="+requested,actual==style.antiAliasing,new{requested,actual,other=style.antiAliasing});
    var read=new Texture2D(N,N,TextureFormat.RGBA32,false,true){hideFlags=HideFlags.HideAndDontSave};owned.Add(read);
    Func<RenderTexture,Color32[]> pixels=rt=>{RenderTexture.active=rt;read.ReadPixels(new Rect(0,0,N,N),0,0,false);read.Apply(false,false);return read.GetPixels32();};
    foreach(string family in new[]{"hardware","customDepth","sprite","spineSpriteLit","grass"})
    {
     string name=family=="spineSpriteLit"?"Hidden/Ossastra/Rooms/PositiveSpriteLitTruth":family=="grass"?"Ossastra/Vegetation/GrassLayer":"Hidden/Ossastra/Rooms/PositiveTruth";
     var shader=Shader.Find(name);if(shader==null)throw new InvalidOperationException("Missing production shader: "+name);
     var mat=new Material(shader){hideFlags=HideFlags.HideAndDontSave};owned.Add(mat);
     mat.SetTexture("_MainTex",Texture2D.whiteTexture);if(mat.HasProperty("_BaseMap"))mat.SetTexture("_BaseMap",Texture2D.whiteTexture);
     mat.SetColor("_Color",Color.white);if(mat.HasProperty("_BaseColor"))mat.SetColor("_BaseColor",Color.white);
     if(mat.HasProperty("_RendererColor"))mat.SetColor("_RendererColor",Color.white);if(mat.HasProperty("_Flip"))mat.SetVector("_Flip",Vector4.one);
     if(mat.HasProperty("_Cull"))mat.SetFloat("_Cull",0);if(mat.HasProperty("_Cutoff"))mat.SetFloat("_Cutoff",.05f);
     if(mat.HasProperty("_RoomSurfaceMode"))mat.SetFloat("_RoomSurfaceMode",family=="customDepth"?4:0);
     if(mat.HasProperty("_VisualDepthEnabled"))mat.SetFloat("_VisualDepthEnabled",0);
     mat.SetFloat("_RoomPositivePartial",0);mat.SetFloat("_RoomPositiveCutRoof",0);mat.SetFloat("_RoomPositiveDepthWrite",0);mat.SetColor("_RoomPositiveColor",Color.white);
     if(family=="grass"){mat.DisableKeyword("_GRASS_WIND");mat.SetVector("_GrassCameraLod",new Vector4(0,0,1000,1001));}
     int pass=family=="hardware"?3:family=="sprite"?1:family=="grass"?mat.FindPass("RoomTruth"):0;
     Func<bool,bool,bool,bool,Pair> render=(enabled,environment,actor,indoor)=>
     {
      if(enabled)mat.EnableKeyword("_ROOM_STYLE_COVERAGE");else mat.DisableKeyword("_ROOM_STYLE_COVERAGE");
      if(!Ossastra.Rendering.RenderPassReadiness.Ensure(mat,pass))throw new InvalidOperationException("Actual pass/keyword not ready: "+family+" "+enabled);
      var block=new MaterialPropertyBlock();block.SetInteger("_RoomSightSeedCount",0);block.SetInteger("_RoomSightBuildingCount",indoor?1:0);block.SetInteger("_RoomSightDomainCount",0);block.SetInteger("_RoomSightBlockerCount",0);
      block.SetInteger("_RoomSightOwnerCount",0);block.SetInteger("_RoomSightOwnerExterior",0);block.SetInteger("_RoomSightExteriorView",1);block.SetInteger("_RoomSightWindowCount",0);block.SetInteger("_RoomSightCameraBlockerCount",0);block.SetInteger("_RoomSightPassMode",0);block.SetInteger("_RoomFogHistoryCount",0);
      block.SetFloat("_RoomPositiveCoverageMode",0);block.SetFloat("_RoomPositivePermissionMode",1);block.SetFloat("_RoomWorldLookEnvironment",environment?1:0);block.SetFloat("_RoomStyleEnvironment",environment?1:0);block.SetFloat("_RoomStyleCharacter",actor?1:0);
      block.SetVector("_RoomSightTargetSize",new Vector4(N,N,1f/N,1f/N));block.SetMatrix("_RoomSightInverseViewProjection",inverse);block.SetMatrixArray("_RoomSightBuildingMatrices",new[]{Matrix4x4.identity});block.SetVectorArray("_RoomSightBuildingExtents",new[]{new Vector4(10,10,10,0)});
      var cmd=new CommandBuffer{name="StyleCoverage private production MRT"};
      try
      {
       cmd.SetRenderTarget(style);cmd.ClearRenderTarget(false,true,Color.clear);
       cmd.SetRenderTarget(label);cmd.ClearRenderTarget(true,true,Color.black);
       if(enabled)cmd.SetRenderTarget(new RenderTargetIdentifier[]{label,style},label);else cmd.SetRenderTarget(label);
       cmd.SetViewport(new Rect(0,0,N,N));cmd.SetViewProjectionMatrices(view,projection);cmd.DrawMesh(mesh,Matrix4x4.identity,mat,0,pass,block);
       if(actual>1){cmd.ResolveAntiAliasedSurface(label);cmd.ResolveAntiAliasedSurface(style);}Graphics.ExecuteCommandBuffer(cmd);
      }
      finally{cmd.Release();}
      return new Pair(pixels(label),pixels(style));
     };
     var legacy=render(false,true,false,false);var env=render(true,true,false,false);var actorResult=render(true,false,true,false);var info=render(true,false,false,false);var indoorResult=render(true,true,false,true);
     int center=(N/2)*N+N/2;string key=family+" samples="+actual;
     check(key+" target0 byte-identical to legacy",legacy.Item1.Zip(env.Item1,(a,b)=>Eq(a,b)).All(v=>v),null);
     check(key+" environment=(1,0,1,1)",Near(env.Item2[center],255,0,255,255),env.Item2[center]);
     check(key+" actor=(1,1,1,1)",Near(actorResult.Item2[center],255,255,255,255),actorResult.Item2[center]);
     check(key+" information clears all coverage",Near(info.Item2[center],0,0,0,0),info.Item2[center]);
     check(key+" indoor has no outdoor affinity",Near(indoorResult.Item2[center],255,0,0,255),indoorResult.Item2[center]);
     check(key+" untouched background remains invalid",Near(env.Item2[0],0,0,0,0),env.Item2[0]);
    }
   }
  }
  catch(Exception e){exception=e.ToString();failures.Add(exception);}
  finally
  {
   var restore=new CommandBuffer{name="Restore StyleCoverage private view state"};try{restore.SetViewProjectionMatrices(oldV,oldP);restore.SetGlobalMatrix("unity_MatrixV",oldV);restore.SetGlobalMatrix("unity_MatrixP",oldP);restore.SetGlobalMatrix("unity_MatrixVP",oldVP);Graphics.ExecuteCommandBuffer(restore);}finally{restore.Release();RenderTexture.active=oldActive;}
   foreach(var rt in rts)if(rt!=null){rt.Release();UnityEngine.Object.DestroyImmediate(rt);}foreach(var item in owned)if(item!=null)UnityEngine.Object.DestroyImmediate(item);
  }
  ctx.ReturnValue=new{success=failures.Count==0,checks=rows,failures,exception,scope="Private 48x48 MRT, real hardware/custom-depth/Sprite/Spine SpriteLit/grass truth passes, MSAA1 and requested4 explicit resolve. Tests output/channel/Target0 invariance; actual Main classification, occlusion and final composition require copied-scene checks. No Main frame publish, source globals, camera render or scene save."};
 }
}
