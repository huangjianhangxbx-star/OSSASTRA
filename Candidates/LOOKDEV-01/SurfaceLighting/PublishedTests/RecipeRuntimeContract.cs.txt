using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Stylization;using Funplay.Editor.Tools.Scripting;
public sealed class SurfaceRecipeRuntimeContract:IFunplayCommand
{
 public void Execute(ExecutionContext ctx)
 {
  if(!Application.isPlaying||SceneManager.GetActiveScene().name!="SC_SurfaceLightingLab")throw new Exception("SurfaceLighting candidate Play required");var candidates=SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<SurfaceLightingRecipe>(true)).ToArray();if(candidates.Length!=1)throw new Exception("Expected one explicit recipe");var r=candidates[0];if(r.State!=SurfaceLightingState.Previous)throw new Exception("Begin this finite contract in complete A to preserve exact baseline");var author=r.recipe;var styleBefore=new[]{author.previous,author.neutral,author.blue}.Select(JsonUtility.ToJson).ToArray();var dayBefore=new[]{author.neutralDay,author.blueDay}.Select(JsonUtility.ToJson).ToArray();var aDay=r.environment.dayCycle;float aSun=r.environment.sunIntensityMultiplier;var results=new System.Collections.Generic.List<object>();int failed=0;Action<string,bool> check=(n,v)=>{results.Add(new{name=n,passed=v});if(!v)failed++;};
  try{
   check("A retains actual full v1.1 profile",r.RuntimeProfile!=null&&r.RuntimeProfile.pixelSize==author.previous.pixelSize&&r.RuntimeProfile.bandsEnabled==author.previous.bandsEnabled&&r.RuntimeProfile.environmentStrength==author.previous.environmentStrength);
   bool b=r.TrySetState(SurfaceLightingState.MaterialLighting,out var error);check("B commits authored day and uncolored single-ToneMap profile",b&&r.State==SurfaceLightingState.MaterialLighting&&r.environment.dayCycle==author.neutralDay&&r.RuntimeProfile!=null&&r.RuntimeProfile.themeMode==StylizedThemeMode.LightingLed&&r.RuntimeProfile.ledEnvironmentStrength==0&&!r.RuntimeProfile.airEnabled&&!r.RuntimeProfile.bloomEnabled&&r.RuntimeProfile.pixelSize==1);
   bool c=r.TrySetState(SurfaceLightingState.BlueComplete,out error);check("C commits blue environment and safe Bloom without pixels",c&&r.State==SurfaceLightingState.BlueComplete&&r.environment.dayCycle==author.blueDay&&r.RuntimeProfile!=null&&r.RuntimeProfile.visibleSourceBloom&&r.RuntimeProfile.airEnabled&&r.RuntimeProfile.bloomEnabled&&!r.RuntimeProfile.bandsEnabled&&r.RuntimeProfile.pixelSize==1);
   var profileRef=r.look.stylizedLook;var envRef=r.environment.dayCycle;var invalid=UnityEngine.Object.Instantiate(author);invalid.hideFlags=HideFlags.HideAndDontSave;invalid.neutral=null;r.recipe=invalid;try{bool rejected=!r.TrySetState(SurfaceLightingState.MaterialLighting,out error);check("missing candidate rejects without partial refs or state",rejected&&r.look.stylizedLook==profileRef&&r.environment.dayCycle==envRef&&r.State==SurfaceLightingState.BlueComplete);}finally{r.recipe=author;UnityEngine.Object.DestroyImmediate(invalid);}
   check("main off diagnostic is owned by original root",r.SetLightContributions(false,true)&&r.environment.sunIntensityMultiplier==0&&r.environment.Current!=null&&r.environment.Current.SunIntensity==0);
   check("combined reset uses authored recipe values",r.SetLightContributions(true,true)&&r.environment.sunIntensityMultiplier==author.blueLighting.sunMultiplier&&r.environment.ambientMultiplier==author.blueLighting.ambientMultiplier);
   float external=r.environment.sunIntensityMultiplier+.137f;r.environment.sunIntensityMultiplier=external;bool conflict=!r.TrySetState(SurfaceLightingState.MaterialLighting,out error);check("external root change rejects stale full switch",conflict&&r.environment.sunIntensityMultiplier==external&&r.State==SurfaceLightingState.BlueComplete);r.Release();check("release preserves changed scalar and restores owned author reference",!r.HasLease&&r.environment.sunIntensityMultiplier==external&&r.environment.dayCycle==aDay);
   r.environment.sunIntensityMultiplier=aSun;r.environment.RefreshNow();check("A can reacquire after release",r.TrySetState(SurfaceLightingState.Previous,out error)&&r.State==SurfaceLightingState.Previous&&r.environment.dayCycle==aDay);
   check("style and lighting authors unchanged",new[]{author.previous,author.neutral,author.blue}.Select(JsonUtility.ToJson).SequenceEqual(styleBefore)&&new[]{author.neutralDay,author.blueDay}.Select(JsonUtility.ToJson).SequenceEqual(dayBefore));
  }finally{r.recipe=author;if(r.HasLease)r.Release();r.environment.sunIntensityMultiplier=aSun;r.environment.RefreshNow();r.TrySetState(SurfaceLightingState.Previous,out _);}
  ctx.ReturnValue=new{passed=failed==0,checks=results.Count,failed,results};
 }
}
