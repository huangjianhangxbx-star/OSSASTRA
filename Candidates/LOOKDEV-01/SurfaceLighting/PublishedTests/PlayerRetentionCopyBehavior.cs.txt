// Native behavioral contract, stable copied-scene Edit only. Own hidden objects/materials; no target material-slot writes.
using System;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Stylization;using Funplay.Editor.Tools.Scripting;
public sealed class SurfacePlayerRetentionBehaviorContract:IFunplayCommand
{
 static Material Copy(Material source,Shader wrapper,bool outline){var m=new Material(wrapper){hideFlags=HideFlags.HideAndDontSave};m.CopyPropertiesFromMaterial(source);m.shaderKeywords=source.shaderKeywords.Where(k=>wrapper.keywordSpace.FindKeyword(k).isValid&&k!="N_F_O_ON").ToArray();m.SetFloat("_N_F_O",outline?1:0);if(outline)m.EnableKeyword("N_F_O_ON");m.SetShaderPassEnabled("SRPDefaultUnlit",outline);return m;}
 public void Execute(ExecutionContext ctx){var scene=SceneManager.GetActiveScene();if(Application.isPlaying||scene.name!="SC_SurfaceLightingLab")throw new Exception("Exact copied SurfaceLighting Edit required");var owner=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<LookDevFiniteToon>(true)).Single();var binding=owner.Bindings.FirstOrDefault(b=>b!=null&&b.renderer!=null&&b.realToon!=null);if(binding==null)throw new Exception("Actual finite binding needed");var wrapper=Shader.Find("Hidden/Ossastra/LookDev/RealToonRoomLocal");if(wrapper==null)throw new Exception("Actual local wrapper required");var before=binding.renderer.sharedMaterials;var go=new GameObject("Owned retention contract"){hideFlags=HideFlags.HideAndDontSave};var materials=new List<Material>();LookDevFiniteToon fixture=null;var field=typeof(LookDevFiniteToon).GetField("bindings",BindingFlags.Instance|BindingFlags.NonPublic);var rows=new List<object>();int failed=0;Action<string,bool> check=(name,pass)=>{rows.Add(new{name,pass});if(!pass)failed++;};try{fixture=go.AddComponent<LookDevFiniteToon>();var source=new Material(binding.realToon){hideFlags=HideFlags.HideAndDontSave};source.SetOverrideTag("LookDevFiniteToon","LOOKDEV-01");materials.Add(source);field.SetValue(fixture,new[]{new LookDevToonBinding{renderer=binding.renderer,materialSlot=binding.materialSlot,original=binding.original,threeBand=binding.threeBand,realToon=source,outline=true}});check("default does not prepare Player",!fixture.PlayerRetentionReady);check("null retention rejected",!fixture.ConfigurePlayerRetention(null,out _));check("empty retention rejected",!fixture.ConfigurePlayerRetention(Array.Empty<Material>(),out _));check("vendor source alone cannot retain wrapper",!fixture.ConfigurePlayerRetention(new[]{source},out _));var forward=Copy(source,wrapper,false);var outline=Copy(source,wrapper,true);materials.Add(forward);materials.Add(outline);check("forward-only misses actual Outline signature",!fixture.ConfigurePlayerRetention(new[]{forward},out _));check("duplicate signature rejected",!fixture.ConfigurePlayerRetention(new[]{forward,outline,outline},out _));check("oversized list rejected",!fixture.ConfigurePlayerRetention(Enumerable.Repeat(forward,49).ToArray(),out _));var retained=new[]{forward,outline};bool configured=fixture.ConfigurePlayerRetention(retained,out var configuredError);check("exact actual forward and outline signatures prepare",configured&&fixture.PlayerRetentionReady);retained[0]=source;check("configuration defensively copies array",fixture.RefreshPlayerRetention(out _));check("identity identifies expected wrapper and actual passes",fixture.GetRealToonRuntimeIdentity().Contains("Hidden/Ossastra/LookDev/RealToonRoomLocal")&&fixture.GetRealToonRuntimeIdentity().Contains("Forward=0")&&fixture.GetRealToonRuntimeIdentity().Contains("Outline=6"));source.EnableKeyword("N_F_RL_ON");check("changed source keywords invalidate readiness",!fixture.RefreshPlayerRetention(out _)&&!fixture.PlayerRetentionReady);check("no actual renderer slot references changed",binding.renderer.sharedMaterials.SequenceEqual(before));ctx.ReturnValue=new{passed=failed==0,failed,checks=rows,configuredError,sourceShader=source.shader.name,sourceKeywords=source.shaderKeywords,forwardKeywords=forward.shaderKeywords,outlineKeywords=outline.shaderKeywords,retentionError=fixture.PlayerRetentionError,retainedFacts=new[]{forward,outline}.Select(m=>new{name=m.name,shader=m.shader.name,shaderSame=m.shader==wrapper,forward=m.FindPass("ForwardLit"),outline=m.FindPass("Outline"),roof=m.HasProperty("_RoomSightRoofMode"),full=m.HasProperty("_EnvironmentFullSurface"),tag=m.GetTag("LookDevFiniteToon",false,""),hide=m.hideFlags.ToString()}).ToArray(),playerVisualCertified=false};}finally{if(fixture!=null)field.SetValue(fixture,Array.Empty<LookDevToonBinding>());UnityEngine.Object.DestroyImmediate(go);foreach(var m in materials)if(m!=null)UnityEngine.Object.DestroyImmediate(m);}}
}
