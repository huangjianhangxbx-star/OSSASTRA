// Real production API + GPU, private targets only. Baseline must fail NEW opt-in
// behavior while retaining legacy checks. No scene/author/asset writes.
using System;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEngine.Rendering;using Ossastra.WorldRendering.Stylization;using Funplay.Editor.Tools.Scripting;
public sealed class SurfaceStageContract:IFunplayCommand
{
 readonly List<UnityEngine.Object> objects=new List<UnityEngine.Object>();readonly List<RTHandle> handles=new List<RTHandle>();
 static bool Near(float a,float b)=>Mathf.Abs(a-b)<.015f;
 static float Y(Color c)=>c.r*.2126f+c.g*.7152f+c.b*.0722f;
 public void Execute(ExecutionContext context)
 {
  var rows=new List<object>();int failed=0;Action<string,bool,string> ck=(n,p,d)=>{rows.Add(new{name=n,pass=p,detail=d});if(!p)failed++;};var prev=RenderTexture.active;
  try{
   var p=ScriptableObject.CreateInstance<StylizedLookProfileSO>();objects.Add(p);var pt=p.GetType();var mt=pt.GetField("themeMode");var vb=pt.GetField("visibleSourceBloom");var ls=pt.GetField("ledEnvironmentStrength");
   ck("new opt-in author fields exist",mt!=null&&vb!=null&&ls!=null,"themeMode/visibleSourceBloom/ledEnvironmentStrength");
   if(mt!=null&&vb!=null&&ls!=null){ck("legacy author defaults retained",Convert.ToInt32(mt.GetValue(p))==0&&!(bool)vb.GetValue(p),"new instance defaults");mt.SetValue(p,Enum.ToObject(mt.FieldType,1));vb.SetValue(p,true);ls.SetValue(p,.35f);var snap=StylizedLookSettings.Capture(p);var st=snap.GetType();var sm=st.GetProperty("ThemeMode");var sb=st.GetProperty("VisibleSourceBloom");var sl=st.GetProperty("LedEnvironmentStrength");ck("immutable snapshot exposes opted values",sm!=null&&sb!=null&&sl!=null&&Convert.ToInt32(sm.GetValue(snap))==1&&(bool)sb.GetValue(snap),"snapshot properties");ls.SetValue(p,.9f);ck("snapshot not live author reference",sl!=null&&Near((float)sl.GetValue(snap),.35f),"author strength mutated after Capture");}
   var shader=Shader.Find("Hidden/Ossastra/WorldRendering/StylizedLook");var bloom=Shader.Find("Hidden/Ossastra/WorldRendering/StylizedBloom");if(shader==null||bloom==null)throw new Exception("Missing production style Shader");var m=Mat(shader);var b=Mat(bloom);var valid=Fill(4,new Color(1,0,1,1));var dst=Target(2,2);var one=Target(1,1);
   m.SetVector("_LD_Size",new Vector4(2,2,.5f,.5f));m.SetVector("_LD_Features",new Vector4(1,0,0,0));m.SetVector("_LD_PixelDebug",new Vector4(1,0,0,0));m.SetVector("_LD_Mode",new Vector4(1,1,0,0));m.SetVector("_LD_Grade",new Vector4(.35f,.15f,0,0));m.SetColor("_LD_GradeShadow",new Color(.72f,.88f,1.22f,1));m.SetColor("_LD_GradeMid",new Color(.85f,1,1.12f,1));m.SetColor("_LD_GradeLight",new Color(1.05f,1.04f,.98f,1));m.SetVector("_LD_Theme",new Vector4(1,1,.18f,1));m.SetVector("_LD_Warm",new Vector4(0,1.5f,3,0));m.SetColor("_LD_Shadow",Color.blue);m.SetColor("_LD_Mid",Color.blue);m.SetColor("_LD_Light",Color.blue);m.SetTexture("_LD_Coverage",Tex(2,2,valid));
   var colors=new[]{new Color(.5f,.1f,.1f,1),new Color(.1f,.2f,.28864265f,1),Color.black,new Color(8,4,.2f,1)};Draw(Input(2,2,colors),dst,m,0);var grade=Read(dst);ck("LightingLed retains different source channel relationships",Mathf.Abs(grade[0].r-grade[1].r)>.2f&&grade[0].r>grade[0].b&&grade[1].b>grade[1].r,grade[0]+" / "+grade[1]);ck("LightingLed retains linear HDR luminance",Near(Y(grade[0]),Y(colors[0]))&&Near(Y(grade[1]),Y(colors[1]))&&Near(Y(grade[3]),Y(colors[3])),"per-pixel original Y");ck("LightingLed keeps black finite",grade[2].maxColorComponent<.001f&&!float.IsNaN(grade[2].r),grade[2].ToString());
   m.SetVector("_LD_Mode",Vector4.zero);Draw(Input(2,2,Fill(4,new Color(.18f,.18f,.18f,1))),dst,m,0);ck("legacy palette unchanged",Near(Read(dst)[0].b,.18f/.0722f),"legacy mode0 blue ramp");
   b.SetVector("_LD_Mode",new Vector4(1,1,0,0));b.SetVector("_LD_SourceSize",new Vector4(2,2,1,1));b.SetVector("_LD_Bloom",new Vector4(1.1f,0,8,.2f));b.SetTexture("_LD_Coverage",Tex(2,2,valid));var tiny=Fill(4,Color.black);for(int i=0;i<4;i++)tiny[i].a=1;tiny[0]=new Color(4,4,4,1);Draw(Input(2,2,tiny),one,b,0);var seed=Read(one)[0];ck("prefilter before area mean preserves one-pixel HDR source",Near(seed.r,.725f),seed.ToString("F5"));
   var edge=Fill(4,new Color(1,0,1,1));edge[3]=Color.clear;b.SetTexture("_LD_Coverage",Tex(2,2,edge));var mixed=Fill(4,new Color(4,2,1,1));mixed[3]=new Color(100,100,100,0);Draw(Input(2,2,mixed),one,b,0);var edgeA=Read(one)[0];mixed[3]=new Color(0,0,0,0);Draw(Input(2,2,mixed),one,b,0);var edgeB=Read(one)[0];ck("legal source survives invalid neighbour without hidden contribution",edgeA.r>1&&Near(edgeA.r,edgeB.r),edgeA+" / "+edgeB);
   b.SetTexture("_LD_Coverage",Tex(2,2,Fill(4,Color.clear)));Draw(Input(2,2,Fill(4,new Color(100,100,100,1))),one,b,0);var zero=Read(one)[0];ck("zero source permission has no residual energy",zero.r<.001f&&zero.g<.001f&&zero.b<.001f,zero.ToString());
   b.SetTexture("_LD_Coverage",Tex(2,2,edge));mixed[3]=new Color(float.NaN,float.PositiveInfinity,100,0);Draw(Input(2,2,mixed),one,b,0);var finite=Read(one)[0];ck("hidden non-finite texel cannot poison legal Bloom",!float.IsNaN(finite.r)&&!float.IsInfinity(finite.r)&&Near(finite.r,edgeB.r),finite.ToString());
   var level=Input(2,2,Fill(4,new Color(2,1,.5f,0)));b.SetVector("_LD_SourceSize",new Vector4(2,2,2,2));b.SetVector("_LD_LowSize",new Vector4(2,2,.5f,.5f));b.SetTexture("_LD_LowTexture",level.rt);b.SetFloat("_LD_Scatter",.65f);Draw(level,dst,b,2);ck("new intermediate RGB energy not vetoed by debug alpha",Read(dst)[0].r>1,"upsample RGB with alpha0");
   m.SetVector("_LD_Features",Vector4.zero);m.SetVector("_LD_Mode",new Vector4(1,1,0,0));m.SetVector("_LD_PixelDebug",new Vector4(1,0,1,0));m.SetTexture("_LD_BloomTexture",level.rt);m.SetVector("_LD_BloomSize",new Vector4(2,2,.5f,.5f));m.SetVector("_LD_Clip",new Vector4(.05f,1,0,0));m.SetTexture("_LD_Coverage",Tex(2,2,valid));var dark=Input(2,2,Fill(4,new Color(.1f,.1f,.1f,1)));Draw(dark,dst,m,2);ck("new final accepts legal recipient without min-alpha erosion",Near(Read(dst)[0].r,2.1f),Read(dst)[0].ToString());m.SetTexture("_LD_Coverage",Tex(2,2,Fill(4,Color.clear)));Draw(dark,dst,m,2);ck("unknown recipient never receives Bloom",Near(Read(dst)[0].r,.1f),Read(dst)[0].ToString());
   context.ReturnValue=new{success=failed==0,failed,device=SystemInfo.graphicsDeviceName,checks=rows};
  }catch(Exception e){context.ReturnValue=new{success=false,error=e.ToString(),checks=rows};}
  finally{RenderTexture.active=prev;foreach(var h in handles)h.Release();foreach(var o in objects)UnityEngine.Object.DestroyImmediate(o);}
 }
 static Color[] Fill(int n,Color c){var a=new Color[n];for(int i=0;i<n;i++)a[i]=c;return a;}
 Material Mat(Shader s){var m=new Material(s);objects.Add(m);return m;}
 Texture2D Tex(int w,int h,Color[] c){var t=new Texture2D(w,h,TextureFormat.RGBAFloat,false,true);objects.Add(t);t.filterMode=FilterMode.Point;t.wrapMode=TextureWrapMode.Clamp;t.SetPixels(c);t.Apply(false,false);return t;}
 RTHandle Target(int w,int h){var t=RTHandles.Alloc(w,h,colorFormat:UnityEngine.Experimental.Rendering.GraphicsFormat.R16G16B16A16_SFloat,filterMode:FilterMode.Point,name:"SurfaceLighting private GPU fixture");handles.Add(t);return t;}
 RTHandle Input(int w,int h,Color[] c){var t=Target(w,h);Graphics.Blit(Tex(w,h,c),t.rt);return t;}
 static void Draw(RTHandle s,RTHandle t,Material m,int pass){var cmd=new CommandBuffer{name="SurfaceLighting private GPU contract"};try{using(new Ossastra.Rendering.RenderCommandScope(cmd))Blitter.BlitCameraTexture(cmd,s,t,m,pass);Graphics.ExecuteCommandBuffer(cmd);}finally{cmd.Release();}}
 Color[] Read(RTHandle t){RenderTexture.active=t.rt;var o=new Texture2D(t.rt.width,t.rt.height,TextureFormat.RGBAFloat,false,true);objects.Add(o);o.ReadPixels(new Rect(0,0,o.width,o.height),0,0);o.Apply(false,false);return o.GetPixels();}
}
