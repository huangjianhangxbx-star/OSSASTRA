using System;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Stylization;using Funplay.Editor.Tools.Scripting;
public sealed class SurfaceStartupLifecycleContract:IFunplayCommand{
 public void Execute(ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Stable candidate Play required");var r=SceneManager.GetActiveScene().GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<SurfaceLightingRecipe>(true)).Single();var t=typeof(SurfaceLightingRecipe);var flags=BindingFlags.Instance|BindingFlags.NonPublic;var callback=t.GetMethod("OnStartupCameraRendered",flags);var hook=t.GetField("startupCameraHooked",flags);var pending=t.GetField("pendingInitial",flags);var observed=t.GetField("startupCameraObserved",flags);if(callback==null||hook==null||pending==null||observed==null){ctx.ReturnValue=new{passed=false,stage="RED",reason="Startup camera barrier missing"};return;}var rows=new List<object>();int failed=0;Action<string,bool> check=(name,passed)=>{rows.Add(new{name,passed});if(!passed)failed++;};var starting=r.State;string runtimeJson=r.RuntimeProfile!=null?JsonUtility.ToJson(r.RuntimeProfile):null;GameObject wrong=null;string error="";Action<string> invoke=name=>t.GetMethod(name,flags).Invoke(r,null);Func<FieldInfo,bool> read=f=>(bool)f.GetValue(r);
 try{r.Release();invoke("OnEnable");check("OnEnable installs one pending camera barrier",read(hook)&&read(pending)&&!read(observed));invoke("Update");check("first Update cannot attempt before the explicit camera rendered",!r.HasLease&&read(hook)&&read(pending));callback.Invoke(r,new object[]{default(ScriptableRenderContext),null});check("null camera grants no permit",read(hook)&&!read(observed));wrong=new GameObject("__StartupWrongCameraTransient"){hideFlags=HideFlags.HideAndDontSave};var c=wrong.AddComponent<Camera>();c.enabled=false;callback.Invoke(r,new object[]{default(ScriptableRenderContext),c});check("another camera grants no permit",read(hook)&&!read(observed));callback.Invoke(r,new object[]{default(ScriptableRenderContext),r.environment.gameplayCamera});check("one explicit camera end removes the hook without applying in render callback",!read(hook)&&read(observed)&&read(pending)&&!r.HasLease);invoke("Update");check("next Update consumes permit and attempts the initial state once",!read(pending)&&!read(hook)&&r.HasLease&&r.State==r.initialState);r.Release();check("Release clears pending hook and permit",!read(hook)&&!read(pending)&&!read(observed)&&!r.HasLease);invoke("OnEnable");invoke("OnDisable");check("Disable clears a still-waiting subscription",!read(hook)&&!read(pending)&&!read(observed));}
 catch(Exception e){failed++;rows.Add(new{name="Native lifecycle exception",passed=false});error=e.ToString();}
 finally{if(wrong!=null)UnityEngine.Object.DestroyImmediate(wrong);r.Release();if(!r.TrySetState(starting,out var restoreError)){failed++;rows.Add(new{name="restore starting state",passed=false,error=restoreError});}else if(runtimeJson!=null&&r.RuntimeProfile!=null)JsonUtility.FromJsonOverwrite(runtimeJson,r.RuntimeProfile);}
 ctx.ReturnValue=new{passed=failed==0,checks=rows.Count,failed,rows,error,restored=r.State.ToString(),note="Synthetic callback proves lifecycle; true first Player camera is separately verified by the final binary."};}
}
