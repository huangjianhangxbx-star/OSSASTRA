using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Ossastra.Audio;using Ossastra.CharacterCore;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class DeathSoundBoundary:IFunplayCommand
{
 public static object Result;static Ossastra.Audio.GameAudioRoot root;static Ossastra.Characters.CharacterRoot actor;static double end;static List<Ossastra.Audio.GameAudioRoot.SfxObservation> observed;static Action<Ossastra.Audio.GameAudioRoot.SfxObservation> onSound;static bool killed;
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){root=Ossastra.Audio.GameAudioRoot.Instance;actor=UnityEngine.Object.FindObjectOfType<Ossastra.Characters.CharacterControlCoordinator>()?.Current;var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(root==null||actor==null||enemy==null)throw new Exception("Real world missing");actor.SendMessage("OnApplicationFocus",true);observed=new List<Ossastra.Audio.GameAudioRoot.SfxObservation>();Result=null;onSound=o=>{if(o.Request.SourceId==actor.Vitals.Health.Id+"@"+actor.GetInstanceID())observed.Add(o);};root.SfxObserved+=onSound;var release=AttackRelease.ForEmission(Guid.NewGuid(),0,enemy.Vitals.Health.Id,1000,0);var receipts=CombatContactGateway.Resolve(release,enemy.Vitals,new IHitTarget[]{actor.Vitals},actor.transform.position);killed=receipts.Any(r=>r.Reason==HitReason.Applied&&r.HpAfter==0);end=EditorApplication.timeSinceStartup+.8;EditorApplication.update+=Poll;c.ReturnValue=new{started=true,killed,scope="DEVELOPMENT_ONLY real gateway lethal boundary, no raw HP writes; fresh Play reset follows"};}
 static void Poll(){if(Application.isPlaying&&EditorApplication.timeSinceStartup<end)return;EditorApplication.update-=Poll;if(root!=null)root.SfxObserved-=onSound;Result=new{killed,events=observed.Select(o=>new{id=o.Request.EventId,o.SourcePlayIssued}).ToArray(),checks=new[]{new{name="lethal receipt remains one actual death",pass=killed&&actor.Vitals.DeathCount==1},new{name="death uses prehandoff player role",pass=observed.Any(o=>o.Request.EventId=="player.body.death"&&o.SourcePlayIssued)},new{name="lethal contact does not stack hurt cry with death cry",pass=!observed.Any(o=>o.Request.EventId.EndsWith("body.hurt")&&o.SourcePlayIssued)}}};}
}
