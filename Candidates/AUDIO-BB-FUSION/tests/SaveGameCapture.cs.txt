using System;using System.IO;using System.Linq;using System.Collections;using UnityEngine;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public class CompactVectorConverter:JsonConverter{public override bool CanConvert(Type t)=>t==typeof(Vector3)||t==typeof(Vector2);public override void WriteJson(JsonWriter w,object value,JsonSerializer s){w.WriteStartArray();if(value is Vector3 a){w.WriteValue(a.x);w.WriteValue(a.y);w.WriteValue(a.z);}else{var b=(Vector2)value;w.WriteValue(b.x);w.WriteValue(b.y);}w.WriteEndArray();}public override object ReadJson(JsonReader r,Type t,object value,JsonSerializer s)=>throw new NotSupportedException();}

{
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
 {

 }
