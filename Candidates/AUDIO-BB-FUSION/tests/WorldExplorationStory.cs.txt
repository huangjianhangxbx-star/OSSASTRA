/* 自有Play场真实左摇杆/Tab输入；只暂移实体并持有测试设备，不写HP/速度/音频事实。
 * 区域停止探针只用区域自己的来源许可，验证终止后节制重试；不冒充预算抢占。
 */
using System;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Funplay.Editor.Tools.Scripting;
using A=Ossastra.Audio;using C=Ossastra.Characters;using E=Ossastra.Enemies;
public class WorldExplorationStory:IFunplayCommand{
public static object Report;static A.GameAudioRoot root;static C.CharacterControlCoordinator cc;static C.CharacterRoot actor;static Gamepad pad;static Keyboard keys;static double start;static int next,frame;static bool mixFocus;static string error;static readonly List<(double at,Action action)> steps=new List<(double,Action)>();static readonly List<object> checks=new List<object>();static readonly List<string> events=new List<string>();static readonly Dictionary<Transform,Vector3> poses=new Dictionary<Transform,Vector3>();static readonly List<(Component c,bool focused)> focus=new List<(Component,bool)>();static GameObject wall;static int blockedBefore,zoneBefore;static A.AudioEnvironmentZone zone;static Action<A.GameAudioRoot.SfxObservation> observed;
public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){try{if(!Application.isPlaying)throw new Exception("Play required");root=A.GameAudioRoot.Instance;cc=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();actor=cc.Current;steps.Clear();checks.Clear();events.Clear();poses.Clear();focus.Clear();next=0;frame=-1;error="";Focus(cc);foreach(var a in cc.Characters){poses[a.transform]=a.transform.position;Focus(a);}foreach(var e in UnityEngine.Object.FindObjectsOfType<E.EnemyRoot>()){poses[e.transform]=e.transform.position;Focus(e);Place(e.transform,new Vector3(80,0,80));}mixFocus=root.Mix.Focused;root.SendMessage("OnApplicationFocus",true);pad=InputSystem.AddDevice<Gamepad>("FUSION_WorldPad");keys=InputSystem.AddDevice<Keyboard>("FUSION_WorldKeys");observed=o=>{if(o.SourcePlayIssued)events.Add(o.Request.EventId);};root.SfxObserved+=observed;
At(.2,()=>{Place(actor.transform,new Vector3(-5,0,2));Pad(0);});
At(1,()=>{zone=UnityEngine.Object.FindObjectsOfType<A.AudioEnvironmentZone>().First(z=>z.ZoneId=="wood_room");zoneBefore=zone.AcceptedStarts;var source=(A.SfxSourceHandle)typeof(A.AudioEnvironmentZone).GetField("source",BindingFlags.NonPublic|BindingFlags.Instance).GetValue(zone);var voice=root.Sfx.ActiveVoices.FirstOrDefault(v=>v.Request.SourceId==source?.SourceId);Check("zone entered via actual controlled actor",zone.IsInside&&zone.ActiveVoiceCount==1);if(voice!=null)source.StopVoice(voice.VoiceId);});
At(2,()=>Check("terminated loop retries using actual permit liveness",zone.AcceptedStarts>zoneBefore&&zone.ActiveVoiceCount==1));
At(2.2,()=>Pad(.55f));At(10.5,()=>{Pad(0);Check("walk across wood stone dirt",new[]{"wood","stone","dirt"}.All(s=>events.Contains("player.footstep."+s+".walk")));Place(actor.transform,new Vector3(-5,0,2));});
At(11,()=>Pad(1));At(14.7,()=>{Pad(0);Check("run across wood stone dirt",new[]{"wood","stone","dirt"}.All(s=>events.Contains("player.footstep."+s+".run")));Place(actor.transform,new Vector3(0,0,3));wall=new GameObject("DEVELOPMENT_ONLY_wall");wall.transform.position=new Vector3(.6f,1,3);wall.AddComponent<BoxCollider>().size=new Vector3(.2f,2,2);Physics.SyncTransforms();Pad(1);});
At(15.8,()=>blockedBefore=events.Count(x=>x.StartsWith("player.footstep.")));At(16.5,()=>{Check("holding movement against real wall creates no new footsteps",events.Count(x=>x.StartsWith("player.footstep."))==blockedBefore);Pad(0);if(wall!=null)UnityEngine.Object.Destroy(wall);InputSystem.QueueStateEvent(keys,new KeyboardState(Key.Tab));});
At(16.8,()=>InputSystem.QueueStateEvent(keys,new KeyboardState()));At(17.5,()=>InputSystem.QueueStateEvent(keys,new KeyboardState(Key.Tab)));At(17.8,()=>InputSystem.QueueStateEvent(keys,new KeyboardState()));
At(18.4,()=>{Check("real Tab inventory open and close edges issue audio",events.Contains("ui.inventory.open")&&events.Contains("ui.inventory.close"));Place(actor.transform,new Vector3(40,0,40));Pad(1);});
At(19.3,()=>blockedBefore=events.Count(x=>x.StartsWith("player.footstep.")));At(20,()=>{Check("moving outside real floor emits no footsteps",events.Count(x=>x.StartsWith("player.footstep."))==blockedBefore);Check("all three real ambience zones entered",new[]{"ambience.torch","ambience.fire","ambience.rain"}.All(events.Contains));Check("leaving all zones releases loops after fade",UnityEngine.Object.FindObjectsOfType<A.AudioEnvironmentZone>().All(z=>!z.IsInside&&z.ActiveVoiceCount==0));Cleanup();});
start=EditorApplication.timeSinceStartup;Report=null;EditorApplication.update+=Tick;c.ReturnValue=new{started=true,seconds=21};}catch(Exception e){error=e.ToString();Cleanup();c.ReturnValue=Report;}}
static void At(double t,Action a)=>steps.Add((t,a));static void Check(string name,bool pass)=>checks.Add(new{name,pass});static void Pad(float x){if(pad!=null)InputSystem.QueueStateEvent(pad,new GamepadState{leftStick=new Vector2(x,0),rightStick=Vector2.right});}
static void Place(Transform t,Vector3 p){var b=t.GetComponent<CharacterController>();bool e=b!=null&&b.enabled;if(e)b.enabled=false;t.position=p;if(e)b.enabled=true;Physics.SyncTransforms();}
static void Focus(Component c){var f=c.GetType().GetField("focused",BindingFlags.Instance|BindingFlags.NonPublic);if(f==null)return;focus.Add((c,(bool)f.GetValue(c)));c.SendMessage("OnApplicationFocus",true);}
static void Tick(){if(frame==Time.frameCount)return;frame=Time.frameCount;try{if(!Application.isPlaying||EditorApplication.timeSinceStartup-start>30)throw new Exception("Story deadline");if(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].at)steps[next++].action();}catch(Exception e){error=e.ToString();Cleanup();}}
static void Cleanup(){EditorApplication.update-=Tick;Pad(0);if(root!=null){root.SfxObserved-=observed;root.SendMessage("OnApplicationFocus",mixFocus);}foreach(var p in poses)if(p.Key!=null)Place(p.Key,p.Value);foreach(var p in focus)if(p.c!=null)p.c.SendMessage("OnApplicationFocus",p.focused);if(pad!=null)InputSystem.RemoveDevice(pad);if(keys!=null)InputSystem.RemoveDevice(keys);if(wall!=null)UnityEngine.Object.Destroy(wall);Report=new{done=true,error,checks=checks.ToArray(),events=events.ToArray(),devicesRemaining=InputSystem.devices.Count(d=>d.name.StartsWith("FUSION_World"))};}}
