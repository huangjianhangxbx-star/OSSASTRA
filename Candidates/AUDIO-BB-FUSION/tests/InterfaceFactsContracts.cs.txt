/* 开发合同：InterfaceFactsContracts
 * 功能：验证实际界面状态边沿，首帧/重连不响，稳定刷新不响，容器切换保留关闭与打开两个事实。
 * 边界：只反射调用纯状态观察器，不创建界面、不提交声音、不改变库存或Unity现场。
 */
using System;
using System.Collections;
using System.Collections.Generic;
using System.Linq;
using System.Reflection;
using Funplay.Editor.Tools.Scripting;

public sealed class InterfaceFactsContracts : IFunplayCommand
{
    public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
    {
        var rows = new List<object>();
        int passed = 0, failed = 0;
        void Check(string name, bool ok, object detail = null)
        { if (ok) passed++; else failed++; rows.Add(new { name, passed = ok, detail }); }
        try
        {
            var type = AppDomain.CurrentDomain.GetAssemblies()
                .Select(assembly => assembly.GetType("Ossastra.Audio.InterfaceAudioEdges", false))
                .FirstOrDefault(candidate => candidate != null);
            Check("真实界面边沿观察器已实现", type != null);
            if (type != null)
            {
                var observe = type.GetMethod("Observe", new[] { typeof(bool), typeof(string) });
                var reset = type.GetMethod("Reset", BindingFlags.Instance | BindingFlags.Public);
                Check("公开状态观察与重采基线入口存在", observe != null && reset != null);
                if (observe != null && reset != null)
                {
                    var instance = Activator.CreateInstance(type);
                    string[] Read(bool open, string external)
                        => ((IEnumerable)observe.Invoke(instance, new object[] { open, external })).Cast<string>().ToArray();
                    bool Is(string[] actual, params string[] expected) => actual.SequenceEqual(expected);

                    Check("初次关闭状态只建立基线", Is(Read(false, null)));
                    Check("相同关闭状态不产生声音", Is(Read(false, null)));
                    Check("真正打开背包只产生一次打开事实", Is(Read(true, null), "ui.inventory.open"));
                    Check("界面刷新而开关不变时不产生事实", Is(Read(true, null)));
                    Check("合法容器打开产生容器事实", Is(Read(true, "external"), "interaction.container.open"));
                    Check("容器不变不重复播放", Is(Read(true, "external")));
                    Check("容器更换先关闭原容器再打开新容器", Is(Read(true, "ground:1"), "interaction.container.close", "interaction.container.open"));
                    Check("关包同时关闭容器且各只产生一次", Is(Read(false, null), "interaction.container.close", "ui.inventory.close"));
                    Check("重复关闭保持安静", Is(Read(false, null)));

                    reset.Invoke(instance, null);
                    Check("世界换代时已打开界面只重采基线", Is(Read(true, "external")));
                    Check("重连后相同状态保持安静", Is(Read(true, "external")));
                    Check("重连后的真实关闭仍可观察", Is(Read(false, ""), "interaction.container.close", "ui.inventory.close"));

                    var initiallyOpen = Activator.CreateInstance(type);
                    var initial = ((IEnumerable)observe.Invoke(initiallyOpen, new object[] { true, "external" })).Cast<string>().ToArray();
                    Check("首次发现已打开的界面不补播历史事实", initial.Length == 0);
                }
            }
        }
        catch (Exception error) { Check("界面事实合同执行无异常", false, error.ToString()); }
        ctx.ReturnValue = new { passed, failed, rows, scope = "纯界面状态边沿；不伪造库存事务或发声" };
    }
}
