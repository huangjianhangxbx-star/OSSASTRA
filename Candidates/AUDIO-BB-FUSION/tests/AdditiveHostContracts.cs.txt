using System;using UnityEngine;using UnityEngine.SceneManagement;using Ossastra.Audio;using Funplay.Editor.Tools.Scripting;
public class AdditiveHostContracts:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){var root=Ossastra.Audio.GameAudioRoot.Instance;if(root==null||!Application.isPlaying)throw new Exception("Play required");var old=UnityEngine.Object.FindObjectOfType<BgmContextHost>();var original=SceneManager.GetActiveScene();var extra=SceneManager.CreateScene("DEVELOPMENT_ONLY_AUDIO_ADDITIVE");var go=new GameObject("SecondaryAudioHost");SceneManager.MoveGameObjectToScene(go,extra);var child=go.AddComponent<BgmContextHost>();bool stayed=root.IsBoundHost(old);var focus=old.AcquireDialogueFocus();SceneManager.SetActiveScene(extra);bool childBound=root.IsBoundHost(child);bool oldFocusReleased=root.Mix.SnapshotCount==0;SceneManager.SetActiveScene(original);focus.Dispose();UnityEngine.Object.DestroyImmediate(go);SceneManager.UnloadSceneAsync(extra);c.ReturnValue=new{checks=new[]{new{name="nonactive additive host does not steal music context",pass=stayed},new{name="explicit active change binds its actual host",pass=childBound},new{name="active change releases previous dialogue scope",pass=oldFocusReleased},new{name="unloading secondary retains primary host",pass=root.IsBoundHost(old)},new{name="active scene restored",pass=SceneManager.GetActiveScene()==original}}};}}
