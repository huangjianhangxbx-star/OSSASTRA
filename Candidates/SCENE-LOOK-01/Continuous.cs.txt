// 功能：用原昼夜/相机入口连续推进16.5→19时并旋转45度，用于录屏；明确18秒终止和归还镜头。
using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using Funplay.Editor.Tools.Scripting;using Ossastra.EnvironmentSimulation.Rendering;using Ossastra.MapResearch;using Ossastra.WorldRendering.VoxelSurface;
public class SceneLookContinuous:IFunplayCommand{
static double started;static bool rotated;static EnvironmentSkyBinding sky;static MapResearchRoot map;static int frames;static void Render(UnityEngine.Rendering.ScriptableRenderContext ctx,Camera camera){if(map!=null&&camera==map.gameplayCamera)frames++;}
public void Execute(ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Play only");sky=UnityEngine.Object.FindObjectOfType<EnvironmentSkyBinding>();map=UnityEngine.Object.FindObjectOfType<MapResearchRoot>();var c=UnityEngine.Object.FindObjectOfType<VoxelSurfaceComparison>();if(sky==null||map==null||c==null)throw new Exception("Explicit roots missing");c.ApplyLook(c.lookPresets.Single(l=>l.name=="RichTimberHall"));c.SetSessionAdjustments(.15f,0);c.FlushPendingParameters();sky.sceneEnvironmentEnabled=true;sky.overrideSkyColors=false;sky.environment.SetManualSun(false,225,15);sky.environment.SetContribution(Ossastra.EnvironmentSimulation.EnvironmentContribution.Combined);sky.environment.SetClockRunning(false);UnityEngine.Object.FindObjectOfType<Ossastra.Development.SceneVisibility.RoomVisibilityLabPanel>().PlaceCurrent(new Vector3(4,.3f,-17));map.followParticipant=false;map.fixedTarget=new Vector3(4,.1f,-10);map.distance=30;started=EditorApplication.timeSinceStartup;rotated=false;frames=0;EditorApplication.update+=Tick;UnityEngine.Rendering.RenderPipelineManager.endCameraRendering+=Render;ctx.ReturnValue="18-second continuous natural dusk + one original45degree camera step scheduled";}
static void Tick(){double seconds=EditorApplication.timeSinceStartup-started;if(!Application.isPlaying||seconds>=18){EditorApplication.update-=Tick;UnityEngine.Rendering.RenderPipelineManager.endCameraRendering-=Render;if(sky!=null)sky.environment.SetTimeHours(17);if(rotated&&map!=null)map.RequestRotationStep(-1);File.WriteAllText("artifacts/SCENE-LOOK-01/continuous-result.json",JsonUtility.ToJson(new Result{frames=frames,seconds=(float)seconds,rotated=rotated},true));return;}sky.environment.SetTimeHours(16.5+seconds/18*2.5);if(seconds>=8&&!rotated){map.RequestRotationStep(1);rotated=true;}EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();}
[Serializable]class Result{public int frames;public float seconds;public bool rotated;}}
