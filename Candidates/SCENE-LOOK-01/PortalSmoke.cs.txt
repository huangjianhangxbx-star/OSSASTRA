using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Funplay.Editor.Tools.Scripting;
using Ossastra.EnvironmentSimulation.Rendering;using Ossastra.SceneVisibility.Rooms.Rendering;
public class SceneLookPortalSmoke:IFunplayCommand{
static Ossastra.Development.SceneVisibility.RoomVisibilityLabPanel lab;static EnvironmentSkyBinding sky;static double since;static int stage;static bool captured;static bool oldWindow,oldDoor;static Vector3 oldPos;static readonly List<object> rows=new List<object>();
public void Execute(ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Isolated Play required");var c=UnityEngine.Object.FindObjectOfType<Ossastra.Commerce.Runtime.CommerceService>();if(c==null||!Path.GetFullPath(c.saveDirectory).Replace('\\','/').StartsWith("E:/WORLDCREATOR/XingHaiHuiLang/研究/SCENE-LOOK-01/"))throw new Exception("Isolated store only");lab=UnityEngine.Object.FindObjectOfType<Ossastra.Development.SceneVisibility.RoomVisibilityLabPanel>();sky=UnityEngine.Object.FindObjectOfType<EnvironmentSkyBinding>();if(lab==null||lab.shutterPortal==null||lab.doorPortal==null||sky==null)throw new Exception("Original author portal fixtures missing");oldWindow=lab.shutterPortal.open;oldDoor=lab.doorPortal.open;oldPos=lab.visibility.coordinator.Current.transform.position;rows.Clear();stage=0;sky.sceneEnvironmentEnabled=true;sky.environment.SetTimeHours(17);sky.environment.SetManualSun(false,225,15);sky.environment.SetContribution(Ossastra.EnvironmentSimulation.EnvironmentContribution.Combined);lab.PlaceCurrent(lab.knownRoomPosition);lab.map.followParticipant=false;lab.map.fixedTarget=lab.knownRoomPosition;lab.map.distance=20;Apply();EditorApplication.update+=Tick;ctx.ReturnValue="Two-state original portal smoke scheduled";}
static void Apply(){lab.SetShutterOpen(stage==0);lab.SetDoorOpen(stage==0);since=EditorApplication.timeSinceStartup;captured=false;}
static void Tick(){try{if(!Application.isPlaying){Finish("Play ended");return;}EditorApplication.QueuePlayerLoopUpdate();UnityEditorInternal.InternalEditorUtility.RepaintAllViews();if(EditorApplication.timeSinceStartup-since<3)return;if(captured){if(EditorApplication.timeSinceStartup-since<4.5)return;stage++;if(stage==2){Finish(null);return;}Apply();return;}RoomMaskCameraRegistry.TryGetStatistics(lab.visibility.gameplayCamera,out var stats);rows.Add(new{stage,window=lab.shutterPortal.open,door=lab.doorPortal.open,active=stats!=null&&stats.Active,unsupported=stats==null?-1:stats.UnsupportedMaterialCount,error=stats==null?"missing":stats.Error});ScreenCapture.CaptureScreenshot("artifacts/SCENE-LOOK-01/final/portal-"+(stage==0?"open":"closed")+".png");captured=true;}catch(Exception e){Finish(e.ToString());}}
static void Finish(string error){EditorApplication.update-=Tick;if(lab!=null&&Application.isPlaying){lab.SetShutterOpen(oldWindow);lab.SetDoorOpen(oldDoor);lab.PlaceCurrent(oldPos);}File.WriteAllText("artifacts/SCENE-LOOK-01/portal-smoke.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{error,rows},Newtonsoft.Json.Formatting.Indented));}}
