using System;using System.Linq;using UnityEngine;using UnityEngine.UI;using Funplay.Editor.Tools.Scripting;
public class SceneLookPanelContract:IFunplayCommand{public void Execute(ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Play only");var p=UnityEngine.Object.FindObjectOfType<Ossastra.Development.Environment.EnvironmentSimulationPanel>();var c=UnityEngine.Object.FindObjectOfType<Ossastra.WorldRendering.VoxelSurface.VoxelSurfaceComparison>();if(p==null||p.skyBinding==null||c==null)throw new Exception("Explicit panel binding missing");bool oldExpanded=p.Expanded,old=p.skyBinding.sceneEnvironmentEnabled;var look=c.ActiveLook;double hours=p.environment.timeHours;p.SetExpanded(true);Canvas.ForceUpdateCanvases();var row=p.Surface.GetComponentsInChildren<RectTransform>(true).Single(t=>t.name=="SceneEnvironmentAB");var b=row.GetComponentsInChildren<UnityEngine.UI.Button>(true);if(b.Length!=2)throw new Exception("Actual two buttons missing");bool a=false,z=false;try{b[0].onClick.Invoke();a=!p.skyBinding.sceneEnvironmentEnabled;b[1].onClick.Invoke();z=p.skyBinding.sceneEnvironmentEnabled;ctx.ReturnValue=new{pass=a&&z&&c.ActiveLook==look&&hours==p.environment.timeHours,buttonA=a,buttonB=z,materialLookUnchanged=c.ActiveLook==look,timeUnchanged=hours==p.environment.timeHours};}finally{p.skyBinding.sceneEnvironmentEnabled=old;p.SetExpanded(oldExpanded);}}}
