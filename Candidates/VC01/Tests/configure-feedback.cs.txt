using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Ossastra.UserInterface;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConfigureFeedback:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var s=EditorSceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/Scenes/SC_CharacterVerticalSlice.unity")throw new Exception("Own Edit scene required");s=EditorSceneManager.OpenScene(s.path);var hud=UnityEngine.Object.FindObjectOfType<CharacterHudAdapter>();var feedback=hud.GetComponent<CombatFeedbackView>()??Undo.AddComponent<CombatFeedbackView>(hud.gameObject);ctx.RegisterObjectModification(feedback);var so=new SerializedObject(feedback);foreach(var name in new[]{"release","hit","hurt"})so.FindProperty(name).objectReferenceValue=AssetDatabase.LoadAssetAtPath<AudioClip>("Assets/OSSASTRA/Art/AudioReferenceLocal/"+name+".wav");so.FindProperty("bottle").objectReferenceValue=UnityEngine.Object.FindObjectOfType<HealingBottleService>();so.ApplyModifiedProperties();if(UnityEngine.Object.FindObjectOfType<AudioListener>()==null)Undo.AddComponent<AudioListener>(Camera.main.gameObject);EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue="feedback clips wired to original local reference samples";}}
