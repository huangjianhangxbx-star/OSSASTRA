using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Funplay.Editor.Tools.Scripting;
public class ProjectileSpriteProbe:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var a=c.Current;a.SendMessage("OnApplicationFocus",true);c.SendMessage("OnApplicationFocus",true);var equipment=a.GetComponent<CharacterEquipment>();equipment.TryEquip(OffhandKind.Bomb);var offhand=a.GetComponent<CharacterOffhand>();var expected=equipment.LeftDefinition.worldSprite;var accepted=offhand.Request(a.transform.position+Vector3.right,true);var id=offhand.LastActionId;var rows=new List<object>();int failed=0,stage=0;var start=EditorApplication.timeSinceStartup;EditorApplication.CallbackFunction tick=null;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}tick=()=>{if(stage==0&&EditorApplication.timeSinceStartup-start>.3){var renderer=UnityEngine.Object.FindObjectsOfType<SpriteRenderer>().FirstOrDefault(r=>r.name=="OffhandProjectile_"+id.ToString("N"));Check("accepted independent bomb uses frozen actual sprite",accepted&&renderer!=null&&renderer.sprite==expected&&offhand.ReleasedCount==1);stage++;}if(stage==1&&EditorApplication.timeSinceStartup-start>.9){Check("landing cleans actual projectile sprite",offhand.LandedCount==1&&!UnityEngine.Object.FindObjectsOfType<SpriteRenderer>().Any(r=>r.name=="OffhandProjectile_"+id.ToString("N")));EditorApplication.update-=tick;SessionState.SetString("VC01_PROJECTILE_SPRITE",Newtonsoft.Json.JsonConvert.SerializeObject(new{state="completed",failed,rows}));}};EditorApplication.update+=tick;}}
