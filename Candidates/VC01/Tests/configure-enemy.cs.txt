using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Spine.Unity;using Ossastra.AnimationCore;using Ossastra.AnimationPresentation;using Funplay.Editor.Tools.Scripting;
public class ConfigureEnemy:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var s=EditorSceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/Scenes/SC_CharacterVerticalSlice.unity")throw new Exception("Own Edit scene required");s=EditorSceneManager.OpenScene(s.path);var folder="Assets/OSSASTRA/Configs/Enemies";if(!AssetDatabase.IsValidFolder(folder))AssetDatabase.CreateFolder("Assets/OSSASTRA/Configs","Enemies");var media="Assets/OSSASTRA/Art/EnemyReferenceLocal/";var mat=AssetDatabase.LoadAssetAtPath<Material>(folder+"/Zombie.mat");if(mat==null){mat=new Material(Shader.Find("Spine/Skeleton"));mat.name="僵尸1";mat.mainTexture=AssetDatabase.LoadAssetAtPath<Texture2D>(media+"僵尸1.png");ctx.RegisterObjectCreation(mat);AssetDatabase.CreateAsset(mat,folder+"/Zombie.mat");}var atlas=AssetDatabase.LoadAssetAtPath<SpineAtlasAsset>(folder+"/ZombieAtlas.asset");if(atlas==null){atlas=SpineAtlasAsset.CreateRuntimeInstance(AssetDatabase.LoadAssetAtPath<TextAsset>(media+"ZombieAtlas.txt"),new[]{mat},true);ctx.RegisterObjectCreation(atlas);AssetDatabase.CreateAsset(atlas,folder+"/ZombieAtlas.asset");}var data=AssetDatabase.LoadAssetAtPath<SkeletonDataAsset>(folder+"/ZombieSkeleton.asset");if(data==null){data=SkeletonDataAsset.CreateRuntimeInstance(AssetDatabase.LoadAssetAtPath<TextAsset>(media+"ZombieSkeleton.txt"),atlas,true,.008f);if(data.GetSkeletonData(false)==null)throw new Exception("Actual skeleton load failed");ctx.RegisterObjectCreation(data);AssetDatabase.CreateAsset(data,folder+"/ZombieSkeleton.asset");}var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();var skeleton=enemy.GetComponentInChildren<SkeletonAnimation>();ctx.RegisterObjectModification(skeleton);ctx.RegisterObjectModification(skeleton.transform);skeleton.skeletonDataAsset=data;skeleton.AnimationName="_stand";skeleton.loop=true;skeleton.transform.localScale=Vector3.one;skeleton.Initialize(true);var presenter=enemy.GetComponentInChildren<EntitySpinePresenter>();ctx.RegisterObjectModification(presenter);var so=new SerializedObject(presenter);so.FindProperty("idle").stringValue="_stand";var list=so.FindProperty("catalog");var clips=new[]{"_stand","run","attack","_damaged","_stun","_die"};var semantics=new[]{AnimationSemantic.Idle,AnimationSemantic.Move,AnimationSemantic.BasicAttack,AnimationSemantic.HitReaction,AnimationSemantic.PostureBroken,AnimationSemantic.Death};list.arraySize=clips.Length;for(int i=0;i<clips.Length;i++){if(data.GetSkeletonData(false).FindAnimation(clips[i])==null)throw new Exception("Missing clip "+clips[i]);var item=list.GetArrayElementAtIndex(i);item.FindPropertyRelative("semantic").intValue=(int)semantics[i];item.FindPropertyRelative("clip").stringValue=clips[i];item.FindPropertyRelative("loop").boolValue=i==0||i==1||i==4;item.FindPropertyRelative("mixDuration").floatValue=.06f;}so.ApplyModifiedProperties();EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=new{model=data.GetSkeletonData(false).Version,height=data.GetSkeletonData(false).Height,enemy=enemy.ActorId};}}
