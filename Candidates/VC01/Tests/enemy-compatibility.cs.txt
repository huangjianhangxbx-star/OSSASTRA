using System;using System.IO;using System.Linq;using Funplay.Editor.Tools.Scripting;
public class EnemyAtlasCompatibility:IFunplayCommand{private class Loader:Spine.TextureLoader{public void Load(Spine.AtlasPage page,string path){}public void Unload(object texture){}}public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var dir="E:/WORLDCREATOR/XingHaiHuiLang/Origin/work/AL-01/assets/zombie";try{var atlas=new Spine.Atlas(new StringReader(File.ReadAllText(dir+"/unit.atlas")),dir,new Loader());var loader=new Spine.SkeletonJson(new Spine.AtlasAttachmentLoader(atlas));var data=loader.ReadSkeletonData(new StringReader(File.ReadAllText(dir+"/unit.json")));var skeleton=new Spine.Skeleton(data);var state=new Spine.AnimationState(new Spine.AnimationStateData(data));foreach(var animation in data.Animations){state.SetAnimation(0,animation,true);state.Update(.1f);state.Apply(skeleton);skeleton.UpdateWorldTransform(Spine.Physics.Update);}ctx.ReturnValue=new{loaded=true,version=data.Version,animations=data.Animations.Select(a=>a.Name).ToArray(),bones=data.Bones.Count};}catch(Exception e){ctx.ReturnValue=new{loaded=false,error=e.ToString()};}}}
