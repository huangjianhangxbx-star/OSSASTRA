using System;using System.Reflection;using UnityEditor;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class SetQaResolution:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var flags=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;var asm=typeof(Editor).Assembly;var sizesType=asm.GetType("UnityEditor.GameViewSizes");var sizes=sizesType.BaseType.GetProperty("instance",BindingFlags.Public|BindingFlags.Static).GetValue(null);var group=sizesType.GetProperty("currentGroup",flags).GetValue(sizes);var groupType=group.GetType();var viewType=asm.GetType("UnityEditor.GameView");var view=EditorWindow.GetWindow(viewType);var selected=viewType.GetProperty("selectedSizeIndex",flags);int count=(int)groupType.GetMethod("GetTotalCount",flags).Invoke(group,null);if(!SessionState.GetBool("VC01_QA_SIZES",false)){SessionState.SetInt("VC01_QA_ORIGINAL",(int)selected.GetValue(view));SessionState.SetInt("VC01_QA_FIRST",count);var sizeType=asm.GetType("UnityEditor.GameViewSize");var kind=Enum.Parse(asm.GetType("UnityEditor.GameViewSizeType"),"FixedResolution");var dimensions=new[]{new[]{1920,1080},new[]{1440,900},new[]{1280,720}};foreach(var pair in dimensions){var size=Activator.CreateInstance(sizeType,new object[]{kind,pair[0],pair[1],"VC01 QA "+pair[0]});groupType.GetMethod("AddCustomSize",flags).Invoke(group,new[]{size});}SessionState.SetBool("VC01_QA_SIZES",true);}selected.SetValue(view,SessionState.GetInt("VC01_QA_FIRST",0)+QA_INDEX);view.Repaint();ctx.ReturnValue="QA resolution selected";}}
