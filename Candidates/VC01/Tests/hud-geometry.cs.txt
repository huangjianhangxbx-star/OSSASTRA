using System;using System.Linq;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class HudGeometry:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var k=Mathf.Min(Screen.width/1920f,Screen.height/1080f);var names=new[]{"HeroCluster","CompanionCluster","WorldCluster","SkillCluster","VitalityCluster","QuickCluster"};var expected=new[]{new Rect(34*k,41*k,529*k,163*k),new Rect(3*k,Screen.height-220*k,377*k,220*k),new Rect(Screen.width-272*k,17*k,259*k,117*k),new Rect(Screen.width-255*k,Screen.height-320*k,254*k,273*k),new Rect(Screen.width-352*k,Screen.height-30*k,350*k,30*k),new Rect(Screen.width*.5f-410*k,Screen.height-115*k,866*k,110*k)};var rows=names.Select((name,i)=>{var r=GameObject.Find(name).GetComponent<RectTransform>();var corners=new Vector3[4];r.GetWorldCorners(corners);var actual=new Rect(corners[1].x,Screen.height-corners[1].y,corners[2].x-corners[1].x,corners[1].y-corners[0].y);var e=expected[i];bool pass=Math.Abs(actual.x-e.x)<1.5&&Math.Abs(actual.y-e.y)<1.5&&Math.Abs(actual.width-e.width)<1.5&&Math.Abs(actual.height-e.height)<1.5;return new{name,pass,actual=actual.ToString(),expected=e.ToString()};}).ToArray();ctx.ReturnValue=new{width=Screen.width,height=Screen.height,scale=k,failed=rows.Count(r=>!r.pass),rows};}}
