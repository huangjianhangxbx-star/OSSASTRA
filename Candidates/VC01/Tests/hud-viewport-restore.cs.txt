using System;using System.Reflection;using UnityEditor;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class RestoreQaResolution:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(!SessionState.GetBool("VC01_QA_SIZES",false)){ctx.ReturnValue="not active";return;}var f=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;var asm=typeof(Editor).Assembly;var t=asm.GetType("UnityEditor.GameViewSizes");var sizes=t.BaseType.GetProperty("instance",BindingFlags.Static|BindingFlags.Public).GetValue(null);var group=t.GetProperty("currentGroup",f).GetValue(sizes);var gt=group.GetType();var vt=asm.GetType("UnityEditor.GameView");var view=EditorWindow.GetWindow(vt);vt.GetProperty("selectedSizeIndex",f).SetValue(view,SessionState.GetInt("VC01_QA_ORIGINAL",0));var total=(int)gt.GetMethod("GetTotalCount",f).Invoke(group,null);var first=SessionState.GetInt("VC01_QA_FIRST",0);if(total!=first+3)throw new Exception("Custom size list changed; original selected, preserve list");var builtins=(int)gt.GetMethod("GetBuiltinCount",f).Invoke(group,null);for(int i=2;i>=0;i--)gt.GetMethod("RemoveCustomSize",f).Invoke(group,new object[]{first+i-builtins});SessionState.SetBool("VC01_QA_SIZES",false);view.Repaint();ctx.ReturnValue="original resolution restored, three own QA sizes removed";}}
