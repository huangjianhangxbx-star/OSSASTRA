using System;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.InputSystem;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConfigureBindings:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var s=EditorSceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/Scenes/SC_CharacterVerticalSlice.unity")throw new Exception("Own Edit scene required");s=EditorSceneManager.OpenScene(s.path);var coordinator=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();ctx.RegisterObjectModification(coordinator);for(int i=0;i<coordinator.switchAction.bindings.Count;i++)if(coordinator.switchAction.bindings[i].path=="<Keyboard>/tab")coordinator.switchAction.ChangeBinding(i).WithPath("<Keyboard>/z");var camp=UnityEngine.Object.FindObjectOfType<PrototypeCampfire>();ctx.RegisterObjectModification(camp);for(int i=0;i<camp.interactAction.bindings.Count;i++)if(camp.interactAction.bindings[i].path.StartsWith("<Keyboard>"))camp.interactAction.ChangeBinding(i).WithPath("<Keyboard>/f");EditorUtility.SetDirty(coordinator);EditorUtility.SetDirty(camp);EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=new{switchKey=coordinator.switchAction.bindings[0].path,campfire=camp.interactAction.bindings[0].path};}}
