using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Ossastra.Characters;using Ossastra.CharacterCore;using Funplay.Editor.Tools.Scripting;
public class ConfigureEquipment:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(Application.isPlaying||scene.path!="Assets/OSSASTRA/Scenes/SC_CharacterVerticalSlice.unity"||scene.isDirty)throw new Exception("Own clean Edit scene required");var folder="Assets/OSSASTRA/Configs/Equipment";if(!AssetDatabase.IsValidFolder(folder))AssetDatabase.CreateFolder("Assets/OSSASTRA/Configs","Equipment");EquipmentDefinitionSO Make(string id,string name,bool main,OffhandKind kind,string art,string bone,float size){var path=folder+"/"+id+".asset";var d=AssetDatabase.LoadAssetAtPath<EquipmentDefinitionSO>(path);if(d!=null)return d;d=ScriptableObject.CreateInstance<EquipmentDefinitionSO>();d.definitionId=id;d.displayName=name;d.mainHand=main;d.offhandKind=kind;d.boneName=bone;d.worldScale=size;if(art!=null)d.worldSprite=AssetDatabase.LoadAssetAtPath<Sprite>(art);d.offset=new Vector3(0,0,-.015f);if(main)d.actions=Enumerable.Range(1,4).Select(i=>AssetDatabase.LoadAssetAtPath<ActionDefinitionSO>("Assets/OSSASTRA/Configs/Actions/Rozeul_Basic"+i+".asset")).ToArray();ctx.RegisterObjectCreation(d);AssetDatabase.CreateAsset(d,path);AssetDatabase.SaveAssetIfDirty(d);return d;}var blade=Make("Rozeul_Blade","近战刀",true,OffhandKind.None,null,"Rhand",1);var shield=Make("Rozeul_Shield","巡行盾",false,OffhandKind.Shield,"Assets/OSSASTRA/Art/Equipment/RozeulShield_v1.png","Lhand",.55f);var bomb=Make("Rozeul_Bomb","投掷炸弹",false,OffhandKind.Bomb,"Assets/Plugins/CW/Destructible2D/Examples/Textures/Bomb.png","Lhand",.35f);foreach(var root in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>()){var equipment=root.GetComponent<CharacterEquipment>();ctx.RegisterObjectModification(equipment);var so=new SerializedObject(equipment);so.FindProperty("rightDefinition").objectReferenceValue=blade;var list=so.FindProperty("ownedOffhands");list.arraySize=2;list.GetArrayElementAtIndex(0).objectReferenceValue=shield;list.GetArrayElementAtIndex(1).objectReferenceValue=bomb;so.ApplyModifiedProperties();if(root.GetComponent<CharacterEquipmentView>()==null)Undo.AddComponent<CharacterEquipmentView>(root.gameObject);}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=new{shield=shield.worldSprite!=null,bomb=bomb.worldSprite!=null,roots=UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>().Length};}}
