 * 不直接调用Paint/Erase/BoxFill；SendEvent是否消费不算通过，必须实际Tile/GetSprite变化。
 * 约10秒异步执行，超时30秒。通过反射轮询FrontierActualScenePersistence.ResultJson。
 * 实际GUI拖刷/擦除Undo后保存同一scene，两次Close/Open按sourceId重绑，再重建显示/重置自有Demo并保存；不对旧引用执行Undo回退。
 * 实际菜单/UIToolkit的物理鼠标命中仍由root另外验；本候选只证明原生GUI事件与同源持久性。

    {

    }
    public void Execute(ExecutionContext ctx)
    {
        if (active != null) { ctx.ReturnValue = new { status = "ALREADY_RUNNING", pollType = "FrontierActualScenePersistence", pollField = "ResultJson" }; return; }
        active = new Runner();
        try { active.Start(); }
        catch (Exception e) { active.End("ACTUAL_SCENEVIEW_BRUSH_SETUP_FAILED", e.ToString()); }
        ctx.ReturnValue = new { status = active == null ? "FINISHED_OR_SETUP_FAILED" : "ACTUAL_SCENEVIEW_BRUSH_STARTED", pollType = "FrontierActualScenePersistence", pollField = "ResultJson", assembly = typeof(FrontierActualScenePersistence).Assembly.FullName, result = ResultJson };
    }

        static Vector3Int[] EmptyHorizontalRoute(Tilemap target, BoundsInt bounds)
        {
            if (bounds.size.x < 9 || bounds.size.y < 3) return null;
            for (int y = bounds.yMin + 1; y < bounds.yMax - 1; y++)
                for (int x = bounds.xMin + 1; x + 6 < bounds.xMax - 1; x++)
                    if (Enumerable.Range(0, 7).All(k => target.GetTile(new Vector3Int(x + k, y, 0)) == null))
                        return new[] { new Vector3Int(x, y, 0), new Vector3Int(x + 3, y, 0), new Vector3Int(x + 6, y, 0) };
            return null;
        }
        Vector3Int[] RouteCells() => Enumerable.Range(path[0].x, 7).Select(x => new Vector3Int(x, path[0].y, 0)).ToArray();

        Vector2 WindowPoint(Vector3 world)
        {
            var camera = view.camera; var pixel = camera.WorldToScreenPoint(world); float pp = EditorGUIUtility.pixelsPerPoint;
            float viewportTop = view.position.height - (camera.pixelRect.y + camera.pixelRect.height) / pp;
            return new Vector2(pixel.x / pp, viewportTop + (camera.pixelRect.y + camera.pixelRect.height - pixel.y) / pp);
        }
        void DuringSceneGui(SceneView current)
        {
            if (current != view) return;
            if(Event.current!=null)
            {
                lastReceivedX=Event.current.mousePosition.x;lastReceivedY=Event.current.mousePosition.y;
                var plane=new Plane(-map.transform.up,map.transform.position);var ray=HandleUtility.GUIPointToWorldRay(Event.current.mousePosition);
                if(plane.Raycast(ray,out float distance))lastPublicRayCell=map.WorldToCell(ray.GetPoint(distance)).ToString();
            }
            guiSamplesPerPhase.TryGetValue(phase,out int phaseSamples);
            if (phaseSamples < 6)
            { guiStates.Add(Observe("actual-scene-gui", true)); guiSamplesPerPhase[phase]=phaseSamples+1; lastGuiPhase = phase; }
            if (Event.current.type != EventType.Repaint) return;
            try
            {
                // HandleUtility坐标属于Scene相机的GUI clip；SendEvent要求EditorWindow根坐标。
                if(!nativeCalibrated)gui = path.Select(x => WindowPoint(map.GetCellCenterWorld(x))).ToArray();
                var safe = new Rect(35, 45, view.position.width - 70, view.position.height - 90);
                if (!gui.All(x => !float.IsNaN(x.x) && !float.IsNaN(x.y) && safe.Contains(x))) { guiError = "Native GUI projection outside safe scene area: " + string.Join(";", gui.Select(x => x.ToString()).ToArray()); return; }
                coordinatesReady = true;
            }
            catch (Exception e) { guiError = e.ToString(); }
        }

                    previewCount=preview.Count,nativePreviewX=observed.x,nativePreviewY=observed.y,wantX=path[0].x,wantY=path[0].y,

                if(observed==path[0]){success=true;nativeCalibrated=true;break;}

        {

                foreach (var cell in baseline.bounds.allPositionsWithin)

        void Navigation(string name, int button, EventModifiers modifiers)
        {
            Step(name + "-frame", () => FrameRoute());
            Step(name + "-enter", () => Enter(), true);
            Step(name + "-down", () => Send(EventType.MouseDown, gui[0], button, modifiers));
            Step(name + "-drag", () => Send(EventType.MouseDrag, gui[1], button, modifiers, gui[1] - gui[0]));
            Step(name + "-up", () => Send(EventType.MouseUp, gui[1], button, modifiers));
            Step(name + "-check", () => Require(name + "-navigation-does-not-write-author-data", initialMaps.All(x => x.Matches()) && !brush.IsStrokeActive));
        }
        void Tick()
        {
            if (ending || sceneSwitching) return;
            try
            {
                double now = EditorApplication.timeSinceStartup;
                if (now - started > 30) throw new TimeoutException("SceneView events/projection timed out at " + phase + "; " + guiError);
                if (Application.isPlaying || EditorApplication.isCompiling || scene != SceneManager.GetActiveScene()) throw new InvalidOperationException("Editor lifecycle changed during fixture.");
                if (now < nextTick) return;
                if (steps.Count == 0) { End("ACTUAL_SCENEVIEW_BRUSH_COMPLETE", null); return; }
                var step = steps.Peek(); phase = step.name;
                ResultJson = PureJson(new { status = "RUNNING", phase = phase, scene = scene.name, passed = rows.Count(x => x.passed), failed = rows.Count(x => !x.passed) });
                if (step.awaitCoordinates && EditorWindow.focusedWindow != view) { view.Show(); view.Focus(); view.Repaint(); return; }
                if (step.awaitCoordinates && !coordinatesReady) { view.Repaint(); return; }
                steps.Dequeue(); step.run(); nextTick = now + 0.12;
            }
            catch (Exception e) { End("ACTUAL_SCENEVIEW_BRUSH_FAILED", e.ToString()); }
        }
        void ReopenAndBind()
        {
            sceneSwitching=true;SceneView.duringSceneGui-=DuringSceneGui;oldMapInstance=map.GetInstanceID();
            try
            {
                GridPaintingState.scenePaintTarget=null;ToolManager.RestorePreviousTool();
                // 空内存scene只为合法关闭最后一个scene，不保存第二个场景文件。
                lifeboat=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);SceneManager.SetActiveScene(lifeboat);
                Require("actually-closes-saved-native-scene-cycle-"+(reopenCycles+1),EditorSceneManager.CloseScene(scene,true));
                scene=EditorSceneManager.OpenScene(FrontierGroundSceneBuilder.ScenePath,OpenSceneMode.Single);
                surface=scene.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<GroundResearchSurface>(true)).Single();
                layer=surface.layers.Single(l=>l!=null&&l.sourceId==expectedSourceId);map=layer.tilemap;reopened=true;reopenCycles++;
                Require("reopened-map-new-instance-cycle-"+reopenCycles,map!=null&&map.GetInstanceID()!=oldMapInstance);
                Require("reopened-surface-same-owned-path-cycle-"+reopenCycles,scene.path==FrontierGroundSceneBuilder.ScenePath&&surface.Owns(map));
                brush.Configure(surface,Array.IndexOf(surface.layers,layer),1);GridPaintingState.gridBrush=brush;GridPaintingState.scenePaintTarget=map.gameObject;
            }
            finally{sceneSwitching=false;SceneView.duringSceneGui+=DuringSceneGui;}
        }
        void RestoreDemo()
        {
            if(surface==null||!surface.gameObject.scene.IsValid()||surface.gameObject.scene.path!=FrontierGroundSceneBuilder.ScenePath)throw new InvalidOperationException("Refusing reset outside owned Lab.");
            FrontierGroundSceneBuilder.ResetDemonstration(surface);
            Require("root-owned-fixture-demo-restored-and-saved",EditorSceneManager.SaveScene(surface.gameObject.scene,FrontierGroundSceneBuilder.ScenePath));
            demoRestored=true;
        }
        public void End(string status,string error)
        {
            if(ending)return;ending=true;EditorApplication.update-=Tick;SceneView.duringSceneGui-=DuringSceneGui;
            try
            {
                if(brush!=null)brush.FinishStroke(false);
                if(fixtureGroup>=0&&!demoRestored&&!EditorApplication.isPlayingOrWillChangePlaymode&&!EditorApplication.isCompiling)
                {
                    if((surface==null||!surface.gameObject.scene.IsValid())&&persisted&&SceneManager.GetActiveScene()==lifeboat)
                    {
                        scene=EditorSceneManager.OpenScene(FrontierGroundSceneBuilder.ScenePath,OpenSceneMode.Single);
                        surface=scene.GetRootGameObjects().SelectMany(o=>o.GetComponentsInChildren<GroundResearchSurface>(true)).Single();
                    }
                    RestoreDemo();
                }
                // 不对已Close/Open的旧对象执行Undo回退；只重新解析已保存GlobalObjectId。
                if(stateCaptured)
                {
                    var restoredTarget=priorTargetId.Equals(default(GlobalObjectId))?null:GlobalObjectId.GlobalObjectIdentifierToObjectSlow(priorTargetId) as GameObject;
                    var restoredSelection=priorSelectionIds.Select(GlobalObjectId.GlobalObjectIdentifierToObjectSlow).Where(o=>o!=null).ToArray();
                    if(priorBrush is FrontierGroundBrush custom && !string.IsNullOrEmpty(priorCustomLayerId)&&surface!=null)
                    {
                        int index=Array.FindIndex(surface.layers,l=>l!=null&&l.sourceId==priorCustomLayerId);
                        if(index>=0)custom.Configure(surface,index,priorCustomDiameter);
                    }
                    GridPaintingState.gridBrush=priorBrush==null?null:priorBrush;GridPaintingState.scenePaintTarget=restoredTarget;Selection.objects=restoredSelection;
                    if(priorTool!=null)ToolManager.SetActiveTool(priorTool);else ToolManager.RestorePreviousTool();
                    Check("prior-target-restored-by-global-id-not-destroyed-reference",restoredTarget==null||restoredTarget.scene.IsValid()||EditorUtility.IsPersistent(restoredTarget));
                    if(ownsView&&view!=null)view.Close();
                    if(palette!=null&&!previousWindows.Contains(palette.GetInstanceID()))palette.Close();
                    if(priorFocus!=null)priorFocus.Focus();
                }
            }
            catch(Exception e){Check("persistence-cleanup-exception",false,e.ToString());if(error==null)error=e.ToString();}
            finally
            {
                if(brush!=null)UnityEngine.Object.DestroyImmediate(brush);
                if(status=="ACTUAL_SCENEVIEW_BRUSH_COMPLETE")status=rows.Any(r=>!r.passed)?"ACTUAL_SCENEVIEW_PERSISTENCE_FAILED":"ACTUAL_SCENEVIEW_PERSISTENCE_GREEN";
                ResultJson=PureJson(new{status,error,phase,scene=scene.IsValid()?scene.path:"closed",sourceId=expectedSourceId,tilePath=expectedTilePath,persisted,reopened,reopenCycles,demoRestored,chunkBefore,chunkAfter,
                    route=path==null?null:path.Select(p=>new[]{p.x,p.y,p.z}).ToArray(),calibration=calibrations.ToArray(),
                    rows=rows.Select(r=>new{r.name,r.passed,r.evidence}).ToArray(),passed=rows.Count(r=>r.passed),failed=rows.Count(r=>!r.passed),
                    boundary="Actual native GUI-created stroke saved/closed/reopened in the same Lab; new references resolved by sourceId/GlobalObjectId; no direct Paint/SetTile, no duplicate scene file, no stale-reference Undo. Root explicitly authorizes restoring its own demo after verification."},Newtonsoft.Json.Formatting.Indented);
                active=null;
            }
        }
