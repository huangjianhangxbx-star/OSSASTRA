    static string PureJson(object value, Newtonsoft.Json.Formatting formatting = Newtonsoft.Json.Formatting.None)
    {
        var serializer = Newtonsoft.Json.JsonSerializer.Create(new Newtonsoft.Json.JsonSerializerSettings {
            Formatting = formatting, TypeNameHandling = Newtonsoft.Json.TypeNameHandling.None,
            PreserveReferencesHandling = Newtonsoft.Json.PreserveReferencesHandling.None,
            ContractResolver = new Newtonsoft.Json.Serialization.DefaultContractResolver()
        });
        using(var writer=new System.IO.StringWriter(System.Globalization.CultureInfo.InvariantCulture)) { serializer.Serialize(writer,value); return writer.ToString(); }
    }
    public void Execute(ExecutionContext ctx)
    {
        if (active != null) { ctx.ReturnValue = new { status = "ALREADY_RUNNING", pollType = "FrontierActualSceneEdgeContracts", pollField = "ResultJson" }; return; }
        active = new Runner();
        try { active.Start(); }
        catch (Exception e) { active.End("ACTUAL_SCENEVIEW_BRUSH_SETUP_FAILED", e.ToString()); }
        ctx.ReturnValue = new { status = active == null ? "FINISHED_OR_SETUP_FAILED" : "ACTUAL_SCENEVIEW_BRUSH_STARTED", pollType = "FrontierActualSceneEdgeContracts", pollField = "ResultJson", assembly = typeof(FrontierActualSceneEdgeContracts).Assembly.FullName, result = ResultJson };
    }

        static Vector3Int[] EmptyHorizontalRoute(Tilemap target, BoundsInt bounds)
        {
            if (bounds.size.x < 9 || bounds.size.y < 3) return null;
            for (int y = bounds.yMin + 1; y < bounds.yMax - 1; y++)
                for (int x = bounds.xMin + 1; x + 6 < bounds.xMax - 1; x++)
                    if (Enumerable.Range(0, 7).All(k => target.GetTile(new Vector3Int(x + k, y, 0)) == null))
                        return new[] { new Vector3Int(x, y, 0), new Vector3Int(x + 3, y, 0), new Vector3Int(x + 6, y, 0) };
            return null;
        }

        void FrameRoute()
        {
            coordinatesReady = false; guiError = null; nativeCalibrated=false;
            view.in2DMode = false;
            view.orthographic = true;
            view.LookAtDirect(map.GetCellCenterWorld(path[1]), Quaternion.Euler(90f, 0, 0), Mathf.Max(10f, surface.WorldBounds.size.magnitude * 0.6f));
            view.Focus(); view.Repaint();
            TilemapEditorTool.SetActiveEditorTool(typeof(PaintTool));
        }
        Vector2 WindowPoint(Vector3 world)
        {
            var camera = view.camera; var pixel = camera.WorldToScreenPoint(world); float pp = EditorGUIUtility.pixelsPerPoint;
            float viewportTop = view.position.height - (camera.pixelRect.y + camera.pixelRect.height) / pp;
            return new Vector2(pixel.x / pp, viewportTop + (camera.pixelRect.y + camera.pixelRect.height - pixel.y) / pp);
        }
        void DuringSceneGui(SceneView current)
        {
            if (current != view) return;
            if(Event.current!=null)
            {
                lastReceivedX=Event.current.mousePosition.x;lastReceivedY=Event.current.mousePosition.y;
                var plane=new Plane(-map.transform.up,map.transform.position);var ray=HandleUtility.GUIPointToWorldRay(Event.current.mousePosition);
                if(plane.Raycast(ray,out float distance))lastPublicRayCell=map.WorldToCell(ray.GetPoint(distance)).ToString();
            }
            if (Event.current.rawType == EventType.MouseLeaveWindow) actualMouseLeaveCount++;
            observedHot = GUIUtility.hotControl; hotEvent = Event.current.rawType.ToString();
            if (Event.current.type != EventType.Repaint) return;
            try
            {
                // HandleUtility坐标属于Scene相机的GUI clip；SendEvent要求EditorWindow根坐标。
                if(!nativeCalibrated)gui = path.Select(x => WindowPoint(map.GetCellCenterWorld(x))).ToArray();
                var safe = new Rect(35, 45, view.position.width - 70, view.position.height - 90);
                if (!gui.All(x => !float.IsNaN(x.x) && !float.IsNaN(x.y) && safe.Contains(x))) { guiError = "Native GUI projection outside safe scene area: " + string.Join(";", gui.Select(x => x.ToString()).ToArray()); return; }
                coordinatesReady = true;
            }
            catch (Exception e) { guiError = e.ToString(); }
        }
        void Enter()
        {
            view.Focus();
            var originalTool=ToolManager.activeToolType; var calibrationBefore=initialMaps.Select(m=>new MapSnapshot{map=m.map,bounds=m.bounds,tiles=m.map.GetTilesBlock(m.bounds)}).ToArray();
            bool eligible=brush.CanPaintTarget(GridPaintingState.scenePaintTarget);
            if(eligible)TilemapEditorTool.SetActiveEditorTool(typeof(PaintTool));
            bool success=!eligible;
            if(eligible)
            {
                // Native OnMouseLeave会清预览但不清hash；公开MouseMove到相邻格再回来，让Native正常刷新。
                var from=view.camera.WorldToScreenPoint(map.GetCellCenterWorld(path[0]));var to=view.camera.WorldToScreenPoint(map.GetCellCenterWorld(path[0]+Vector3Int.right));
                float pp=EditorGUIUtility.pixelsPerPoint;var nudge=new Vector2((to.x-from.x)/pp,-(to.y-from.y)/pp);
                Send(EventType.MouseEnterWindow,gui[0]);Send(EventType.MouseMove,gui[0]+nudge);Send(EventType.Layout,gui[0]+nudge);
            }
            for(int attempt=0;attempt<4;attempt++)
            {
                var sent=gui[0];Send(EventType.MouseEnterWindow,sent);Send(EventType.MouseMove,sent);Send(EventType.Layout,sent);
                var preview=new List<Vector3Int>();foreach(var cell in surface.EditableCells.allPositionsWithin)if(map.GetEditorPreviewTile(cell)==layer.paintTile)preview.Add(cell);
                var observed=preview.Count==1?preview[0]:new Vector3Int(int.MinValue,int.MinValue,0);
                calibrations.Add(new{phase,attempt,eligible,sentWindowX=sent.x,sentWindowY=sent.y,receivedGuiX=lastReceivedX,receivedGuiY=lastReceivedY,publicRayCell=lastPublicRayCell,
                    previewCount=preview.Count,nativePreviewX=observed.x,nativePreviewY=observed.y,wantX=path[0].x,wantY=path[0].y,
                    actualAuthorDataUnchanged=calibrationBefore.All(x=>x.Matches()),cameraPixelWidth=view.camera.pixelRect.width,cameraPixelHeight=view.camera.pixelRect.height,windowHeight=view.position.height,pp=EditorGUIUtility.pixelsPerPoint});
                if(!eligible){success=true;break;}
                Require("calibration-received-unique-native-paint-preview",preview.Count==1,"attempt="+attempt+"; preview count="+preview.Count+"; publicRay="+lastPublicRayCell);
                if(observed==path[0]){success=true;nativeCalibrated=true;break;}
                var wantedPixel=view.camera.WorldToScreenPoint(map.GetCellCenterWorld(path[0]));var observedPixel=view.camera.WorldToScreenPoint(map.GetCellCenterWorld(observed));
                float pp=EditorGUIUtility.pixelsPerPoint;var correction=new Vector2((wantedPixel.x-observedPixel.x)/pp,-(wantedPixel.y-observedPixel.y)/pp);
                for(int i=0;i<gui.Length;i++)gui[i]+=correction;
                nativeCalibrated=true;
            }
            Require("calibration-native-preview-matches-author-target",success,"Public Ray is diagnostic only; actual native preview must match before MouseDown.");
            Require("calibration-never-writes-author-data",calibrationBefore.All(x=>x.Matches()));
            if(originalTool!=null&&originalTool!=typeof(PaintTool))TilemapEditorTool.SetActiveEditorTool(originalTool);
            Send(EventType.MouseMove,gui[0]);Send(EventType.Layout,gui[0]);
        }
        void Send(EventType type, Vector2 point, int button = 0, EventModifiers modifiers = EventModifiers.None, Vector2 delta = default)
        {
            var e = new Event { type = type, mousePosition = point, button = button, modifiers = modifiers, delta = delta, clickCount = type == EventType.MouseDown ? 1 : 0 };
            bool used = view.SendEvent(e);
            dispatches.Add(new Dispatch { type = type.ToString(), button = button, modifiers = modifiers.ToString(), point = point, consumed = used });
        }
        void Navigation(string name, int button, EventModifiers modifiers)
        {
            Step(name + "-frame", () => FrameRoute());
            Step(name + "-enter", () => Enter(), true);
            Step(name + "-down", () => Send(EventType.MouseDown, gui[0], button, modifiers));
            Step(name + "-drag", () => Send(EventType.MouseDrag, gui[1], button, modifiers, gui[1] - gui[0]));
            Step(name + "-up", () => Send(EventType.MouseUp, gui[1], button, modifiers));
            Step(name + "-check", () => Require(name + "-navigation-does-not-write-author-data", initialMaps.All(x => x.Matches()) && !brush.IsStrokeActive));
        }

        void ClearCase()
        {
            // 仅病例之间的夹具清理；不以直接提交/回滚冒充被测的GUI事件结果。
            brush.FinishStroke(true); Undo.RevertAllDownToGroup(fixtureGroup); Undo.IncrementCurrentGroup();
            Require("case-cleanup-restores-native-author-data", initialMaps.All(x => x.Matches()));
            TilemapEditorTool.SetActiveEditorTool(typeof(PaintTool));
            Send(EventType.MouseUp, gui[2]); Send(EventType.Layout, gui[0]);
        }
        void Tick()
        {
            if (ending) return;
            try
            {
                double now = EditorApplication.timeSinceStartup;
                if (now - started > 30) throw new TimeoutException("SceneView events/projection timed out at " + phase + "; " + guiError);
                if (Application.isPlaying || EditorApplication.isCompiling || scene != SceneManager.GetActiveScene()) throw new InvalidOperationException("Editor lifecycle changed during fixture.");
                if (now < nextTick) return;
                if (steps.Count == 0) { End("ACTUAL_SCENEVIEW_BRUSH_COMPLETE", null); return; }
                var step = steps.Peek(); phase = step.name;
                ResultJson = PureJson(new { status = "RUNNING", phase = phase, scene = scene.name, passed = rows.Count(x => x.passed), failed = rows.Count(x => !x.passed) });
                if (step.awaitCoordinates && EditorWindow.focusedWindow != view) { view.Show(); view.Focus(); view.Repaint(); return; }
                if (step.awaitCoordinates && !coordinatesReady) { view.Repaint(); return; }
                steps.Dequeue(); step.run(); nextTick = now + 0.12;
            }
            catch (Exception e) { End("ACTUAL_SCENEVIEW_BRUSH_FAILED", e.ToString()); }
        }
        public void End(string status, string error)
        {
            if (ending) return; ending = true;
            EditorApplication.update -= Tick; SceneView.duringSceneGui -= DuringSceneGui;
            try
            {
                // 事务API只用于夹具清理，所有成功路径的绘制/取消/提交已由真实GUI事件证明。
                if (brush != null) brush.FinishStroke(true);
                if (fixtureGroup >= 0) { Undo.RevertAllDownToGroup(fixtureGroup); Undo.IncrementCurrentGroup(); }
                if (initialMaps != null) Check("cleanup-restores-only-fixture-author-data", initialMaps.All(x => x.Matches()));
                if (stateCaptured)
                {
                    GridPaintingState.gridBrush = priorBrush; GridPaintingState.scenePaintTarget = priorTarget;
                    Selection.objects = priorSelection;
                    if (priorTool != null) ToolManager.SetActiveTool(priorTool); else ToolManager.RestorePreviousTool();
                    if (originalView != null) Check("original-sceneview-object-remains-loaded", Resources.FindObjectsOfTypeAll<SceneView>().Contains(originalView),
                        "Before pivot=" + priorPivot.ToString("F6") + "; after=" + originalView.pivot.ToString("F6") + "; rotation before=" + priorRotation.ToString("F6") + "; after=" + originalView.rotation.ToString("F6") + "; size before=" + priorSize + "; after=" + originalView.size + "; ortho before=" + priorOrtho + "; after=" + originalView.orthographic + "; 2D before=" + prior2D + "; after=" + originalView.in2DMode + "; dock before=" + priorDocked + "; after=" + originalView.docked + "; rect before=" + priorViewRect + "; after=" + originalView.position + "; camera/dock readback only: public Show may restore minimized main window or settle prior camera animation.");
                    if (ownsView && view != null) view.Close();
                    if (palette != null && !previousWindows.Contains(palette.GetInstanceID())) palette.Close();
                    if (priorFocus != null) priorFocus.Focus();
                }
            }
            catch (Exception e) { Check("cleanup-exception", false, e.ToString()); if (error == null) error = e.ToString(); }
            finally
            {
                if (sentinel != null) UnityEngine.Object.DestroyImmediate(sentinel); if (brush != null) UnityEngine.Object.DestroyImmediate(brush);
                if (status == "ACTUAL_SCENEVIEW_BRUSH_COMPLETE") status = rows.Any(x => !x.passed) ? "ACTUAL_SCENEVIEW_EDGE_CONTRACTS_RED" : "ACTUAL_SCENEVIEW_EDGE_CONTRACTS_GREEN";
                ResultJson = PureJson(new { status, error, phase, scene = scene.IsValid() ? scene.name : "none", layer = layer == null ? "none" : layer.displayName,
                    route = path == null ? null : path.Select(p => new[] { p.x, p.y, p.z }).ToArray(), paintedSprites = sprites, calibration=calibrations.ToArray(),
                    rows = rows.Select(r => new { r.name, r.passed, r.evidence }).ToArray(),
                    dispatches = dispatches.Select(d => new { d.type, d.modifiers, d.button, point = new[] { d.point.x, d.point.y }, d.consumed }).ToArray(),
                    passed = rows.Count(x => x.passed), failed = rows.Count(x => !x.passed), initialDirty, finalDirty = scene.IsValid() && scene.isDirty,
                    boundary = "Actual SceneView two-box/focus and owned native Undo sentinel; only cleanup calls Finish/Revert. No direct Paint/SetTile. Full author bounds and exclusive root Editor lease required." }, Newtonsoft.Json.Formatting.Indented);
                active = null;
            }
        }
    }
