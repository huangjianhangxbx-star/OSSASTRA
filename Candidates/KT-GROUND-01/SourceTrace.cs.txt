using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.Tilemaps;using Funplay.Editor.Tools.Scripting;using G=Ossastra.GroundResearch;using E=Ossastra.GroundResearch.Editor;
public class GroundSourceTrace:IFunplayCommand{public void Execute(ExecutionContext ctx){var s=Object.FindObjectOfType<G.GroundResearchSurface>();var all=s.EditableCells;var cells=new System.Collections.Generic.List<Vector3Int>();foreach(var c in all.allPositionsWithin)cells.Add(c);var cases=new[]{new{label="base-only grass",index=0,cell=cells.First(c=>s.layers[0].tilemap.HasTile(c)&&s.layers.Skip(1).All(l=>!l.tilemap.HasTile(c)))},new{label="road",index=6,cell=cells.First(c=>s.layers[6].tilemap.HasTile(c))},new{label="complex edge",index=2,cell=cells.First(c=>s.layers[2].tilemap.HasTile(c)&&s.layers.Count(l=>l.tilemap.HasTile(c))>=3)}};var brush=ScriptableObject.CreateInstance<E.FrontierGroundBrush>();try{ctx.ReturnValue=cases.Select(x=>{brush.Configure(s,x.index,1);var l=s.layers[x.index];brush.Pick(s.grid,l.tilemap.gameObject,new BoundsInt(x.cell,Vector3Int.one),x.cell);var tile=l.tilemap.GetTile(x.cell);var sprite=l.tilemap.GetSprite(x.cell);return new{use=x.label,picked=brush.LastPickedMap==l.tilemap&&brush.LastPickedCell==x.cell,layer=l.displayName,cell=x.cell.ToString(),tile=AssetDatabase.GetAssetPath(tile),sprite=sprite.name,atlas=AssetDatabase.GetAssetPath(sprite),detail=AssetDatabase.GetAssetPath(l.detailTexture),sourceContainer=l.sourceContainer,sourceObjectId=l.sourceObjectId,evidence=l.evidence};}).ToArray();}finally{Object.DestroyImmediate(brush);}}}
