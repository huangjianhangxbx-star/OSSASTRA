using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Tilemaps;using Funplay.Editor.Tools.Scripting;using G=Ossastra.GroundResearch;
public class GroundVisibleCoverage:IFunplayCommand{public void Execute(ExecutionContext ctx){var s=UnityEngine.Object.FindObjectOfType<G.GroundResearchSurface>();var map=s.layers[0].tilemap;var mat=map.GetComponent<TilemapRenderer>().sharedMaterial;var clip=mat.GetVector("_GroundBounds");var rects=new List<Rect>();bool fullQuads=true;foreach(var cell in s.EditableCells.allPositionsWithin){var sprite=map.GetSprite(cell);if(sprite==null){fullQuads=false;continue;}var vertices=sprite.vertices;fullQuads&=vertices.Length==4;var center=map.GetCellCenterWorld(cell);var points=vertices.Select(v=>center+map.transform.TransformVector(map.orientationMatrix.MultiplyVector(new Vector3(v.x,v.y,0)))).ToArray();float x0=Mathf.Max(clip.x,points.Min(p=>p.x)),x1=Mathf.Min(clip.z,points.Max(p=>p.x)),z0=Mathf.Max(clip.y,points.Min(p=>p.z)),z1=Mathf.Min(clip.w,points.Max(p=>p.z));if(x1>x0&&z1>z0)rects.Add(Rect.MinMaxRect(x0,z0,x1,z1));}var xs=rects.SelectMany(r=>new[]{r.xMin,r.xMax}).Distinct().OrderBy(x=>x).ToArray();double area=0;for(int i=1;i<xs.Length;i++){float lo=xs[i-1],hi=xs[i],mid=(lo+hi)*.5f;var spans=rects.Where(r=>r.xMin<=mid&&r.xMax>=mid).OrderBy(r=>r.yMin).ToArray();float length=0,end=float.NegativeInfinity;foreach(var r in spans){if(r.yMin>end){length+=r.height;end=r.yMax;}else if(r.yMax>end){length+=r.yMax-end;end=r.yMax;}}area+=(hi-lo)*length;}bool pass=fullQuads&&Math.Abs(area-3072)<.01&&Mathf.Approximately(mat.GetFloat("_BaseCoverage"),1);ctx.ReturnValue=new{passed=pass?1:0,failed=pass?0:1,baseQuadCount=rects.Count,fullNativeSpriteQuads=fullQuads,effectiveClippedUnionArea=area,oldMeasuredArea=1024,ratio=area/1024.0,method="Native Sprite.vertices + Tilemap cell center/orientation/world transform; rectangle XZ union clipped by actual own shader bounds. Full base alpha is shader coverage=1."};}}
