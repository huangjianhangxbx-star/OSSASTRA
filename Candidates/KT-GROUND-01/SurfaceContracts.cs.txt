/*
 * 候选验证：地表范围、世界面积与图层归属。
 * 用途：由唯一 Unity writer 通过 execute_code 执行；正式类型缺失时明确返回 RED。
 * 边界：只创建并关闭独立 PreviewScene，不保存资产，不修改现有场景或角色。
 * 失败意图：捕获遗漏 Grid swizzle/TRS、硬编码面积、范围越界及借用外部 Tilemap 的错误。
 */
using System;
using System.Collections.Generic;
using System.Reflection;
using Funplay.Editor.Tools.Scripting;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.Tilemaps;

public sealed class GroundSurfaceContracts : IFunplayCommand
{
    public void Execute(ExecutionContext ctx)
    {
        Type surfaceType = FindType("Ossastra.GroundResearch.GroundResearchSurface");
        Type layerType = FindType("Ossastra.GroundResearch.GroundLayer");
        if (surfaceType == null || layerType == null)
        {
            ctx.ReturnValue = new
            {
                status = "RED",
                reason = "Missing production GroundResearchSurface/GroundLayer; world geometry and ownership contracts are not implemented.",
                passed = 0,
                failed = 1
            };
            return;
        }

        var results = new List<string>();
        int passed = 0;
        int failed = 0;
        Scene previewScene = default(Scene);
        Tile paintTile = null;
        try
        {
            previewScene = EditorSceneManager.NewPreviewScene();
            var root = CreateObject("SurfaceContractsRoot", previewScene, null);
            var grid = root.AddComponent<Grid>();
            grid.cellSwizzle = GridLayout.CellSwizzle.XZY;
            grid.cellSize = Vector3.one;
            grid.cellGap = Vector3.zero;
            root.transform.position = new Vector3(10f, 3f, -4f);
            root.transform.rotation = Quaternion.Euler(0f, 90f, 0f);
            root.transform.localScale = new Vector3(2f, 1f, 3f);
            var surface = root.AddComponent(surfaceType);
            var map = CreateObject("OwnedTilemap", previewScene, root.transform).AddComponent<Tilemap>();
            var unlisted = CreateObject("UnlistedTilemap", previewScene, root.transform).AddComponent<Tilemap>();
            var foreign = CreateObject("ForeignTilemap", previewScene, null).AddComponent<Tilemap>();
            var nestedGrid = CreateObject("OtherGrid", previewScene, root.transform).AddComponent<Grid>();
            var nestedMap = CreateObject("OtherGridTilemap", previewScene, nestedGrid.transform).AddComponent<Tilemap>();
            paintTile = ScriptableObject.CreateInstance<Tile>();
            object layer = Activator.CreateInstance(layerType);
            Set(layer, "displayName", "合同样本");
            Set(layer, "tilemap", map);
            Set(layer, "paintTile", paintTile);
            Array layers = Array.CreateInstance(layerType, 1);
            layers.SetValue(layer, 0);
            Set(surface, "grid", grid);
            Set(surface, "layers", layers);
            BoundsInt originalCells = new BoundsInt(-2, -1, 0, 4, 3, 1);
            Set(surface, "editableCells", originalCells);

            Check(Contains(surface, new Vector3Int(-2, -1, 0)), "minimum XY cell is editable", results, ref passed, ref failed);
            Check(Contains(surface, new Vector3Int(1, 1, 0)), "last XY cell is editable", results, ref passed, ref failed);
            Check(!Contains(surface, new Vector3Int(-3, 0, 0)), "cell below xMin is rejected", results, ref passed, ref failed);
            Check(!Contains(surface, new Vector3Int(2, 0, 0)), "exclusive xMax is rejected", results, ref passed, ref failed);
            Check(!Contains(surface, new Vector3Int(0, -2, 0)), "cell below yMin is rejected", results, ref passed, ref failed);
            Check(!Contains(surface, new Vector3Int(0, 2, 0)), "exclusive yMax is rejected", results, ref passed, ref failed);
            Check(!Contains(surface, new Vector3Int(0, 0, -1)) && !Contains(surface, new Vector3Int(0, 0, 1)), "logical cell z must remain zero", results, ref passed, ref failed);
            Check(Owns(surface, map), "registered descendant Tilemap is owned", results, ref passed, ref failed);
            Check(!Owns(surface, unlisted), "unregistered descendant Tilemap is rejected", results, ref passed, ref failed);
            Check(!Owns(surface, null), "null Tilemap is rejected", results, ref passed, ref failed);

            // 手算：逻辑宽4×世界步长2，逻辑高3×世界步长3；旋转只交换世界轴，面积=72。
            Check(Near(Area(surface), 72f), "XZY grid and nonuniform world scale derive area 72", results, ref passed, ref failed);
            Bounds worldBounds = (Bounds)Get(surface, "WorldBounds");
            Check(Vector3.Distance(worldBounds.center, new Vector3(11.5f, 3f, -4f)) < 0.001f &&
                  Vector3.Distance(worldBounds.size, new Vector3(9f, 0f, 8f)) < 0.001f,
                "rotated grid world bounds match four hand-checked corners", results, ref passed, ref failed);
            Set(surface, "editableCells", new BoundsInt(-2, -1, 0, 8, 3, 1));
            Check(Near(Area(surface), 144f), "changing editable width changes derived area", results, ref passed, ref failed);
            Set(surface, "editableCells", originalCells);
            grid.cellSize = Vector3.one * 0.5f;
            Check(Near(Area(surface), 18f), "changing Grid cell size changes derived area", results, ref passed, ref failed);
            grid.cellSize = Vector3.one;
            root.transform.rotation = Quaternion.Euler(60f, 90f, 0f);
            Check(Near(Area(surface), 36f), "tilted grid reports XZ footprint rather than sloped area", results, ref passed, ref failed);
            root.transform.rotation = Quaternion.Euler(0f, 90f, 0f);

            Set(layer, "tilemap", foreign);
            Check(!Owns(surface, foreign), "registered but foreign Tilemap is rejected", results, ref passed, ref failed);
            Set(layer, "tilemap", nestedMap);
            Check(!Owns(surface, nestedMap), "Tilemap on another nested Grid is rejected", results, ref passed, ref failed);
            Set(layer, "tilemap", map);
            Set(surface, "layers", Array.CreateInstance(layerType, 0));
            Check(!Contains(surface, Vector3Int.zero) && !Owns(surface, map) && Near(Area(surface), 0f),
                "empty layer configuration exposes no editable surface", results, ref passed, ref failed);
            Set(surface, "layers", layers);
            Set(surface, "grid", null);
            Check(!Contains(surface, Vector3Int.zero) && !Owns(surface, map) && Near(Area(surface), 0f),
                "missing Grid exposes no editable surface", results, ref passed, ref failed);
            Set(surface, "grid", grid);
            Set(surface, "editableCells", new BoundsInt(0, 0, 0, 0, 3, 1));
            Check(!Contains(surface, Vector3Int.zero) && !Owns(surface, map) && Near(Area(surface), 0f),
                "zero-width authoring range is rejected", results, ref passed, ref failed);
            Set(surface, "editableCells", new BoundsInt(0, 0, 1, 4, 3, 1));
            Check(!Contains(surface, new Vector3Int(0, 0, 1)) && Near(Area(surface), 0f),
                "nonzero logical z authoring range is rejected", results, ref passed, ref failed);
            Set(surface, "editableCells", new BoundsInt(0, 0, 0, 4, 3, 2));
            Check(!Contains(surface, Vector3Int.zero) && Near(Area(surface), 0f),
                "multiple logical z layers are rejected", results, ref passed, ref failed);
        }
        catch (Exception exception)
        {
            failed++;
            results.Add("ERROR fixture/API: " + exception.GetBaseException().Message);
        }
        finally
        {
            if (paintTile != null) UnityEngine.Object.DestroyImmediate(paintTile);
            if (previewScene.IsValid()) EditorSceneManager.ClosePreviewScene(previewScene);
        }
        ctx.ReturnValue = new { status = failed == 0 ? "GREEN" : "RED", passed, failed, checks = results.ToArray() };
    }

    private static Type FindType(string fullName)
    {
        foreach (Assembly assembly in AppDomain.CurrentDomain.GetAssemblies())
        {
            Type type = assembly.GetType(fullName, false);
            if (type != null) return type;
        }
        return null;
    }

    private static GameObject CreateObject(string name, Scene scene, Transform parent)
    {
        var instance = new GameObject(name);
        SceneManager.MoveGameObjectToScene(instance, scene);
        if (parent != null) instance.transform.SetParent(parent, false);
        return instance;
    }

    private static void Set(object target, string name, object value)
    {
        FieldInfo field = target.GetType().GetField(name, BindingFlags.Public | BindingFlags.Instance);
        if (field == null) throw new InvalidOperationException("Missing public field " + target.GetType().Name + "." + name);
        field.SetValue(target, value);
    }

    private static object Get(object target, string name)
    {
        PropertyInfo property = target.GetType().GetProperty(name, BindingFlags.Public | BindingFlags.Instance);
        if (property == null) throw new InvalidOperationException("Missing property " + target.GetType().Name + "." + name);
        return property.GetValue(target);
    }

    private static bool Contains(object surface, Vector3Int cell)
    {
        return (bool)surface.GetType().GetMethod("ContainsCell").Invoke(surface, new object[] { cell });
    }

    private static bool Owns(object surface, Tilemap map)
    {
        return (bool)surface.GetType().GetMethod("Owns").Invoke(surface, new object[] { map });
    }

    private static float Area(object surface) { return Convert.ToSingle(Get(surface, "WorldArea")); }
    private static bool Near(float actual, float expected) { return Mathf.Abs(actual - expected) < 0.001f; }

    private static void Check(bool success, string name, List<string> results, ref int passed, ref int failed)
    {
        results.Add((success ? "PASS " : "FAIL ") + name);
        if (success) passed++; else failed++;
    }
}
