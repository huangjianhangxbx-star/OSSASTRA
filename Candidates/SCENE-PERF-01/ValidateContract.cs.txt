/* 功能：在独立临时RoomPositivePass上调用真实Validate(out error)，固定局部材质扫描去重前后的接管语义。
 * 边界：只从主相机Pass读取frame/feature/Shader，frame浅复制后仅替换自有副本PositiveSurfaces；不调用live Validate、不提交GPU。
 * 生命周期：同步八边界；finally销毁临时Pass/Camera/Renderer/Mesh/材质，守护主帧字段、活动场与dirty状态。
 * 证据：真实Validate返回值和原错误消息；不构造性能失败、不冒称已减少调用次数或提升整帧。
 */
using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using Ossastra.MapResearch;
using Ossastra.SceneVisibility.Rooms.Rendering;
using Funplay.Editor.Tools.Scripting;
using Object=UnityEngine.Object;

public sealed class ValidateContract : IFunplayCommand
{
    const BindingFlags Members=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;
    const string MainScene="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity";
    public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext context)
    {
        var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();
        if(!Application.isPlaying||scene.path!=MainScene||Application.dataPath.Replace('\\','/')!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new InvalidOperationException("Actual isolated ANKALUTE main Play required");
        var map=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<MapResearchRoot>(true)).Single();
        var mainCamera=map.gameplayCamera;var data=mainCamera==null?null:mainCamera.GetComponent<UniversalAdditionalCameraData>();
        var features=Read(data==null?null:data.scriptableRenderer,"rendererFeatures") as IEnumerable;
        if(features==null)throw new InvalidOperationException("Actual gameplay renderer features unavailable");
        var owners=features.Cast<object>().Where(o=>o!=null&&o.GetType().Name=="RoomMaskRendererFeature").ToArray();
        if(owners.Length!=1)throw new InvalidOperationException("Exactly one actual RoomMaskRendererFeature required");
        var registry=Read(owners[0],"positiveCameras") as IDictionary;
        if(registry==null||!registry.Contains(mainCamera.GetInstanceID()))throw new InvalidOperationException("Actual main camera pass not ready");
        object live=registry[mainCamera.GetInstanceID()];Type passType=live.GetType();
        var liveFrame=Read(live,"frame") as RoomMaskFrame;object feature=Read(live,"feature");var truth=Read(live,"truthShader") as Shader;
        var ctor=passType.GetConstructor(Members,null,new[]{typeof(Camera)},null);var validate=passType.GetMethod("Validate",Members,null,new[]{typeof(string).MakeByRefType()},null);
        if(liveFrame==null||feature==null||truth==null||!truth.isSupported||ctor==null||validate==null)throw new InvalidOperationException("Actual frame/feature/Shader/Validate seam missing");
        if(!liveFrame.PositiveRenderingEnabled||!liveFrame.Enabled)throw new InvalidOperationException("Live positive frame must be enabled before read-only fixture");
        Shader lit=Shader.Find("Universal Render Pipeline/Lit"),unlit=Shader.Find("Universal Render Pipeline/Unlit"),unknown=Shader.Find("UI/Default");
        if(unknown==null||!unknown.isSupported)unknown=Shader.Find("Hidden/InternalErrorShader");
        if(lit==null||unlit==null||unknown==null||!lit.isSupported||!unlit.isSupported||!unknown.isSupported)throw new InvalidOperationException("Fixture shaders unavailable: Lit="+(lit==null?"missing":lit.name+"/"+lit.isSupported)+"; Unlit="+(unlit==null?"missing":unlit.name+"/"+unlit.isSupported)+"; unknown="+(unknown==null?"missing":unknown.name+"/"+unknown.isSupported));
        var publicFields=typeof(RoomMaskFrame).GetFields(BindingFlags.Public|BindingFlags.Instance);object[] before=publicFields.Select(f=>f.GetValue(liveFrame)).ToArray();
        var cloneMethod=typeof(object).GetMethod("MemberwiseClone",BindingFlags.Instance|BindingFlags.NonPublic);
        var copy=(RoomMaskFrame)cloneMethod.Invoke(liveFrame,null);copy.TruthMode=false;
        string path=Path.GetFullPath(Path.Combine(Application.dataPath,"..","artifacts","SCENE-PERF-01","validate-contract-"+DateTime.UtcNow.ToString("yyyyMMdd-HHmmss-fffffff")+".json"));
        var checks=new List<Check>();var observations=new List<Observation>();var errors=new List<string>();
        var report=new Report{scenePath=scene.path,sceneDirtyBefore=scene.isDirty,liveCameraId=mainCamera.GetInstanceID()};
        GameObject cameraObject=null,meshObject=null;Mesh mesh=null;Material a=null,b=null,c=null;object pass=null;
        try
        {
            cameraObject=new GameObject("SCENE-PERF-01 Validate temporary Camera"){hideFlags=HideFlags.HideAndDontSave};cameraObject.transform.position=new Vector3(10000,10000,10000);
            var camera=cameraObject.AddComponent<Camera>();camera.enabled=false;camera.cullingMask=0;
            meshObject=new GameObject("SCENE-PERF-01 Validate temporary Mesh"){hideFlags=HideFlags.HideAndDontSave};meshObject.transform.position=new Vector3(10000,10000,10010);
            var filter=meshObject.AddComponent<MeshFilter>();var renderer=meshObject.AddComponent<MeshRenderer>();
            renderer.shadowCastingMode=ShadowCastingMode.Off;renderer.receiveShadows=false;
            mesh=new Mesh{name="Validate temporary triangle",hideFlags=HideFlags.HideAndDontSave};mesh.vertices=new[]{Vector3.zero,Vector3.right,Vector3.up};mesh.triangles=new[]{0,1,2};mesh.RecalculateBounds();filter.sharedMesh=mesh;
            a=new Material(lit){name="Validate Lit",hideFlags=HideFlags.HideAndDontSave};b=new Material(unlit){name="Validate Unlit",hideFlags=HideFlags.HideAndDontSave};c=new Material(unknown){name="Validate unsupported",hideFlags=HideFlags.HideAndDontSave};
            a.SetFloat("_Surface",0);b.SetFloat("_Surface",0);renderer.sharedMaterial=a;
            pass=ctor.Invoke(new object[]{camera});passType.GetField("frame",Members).SetValue(pass,copy);passType.GetField("feature",Members).SetValue(pass,feature);passType.GetField("truthShader",Members).SetValue(pass,truth);
            var surface=new RoomPositiveSurface{Renderer=renderer,Kind=RoomPositiveSurfaceKind.Partial,StaticEnvironment=true,DrawMask=RoomPositiveRenderingLayers.PartialLitBit};
            Action<string,bool,string> sample=(name,expected,messagePart)=>
            {
                copy.PositiveSurfaces=new[]{surface};object[] args={null};bool actual=(bool)validate.Invoke(pass,args);string error=args[0] as string??"";
                observations.Add(new Observation{name=name,expected=expected,actual=actual,error=error});
                checks.Add(new Check{name=name,pass=actual==expected&&(expected?error.Length==0:error.Contains(messagePart))});
            };
            sample("static opaque Lit valid",true,"");
            renderer.sharedMaterials=new[]{a,b};sample("mixed Lit and Unlit rejected",false,"未支持或混合");
            renderer.sharedMaterials=new[]{c};sample("unknown shader family rejected",false,"未支持或混合");
            renderer.sharedMaterials=Enumerable.Repeat(a,17).ToArray();sample("seventeen material slots rejected",false,"16");
            renderer.sharedMaterials=new[]{b};surface.Kind=RoomPositiveSurfaceKind.Roof;surface.DrawMask=RoomPositiveRenderingLayers.RoofBit;sample("non-Lit roof rejected",false,"顶盖");
            renderer.sharedMaterials=new[]{a};surface.Kind=RoomPositiveSurfaceKind.Partial;surface.DrawMask=RoomPositiveRenderingLayers.PartialLitBit;
            surface.OwnerVolumes=new[]{new RoomMaskVolume{WorldToLocal=Matrix4x4.identity,HalfExtents=new Vector3(-1,1,1)}};sample("invalid owner volume rejected",false,"所属空间");
            surface.OwnerVolumes=null;filter.sharedMesh=null;renderer.sharedMaterials=new[]{c};sample("empty Mesh Renderer with unknown material allowed",true,"");
            surface.Renderer=null;sample("null Renderer entry allowed",true,"");
        }
        catch(Exception failure){errors.Add((failure is TargetInvocationException&&failure.InnerException!=null?failure.InnerException:failure).ToString());}
        finally
        {
            Cleanup(()=>{if(pass is IDisposable d)d.Dispose();},errors,"private Pass");
            Cleanup(()=>{if(cameraObject!=null)Object.DestroyImmediate(cameraObject);},errors,"Camera GO");Cleanup(()=>{if(meshObject!=null)Object.DestroyImmediate(meshObject);},errors,"Renderer GO");
            Cleanup(()=>{if(mesh!=null)Object.DestroyImmediate(mesh);},errors,"Mesh");Cleanup(()=>{if(a!=null)Object.DestroyImmediate(a);},errors,"Lit");Cleanup(()=>{if(b!=null)Object.DestroyImmediate(b);},errors,"Unlit");Cleanup(()=>{if(c!=null)Object.DestroyImmediate(c);},errors,"unknown");
            report.cleanupComplete=cameraObject==null&&meshObject==null&&mesh==null&&a==null&&b==null&&c==null;
            checks.Add(new Check{name="all temporary resources released",pass=report.cleanupComplete});
            bool unchanged=ReferenceEquals(Read(live,"frame"),liveFrame)&&ReferenceEquals(Read(live,"feature"),feature)&&ReferenceEquals(registry[mainCamera.GetInstanceID()],live)&&publicFields.Select((f,i)=>Equals(f.GetValue(liveFrame),before[i])).All(v=>v);
            checks.Add(new Check{name="live frame fields, feature and registry entry unchanged",pass=unchanged});
            report.sceneDirtyAfter=scene.IsValid()&&scene.isDirty;checks.Add(new Check{name="active scene identity and author dirty state preserved",pass=scene.IsValid()&&scene==UnityEngine.SceneManagement.SceneManager.GetActiveScene()&&scene.isDirty==report.sceneDirtyBefore});
        }
        report.checks=checks.ToArray();report.observations=observations.ToArray();report.errors=errors.ToArray();report.red=checks.Any(v=>!v.pass);report.pass=!report.red&&errors.Count==0&&observations.Count==8&&report.cleanupComplete;report.status=errors.Count>0?"invalid":report.pass?"GREEN":"RED";
        Directory.CreateDirectory(Path.GetDirectoryName(path));File.WriteAllText(path,Newtonsoft.Json.JsonConvert.SerializeObject(report,Newtonsoft.Json.Formatting.Indented));
        context.ReturnValue=new{path,report.status,report.pass,report.red,report.cleanupComplete,report.checks,report.observations,report.errors,report.scope};
    }
    static object Read(object target,string name)
    {
        for(var t=target==null?null:target.GetType();t!=null;t=t.BaseType)
        {var f=t.GetField(name,Members|BindingFlags.DeclaredOnly);if(f!=null)return f.GetValue(target);var p=t.GetProperty(name,Members|BindingFlags.DeclaredOnly);if(p!=null&&p.GetIndexParameters().Length==0)return p.GetValue(target,null);}return null;
    }
    static void Cleanup(Action action,List<string> errors,string label){try{action();}catch(Exception e){errors.Add("cleanup "+label+": "+e);}}
    public sealed class Check{public string name;public bool pass;}
    public sealed class Observation{public string name,error;public bool expected,actual;}
    public sealed class Report
    {
        public int schemaVersion=1,liveCameraId;public string scenePath,status;public bool sceneDirtyBefore,sceneDirtyAfter,cleanupComplete,red,pass;public Check[] checks;public Observation[] observations;public string[] errors;
        public string scope="Actual private Validate(out error) on an independent temporary Pass with a shallow copy of current main frame and its feature. Only copy.PositiveSurfaces/TruthMode differ. No live Validate, GPU, authority mutation, timing threshold or inferred call-count reduction.";
    }
}
