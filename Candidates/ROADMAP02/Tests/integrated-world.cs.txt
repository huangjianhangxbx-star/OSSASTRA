using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Enemies;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class IntegratedWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");
var service=UnityEngine.Object.FindObjectOfType<PartyRespawnCoordinator>();var debug=UnityEngine.Object.FindObjectOfType<CombatDeathDebugView>();var d=UnityEngine.Object.FindObjectOfType<CompanionFollowDirector>();var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(service==null||debug==null||d==null||e==null)throw new Exception("Missing world");
var c=d.Coordinator;var a=c.Characters[0];var b=c.Characters[1];var roots=new[]{a,b};var inputs=roots.Select(x=>x.GetComponent<CharacterInputHandler>()).ToArray();var healths=roots.Select(x=>x.Vitals.Health).ToArray();var initialGeneration=a.LifeGeneration;
var originalSettings=InputSystem.settings;var testSettings=UnityEngine.Object.Instantiate(originalSettings);testSettings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;testSettings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=testSettings;var keyboard=InputSystem.AddDevice<Keyboard>("M02B1Keyboard");var mouse=InputSystem.AddDevice<Mouse>("M02B1Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in inputs.SelectMany(i=>new[]{i.movementAction,i.mouseAction,i.attackAction,i.cancelAction,i.pauseAction}).Concat(new[]{c.switchAction})){overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var path=action.bindings[i].path;if(path==null||action.bindings[i].isComposite)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(i,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(i,path.Replace("<Mouse>",mouse.path));}}
var game=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");if(game!=null)EditorWindow.GetWindow(game).Focus();
var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();var records=new List<object>();var steps=new List<(double at,Action run)>();int next=0,failed=0;var start=EditorApplication.timeSinceStartup;int initialA=0,initialB=0,generation=0;bool died=false;double diedAt=0;bool warnSeen=false;
void Flush(string state){SessionState.SetString("ROADMAP_S8_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,records},Newtonsoft.Json.Formatting.Indented));}
void Check(string label,bool pass){if(!pass)failed++;records.Add(new{label,pass,hpE=e.Vitals.Health.Current,hpA=a.Vitals.Health.Current,hpB=b.Vitals.Health.Current,current=c.Current==null?"none":c.Current.ActorId,a=a.ReleaseCount,b=b.ReleaseCount,stage=a.AttackStage,gen=a.LifeGeneration,time=service.SimulationTime});Flush("running");}
void Keys(params Key[] keys){InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));}
void Mouse(bool held){var p=Camera.main.WorldToScreenPoint(e.transform.position);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(p.x,p.y),buttons=(ushort)(held?1:0)});}
void Add(double at,Action run){steps.Add((at,run));}
void Reset(){debug.RestartForDevelopment();e.AttacksEnabled=true;foreach(var r in roots){r.SendMessage("OnApplicationFocus",true);r.SetPaused(false);}c.SendMessage("OnApplicationFocus",true);e.SendMessage("OnApplicationFocus",true);service.SendMessage("OnApplicationFocus",true);a.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.left*1.15f);b.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.back*2.1f);Keys();Mouse(false);Physics.SyncTransforms();}
Reset();initialA=a.ReleaseCount;initialB=b.ReleaseCount;
Add(.3,()=>Mouse(true));Add(.5,()=>Mouse(false));Add(1.01,()=>Mouse(true));Add(1.35,()=>Mouse(false));
Add(1.85,()=>{Check("natural-two-stage-releases",a.ReleaseCount==initialA+2&&a.AttackStage==2);Check("natural-partner-joins",b.ReleaseCount>initialB);});
Add(8.8,()=>{Check("natural-enemy-killed",!e.Vitals.IsAlive&&!e.ActionActive);Check("safe-follow-resumes",!engagement.Engaged);});
Add(9,()=>{Reset();d.enabled=false;foreach(var r in roots)r.GetComponent<CompanionCombatBrain>().enabled=false;b.GetComponent<CharacterMovement>().Teleport(new Vector3(-4,.03f,-3));generation=a.LifeGeneration;});
Add(21,()=>{Check("natural-enemy-attacks-kill-player",died);Check("death-auto-handover",c.Current==b);Check("warning-observed",warnSeen);});
Add(32,()=>{Check("timed-respawn-real-life",a.IsAlive&&a.LifeGeneration==generation+1);Check("respawn-keeps-enemy-hp",e.Vitals.Health.Current==72);Check("respawn-keeps-current",c.Current==b);Check("no-stale-action",!a.ActionActive);Check("npc-still-independent",UnityEngine.Object.FindObjectOfType<Ossastra.Npcs.NpcRoot>().GetComponent<CombatVitals>()==null);});
EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;e.AttacksEnabled=false;foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=originalSettings;UnityEngine.Object.DestroyImmediate(testSettings);}
tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");var elapsed=EditorApplication.timeSinceStartup-start;if(e.WarningVisible)warnSeen=true;if(elapsed>9&&!a.IsAlive&&!died){died=true;diedAt=elapsed;e.AttacksEnabled=false;}while(next<steps.Count&&elapsed>=steps[next].at)steps[next++].run();if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;records.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.Log("S8 natural 32 seconds, no development damage");}}
