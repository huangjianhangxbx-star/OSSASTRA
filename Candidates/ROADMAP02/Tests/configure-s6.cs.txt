using System;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class AddWarning:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(Application.isPlaying||s.isDirty||s.path!="Assets/OSSASTRA/ROADMAP02/SC_ROADMAP02_TeamCombat.unity")throw new Exception("Wrong scene");var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(e==null)throw new Exception("No enemy");ctx.RegisterObjectModification(e.gameObject);if(e.GetComponent<Ossastra.Enemies.EnemyAttackWarning>()==null)e.gameObject.AddComponent<Ossastra.Enemies.EnemyAttackWarning>();EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=s.path;}}
