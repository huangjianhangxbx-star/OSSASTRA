using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Enemies;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class WarningWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");
var service=UnityEngine.Object.FindObjectOfType<PartyRespawnCoordinator>();var debug=UnityEngine.Object.FindObjectOfType<CombatDeathDebugView>();var d=UnityEngine.Object.FindObjectOfType<CompanionFollowDirector>();var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(service==null||debug==null||d==null||e==null)throw new Exception("Missing world");
var c=d.Coordinator;var a=c.Characters[0];var b=c.Characters[1];var roots=new[]{a,b};var inputs=roots.Select(x=>x.GetComponent<CharacterInputHandler>()).ToArray();var healths=roots.Select(x=>x.Vitals.Health).ToArray();var initialGeneration=a.LifeGeneration;
var originalSettings=InputSystem.settings;var testSettings=UnityEngine.Object.Instantiate(originalSettings);testSettings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;testSettings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=testSettings;var keyboard=InputSystem.AddDevice<Keyboard>("M02B1Keyboard");var mouse=InputSystem.AddDevice<Mouse>("M02B1Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in inputs.SelectMany(i=>new[]{i.movementAction,i.mouseAction,i.attackAction,i.cancelAction,i.pauseAction}).Concat(new[]{c.switchAction})){overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var path=action.bindings[i].path;if(path==null||action.bindings[i].isComposite)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(i,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(i,path.Replace("<Mouse>",mouse.path));}}
var game=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");if(game!=null)EditorWindow.GetWindow(game).Focus();
var warning=e.GetComponent<EnemyAttackWarning>();var records=new List<object>();int failed=0,phase=0,accepts=0,releases=0;bool seen=false;var start=EditorApplication.timeSinceStartup;GameObject wall=null;
void Flush(string state){SessionState.SetString("ROADMAP_S6_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,records},Newtonsoft.Json.Formatting.Indented));}
void Check(string label,bool pass){if(!pass)failed++;records.Add(new{label,pass,hp=a.Vitals.Health.Current,warning=e.WarningVisible,visible=warning.Visible,last=e.LastFact,hit=e.LastHit});Flush("running");}
void Fact(ActionFact f){if(f.Kind==ActionFactKind.Accepted){accepts++;Check("accepted-tracking-"+phase,e.WarningVisible);if(phase==2){wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.transform.position=(a.transform.position+e.transform.position)*.5f+Vector3.up*.8f;wall.transform.localScale=new Vector3(.1f,1.6f,2);Physics.SyncTransforms();}}if(f.Kind==ActionFactKind.Released)releases++;}
e.ActionObserved+=Fact;
void Reset(bool view){debug.RestartForDevelopment();d.enabled=false;foreach(var r in roots){r.GetComponent<CompanionCombatBrain>().enabled=false;r.SendMessage("OnApplicationFocus",true);}c.SendMessage("OnApplicationFocus",true);e.SendMessage("OnApplicationFocus",true);e.SetPaused(false);e.AttacksEnabled=true;warning.enabled=view;a.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.left*1.15f);b.GetComponent<CharacterMovement>().Teleport(new Vector3(-4,.03f,-3));Physics.SyncTransforms();accepts=releases=0;seen=false;}
Reset(true);
EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;e.ActionObserved-=Fact;e.AttacksEnabled=false;if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=originalSettings;UnityEngine.Object.DestroyImmediate(testSettings);}
tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");var t=EditorApplication.timeSinceStartup-start;if(warning.Visible)seen=true;
if(phase==0&&t>3){Check("visible-real-warning",seen);Check("one-release-real-hit",releases==1&&a.Vitals.Health.Current==80&&!e.WarningVisible);phase=1;Reset(false);}
else if(phase==1&&t>6){Check("visual-off-same-damage",releases==1&&a.Vitals.Health.Current==80&&!warning.Visible);phase=2;Reset(true);}
else if(phase==2&&t>9){Check("wall-cancel-no-hit",accepts>=1&&releases==0&&a.Vitals.Health.Current==100&&!e.WarningVisible&&!warning.Visible);if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);wall=null;phase=3;Reset(true);}
else if(phase==3&&e.WarningVisible){e.SetPaused(true);Check("pause-clears-snapshot",!e.WarningVisible);phase=4;}
else if(phase==4&&t>12){Check("pause-clears-renderer",!warning.Visible);e.enabled=false;Check("disable-clears",!e.WarningVisible);Check("reused-two-lines",e.GetComponentsInChildren<LineRenderer>(true).Length==2);Cleanup();Flush("completed");}
}catch(Exception ex){failed++;records.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.Log("S6 12 seconds");}}
