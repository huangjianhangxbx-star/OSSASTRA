using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Enemies;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class TeamWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");
var service=UnityEngine.Object.FindObjectOfType<PartyRespawnCoordinator>();var debug=UnityEngine.Object.FindObjectOfType<CombatDeathDebugView>();var d=UnityEngine.Object.FindObjectOfType<CompanionFollowDirector>();var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(service==null||debug==null||d==null||e==null)throw new Exception("Missing world");
var c=d.Coordinator;var a=c.Characters[0];var b=c.Characters[1];var roots=new[]{a,b};var inputs=roots.Select(x=>x.GetComponent<CharacterInputHandler>()).ToArray();var healths=roots.Select(x=>x.Vitals.Health).ToArray();var initialGeneration=a.LifeGeneration;
var originalSettings=InputSystem.settings;var testSettings=UnityEngine.Object.Instantiate(originalSettings);testSettings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;testSettings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=testSettings;var keyboard=InputSystem.AddDevice<Keyboard>("M02B1Keyboard");var mouse=InputSystem.AddDevice<Mouse>("M02B1Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in inputs.SelectMany(i=>new[]{i.movementAction,i.mouseAction,i.attackAction,i.cancelAction,i.pauseAction}).Concat(new[]{c.switchAction})){overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var path=action.bindings[i].path;if(path==null||action.bindings[i].isComposite)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(i,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(i,path.Replace("<Mouse>",mouse.path));}}
var game=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");if(game!=null)EditorWindow.GetWindow(game).Focus();
var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();var brainA=a.GetComponent<CompanionCombatBrain>();var brainB=b.GetComponent<CompanionCombatBrain>();
var records=new List<object>();var steps=new List<(double at,Action run)>();int next=0,failed=0;var start=EditorApplication.timeSinceStartup;int initialA=0,initialB=0;Vector3 initialPos=default;var facts=new List<string>();
void Fact(ActionFact f){if(f.Kind==ActionFactKind.Accepted)facts.Add(f.Source);}
a.ActionObserved+=Fact;b.ActionObserved+=Fact;
void Flush(string state){SessionState.SetString("ROADMAP_S5_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,records,facts},Newtonsoft.Json.Formatting.Indented));}
void Check(string label,bool pass){if(!pass)failed++;records.Add(new{label,pass,hpE=e.Vitals.Health.Current,hpA=a.Vitals.Health.Current,hpB=b.Vitals.Health.Current,engaged=engagement.Engaged,current=c.Current==null?"none":c.Current.ActorId,a=a.ReleaseCount,b=b.ReleaseCount,ai=brainB.Reason,target=brainB.TargetId,pos=b.transform.position.ToString(),last=b.LastFact});Flush("running");}
void Keys(params Key[] keys){InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));}
void Mouse(bool held){var p=Camera.main.WorldToScreenPoint(e.transform.position);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(p.x,p.y),buttons=(ushort)(held?1:0)});}
void Add(double at,Action run){steps.Add((at,run));}
debug.RestartForDevelopment();e.AttacksEnabled=false;foreach(var r in roots){r.SendMessage("OnApplicationFocus",true);r.SetPaused(false);}c.SendMessage("OnApplicationFocus",true);service.SendMessage("OnApplicationFocus",true);Keys();Mouse(false);
Add(.1,()=>{engagement.enabled=false;engagement.enabled=true;a.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.left*1.15f);b.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.back*2.1f);Physics.SyncTransforms();initialA=a.ReleaseCount;initialB=b.ReleaseCount;initialPos=b.transform.position;});
Add(.6,()=>{Check("safe-no-auto",!engagement.Engaged&&b.ReleaseCount==initialB);Mouse(true);});
Add(.78,()=>{Check("real-player-accepted-opens",engagement.Engaged&&a.ActionActive);Mouse(false);});
Add(1.8,()=>{Check("partner-real-approach",Vector3.Distance(initialPos,b.transform.position)>.2f);Check("partner-stage1-release-hit",b.ReleaseCount>initialB&&e.Vitals.Health.Current<60&&b.AttackStage==1);});
Add(2.2,()=>{Keys(Key.Tab);});Add(2.35,()=>{Keys();Check("switch-B-owns-input",c.Current==b&&brainB.Decision==CompanionCombatDecision.Inactive);initialB=b.ReleaseCount;});
Add(3.8,()=>{Check("old-current-A-autonomous",facts.Count(x=>x=="companion:Stage1")>=2);Check("no-auto-stage2",!facts.Contains("companion:Stage2")&&!b.HasBufferedAttack);Check("B-no-new-ai-request",b.ReleaseCount<=initialB+1);});
Add(4,()=>{a.SetPaused(true);b.SetPaused(true);initialA=a.ReleaseCount;initialB=b.ReleaseCount;});Add(4.8,()=>Check("pause-no-release",a.ReleaseCount==initialA&&b.ReleaseCount==initialB));
Add(5,()=>{a.SetPaused(false);b.SetPaused(false);engagement.enabled=false;initialA=a.ReleaseCount;});Add(6.4,()=>Check("session-disabled-no-new-ai",a.ReleaseCount<=initialA+1));
Add(6.6,()=>{brainA.enabled=false;brainB.enabled=false;Check("disabled-brain-neutral",brainA.Decision==CompanionCombatDecision.Inactive&&brainB.Decision==CompanionCombatDecision.Inactive);});
EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;a.ActionObserved-=Fact;b.ActionObserved-=Fact;foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=originalSettings;UnityEngine.Object.DestroyImmediate(testSettings);}
tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");while(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].at)steps[next++].run();if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;records.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.Log("Scheduled S5 real input world");}}
