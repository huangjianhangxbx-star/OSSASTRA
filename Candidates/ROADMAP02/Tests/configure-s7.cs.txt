using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Ossastra.AnimationCore;using Ossastra.AnimationPresentation;using Funplay.Editor.Tools.Scripting;
public class ConfigureVisual:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(Application.isPlaying||s.isDirty||s.path!="Assets/OSSASTRA/ROADMAP02/SC_ROADMAP02_TeamCombat.unity")throw new Exception("Clean new scene required");
var player=UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>().FirstOrDefault();var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();var npc=UnityEngine.Object.FindObjectOfType<Ossastra.Npcs.NpcRoot>();if(player==null||enemy==null||npc==null)throw new Exception("Missing roots");
var source=player.GetComponentInChildren<Spine.Unity.SkeletonAnimation>();if(source==null)throw new Exception("No visual");
void Maps(SerializedObject so){var a=so.FindProperty("catalog");var semantics=new[]{AnimationSemantic.Idle,AnimationSemantic.Move,AnimationSemantic.BasicAttack,AnimationSemantic.BasicAttackSecond,AnimationSemantic.HitReaction,AnimationSemantic.PostureBroken,AnimationSemantic.Death};var clips=new[]{"idle1_1","run","atk1","atk2","upper_hitted","dizzy_loop","death"};a.arraySize=clips.Length;for(int i=0;i<clips.Length;i++){if(source.skeletonDataAsset.GetSkeletonData(true).FindAnimation(clips[i])==null)throw new Exception("Missing clip "+clips[i]);var e=a.GetArrayElementAtIndex(i);e.FindPropertyRelative("semantic").enumValueIndex=(int)semantics[i];e.FindPropertyRelative("clip").stringValue=clips[i];e.FindPropertyRelative("loop").boolValue=i==0||i==1||i==5;e.FindPropertyRelative("mixDuration").floatValue=.08f;}so.ApplyModifiedPropertiesWithoutUndo();}
foreach(var p in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterSpineView>()){ctx.RegisterObjectModification(p);Maps(new SerializedObject(p));}
void Visual(GameObject root,bool combat){if(root.transform.Find("TEMP_2D_VISUAL")!=null)throw new Exception("Already configured");ctx.RegisterObjectModification(root);foreach(var renderer in root.GetComponentsInChildren<Renderer>())renderer.enabled=false;var go=UnityEngine.Object.Instantiate(source.transform.parent.gameObject,root.transform);go.name="TEMP_2D_VISUAL";ctx.RegisterObjectCreation(go);go.transform.localPosition=Vector3.zero;go.transform.localRotation=Quaternion.identity;go.transform.localScale=Vector3.one*(combat?.75f:.8f);foreach(var old in go.GetComponentsInChildren<Ossastra.Characters.CharacterSpineView>())ctx.DestroyObject(old);var skeleton=go.GetComponentInChildren<Spine.Unity.SkeletonAnimation>();var presenter=go.AddComponent<EntitySpinePresenter>();var ps=new SerializedObject(presenter);ps.FindProperty("skeleton").objectReferenceValue=skeleton;ps.ApplyModifiedPropertiesWithoutUndo();Maps(ps);MonoBehaviour view=combat?(MonoBehaviour)root.AddComponent<Ossastra.Enemies.EnemySpineView>():root.AddComponent<Ossastra.Npcs.NpcSpineView>();var vs=new SerializedObject(view);vs.FindProperty("presenter").objectReferenceValue=presenter;vs.ApplyModifiedPropertiesWithoutUndo();}
Visual(enemy.gameObject,true);Visual(npc.gameObject,false);EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=new{scene=s.path,source=AssetDatabase.GetAssetPath(source.skeletonDataAsset),formalEnemyAsset="BLOCKED: local existing character reused as TEMP only"};}}
