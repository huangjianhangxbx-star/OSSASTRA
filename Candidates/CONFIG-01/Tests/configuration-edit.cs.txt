/* 配置样板：真实A Mapping的mix暂改、保存回读、finally恢复；不触及骨架或运行结构。 */
using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEditor;using UnityEngine;using A=Ossastra.CharacterAnimation;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConfigurationEditSample:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var root=UnityEngine.Object.FindObjectsOfType<C.CharacterRoot>().Single(r=>r.ActorId=="temporary-a");var binding=Ossastra.Authoring.CharacterAnimationNavigation.Read(root);var path=AssetDatabase.GetAssetPath(binding.baseMap);var before=File.ReadAllBytes(path);var meta=File.ReadAllBytes(path+".meta");var entry=binding.baseMap.plansMelee.Single(e=>e.logicID==A.EAnimationLogicState.Attack_Normal_1);var old=entry.plan.clips[0].mixDuration;var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}
try{ctx.RegisterObjectModification(binding.baseMap);entry.plan.clips[0].mixDuration=.12f;EditorUtility.SetDirty(binding.baseMap);AssetDatabase.SaveAssetIfDirty(binding.baseMap);AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceUpdate);var read=AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(path);Check("target A mapping saved and reimported mix 0.12",Math.Abs(read.plansMelee.Single(e=>e.logicID==A.EAnimationLogicState.Attack_Normal_1).plan.clips[0].mixDuration-.12f)<.00001f);Check("B mapping remains independent",Math.Abs(Ossastra.Authoring.CharacterAnimationNavigation.Read(UnityEngine.Object.FindObjectsOfType<C.CharacterRoot>().Single(r=>r.ActorId=="temporary-b")).baseMap.GetDefault(A.EAnimationLogicState.Attack_Normal_1).clips[0].mixDuration-old)<.00001f);}
finally{var read=AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(path);read.plansMelee.Single(e=>e.logicID==A.EAnimationLogicState.Attack_Normal_1).plan.clips[0].mixDuration=old;EditorUtility.SetDirty(read);AssetDatabase.SaveAssetIfDirty(read);AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceUpdate);Check("mapping and meta bytes restored exactly",before.SequenceEqual(File.ReadAllBytes(path))&&meta.SequenceEqual(File.ReadAllBytes(path+".meta")));}
var prefab=PrefabUtility.LoadPrefabContents("Assets/OSSASTRA/Prefabs/Characters/PlayerCharacter.prefab");try{var pr=prefab.GetComponent<C.CharacterRoot>();var current=Ossastra.Authoring.CharacterAnimationNavigation.Read(pr);Check("prefab default mapping differs from scene A override",current.baseMap!=null&&current.baseMap!=binding.baseMap);var so=new SerializedObject(pr);so.FindProperty("view").objectReferenceValue=null;so.ApplyModifiedPropertiesWithoutUndo();var missing=Ossastra.Authoring.CharacterAnimationNavigation.Read(pr);Check("missing view diagnosed without guessed child binding",missing.controller==null&&missing.baseMap==null&&!string.IsNullOrEmpty(missing.problem));}finally{PrefabUtility.UnloadPrefabContents(prefab);}ctx.ReturnValue=new{failed,path,oldMix=old,newMix=.12,restoredMix=binding.baseMap.GetDefault(A.EAnimationLogicState.Attack_Normal_1).clips[0].mixDuration,rows};}}
