/* 验证角色根的真实绑定导航，防止把通用Prefab默认值误当成场景覆盖或绘制时修改资产。 */
using System;using System.Linq;using System.Reflection;using System.Collections.Generic;using UnityEngine;using UnityEditor;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConfigurationNavigationProbe:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}var type=AppDomain.CurrentDomain.GetAssemblies().Select(a=>a.GetType("Ossastra.Authoring.CharacterAnimationNavigation")).FirstOrDefault(t=>t!=null);Check("root animation navigation is available",type!=null);if(type!=null){var beforeSelection=Selection.objects;try{foreach(var id in new[]{"temporary-a","temporary-b"}){var role=UnityEngine.Object.FindObjectsOfType<C.CharacterRoot>().Single(r=>r.ActorId==id);var editor=Editor.CreateEditor(role);Check(id+" root uses navigation Inspector",editor.GetType().Name=="CharacterRootEditor");UnityEngine.Object.DestroyImmediate(editor);var read=type.GetMethod("Read").Invoke(null,new object[]{role});var map=(UnityEngine.Object)read.GetType().GetField("baseMap").GetValue(read);Check(id+" resolves actual scene map",AssetDatabase.GetAssetPath(map)=="Assets/OSSASTRA/Configs/Animation/"+id+"_VC02_Animation_Mapping.asset");var targetType=type.Assembly.GetType("Ossastra.Authoring.CharacterAnimationNavigationTarget");foreach(var command in new[]{"Body","BaseMap","Router","Rig"}){var target=Enum.Parse(targetType,command);var result=(bool)type.GetMethod("Navigate").Invoke(null,new object[]{role,target});Check(id+" "+command+" navigates to nonempty real target",result&&Selection.activeObject==(UnityEngine.Object)type.GetMethod("Target").Invoke(null,new object[]{read,target}));}}}finally{Selection.objects=beforeSelection;}}ctx.ReturnValue=new{failed,rows};}}
