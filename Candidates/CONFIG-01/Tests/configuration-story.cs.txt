/* 正常输入小样板：虚拟InputSystem键鼠/手柄→正式消费者；uGUI拖拽→库存事务。非实体手柄体验。 */
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.EventSystems;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using I=Ossastra.Inventory;using U=Ossastra.UserInterface;using Funplay.Editor.Tools.Scripting;
public class ConfigurationStory:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var control=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var actor=control.Current;var input=actor.GetComponent<C.CharacterInputHandler>();var main=actor.GetComponent<C.CharacterMainAction>();var stamina=actor.GetComponent<C.CharacterStamina>();var view=actor.GetComponentInChildren<A.CharacterAnimationController>();var config=view.GetComponent<A.CharacterAnimationConfigManager>();var composer=view.GetComponent<A.CharacterAnimationComposer>();var abilities=actor.GetComponent<C.CharacterAbilities>();var gear=actor.GetComponent<C.CharacterEquipment>();var equipmentView=actor.GetComponent<C.CharacterEquipmentSpineView>();var session=UnityEngine.Object.FindObjectOfType<I.InventorySession>();var ui=UnityEngine.Object.FindObjectOfType<U.InventoryView>();
// 隔离威胁只用于稳定观察配置；不授予物品/技能，不替代正式输入或费用事务。
foreach(var r in control.Characters){r.CancelAction("configuration-story");r.GetComponent<C.CompanionCombatBrain>().enabled=false;r.GetComponent<C.CharacterFollow>().enabled=false;r.SendMessage("OnApplicationFocus",true);}control.SendMessage("OnApplicationFocus",true);foreach(var e in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){e.AttacksEnabled=false;e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;}UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>().enabled=false;
var settingsBefore=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(settingsBefore);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;var pad=InputSystem.AddDevice<Gamepad>("ConfigSamplePad");var keyboard=InputSystem.AddDevice<Keyboard>("ConfigSampleKeys");var overrides=new Dictionary<InputAction,string>();foreach(var action in new[]{input.rightHandAction,input.leftHandAction,input.weaponModeAction,input.gamepadAimAction,input.skillOneAction,ui.toggleAction}){overrides[action]=action.SaveBindingOverridesAsJson();for(int n=0;n<action.bindings.Count;n++){var b=action.bindings[n];if(!b.isComposite&&b.path!=null)action.ApplyBindingOverride(n,b.path.Replace("<Gamepad>",pad.path).Replace("<Keyboard>",keyboard.path));}}EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();
var originalMap=config.baseMap;var temporaryMap=UnityEngine.Object.Instantiate(originalMap);config.baseMap=temporaryMap;var clip=temporaryMap.plansMelee.Single(e=>e.logicID==A.EAnimationLogicState.Attack_Normal_1).plan.clips[0];var originalPosition=actor.transform.position;var oldLeftId=gear.State.LeftInstanceId;var shield=session.Ledger.Items.Single(i=>i.DefinitionId=="vc02.shield-b");var rows=new List<object>();int failed=0,index=0,priorAccepted=0;double since=0;bool entered=false;Guid first=Guid.Empty;
void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,clip=view.CurrentClip,source=view.CurrentSource,rightClip=equipmentView.RightSkeleton?.AnimationState?.GetTrack(0)?.Animation?.Name,leftClip=equipmentView.LeftSkeleton?.AnimationState?.GetTrack(0)?.Animation?.Name,bodyExecution=view.CapturedActionId.ToString(),rightExecution=equipmentView.RightExecutionId.ToString(),leftExecution=equipmentView.LeftExecutionId.ToString(),left=gear.LeftDefinition?.definitionId,ability=abilities.LastResult});}
void Pad(float right=0,float left=0)=>InputSystem.QueueStateEvent(pad,new GamepadState{rightTrigger=right,leftTrigger=left,rightStick=Vector2.right});void Keys(params Key[] keys)=>InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));double Elapsed()=>EditorApplication.timeSinceStartup-since;var steps=new List<(string,Action,Func<bool>,double)>();void Add(string name,Action enter,Func<bool> check,double timeout=5)=>steps.Add((name,enter,check,timeout));void Delay(string name,Action enter,double seconds)=>Add(name,enter,()=>Elapsed()>=seconds,seconds+3);
Delay("neutral physical inputs",()=>{Pad();Keys();stamina.Restore();},.3);
Add("LT+RT selects configured hunter group",()=>Pad(1,1),()=>main.Sequence!=null&&main.SelectedId=="hunter.prototype.I"&&!actor.ActionActive);
Delay("release mode chord",()=>Pad(),.2);
Add("RT accepts main stage and body/equipment play same identity",()=>{stamina.Restore();Pad(1);},()=>{if(!actor.ActionActive||equipmentView.RightExecutionId!=actor.DataBus.ActionId||equipmentView.RightSkeleton?.AnimationState?.GetTrack(0)?.Animation?.Name!="strike"||view.CurrentClip!="atk1")return false;first=actor.DataBus.ActionId;return composer.CurrentBodyIntent?.frozenPlan!=null;});
Add("active action keeps old mix after mapping adjustment",()=>{Pad();clip.mixDuration=.12f;temporaryMap.Revision++;},()=>composer.CurrentBodyIntent?.actionId==first&&Math.Abs(composer.CurrentBodyIntent.frozenPlan.clips[0].mixDuration-.08f)<.00001f,.3);
Delay("let first action and continuation expire",()=>{},1.5);
Add("fresh RT observes adjusted mix on next accepted action",()=>{stamina.Restore();Pad(1);},()=>actor.ActionActive&&actor.DataBus.ActionId!=first&&view.CurrentClip=="atk1"&&Math.Abs((composer.CurrentBodyIntent?.frozenPlan?.clips[0].mixDuration??0)-.12f)<.00001f);
Delay("release attack and finish",()=>Pad(),1.1);
Add("Q body skill plays through formal ability and equipment",()=>{stamina.Restore();priorAccepted=abilities.AcceptedCount;Keys(Key.Q);},()=>abilities.AcceptedCount>priorAccepted&&actor.ActionActive&&view.CapturedActionId==actor.DataBus.ActionId&&equipmentView.RightExecutionId==actor.DataBus.ActionId&&equipmentView.RightSkeleton?.AnimationState?.GetTrack(0)?.Animation?.Name=="strike"&&view.CurrentClip=="atk3");
Delay("release Q and finish",()=>Keys(),1.3);
Add("LT starts configured shield A hold behavior",()=>{stamina.Restore();Pad(0,1);},()=>abilities.IsGuarding&&equipmentView.LeftExecutionId!=Guid.Empty);
Add("LT release stops guarding",()=>Pad(),()=>!abilities.IsGuarding);
Add("Tab opens normal inventory",()=>Keys(Key.Tab),()=>ui.IsOpen);
Delay("release Tab and approach existing chest",()=>{Keys();actor.GetComponent<C.CharacterMovement>().Teleport(session.ContainerPosition("external").Value+Vector3.right);Physics.SyncTransforms();ui.OpenExternal("external");},.3);
Add("uGUI drag existing chest shield B into actual left slot",()=>{Canvas.ForceUpdateCanvases();var targets=UnityEngine.Object.FindObjectsOfType<U.InventoryPointerTarget>();var from=targets.Single(t=>t.ItemId==shield.Id);var to=targets.First(t=>t.ContainerId==I.InventorySession.HandId(actor.ActorId,I.InventoryHand.Left)&&t.ItemId!=shield.Id);Vector2 Center(RectTransform r)=>RectTransformUtility.WorldToScreenPoint(null,r.TransformPoint(r.rect.center));var data=new PointerEventData(EventSystem.current){position=Center((RectTransform)from.transform),button=PointerEventData.InputButton.Left,pointerId=-1};ui.PointerDown(from,data);ui.BeginDrag(from,data);data.position=Center((RectTransform)to.transform);ui.Drag(data);ui.EndDrag(data);},()=>gear.LeftDefinition?.definitionId=="vc02.shield-b");
Add("Tab closes inventory",()=>Keys(Key.Tab),()=>!ui.IsOpen);
Delay("release Tab to rearm gameplay",()=>Keys(),.25);
Add("same LT invokes shield B body attack with left source and Spine",()=>{stamina.Restore();priorAccepted=abilities.AcceptedCount;Pad(0,1);},()=>abilities.AcceptedCount>priorAccepted&&actor.ActionActive&&view.CapturedSource==A.AnimationIntentSourceStateMachine.OffHand&&equipmentView.LeftExecutionId==actor.DataBus.ActionId&&equipmentView.LeftSkeleton?.AnimationState?.GetTrack(0)?.Animation?.Name=="strike"&&!abilities.IsGuarding);
Delay("release final gesture",()=>Pad(),.25);
void Flush(string state)=>SessionState.SetString("CONFIG_01_STORY",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,hardwareExperience=false,rows}));
void Cleanup(){actor.CancelAction("configuration-story");ui.SetOpen(false);if(gear.State.LeftInstanceId!=oldLeftId)session.Move(session.Ledger.Plan(oldLeftId,I.InventorySession.HandId(actor.ActorId,I.InventoryHand.Left)));actor.GetComponent<C.CharacterMovement>().Teleport(originalPosition);config.baseMap=originalMap;UnityEngine.Object.DestroyImmediate(temporaryMap);foreach(var pair in overrides)pair.Key.LoadBindingOverridesFromJson(pair.Value);InputSystem.RemoveDevice(pad);InputSystem.RemoveDevice(keyboard);InputSystem.settings=settingsBefore;UnityEngine.Object.DestroyImmediate(settings);}
EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(index==steps.Count){EditorApplication.update-=tick;Cleanup();Flush("completed");return;}var s=steps[index];if(!entered){entered=true;since=EditorApplication.timeSinceStartup;s.Item2();}if(s.Item3()){Check(s.Item1,true);index++;entered=false;Flush("running");}else if(Elapsed()>s.Item4)throw new Exception("Timeout "+s.Item1+" / "+actor.LastFact+" / "+ui.Notice);}catch(Exception error){failed++;rows.Add(new{error=error.ToString()});EditorApplication.update-=tick;Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="configuration sample scheduled";
}}
