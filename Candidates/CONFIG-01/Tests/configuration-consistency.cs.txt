/* 导出证据对账：确认当前指纹、源/目标GUID、真实事件时线与现场未被作者工具改写。 */
using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Security.Cryptography;using UnityEditor;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class ConfigurationConsistency:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}var scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene();var dirty=scene.isDirty;var selection=Selection.objects;var roots=UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>();var snapshots=roots.ToDictionary(r=>r.GetInstanceID(),r=>EditorJsonUtility.ToJson(r));var count=UnityEngine.Object.FindObjectsOfType<Spine.Unity.SkeletonAnimation>().Length;var report=Ossastra.Authoring.CharacterContentWorkbench.ReportPath;var exported=Ossastra.Authoring.CharacterContentWorkbench.ExportCandidate();Check("export preserves active single clean scene",UnityEngine.SceneManagement.SceneManager.sceneCount==1&&UnityEngine.SceneManagement.SceneManager.GetActiveScene()==scene&&scene.isDirty==dirty);Check("export preserves selection and edit roots without initialization",Selection.objects.SequenceEqual(selection)&&roots.All(r=>r.DataBus==null&&snapshots[r.GetInstanceID()]==EditorJsonUtility.ToJson(r))&&UnityEngine.Object.FindObjectsOfType<Spine.Unity.SkeletonAnimation>().Length==count);
var manifest=File.ReadAllLines(report+"/asset-manifest.tsv").Skip(1).Select(l=>l.Split('\t')).ToArray();bool fingerprints=true;foreach(var r in manifest){if(!File.Exists(r[0]))continue;using(var sha=SHA256.Create())using(var stream=File.OpenRead(r[0]))fingerprints&=r[2]==BitConverter.ToString(sha.ComputeHash(stream)).Replace("-","").ToLowerInvariant();}Check("all existing dependency bytes match SHA256 manifest",fingerprints);var references=File.ReadAllLines(report+"/SO-references.tsv").Skip(1).Select(l=>l.Split('\t')).ToArray();Check("all exported object reference GUIDs match actual paths",references.All(r=>r[1]==AssetDatabase.AssetPathToGUID(r[0])&&r[5]==AssetDatabase.AssetPathToGUID(r[4])));var basic=File.ReadAllLines(report+"/action-contracts.tsv").Single(l=>l.StartsWith("Assets/OSSASTRA/Configs/Actions/Rozeul_Basic1.asset\t")).Split('\t');Check("event timeline author contract exports hit_start and clip duration rather than scalar defaults",Math.Abs(double.Parse(basic[3],System.Globalization.CultureInfo.InvariantCulture)-1d/12)<.00001&&Math.Abs(double.Parse(basic[6],System.Globalization.CultureInfo.InvariantCulture)-1.1)<.00001&&basic[8]=="eventTimeline");ctx.ReturnValue=new{failed,rows,exported,manifestRows=manifest.Length,referenceRows=references.Length,scenePath=scene.path};}}
