using System;using System.Linq;using UnityEngine;using UnityEditor;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class AM02ExportIdentity:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var scripts=AssetDatabase.FindAssets("t:MonoScript",new[]{"Assets/OSSASTRA/Scripts/Animation/Character","Assets/旧代码/Script/Character/Core/Systems/AnimationSystem"}).Select(AssetDatabase.GUIDToAssetPath).Select(p=>new{path=p,guid=AssetDatabase.AssetPathToGUID(p),type=AssetDatabase.LoadAssetAtPath<MonoScript>(p)?.GetClass()?.FullName}).ToArray();
var maps=AssetDatabase.FindAssets("t:ScriptableObject",new[]{"Assets/OSSASTRA/Configs/Animation"}).Select(AssetDatabase.GUIDToAssetPath).Where(p=>AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(p)!=null).Select(p=>{var m=AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(p);var editor=Editor.CreateEditor(m);var editorType=editor.GetType().FullName;UnityEngine.Object.DestroyImmediate(editor);return new{path=p,guid=AssetDatabase.AssetPathToGUID(p),type=m.GetType().FullName,skeleton=AssetDatabase.GetAssetPath(m.skeletonDataAsset),editor=editorType,json=EditorJsonUtility.ToJson(m)};}).ToArray();
var enums=new[]{typeof(A.AnimationLayer),typeof(A.AnimationIntentSourceStateMachine),typeof(A.EAnimSemantic),typeof(A.EAnimationLogicState)}.SelectMany(t=>Enum.GetValues(t).Cast<object>().Select(v=>new{type=t.Name,name=v.ToString(),value=Convert.ToInt32(v)})).ToArray();
ctx.ReturnValue=new{scripts,maps,enums,scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,prefabStage=UnityEditor.SceneManagement.PrefabStageUtility.GetCurrentPrefabStage()?.assetPath,userStageDirty=UnityEditor.SceneManagement.PrefabStageUtility.GetCurrentPrefabStage()?.scene.isDirty};}}
