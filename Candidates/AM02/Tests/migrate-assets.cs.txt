using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Spine.Unity;using A=Ossastra.CharacterAnimation;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class AM02Migration:IFunplayCommand {

 if(Application.isPlaying)throw new Exception("Edit required");var scene=EditorSceneManager.GetActiveScene();if(scene.path!="Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity")throw new Exception("Expected B scene");var user=PrefabStageUtility.GetCurrentPrefabStage();if(user!=null&&user.scene.isDirty)throw new Exception("User prefab dirty; cannot alter stage");
 var actors=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<C.CharacterRoot>(true)).ToArray();if(actors.Length!=2)throw new Exception("Two B actors required");var skeleton=actors[0].GetComponentInChildren<SkeletonAnimation>().skeletonDataAsset;
 var keys=new Dictionary<string,A.EAnimationLogicState>{{"idle",A.EAnimationLogicState.Idle},{"move",A.EAnimationLogicState.Jog},{"dodge",A.EAnimationLogicState.Dodge},{"guard",A.EAnimationLogicState.Guard},{"hurt",A.EAnimationLogicState.HitReact},{"broken",A.EAnimationLogicState.Stagger},{"death",A.EAnimationLogicState.Death},{"basic-1",A.EAnimationLogicState.Attack_Normal_1},{"basic-2",A.EAnimationLogicState.Attack_Normal_2},{"basic-3",A.EAnimationLogicState.Attack_Normal_3},{"basic-4",A.EAnimationLogicState.Attack_Normal_4},{"HunterBreak_Adapted",A.EAnimationLogicState.HeavyAttack},{"RangerRocket_Adapted",A.EAnimationLogicState.RocketAttack},{"rush-prepare",A.EAnimationLogicState.RushPrepare},{"rush-release",A.EAnimationLogicState.RushRelease},{"rush-charge",A.EAnimationLogicState.RushCharge},{"rocket-leap",A.EAnimationLogicState.RocketLeap},{"precision-reload",A.EAnimationLogicState.PrecisionReload},{"magazine-shot",A.EAnimationLogicState.MagazineShot}};
 var actions=AssetDatabase.FindAssets("t:ActionDefinitionSO",new[]{"Assets/OSSASTRA/Configs"}).Select(AssetDatabase.GUIDToAssetPath).Select(p=>AssetDatabase.LoadAssetAtPath<C.ActionDefinitionSO>(p)).Where(a=>a!=null).ToArray();foreach(var a in actions)if(!keys.ContainsKey(a.animationKey))throw new Exception("Unknown action key: "+a.animationKey);
 const string folder="Assets/OSSASTRA/Configs/Animation";if(!AssetDatabase.IsValidFolder(folder))AssetDatabase.CreateFolder("Assets/OSSASTRA/Configs","Animation");var records=new List<object>();var profiles=new Dictionary<C.CharacterAnimationProfileSO,A.AnimationPlanMappingSO>();

 var router=Asset<A.AnimationLogicRouterSO>(folder+"/CharacterLogicRouter.asset");if(router.rulesOther.Count==0){foreach(var pair in new[]{(A.EAnimSemantic.Idle,A.EAnimationLogicState.Idle),(A.EAnimSemantic.Jog,A.EAnimationLogicState.Jog),(A.EAnimSemantic.Guard,A.EAnimationLogicState.Guard),(A.EAnimSemantic.Dodge,A.EAnimationLogicState.Dodge),(A.EAnimSemantic.HitReact,A.EAnimationLogicState.HitReact),(A.EAnimSemantic.Stagger,A.EAnimationLogicState.Stagger),(A.EAnimSemantic.Ability,A.EAnimationLogicState.Ability),(A.EAnimSemantic.Attack_Normal_1,A.EAnimationLogicState.Attack_Normal_1),(A.EAnimSemantic.Attack_Normal_2,A.EAnimationLogicState.Attack_Normal_2),(A.EAnimSemantic.Attack_Normal_3,A.EAnimationLogicState.Attack_Normal_3),(A.EAnimSemantic.Attack_Normal_4,A.EAnimationLogicState.Attack_Normal_4)})router.rulesOther.Add(new A.AnimationLogicRouterSO.RouteRule{inputSemantic=pair.Item1,outputID=pair.Item2});EditorUtility.SetDirty(router);AssetDatabase.SaveAssetIfDirty(router);}
 foreach(var path in AssetDatabase.FindAssets("t:CharacterAnimationProfileSO",new[]{"Assets/OSSASTRA/Configs"}).Select(AssetDatabase.GUIDToAssetPath)){
 var profile=AssetDatabase.LoadAssetAtPath<C.CharacterAnimationProfileSO>(path);if(profile==null)continue;var map=Asset<A.AnimationPlanMappingSO>(folder+"/"+System.IO.Path.GetFileNameWithoutExtension(path)+"_Mapping.asset");map.skeletonDataAsset=skeleton;foreach(var binding in profile.bindings){if(!keys.TryGetValue(binding.key,out var logic))throw new Exception("Unmapped profile key: "+binding.key);Add(map,logic,binding.clip,binding.loop,binding.mixDuration,binding.sourceDuration,binding.semantic==Ossastra.AnimationCore.AnimationSemantic.BasicAttack||binding.semantic==Ossastra.AnimationCore.AnimationSemantic.BasicAttackSecond);}
 Add(map,A.EAnimationLogicState.GuardEnter,"ready_start",false,.08f,0,false);Add(map,A.EAnimationLogicState.GuardExit,"ready_end",false,.08f,0,false);EditorUtility.SetDirty(map);AssetDatabase.SaveAssetIfDirty(map);profiles.Add(profile,map);records.Add(new{kind="body-map",source=path,destination=AssetDatabase.GetAssetPath(map),guid=AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(map))});}
 foreach(var a in actions){if(a.animationLogicId==keys[a.animationKey])continue;a.animationLogicId=keys[a.animationKey];EditorUtility.SetDirty(a);AssetDatabase.SaveAssetIfDirty(a);records.Add(new{kind="action",path=AssetDatabase.GetAssetPath(a),a.animationKey,logic=a.animationLogicId.ToString()});}
 foreach(var path in AssetDatabase.FindAssets("t:EquipmentSpinePresentationSO",new[]{"Assets/OSSASTRA"}).Select(AssetDatabase.GUIDToAssetPath)){
 var profile=AssetDatabase.LoadAssetAtPath<C.EquipmentSpinePresentationSO>(path);if(profile==null)continue;var map=profile.animationMap!=null?profile.animationMap:Asset<A.AnimationPlanMappingSO>(folder+"/"+profile.name+"_Mapping.asset");map.skeletonDataAsset=profile.skeletonData;foreach(var b in profile.animations){var semantic=A.AnimationSemanticTranslation.FromShared(b.semantic);var logic=semantic==A.EAnimSemantic.Idle?A.EAnimationLogicState.Idle:semantic==A.EAnimSemantic.Jog?A.EAnimationLogicState.Jog:semantic==A.EAnimSemantic.Guard?A.EAnimationLogicState.Guard:semantic==A.EAnimSemantic.Dodge?A.EAnimationLogicState.Dodge:semantic==A.EAnimSemantic.HitReact?A.EAnimationLogicState.HitReact:semantic==A.EAnimSemantic.Stagger?A.EAnimationLogicState.Stagger:semantic==A.EAnimSemantic.Death?A.EAnimationLogicState.Death:A.EAnimationLogicState.Ability;Add(map,logic,b.clip,b.loop,b.mixDuration,0,false,b.playbackRate);}EditorUtility.SetDirty(map);AssetDatabase.SaveAssetIfDirty(map);profile.animationMap=map;EditorUtility.SetDirty(profile);AssetDatabase.SaveAssetIfDirty(profile);records.Add(new{kind="equipment-map",source=path,destination=AssetDatabase.GetAssetPath(map),guid=AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(map))});}
 void Wire(C.CharacterRoot root){var old=root.GetComponentInChildren<C.CharacterSpineView>(true);var controller=root.GetComponentInChildren<A.CharacterAnimationController>(true);var body=old!=null?old.gameObject:controller!=null?controller.gameObject:null;if(body==null)throw new Exception("Body adapter missing");if(controller==null)controller=body.AddComponent<A.CharacterAnimationController>();var data=body.GetComponent<SkeletonAnimation>();var so=new SerializedObject(controller);so.FindProperty("skeleton").objectReferenceValue=data;so.ApplyModifiedPropertiesWithoutUndo();var config=body.GetComponent<A.CharacterAnimationConfigManager>();A.AnimationPlanMappingSO map;if(root.Definition==null)map=AssetDatabase.LoadAssetAtPath<A.AnimationPlanMappingSO>(folder+"/Rozeul_Animation_Mapping.asset");else if(!profiles.TryGetValue(root.Definition.animationProfile,out map))throw new Exception("Role profile missing");if(map==null)throw new Exception("Explicit default mapping missing");var cs=new SerializedObject(config);cs.FindProperty("baseMap").objectReferenceValue=map;cs.FindProperty("logicRouter").objectReferenceValue=router;cs.ApplyModifiedPropertiesWithoutUndo();if(PrefabUtility.IsPartOfPrefabInstance(config))PrefabUtility.RecordPrefabInstancePropertyModifications(config);var rs=new SerializedObject(root);rs.FindProperty("view").objectReferenceValue=controller;rs.ApplyModifiedPropertiesWithoutUndo();if(old!=null)ctx.DestroyObject(old);}
 const string prefab="Assets/OSSASTRA/Prefabs/Characters/PlayerCharacter.prefab";var content=PrefabUtility.LoadPrefabContents(prefab);try{Wire(content.GetComponent<C.CharacterRoot>());PrefabUtility.SaveAsPrefabAsset(content,prefab);}finally{PrefabUtility.UnloadPrefabContents(content);}
 actors=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<C.CharacterRoot>(true)).ToArray();foreach(var actor in actors){Wire(actor);var mods=PrefabUtility.GetPropertyModifications(actor.gameObject);PrefabUtility.SetPropertyModifications(actor.gameObject,mods.Where(m=>!(m.target==null&&(m.propertyPath=="profile"||m.propertyPath.StartsWith("catalog.")))).ToArray());}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);
 ctx.ReturnValue=new{records,actors=actors.Select(a=>new{a.ActorId,controllers=a.GetComponentsInChildren<A.CharacterAnimationController>(true).Length,oldViews=a.GetComponentsInChildren<C.CharacterSpineView>(true).Length}).ToArray(),userStage=PrefabStageUtility.GetCurrentPrefabStage()?.assetPath,userDirty=PrefabStageUtility.GetCurrentPrefabStage()?.scene.isDirty};
 }
}
