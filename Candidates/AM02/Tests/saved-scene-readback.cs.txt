using System;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using Funplay.Editor.Tools.Scripting;public class AM02SavedSceneReadback:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(Application.isPlaying)throw new Exception("Edit required");var scene=EditorSceneManager.GetActiveScene();var stage=PrefabStageUtility.GetCurrentPrefabStage();if(scene.isDirty||stage!=null&&stage.scene.isDirty)throw new Exception("Dirty scene/stage preserved");var path=stage?.assetPath;string selection=null;if(stage!=null&&Selection.activeGameObject!=null&&Selection.activeGameObject.scene==stage.scene)selection=AnimationUtility.CalculateTransformPath(Selection.activeGameObject.transform,stage.prefabContentsRoot.transform);try{if(stage!=null)StageUtility.GoToMainStage();var saved=EditorSceneManager.OpenScene("Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity");ctx.ReturnValue=saved.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Ossastra.Characters.CharacterRoot>(true)).Select(r=>new{r.ActorId,map=AssetDatabase.GetAssetPath(r.GetComponentInChildren<Ossastra.CharacterAnimation.CharacterAnimationConfigManager>(true).baseMap),controllers=r.GetComponentsInChildren<Ossastra.CharacterAnimation.CharacterAnimationController>(true).Length,oldViews=r.GetComponentsInChildren<Ossastra.Characters.CharacterSpineView>(true).Length,oldOverrides=PrefabUtility.GetPropertyModifications(r.gameObject).Count(m=>m.target==null&&(m.propertyPath=="profile"||m.propertyPath.StartsWith("catalog.")))}).ToArray();}finally{if(!string.IsNullOrEmpty(path)){var restored=PrefabStageUtility.OpenPrefab(path);if(selection!=null){var selected=string.IsNullOrEmpty(selection)?restored.prefabContentsRoot.transform:restored.prefabContentsRoot.transform.Find(selection);if(selected!=null)Selection.activeGameObject=selected.gameObject;}}}}}
