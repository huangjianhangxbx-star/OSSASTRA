using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Funplay.Editor.Tools.Scripting;
public class AM02HandsStory:IFunplayCommand {
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
 if(!Application.isPlaying)throw new Exception("Play required");var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();if(c==null||c.Current==null)throw new Exception("Missing current");var actor=c.Current;var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
 var keyboard=InputSystem.AddDevice<Keyboard>("AM01Keyboard");var mouse=InputSystem.AddDevice<Mouse>("AM01Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in c.Characters.SelectMany(r=>{var i=r.GetComponent<CharacterInputHandler>();return new[]{i.movementAction,i.mouseAction,i.attackAction,i.dodgeAction,i.offhandAction,i.skillOneAction,i.skillTwoAction,i.skillThreeAction};}).Concat(new[]{c.switchAction}).Where(a=>a!=null).Distinct()){overrides[action]=action.SaveBindingOverridesAsJson();for(int n=0;n<action.bindings.Count;n++){var p=action.bindings[n].path;if(action.bindings[n].isComposite||p==null)continue;action.ApplyBindingOverride(n,p.Replace("<Keyboard>",keyboard.path).Replace("<Mouse>",mouse.path));}}
 foreach(var r in c.Characters)r.SendMessage("OnApplicationFocus",true);c.SendMessage("OnApplicationFocus",true);EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();
 var inventory=UnityEngine.Object.FindObjectOfType<Ossastra.UserInterface.InventoryView>();var rows=new List<object>();var phases=new List<(string,Action,Func<bool>,double)>();int index=0,failed=0;bool entered=false;double since=0;var view=actor.GetComponentInChildren<Ossastra.CharacterAnimation.CharacterAnimationController>();var gear=actor.GetComponent<CharacterEquipmentSpineView>();var ability=actor.GetComponent<CharacterAbilities>();Guid leftId=Guid.Empty;int leftCommits=0;var captured=new List<string>();
 void Keys(params Key[] keys)=>InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));void Mouse(ushort b)=>InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(Screen.width*.6f,Screen.height*.35f),buttons=b});double Elapsed()=>EditorApplication.timeSinceStartup-since;
 void Add(string label,Action entry,Func<bool> check,double timeout=6)=>phases.Add((label,entry,check,timeout));void Delay(string label,Action entry,double duration)=>Add(label,entry,()=>Elapsed()>=duration,duration+3);
 Delay("neutral",()=>{Keys();Mouse(0);actor.GetComponent<CharacterStamina>().Restore();},.25);
 Add("offhand guard accepts own run",()=>Mouse(2),()=>ability.IsGuarding&&gear.LeftExecutionId!=Guid.Empty);
 Add("actual guard start then target loop",()=>{},()=>{if(view.CurrentClip!="ready_loop")return false;leftId=gear.LeftExecutionId;leftCommits=gear.LeftCommitCount;return true;});
 Delay("open actual inventory during accepted guard",()=>inventory.SetOpen(true),.15);
 Add("UI capture preserves original held guard identity",()=>{},()=>ability.IsGuarding&&gear.LeftExecutionId==leftId);
 Delay("close actual inventory while same hand held",()=>inventory.SetOpen(false),.15);
 Add("closing capture cannot synthesize guard release",()=>{},()=>ability.IsGuarding&&gear.LeftExecutionId==leftId);
 Add("main attack accepted while same guard continues",()=>Mouse(3),()=>actor.ActionActive&&view.CurrentSource=="action");
 Add("left animation not restarted by main action replay",()=>{},()=>gear.LeftExecutionId==leftId&&gear.LeftCommitCount==leftCommits&&gear.RightExecutionId==actor.DataBus.ActionId);
 Add("both actual Spine grips aligned",()=>{},()=>gear.LeftGripError<.0001f&&gear.RightGripError<.0001f);
 Add("only one model instance per slot",()=>{},()=>actor.GetComponentsInChildren<Spine.Unity.SkeletonAnimation>().Count(s=>s.name=="LeftEquipmentSpine")==1&&actor.GetComponentsInChildren<Spine.Unity.SkeletonAnimation>().Count(s=>s.name=="RightEquipmentSpine")==1);
 Delay("release both buttons",()=>Mouse(0),1.3);
 Add("new body path is sole active character writer",()=>{},()=>actor.GetComponentsInChildren<CharacterSpineView>().Length==0&&actor.GetComponentsInChildren<Ossastra.CharacterAnimation.AnimationHandleSystem>().Length==3);
 void Flush(string state)=>SessionState.SetString("AM03_GUARD_CAPTURE",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,rows}));void Cleanup(){Keys();Mouse(0);foreach(var pair in overrides)pair.Key.LoadBindingOverridesFromJson(pair.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}
 EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");if(index==phases.Count){EditorApplication.update-=tick;Cleanup();Flush("completed");return;}var p=phases[index];if(!entered){entered=true;since=EditorApplication.timeSinceStartup;p.Item2();}if(p.Item3()){rows.Add(new{name=p.Item1,ok=true,body=view.CurrentClip,left=gear.LeftExecutionId,right=gear.RightExecutionId});index++;entered=false;Flush("running");}else if(Elapsed()>p.Item4)throw new Exception("Timeout "+p.Item1+" / "+view.CurrentClip+" / "+ability.LastResult+" / "+gear.LastResult);}catch(Exception e){failed++;rows.Add(new{error=e.ToString()});EditorApplication.update-=tick;Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="AM02 normal held guard/main attack hand identity story scheduled";
 }
}
