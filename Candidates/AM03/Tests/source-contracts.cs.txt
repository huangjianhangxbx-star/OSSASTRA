}finally{sampler.Dispose();foreach(var a in actions)a.Dispose();foreach(var d in new InputDevice[]{keyboard,mouse,pad,second})if(d.added)InputSystem.RemoveDevice(d);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}ctx.ReturnValue=new{failed,rows};}}
