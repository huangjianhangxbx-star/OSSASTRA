using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.SceneManagement;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class AM03FinalAudit:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(Application.isPlaying)throw new Exception("Edit required");var scene=SceneManager.GetActiveScene();if(scene.path!="Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity")throw new Exception("B scene required");var roots=scene.GetRootGameObjects();var actors=roots.SelectMany(g=>g.GetComponentsInChildren<C.CharacterRoot>(true)).ToArray();
ctx.ReturnValue=new{scene=scene.path,dirty=scene.isDirty,userStage=UnityEditor.SceneManagement.PrefabStageUtility.GetCurrentPrefabStage()?.assetPath,userStageDirty=UnityEditor.SceneManagement.PrefabStageUtility.GetCurrentPrefabStage()?.scene.isDirty,
missingScripts=roots.Sum(g=>g.GetComponentsInChildren<MonoBehaviour>(true).Count(b=>b==null)),actors=actors.Select(a=>{var i=a.GetComponent<C.CharacterInputHandler>();var v=a.GetComponentInChildren<A.CharacterAnimationController>(true);var fields=new SerializedObject(i);return new{a.ActorId,formalView=v!=null,oldViews=a.GetComponentsInChildren<C.CharacterSpineView>(true).Length,map=AssetDatabase.GetAssetPath(v.GetComponent<A.CharacterAnimationConfigManager>().baseMap),threshold=fields.FindProperty("intentThreshold").floatValue,directionChange=fields.FindProperty("directionChangeThreshold").floatValue,layout=AssetDatabase.GetAssetPath(i.inputLayout)};}).ToArray(),devices=InputSystem.devices.Select(d=>new{d.name,type=d.GetType().Name,d.added,d.enabled}).ToArray(),inputSettings=AssetDatabase.GetAssetPath(InputSystem.settings),npcEnemyRoots=roots.SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(b=>b!=null&&(b.GetType().FullName=="Ossastra.Enemies.EnemyRoot"||b.GetType().FullName=="Ossastra.Npcs.NpcRoot")).Select(b=>new{b.name,type=b.GetType().FullName}).ToArray()};}}
