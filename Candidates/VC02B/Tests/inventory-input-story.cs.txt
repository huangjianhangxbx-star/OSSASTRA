using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.UI;using UnityEngine.InputSystem.LowLevel;using UnityEngine.UI;using Ossastra.Characters;using Ossastra.Inventory;using Ossastra.UserInterface;using Funplay.Editor.Tools.Scripting;
public class InventoryInputStory:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
 if(!Application.isPlaying)throw new Exception("Fresh B Play required");var view=UnityEngine.Object.FindObjectOfType<InventoryView>();var session=UnityEngine.Object.FindObjectOfType<InventorySession>();if(view==null||!session.Ready)throw new Exception("B view/session missing");var c=session.Coordinator;var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
 var keyboard=InputSystem.AddDevice<Keyboard>("InventoryStoryKeyboard");var mouse=InputSystem.AddDevice<Mouse>("InventoryStoryMouse");var ui=UnityEngine.Object.FindObjectOfType<InputSystemUIInputModule>();if(ui==null)throw new Exception("UI input module missing");var oldPointer=ui.pointerBehavior;ui.pointerBehavior=UIPointerBehavior.AllPointersAsIs;
 var overrides=new Dictionary<InputAction,string>();foreach(var action in c.Characters.SelectMany(r=>{var i=r.GetComponent<CharacterInputHandler>();return new[]{i.movementAction,i.mouseAction,i.attackAction,i.dodgeAction,i.offhandAction,i.skillOneAction,i.skillTwoAction,i.skillThreeAction};}).Concat(new[]{c.switchAction,view.toggleAction,view.cancelAction,ui.point.action,ui.leftClick.action,ui.rightClick.action}).Where(x=>x!=null).Distinct()){overrides[action]=action.SaveBindingOverridesAsJson();for(int n=0;n<action.bindings.Count;n++){var path=action.bindings[n].path;if(action.bindings[n].isComposite||path==null)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(n,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(n,path.Replace("<Mouse>",mouse.path));}}
 foreach(var r in c.Characters)r.SendMessage("OnApplicationFocus",true);c.SendMessage("OnApplicationFocus",true);EditorWindow.GetWindow(typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView")).Focus();
 var phases=new List<(string,Action,Func<bool>,double)>();var rows=new List<object>();int index=0,failed=0;bool entered=false;double since=0;var actor=c.Current;Vector3 start=actor.transform.position;int release=actor.ReleaseCount;string seed=session.Ledger.Items.Single(i=>i.DefinitionId=="sample-seed").Id;var item=session.Ledger.Find(seed);int quantityBefore=item.Quantity;Vector2 from=default,to=default;
 void Keys(params Key[] keys)=>InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));void Mouse(Vector2 p,ushort buttons)=>InputSystem.QueueStateEvent(mouse,new MouseState{position=p,buttons=buttons});double Elapsed()=>EditorApplication.timeSinceStartup-since;
 void Add(string name,Action action,Func<bool> test,double timeout=5)=>phases.Add((name,action,test,timeout));void Delay(string name,Action action,double seconds)=>Add(name,action,()=>Elapsed()>=seconds,seconds+3);
 Vector2 Center(RectTransform r)=>RectTransformUtility.WorldToScreenPoint(null,r.TransformPoint(r.rect.center));InventoryPointerTarget Find(Func<InventoryPointerTarget,bool> f)=>UnityEngine.Object.FindObjectsOfType<InventoryPointerTarget>().First(f);
 Delay("neutral arm",()=>{Keys();Mouse(new Vector2(800,500),0);},.2);Delay("Tab press",()=>Keys(Key.Tab),.1);Add("normal Tab opens inventory",()=>Keys(),()=>view.IsOpen);
 Add("unselected detail hidden",()=>{},()=>!view.transform.Find("InventoryCanvas/Windows/ItemDetail").gameObject.activeSelf);
 Delay("WASD while open",()=>{start=actor.transform.position;Keys(Key.D);},.25);Add("current moves with inventory",()=>Keys(),()=>Vector3.Distance(start,actor.transform.position)>.1f&&c.Current==actor);
 Delay("world click blocked",()=>Mouse(new Vector2(Screen.width*.55f,Screen.height*.25f),1),.2);Add("no combat through inventory",()=>Mouse(new Vector2(Screen.width*.55f,Screen.height*.25f),0),()=>actor.ReleaseCount==release&&!actor.ActionActive);
 Delay("select seed pointer down",()=>{from=Center((RectTransform)Find(t=>t.ItemId==seed).transform);Mouse(from,1);},.12);Delay("select seed pointer up",()=>Mouse(from,0),.12);Add("selected real details",()=>{},()=>view.SelectedItemId==seed&&view.transform.Find("InventoryCanvas/Windows/ItemDetail").gameObject.activeSelf);
 Delay("drag seed start",()=>{from=Center((RectTransform)Find(t=>t.ItemId==seed).transform);to=Center((RectTransform)Find(t=>t.ContainerId=="safe"&&t.Slot==0).transform);Mouse(from,1);},.12);Delay("drag crosses threshold",()=>Mouse(Vector2.Lerp(from,to,.5f),1),.15);Delay("drag enters safe",()=>Mouse(to,1),.15);Delay("drop safe",()=>Mouse(to,0),.15);Add("actual pointer drag moved whole stack",()=>{},()=>session.Ledger.Find(seed).Location.ContainerId=="safe"&&session.Ledger.Find(seed).Quantity==quantityBefore);
 Delay("close Tab press",()=>Keys(Key.Tab),.12);Add("Tab closes and preserves inventory",()=>Keys(),()=>!view.IsOpen&&session.Ledger.Find(seed).Location.ContainerId=="safe");Delay("reopen Tab press",()=>Keys(Key.Tab),.12);Add("reopen real state without new grant",()=>Keys(),()=>view.IsOpen&&session.Ledger.Items.Count(i=>i.DefinitionId=="sample-seed")==1);
 void Flush(string state)=>SessionState.SetString("VC02B_INPUT",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,rows}));void Cleanup(){foreach(var pair in overrides)pair.Key.LoadBindingOverridesFromJson(pair.Value);ui.pointerBehavior=oldPointer;InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}
 EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");if(index>=phases.Count){EditorApplication.update-=tick;Cleanup();Flush("completed");return;}var p=phases[index];if(!entered){entered=true;since=EditorApplication.timeSinceStartup;p.Item2();}if(p.Item3()){rows.Add(new{name=p.Item1,ok=true,current=c.Current?.ActorId,inventory=view.IsOpen,notice=view.Notice});index++;entered=false;Flush("running");}else if(Elapsed()>p.Item4)throw new Exception("Phase timeout: "+p.Item1+" / "+view.Notice+" / "+session.LastResult);}catch(Exception error){failed++;rows.Add(new{error=error.ToString()});EditorApplication.update-=tick;Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="real inventory input scheduled; world state not injected";}}
