using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Inventory;using Funplay.Editor.Tools.Scripting;
public class InventoryUseWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){if(!Application.isPlaying)throw new Exception("Fresh B Play required");var session=UnityEngine.Object.FindObjectOfType<InventorySession>();var c=session.Coordinator;var actor=c.Current;var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();var consumer=session.GetComponent<InventoryItemUse>();if(consumer==null)throw new Exception("Missing inventory consumer");foreach(var r in c.Characters){r.SendMessage("OnApplicationFocus",true);r.GetComponent<CompanionCombatBrain>().enabled=false;r.CancelAction("item-boundary-fixture");}c.SendMessage("OnApplicationFocus",true);enemy.AttacksEnabled=false;enemy.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;UnityEngine.Object.FindObjectOfType<CombatEngagementSession>().enabled=false;var recovery=actor.GetComponent<CombatRegeneration>();if(recovery==null)throw new Exception("Recovery component missing");
 var rows=new List<object>();var steps=new List<(double,Action)>();int failed=0,index=0;double start=EditorApplication.timeSinceStartup;int pausedHp=0,quantityBefore=0;string medicine=session.Ledger.Items.Single(i=>i.DefinitionId=="sample-al-regen").Id;
 void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,hp=actor.Vitals.Health.Current,quantity=session.Ledger.Find(medicine)?.Quantity??0,result=consumer.LastResult,ability=actor.GetComponent<CharacterAbilities>().LastResult});}
 void Hurt(int amount){CombatContactGateway.Resolve(AttackRelease.ForEmission(Guid.NewGuid(),0,enemy.ActorId,amount,0),enemy.Vitals,new IHitTarget[]{actor.Vitals},enemy.transform.position);}
 steps.Add((.2,()=>{Check("full HP refuses and does not consume",!consumer.TryUse(medicine)&&session.Ledger.Find(medicine).Quantity==2);Hurt(40);quantityBefore=session.Ledger.Find(medicine).Quantity;Check("real inventory medicine accepted",consumer.TryUse(medicine));Check("accept reserves before release",session.Ledger.Find(medicine).Quantity==quantityBefore);Check("same stack second acceptance rejected",!consumer.TryUse(medicine));}));
 steps.Add((.8,()=>{Check("release consumes exactly one",session.Ledger.Find(medicine).Quantity==1);Check("medicine slow effect actually heals",actor.Vitals.Health.Current>60&&actor.Vitals.Health.Current<65&&recovery.Active);actor.SetPaused(true);pausedHp=actor.Vitals.Health.Current;}));
 steps.Add((1.5,()=>{Check("explicit pause freezes slow recovery",actor.Vitals.Health.Current==pausedHp);actor.SetPaused(false);}));
 steps.Add((1.8,()=>{recovery.Clear();Check("accept second item then cancel",consumer.TryUse(medicine));actor.GetComponent<CharacterAbilities>().CancelPending("item-cancel-before-release");}));
 steps.Add((2.1,()=>{Check("cancel releases quantity reservation",session.Ledger.Find(medicine).Quantity==1);Check("cancel leaves no recovery",!recovery.Active);}));
 for(int n=0;n<5;n++){var tier=n;var at=2.5+n*.7;steps.Add((at,()=>{recovery.Clear();actor.Vitals.RestoreForDevelopment();Hurt(50);var food=session.Ledger.Items.Single(i=>i.DefinitionId=="food-"+tier);var q=session.Move(session.Ledger.Plan(food.Id,"team"));Check("take food "+tier+" from real chest",q.Ok);Check("use configured food "+tier,consumer.TryUse(food.Id));}));steps.Add((at+.6,()=>{Check("food "+tier+" actually heals gradually",actor.Vitals.Health.Current>50&&recovery.Active);Check("food "+tier+" consumed original instance",!session.Ledger.Items.Any(i=>i.DefinitionId=="food-"+tier));}));}
 void Flush(string state)=>SessionState.SetString("VC02B_USE",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,boundaryFixture=true,rows}));EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");if(index>=steps.Count){EditorApplication.update-=tick;Flush("completed");return;}if(EditorApplication.timeSinceStartup-start>=steps[index].Item1){steps[index].Item2();index++;Flush("running");}}catch(Exception error){failed++;rows.Add(new{error=error.ToString()});EditorApplication.update-=tick;Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="B3 actual item boundary story scheduled";}}
