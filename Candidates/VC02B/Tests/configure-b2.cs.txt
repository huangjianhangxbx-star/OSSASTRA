using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class ConfigureInventoryUi:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(Application.isPlaying||scene.path!="Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity")throw new Exception("Saved B edit scene required");const string folder="Assets/OSSASTRA/VC02B",art=folder+"/Art/";
 foreach(var file in System.IO.Directory.GetFiles(art,"*.png")){AssetDatabase.ImportAsset(file,ImportAssetOptions.ForceSynchronousImport);var importer=AssetImporter.GetAtPath(file) as TextureImporter;if(importer==null)throw new Exception("Missing texture importer "+file);importer.textureType=TextureImporterType.Sprite;importer.spriteImportMode=SpriteImportMode.Single;importer.mipmapEnabled=false;importer.alphaIsTransparency=true;importer.textureCompression=TextureImporterCompression.Uncompressed;importer.spritePixelsPerUnit=100;importer.spriteBorder=file.Contains("inventory-frame")?new Vector4(12,12,12,12):file.Contains("slot-")?new Vector4(15,15,15,15):Vector4.zero;importer.SaveAndReimport();}
 var session=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();if(session==null)throw new Exception("Missing session");var skin=ScriptableObject.CreateInstance<Ossastra.UserInterface.InventorySkinSO>();skin.frame=AssetDatabase.LoadAssetAtPath<Sprite>(art+"inventory-frame.png");skin.slotFill=AssetDatabase.LoadAssetAtPath<Sprite>(art+"slot-fill.png");skin.slotEdge=AssetDatabase.LoadAssetAtPath<Sprite>(art+"slot-edge.png");ctx.RegisterObjectCreation(skin);AssetDatabase.CreateAsset(skin,folder+"/InventorySkin.asset");
 foreach(var guid in AssetDatabase.FindAssets("t:ItemDefinitionSO",new[]{folder})){var d=AssetDatabase.LoadAssetAtPath<Ossastra.Characters.ItemDefinitionSO>(AssetDatabase.GUIDToAssetPath(guid));if(d==null)continue;var icon=AssetDatabase.LoadAssetAtPath<Sprite>(art+d.definitionId+".png");if(icon!=null){ctx.RegisterObjectModification(d);d.icon=icon;EditorUtility.SetDirty(d);AssetDatabase.SaveAssetIfDirty(d);}}
 if(session.GetComponent<Ossastra.UserInterface.InventoryView>()!=null)throw new Exception("View exists: inspect first");ctx.RegisterObjectModification(session.gameObject);var view=Undo.AddComponent<Ossastra.UserInterface.InventoryView>(session.gameObject);var so=new SerializedObject(view);so.FindProperty("session").objectReferenceValue=session;so.FindProperty("skin").objectReferenceValue=skin;so.ApplyModifiedProperties();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=new{scene=scene.path,frame=skin.frame.border,slot=skin.slotFill.border};}}
