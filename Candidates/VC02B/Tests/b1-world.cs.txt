using System;using System.Linq;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class B1World:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var rows=new System.Collections.Generic.List<object>();void Check(bool ok,string name){rows.Add(new{ok,name});if(!ok)throw new Exception(name);}var s=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();Check(s!=null&&s.Ready,"normal scene initializes actual inventory");var m=s.Ledger;Check(m.Validate(out var reason),"valid world ledger: "+reason);Check(m.Container("team").Columns==5&&m.Container("team").Rows==2,"team 5x2");Check(m.Container("safe").Slots&&m.Container("safe").Columns==2,"safe two entries");Check(m.Container("reserve").Columns==5,"partner reserve five");foreach(var root in s.Coordinator.Characters){var g=root.GetComponent<Ossastra.Characters.CharacterEquipment>();Check(g.InventoryBound,"inventory bound "+root.ActorId);var i=m.Items.Single(x=>x.Location.ContainerId==Ossastra.Inventory.InventorySession.HandId(root.ActorId,Ossastra.Inventory.InventoryHand.Right));Check(g.State.RightInstanceId==i.Id&&ReferenceEquals(g.State.RightRuntime,i.Equipment.Runtime),"sole equipment payload "+root.ActorId);}ctx.ReturnValue=new{checks=rows,items=m.Items.Length,result=s.LastResult};}}
