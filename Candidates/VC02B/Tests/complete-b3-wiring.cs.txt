using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class CompleteConsumerWiring:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(Application.isPlaying||scene.path!="Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity")throw new Exception("B edit required");const string folder="Assets/OSSASTRA/VC02B/";var session=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();if(session==null)throw new Exception("Missing session");var serialized=new SerializedObject(session);var array=serialized.FindProperty("catalog");var catalog=Enumerable.Range(0,array.arraySize).Select(n=>array.GetArrayElementAtIndex(n).objectReferenceValue as Ossastra.Characters.ItemDefinitionSO).ToArray();if(catalog.Any(d=>d==null))throw new Exception("Catalog reference missing");
 T Save<T>(T value,string name)where T:UnityEngine.Object{var path=folder+name+".asset";if(AssetDatabase.LoadMainAssetAtPath(path)!=null)throw new Exception("Expected missing before creation: "+path);ctx.RegisterObjectCreation(value);value.name=name;AssetDatabase.CreateAsset(value,path);return value;}
 foreach(var item in catalog.Where(d=>d.useKind==Ossastra.Characters.InventoryUseKind.Regeneration)){if(item.ability!=null)throw new Exception("Unexpected prior regen binding "+item.name);var executor=ScriptableObject.CreateInstance<Ossastra.Characters.RegenerationAbilitySO>();executor.perSecond=item.regenerationPerSecond;executor.seconds=item.regenerationSeconds;Save(executor,item.definitionId+"_RecoveryExecutor");var a=ScriptableObject.CreateInstance<Ossastra.Characters.AbilityDefinitionSO>();a.definitionId=item.definitionId+".use";a.displayName=item.displayName;a.executor=executor;a.occupancy=Ossastra.CharacterCore.Abilities.AbilityOccupancy.None;a.acceptCost=0;a.cooldown=0;a.finishAt=.1f;a.moveReadyAt=a.attackReadyAt=0;a.releases=new[]{new Ossastra.Characters.AbilityReleaseNode{at=.05f,damage=0}};a.Compile();Save(a,item.definitionId+"_Use");ctx.RegisterObjectModification(item);item.ability=a;EditorUtility.SetDirty(item);AssetDatabase.SaveAssetIfDirty(item);}
 var bomb=AssetDatabase.LoadAssetAtPath<Ossastra.Characters.AbilityDefinitionSO>(folder+"sample-al-bomb_Use.asset");if(bomb==null||bomb.definitionId!="sample-al-bomb.use")throw new Exception("Missing previously read-back bomb candidate");bomb.Compile();var bombItem=catalog.Single(d=>d.definitionId=="sample-al-bomb");ctx.RegisterObjectModification(bombItem);bombItem.useKind=Ossastra.Characters.InventoryUseKind.Ability;bombItem.ability=bomb;EditorUtility.SetDirty(bombItem);AssetDatabase.SaveAssetIfDirty(bombItem);
 foreach(var d in catalog){var icon=AssetDatabase.LoadAssetAtPath<Sprite>(folder+"Art/"+d.definitionId+".png");if(icon!=null){ctx.RegisterObjectModification(d);d.icon=icon;EditorUtility.SetDirty(d);AssetDatabase.SaveAssetIfDirty(d);}}
 if(session.GetComponent<Ossastra.Inventory.InventoryItemUse>()!=null)throw new Exception("Unexpected consumer exists");ctx.RegisterObjectModification(session.gameObject);var use=Undo.AddComponent<Ossastra.Inventory.InventoryItemUse>(session.gameObject);var so=new SerializedObject(use);so.FindProperty("session").objectReferenceValue=session;so.ApplyModifiedProperties();foreach(var root in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>()){ctx.RegisterObjectModification(root.gameObject);if(root.GetComponent<Ossastra.Combat.CombatRegeneration>()==null)Undo.AddComponent<Ossastra.Combat.CombatRegeneration>(root.gameObject);}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=new{recovery=catalog.Count(d=>d.useKind==Ossastra.Characters.InventoryUseKind.Regeneration),icons=catalog.Count(d=>d.icon!=null),bomb=bomb.definitionId};}}
