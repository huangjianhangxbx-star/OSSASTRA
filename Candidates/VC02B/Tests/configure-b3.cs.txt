using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class ConfigureInventoryConsumers:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(Application.isPlaying||scene.path!="Assets/OSSASTRA/Scenes/SC_VC02B_IntegrationReady.unity")throw new Exception("B edit scene required");const string folder="Assets/OSSASTRA/VC02B/";
 T Save<T>(T d,string name)where T:UnityEngine.Object{d.name=name;ctx.RegisterObjectCreation(d);AssetDatabase.CreateAsset(d,folder+name+".asset");return d;}
 var catalog=AssetDatabase.FindAssets("t:ItemDefinitionSO",new[]{"Assets/OSSASTRA/VC02B"}).Select(g=>AssetDatabase.LoadAssetAtPath<Ossastra.Characters.ItemDefinitionSO>(AssetDatabase.GUIDToAssetPath(g))).Where(d=>d!=null).ToArray();
 foreach(var item in catalog.Where(d=>d.useKind==Ossastra.Characters.InventoryUseKind.Regeneration)){var executor=ScriptableObject.CreateInstance<Ossastra.Characters.RegenerationAbilitySO>();executor.perSecond=item.regenerationPerSecond;executor.seconds=item.regenerationSeconds;Save(executor,item.definitionId+"_RecoveryExecutor");var ability=ScriptableObject.CreateInstance<Ossastra.Characters.AbilityDefinitionSO>();ability.definitionId=item.definitionId+".use";ability.displayName=item.displayName;ability.executor=executor;ability.occupancy=Ossastra.CharacterCore.Abilities.AbilityOccupancy.None;ability.acceptCost=0;ability.cooldown=0;ability.finishAt=.1f;ability.moveReadyAt=ability.attackReadyAt=0;ability.releases=new[]{new Ossastra.Characters.AbilityReleaseNode{at=.05f,damage=0}};ability.Compile();Save(ability,item.definitionId+"_Use");ctx.RegisterObjectModification(item);item.ability=ability;EditorUtility.SetDirty(item);AssetDatabase.SaveAssetIfDirty(item);}
 var template=AssetDatabase.FindAssets("t:AbilityDefinitionSO",new[]{"Assets/OSSASTRA/Configs/Abilities"}).Select(g=>AssetDatabase.LoadAssetAtPath<Ossastra.Characters.AbilityDefinitionSO>(AssetDatabase.GUIDToAssetPath(g))).Single(d=>d!=null&&d.definitionId=="vc02.throw");var bomb=UnityEngine.Object.Instantiate(template);bomb.definitionId="sample-al-bomb.use";bomb.displayName="实验消耗型炸弹";bomb.acceptCost=0;bomb.releaseCost=0;bomb.wearOnRelease=false;bomb.cooldown=0;var timeline=ScriptableObject.CreateInstance<Ossastra.Characters.TimelineAbilitySO>();var es=new SerializedObject(timeline);es.FindProperty("usesBodyAction").boolValue=false;es.FindProperty("supportAction").boolValue=true;es.FindProperty("allowsCompanionUse").boolValue=true;es.ApplyModifiedPropertiesWithoutUndo();Save(timeline,"ConsumableDeliveryExecutor");bomb.executor=timeline;bomb.Compile();Save(bomb,"sample-al-bomb_Use");var bombItem=catalog.Single(d=>d.definitionId=="sample-al-bomb");ctx.RegisterObjectModification(bombItem);bombItem.useKind=Ossastra.Characters.InventoryUseKind.Ability;bombItem.ability=bomb;bombItem.description="实验消耗型投弹；由芙蕾茜娅后备按原型安全资格使用，消耗实际数量，不磨损装备。";EditorUtility.SetDirty(bombItem);AssetDatabase.SaveAssetIfDirty(bombItem);
 var session=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();if(session.GetComponent<Ossastra.Inventory.InventoryItemUse>()!=null)throw new Exception("Consumer already exists");ctx.RegisterObjectModification(session.gameObject);var consumer=Undo.AddComponent<Ossastra.Inventory.InventoryItemUse>(session.gameObject);var so=new SerializedObject(consumer);so.FindProperty("session").objectReferenceValue=session;so.ApplyModifiedProperties();foreach(var root in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>()){ctx.RegisterObjectModification(root.gameObject);if(root.GetComponent<Ossastra.Combat.CombatRegeneration>()==null)Undo.AddComponent<Ossastra.Combat.CombatRegeneration>(root.gameObject);}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=new{regenerationDefinitions=catalog.Count(d=>d.useKind==Ossastra.Characters.InventoryUseKind.Regeneration),bomb=bomb.definitionId};}}
