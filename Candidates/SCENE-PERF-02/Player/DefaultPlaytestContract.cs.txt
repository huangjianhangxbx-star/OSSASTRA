 static int Run()
 {

  Directory.CreateDirectory(root);var facts=new List<object>();bool pass=false;string error=null;
  try
  {
   string sourceBefore=Fingerprint(ApprovedSource),destination=Path.Combine(root,"playtest-profile");
   Check(ScenePerformanceDefaultPlaytestIsolation.IsMeasurementInvocation(new[]{"exe","--scene-perf02-cpu","run"}),"cpu CLI excluded");

 static void UpdateValidSavedEnvelope(string folder)
 {
  var store=new CommerceFileSaveStore(folder);var loaded=store.Load(Profile);Check(loaded.Status==CommerceSaveLoadStatus.Loaded,"loaded sandbox snapshot");
  var original=loaded.Snapshot;string session=Guid.NewGuid().ToString("N");store.ActivateSession(Profile,session);
  var next=new CommerceSaveSnapshot(Profile,session,original.StateRevision,original.SaveSequence+1,original.Payload,original.PayloadSchema,original.DefinitionFingerprint);
  store.WriteAsync(next,System.Threading.CancellationToken.None).GetAwaiter().GetResult();Check(store.Load(Profile).Status==CommerceSaveLoadStatus.Loaded,"new valid snapshot");
 }
 static void Check(bool value,string label){if(!value)throw new Exception("FAIL: "+label);}
 static void ExpectFailure(Func<bool> action){bool failed=false;try{action();}catch{failed=true;}Check(failed,"expected rejection");}
 static void CopyAll(string source,string destination){foreach(var p in Directory.GetFiles(source))File.Copy(p,Path.Combine(destination,Path.GetFileName(p)),false);}
 static string Fingerprint(string folder)=>string.Join("|",Directory.GetFiles(folder).OrderBy(Path.GetFileName,StringComparer.Ordinal).Select(p=>Path.GetFileName(p)+":"+Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(p)))));
}
