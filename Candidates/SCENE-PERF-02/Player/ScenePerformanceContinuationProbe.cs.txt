  bool NormalStaticWarmReady(double now)
  {
   bool consecutive=normalStaticLastRenderFrame<0||Time.frameCount==normalStaticLastRenderFrame+1;

   normalStaticLastRenderSeconds=now;normalStaticLastRenderFrame=Time.frameCount;

   {
    var a=normalStaticWarmAnchor;
    same=(mainCamera.transform.position-a.position).sqrMagnitude<=1e-8f&&Quaternion.Angle(mainCamera.transform.rotation,a.rotation)<=.001f&&
     Screen.width==a.width&&Screen.height==a.height&&QualitySettings.GetQualityLevel()==a.quality&&QualitySettings.vSyncCount==a.vsync&&Application.targetFrameRate==a.targetFrameRate&&
     QualitySettings.antiAliasing==a.antiAliasing&&QualitySettings.shadows==normalStaticWarmShadowQuality&&Mathf.Abs(QualitySettings.shadowDistance-a.shadowDistance)<1e-6f&&
     (pipeline==null?a.renderScale<0:Mathf.Abs(pipeline.renderScale-a.renderScale)<1e-6f&&pipeline.msaaSampleCount==a.msaa);
    var projection=mainCamera.projectionMatrix;for(int n=0;n<16;n++)if(Mathf.Abs(projection[n]-normalStaticWarmProjection[n])>1e-6f)same=false;
   }

   {

   }

  }

  void StopCpu()
  {
   if(startupProfilerMode)return; // Engine-started stream is owned by its frame limit/process lifetime, never this component.
   if(!profilerBorrowed)return;
   // Successful capture stops in early Update after two whole unmarked drain frames; explicitly close before restoring binary mode.
   try{Profiler.enabled=false;}catch(Exception e){errors.Add("Profiler stop: "+e.Message);}
   try{Profiler.logFile=string.Empty;}catch(Exception e){errors.Add("binary log close: "+e.Message);}
   try{Profiler.enableBinaryLog=previousBinaryLog;}catch(Exception e){errors.Add("binary restore: "+e.Message);}
   try{Profiler.logFile=previousLogFile;}catch(Exception e){errors.Add("log path restore: "+e.Message);}
   try{Profiler.enabled=previousProfilerEnabled;}catch(Exception e){errors.Add("Profiler enabled restore: "+e.Message);}
   profilerBorrowed=false;cpuRecordingStarted=false;
  }

  void Update()
  {
   if(finished||clock==null)return;

   if(clock.Elapsed.TotalSeconds>(cpuMode?60:seconds+60))Finish(false,"total timeout; warm="+warmed+" sampled="+sampled);
  }
  // CPU-only boundary component; normal timing retains its original Update path and execution order.
  internal void ProcessCpuBoundaryUpdate()
  {
   if(!cpuMode||finished||clock==null)return;
   try
   {
    if(cpuStartRequested){cpuStartRequested=false;StartCpu();return;}
    if(!stopPending)return;
    stopWait++;
    if(stopWait==3){report.cpuCaptureStopFrame=Time.frameCount;report.cpuDrainFrameCount=Math.Max(0,Time.frameCount-report.cpuLastMarkedFrame-1);StopCpu();}
    if(stopWait>=6)Finish(true,"requested measurement complete");
   }
   catch(Exception e){Finish(false,"CPU frame-boundary capture: "+e);}
  }
