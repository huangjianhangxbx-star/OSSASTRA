// 功能：唯一构建在EditorApplication.update达到0.5秒门槛后执行；请求先返回，避免HTTP超时拥有同步构建。
// 此文件只作为显式Native候选执行，不是自动启动脚本；实际Build仍沿上面同一份保护与回执。
public sealed class QueueScenePerformance02PlayerBuild : IFunplayCommand
{
 const string queueKey="SCENE-PERF-02.PlayerBuild.Queued";
 const string requestKey="SCENE-PERF-02.PlayerBuild.QueueRequestId";
 const string queuePath="artifacts/SCENE-PERF-02/player-build-queued.json";
 const string latestPath="artifacts/SCENE-PERF-02/player-build-latest.json";
 static string pendingRequestId,pendingRequestedUtc,pendingConfiguration;
 static double pendingNotBefore;
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext context)
 {
  if(EditorApplication.isPlayingOrWillChangePlaymode||EditorApplication.isCompiling||EditorApplication.isUpdating||BuildPipeline.isBuildingPlayer||SessionState.GetBool(queueKey,false)||SessionState.GetBool("SCENE-PERF-02.PlayerBuild.Active",false)||!string.IsNullOrEmpty(SessionState.GetString(requestKey,"")))throw new InvalidOperationException("Stable Edit and no queued/active owned build required");
  if(Application.dataPath.Replace('\\','/')!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new InvalidOperationException("Actual project required");
  if(SceneManager.GetActiveScene().path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new InvalidOperationException("Actual main required");
  pendingConfiguration=ScenePerformanceBuildModes.Requested();
  Directory.CreateDirectory(Path.GetDirectoryName(Path.GetFullPath(queuePath)));
  pendingRequestId=Guid.NewGuid().ToString("N");pendingRequestedUtc=DateTime.UtcNow.ToString("O");
  // 先以本次身份覆盖latest的旧成功状态；preflight失败也不会被误认成上轮可执行Player。
  ScenePerformanceBuildModes.WriteLatest(latestPath,Newtonsoft.Json.JsonConvert.SerializeObject(new{requestId=pendingRequestId,configuration=pendingConfiguration,requestedUtc=pendingRequestedUtc,result="queued",success=false,callbackRan=false},Newtonsoft.Json.Formatting.Indented),pendingConfiguration,false);
  ScenePerformanceBuildModes.WriteLatest(queuePath,Newtonsoft.Json.JsonConvert.SerializeObject(new{requestId=pendingRequestId,configuration=pendingConfiguration,status="queued",callbackRan=false,requestedUtc=pendingRequestedUtc,note="No build has started until callbackRan; match configuration/requestId; do not replay lost response"},Newtonsoft.Json.Formatting.Indented),pendingConfiguration,true);
  pendingNotBefore=EditorApplication.timeSinceStartup+.5;
  SessionState.SetString(requestKey,pendingRequestId);SessionState.SetBool(queueKey,true);AssemblyReloadEvents.beforeAssemblyReload+=Aborted;EditorApplication.quitting+=Aborted;EditorApplication.update+=RunQueuedUpdate;
  EditorApplication.QueuePlayerLoopUpdate();
  context.ReturnValue=new{requestId=pendingRequestId,configuration=pendingConfiguration,status="queued",notBeforeEditorSeconds=pendingNotBefore,queueReceipt=ScenePerformanceBuildModes.Queue(pendingConfiguration),buildReceipt=ScenePerformanceBuildModes.Manifest(pendingConfiguration),note="Same queued build, fixed per-configuration executable and exact receipt identity."};
 }
 static void RunQueuedUpdate()
 {
  if(EditorApplication.timeSinceStartup<pendingNotBefore)return;
  // 到达门槛先撤回本回调，后续无论构建成功/取消/失败都不会在下一次update重放。
  EditorApplication.update-=RunQueuedUpdate;AssemblyReloadEvents.beforeAssemblyReload-=Aborted;EditorApplication.quitting-=Aborted;
  string id=pendingRequestId,requested=pendingRequestedUtc,configuration=pendingConfiguration;
  if(string.IsNullOrEmpty(id)||SessionState.GetString(requestKey,"")!=id)return; // 旧动态程序集的回调无权覆盖新请求。
  SessionState.SetBool(queueKey,false);
  try
  {
   new BuildScenePerformance02MainCopy().Run(id,requested,configuration);
   ScenePerformanceBuildModes.WriteLatest(queuePath,Newtonsoft.Json.JsonConvert.SerializeObject(new{requestId=id,configuration,status="finished-see-build-receipt",callbackRan=true,finishedUtc=DateTime.UtcNow.ToString("O")},Newtonsoft.Json.Formatting.Indented),configuration,true);
  }
  catch(Exception failure)
  {
   try
   {
    // 尽量保留已生成副本/输出路径供Root精确清理；任何异常都显式撤销本请求成功状态。
    Newtonsoft.Json.Linq.JObject state;
    try{state=Newtonsoft.Json.Linq.JObject.Parse(File.ReadAllText(latestPath));}catch{state=new Newtonsoft.Json.Linq.JObject();}
    if((string)state["requestId"]!=id)state=new Newtonsoft.Json.Linq.JObject();
    state["requestId"]=id;state["configuration"]=configuration;state["requestedUtc"]=requested;state["success"]=false;state["result"]="failed-before-or-during-build";state["error"]=failure.ToString();
    ScenePerformanceBuildModes.WriteLatest(latestPath,state.ToString(Newtonsoft.Json.Formatting.Indented),configuration,false);
   }
   finally{ScenePerformanceBuildModes.WriteLatest(queuePath,Newtonsoft.Json.JsonConvert.SerializeObject(new{requestId=id,configuration,status="failed-before-or-during-build",callbackRan=true,error=failure.ToString()},Newtonsoft.Json.Formatting.Indented),configuration,true);}
  }
  finally{ReleaseQueue(id);}
 }
 static void Aborted()
 {
  EditorApplication.update-=RunQueuedUpdate;AssemblyReloadEvents.beforeAssemblyReload-=Aborted;EditorApplication.quitting-=Aborted;
  string id=pendingRequestId;if(string.IsNullOrEmpty(id)||SessionState.GetString(requestKey,"")!=id)return;
  try
  {
   string json=Newtonsoft.Json.JsonConvert.SerializeObject(new{requestId=id,configuration=pendingConfiguration,status="interrupted-before-native-callback",result="interrupted-before-native-callback",success=false,callbackRan=false},Newtonsoft.Json.Formatting.Indented);
   ScenePerformanceBuildModes.WriteLatest(queuePath,json,pendingConfiguration,true);ScenePerformanceBuildModes.WriteLatest(latestPath,json,pendingConfiguration,false);
  }
  finally{ReleaseQueue(id);}
 }
 static void ReleaseQueue(string id)
 {
  try{if(SessionState.GetString(requestKey,"")==id){SessionState.SetBool(queueKey,false);SessionState.EraseString(requestKey);}}
  finally{pendingRequestId=null;pendingRequestedUtc=null;pendingConfiguration=null;pendingNotBefore=0;}
 }
}

public static class ScenePerformanceBuildModes
{
 public static void Validate(string mode){if(mode!="Development"&&mode!="Release")throw new ArgumentException("Explicit Development or Release configuration required");}
 public static string Requested(){const string path="artifacts/SCENE-PERF-02/player-build-request.json";string mode=File.Exists(path)?(string)Newtonsoft.Json.Linq.JObject.Parse(File.ReadAllText(path))["configuration"]:"Development";Validate(mode);return mode;}
 public static string Folder(string mode){Validate(mode);return Path.GetFullPath("artifacts/SCENE-PERF-02/BuildManifests/"+mode);}
 public static string Manifest(string mode)=>Path.Combine(Folder(mode),"build-latest.json");
 public static string Queue(string mode)=>Path.Combine(Folder(mode),"queue-latest.json");
 public static string Executable(string mode){Validate(mode);return Path.GetFullPath(mode=="Development"?"artifacts/SCENE-PERF-02/Player_20261108_094432_2772224/ScenePerformance02Main.exe":"artifacts/SCENE-PERF-02/Release/ScenePerformance02Main.exe");}
 public static void WriteLatest(string global,string json,string mode,bool queue){Directory.CreateDirectory(Folder(mode));File.WriteAllText(global,json);File.WriteAllText(queue?Queue(mode):Manifest(mode),json);}
}
