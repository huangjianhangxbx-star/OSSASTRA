 {if(value==null||value is string||value is bool||value is int||value is float)return value;var count=value.GetType().GetProperty("Count",Instance);return count!=null?(object)new{type=value.GetType().FullName,count=count.GetValue(value,null)}:value.ToString();}

  state=new[]{"Expanded","showPanel","showOverlay","showDiagnostics","IsCapturingInput","IsOverview","hasClockLease","IsTracing","SampleCount","HookedBodyCount","PreviewReady","previewRequests","Surface"}.ToDictionary(n=>n,n=>Summarize(Read(b,n)))};

  var panels=behaviours.Where(b=>b.GetType().Namespace!=null&&(b.GetType().Namespace.StartsWith("Ossastra.Development",StringComparison.Ordinal)||b.GetType().Namespace=="Ossastra.MapResearchIntegration.Diagnostics")||b is Ossastra.MapResearch.MapResearchRoot||b is Ossastra.SceneVisibility.SceneVisibilityRoot||b is Ossastra.EnvironmentSimulation.Rendering.EnvironmentSkyBinding).Select(Panel).ToArray();

  var callbacks=update==null?null:update.GetValue(null) as Delegate;
  var state=new{utc=DateTime.UtcNow.ToString("O"),Application.isPlaying,EditorApplication.isPlayingOrWillChangePlaymode,scene=scene.path,sceneDirty=scene.isDirty,
   width=Screen.width,height=Screen.height,quality=QualitySettings.GetQualityLevel(),QualitySettings.vSyncCount,Application.targetFrameRate,

  File.WriteAllText(path,Newtonsoft.Json.JsonConvert.SerializeObject(state,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=new{path,state};
 }
