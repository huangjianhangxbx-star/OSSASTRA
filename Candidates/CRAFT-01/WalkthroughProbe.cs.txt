    static void Tick()

                    }

                    break;
                case "between-props": if(age>=.7){propIndex++;if(propIndex<PropIds.Length)StartProp();else {NeedMaterials(6,4,0);Enter("collection-hold");}}break;
                case "collection-hold":

                case "open-station":
                    Need(age<5,"原F未打开装备台A。");
                    if(root.userInterface.IsOpen&&age>=.7)

                    break;
                case "crafted-blade-attack":
                    Need(age<8,"新石刃原左键未令实际HP下降。");Need(Hand(InventoryHand.Right)==bladeId,"主手不再是刚制得的实例。");

                    if(target.Health.Current<targetHpBefore){Add("crafted-blade-real-hit",new{bladeId,target=target.instanceId,hpBefore=targetHpBefore,hpAfter=target.Health.Current,releases=actor.ReleaseCount-releaseBefore});Enter("combat-settle");}
                    else if(age>=.35){pointer=Project(target.WorldPosition);mouseButtons=(age-.35)%1.5<.18?(ushort)1:(ushort)0;}

                case "shield-original-right-hold":
                    Need(Hand(InventoryHand.Left)==shieldId,"副手不再是刚制得的实例。");
                    if(age>=.35&&age<1.35)mouseButtons=2;
                    if(Guarding())guardFrames++;
                    if(age>=1.9){Need(guardFrames>0,"原右鼠标hold未观察到实际Guarding。");Add("crafted-shield-original-right-hold",new{shieldId,guardFrames,holdSeconds=1.0});Enter("before-travel");}break;
                case "before-travel":
                    Need(age<6,"世界奖励/原动作未结清。");
                    if(age>=.8&&!actor.ActionActive&&world.PendingCount==0)
                    {
                        Need(root.ReceiptCount==3&&Count("world-stone")==1&&Count("world-fibre")==0&&Count("craft-binding")==0,"跨场前回执/材料结余不符。");

                    break;
                default: throw new InvalidOperationException("未知FSM状态："+stage);
            }
            if(Running&&now-lastTrace>=.3&&trace.Count<310)
            {
                lastTrace=now;trace.Add(new{seconds=now-began,stage,frame=Time.frameCount,actor=actor==null?null:actor.ActorId,position=actor==null?null:actor.transform.position.ToString("R"),receipts=root==null?-1:root.ReceiptCount,craftUI=root!=null&&root.userInterface.IsOpen,inventoryUI=view!=null&&view.IsOpen,controlRevision=control==null?-1:control.ControlRevision,hp=target==null||target.Health==null?-1:target.Health.Current,guarding=actor!=null&&Guarding(),team=Team()});Publish(false);
            }
        }
        catch(Exception ex){error=ex.ToString();Finish(false);}
    }

    static void StartProp()
    {
        Need(!actor.ActionActive&&actor.CanSimulateIntent,"采集摆位前原动作尚未结束或无世界输入资格。");
        target=world.props.Single(p=>p!=null&&p.instanceId==PropIds[propIndex]);Need(target.IsAlive,"采集物已经被耗尽。");
        targetHpBefore=target.Health.Current;releaseBefore=actor.ReleaseCount;
        Place(target.transform.position+Vector3.right*1.1f,"采集"+target.instanceId+"原输入攻击前摆位");pointer=new Vector2(Screen.width*.5f,Screen.height*.5f);Enter("attack-prop");
    }
    static void Enter(string next){keys=Array.Empty<Key>();mouseButtons=0;stage=next;phaseAt=EditorApplication.timeSinceStartup;stageFrame=Time.frameCount;Add("stage",new{name=stage,frame=stageFrame});}
    static void Pulse(Key key,double age,double warm,double repeat=0)
    {double offset=age-warm;if(offset>=0&&(repeat>0?offset%repeat:offset)<.18)keys=new[]{key};}
    static void Place(Vector3 point,string why)
    {
        Need(motor!=null&&!actor.ActionActive,"Teleport夹具必须经原Motor且原动作已结束。");
        Need(Physics.RaycastAll(point+Vector3.up*.4f,Vector3.down,.9f,~0,QueryTriggerInteraction.Ignore).Any(h=>h.normal.y>.7f&&Mathf.Abs(h.point.y-point.y)<.3f&&h.collider.GetComponentInParent<C.CharacterRoot>()==null),"位置夹具缺同层地面，拒绝摆到样地外："+point);
        var before=actor.transform.position;motor.Teleport(point);
        Add("position-fixture",new{why,method="原CharacterMovement.Teleport",before=before.ToString("R"),requested=point.ToString("R"),actual=actor.transform.position.ToString("R")});
    }
    static Vector2 Project(Vector3 point)
    {var p=root.gameplayCamera.WorldToScreenPoint(point);Need(p.z>0&&p.x>=0&&p.x<Screen.width&&p.y>=0&&p.y<Screen.height,"真实目标不在当前GameView中。");return new Vector2(p.x,p.y);}

    {

        {
            if(dragSource!=null&&dragData!=null&&EventSystem.current!=null)
            {
                // 失败拖拽只释放已有UI手势；不在另一个落点重试、丢物或操作账本。

            if(keyboard!=null)InputSystem.ResetDevice(keyboard);if(mouse!=null)InputSystem.ResetDevice(mouse);
        }catch(Exception ex){error=(error??"")+"\nOwn reset: "+ex.Message;passed=false;}
        if(filters!=null)foreach(var filter in filters)

            catch(Exception ex){error=(error??"")+"\nFilter restore: "+ex.Message;passed=false;}
        try{if(keyboard!=null&&InputSystem.devices.Contains(keyboard))InputSystem.RemoveDevice(keyboard);}catch(Exception ex){error=(error??"")+"\nKeyboard removal: "+ex.Message;passed=false;}
        try{if(mouse!=null&&InputSystem.devices.Contains(mouse))InputSystem.RemoveDevice(mouse);}catch(Exception ex){error=(error??"")+"\nMouse removal: "+ex.Message;passed=false;}
        devicesRemoved=(keyboard==null||!InputSystem.devices.Contains(keyboard))&&(mouse==null||!InputSystem.devices.Contains(mouse));keyboard=null;mouse=null;

        Publish(true);
    }
