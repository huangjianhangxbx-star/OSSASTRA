 static string AuthorHash(GroundWorkshopMap map)
 {
  var text=new StringBuilder(map.transform.localToWorldMatrix.ToString("R")).Append('|').Append(map.surface.EditableCells.ToString());
  foreach(var layer in map.surface.layers)
  {
   var tilemap=layer.tilemap;var bounds=tilemap.cellBounds;Need((long)bounds.size.x*bounds.size.y*bounds.size.z<=262144,"Tile作者范围超限");
   text.Append('|').Append(layer.isLocked).Append('|').Append(tilemap.transform.localToWorldMatrix.ToString("R")).Append('|').Append(tilemap.color.ToString("R"));
   foreach(var cell in bounds.allPositionsWithin)
   {
    if(tilemap.GetTile(cell)==null)continue;var sprite=tilemap.GetSprite(cell);string guid="";long local=0;
    if(sprite!=null)Need(AssetDatabase.TryGetGUIDAndLocalFileIdentifier(sprite,out guid,out local),"原作者Sprite未保存");
    text.Append('|').Append(cell.ToString()).Append(':').Append(guid).Append(':').Append(local).Append(':').Append(tilemap.GetColor(cell).ToString("R"));
    text.Append(':').Append((int)tilemap.GetTileFlags(cell)).Append(':').Append(tilemap.GetTransformMatrix(cell).ToString("R"));
   }
  }
  using(var h=SHA256.Create())return Convert.ToBase64String(h.ComputeHash(Encoding.UTF8.GetBytes(text.ToString())));
 }

    foreach(var layer in map.surface.layers){ctx.RegisterObjectModification(layer.tilemap);ctx.RegisterObjectModification(layer.tilemap.GetComponent<TilemapRenderer>());}

    Need(GroundWorkshopOwnership.IsUnder(AssetDatabase.GetAssetPath(cache.colliderMesh),binding.map.registration.assetFolder),"新cap仍指源缓存");
   }
   Need(protectedFiles.All(p=>File.Exists(p.Key)&&Hash(p.Key)==p.Value),"源场或源Map文件改变");sourceSame=true;
