using System;
using UnityEngine;
using UnityEditor;
using Ossastra.Characters;
using Funplay.Editor.Tools.Scripting;
public class TwoActorContract : IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var prefab=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/OSSASTRA/Prefabs/Characters/PlayerCharacter.prefab");if(!Application.isPlaying||prefab==null)throw new Exception("Own Play prefab required");var parent=new GameObject("R1TwoActorProbe");parent.SetActive(false);try{var a=UnityEngine.Object.Instantiate(prefab,parent.transform);var b=UnityEngine.Object.Instantiate(prefab,parent.transform);a.transform.position=new Vector3(-4,.03f,-3);b.transform.position=new Vector3(4,.03f,-3);var ar=a.GetComponent<Ossastra.Characters.CharacterRoot>();var br=b.GetComponent<Ossastra.Characters.CharacterRoot>();var sa=new SerializedObject(ar);sa.FindProperty("actorId").stringValue="probe-a";sa.ApplyModifiedPropertiesWithoutUndo();var sb=new SerializedObject(br);sb.FindProperty("actorId").stringValue="probe-b";sb.ApplyModifiedPropertiesWithoutUndo();parent.SetActive(true);int count=0;Action<bool,string> check=(ok,label)=>{if(!ok)throw new Exception("FAIL "+label);count++;ctx.Log("PASS "+label);};check(ar.DataBus!=null&&br.DataBus!=null&&!ReferenceEquals(ar.DataBus,br.DataBus),"two actual shared prefabs initialize independent buses");check(ar.DataBus.ActorId=="probe-a"&&br.DataBus.ActorId=="probe-b","two roots retain configured identities");var owner=new object();ar.SetAttackRestriction(owner,true,"probe-block");check(!ar.DataBus.AllowsAttack&&br.DataBus.AllowsAttack,"actual root owner restriction does not cross actor");a.GetComponent<CharacterOrientation>().ResolveDirections(Vector2.left,Vector3.right,Vector2.zero,true);check(ar.DataBus.VisualFacingX<0&&br.DataBus.VisualFacingX>0&&ar.DataBus.AttackDirection.x>0,"actual prefab orientation states do not cross actor");ctx.Log("RESULT "+count+"/4 passed");}finally{UnityEngine.Object.DestroyImmediate(parent);}}}
