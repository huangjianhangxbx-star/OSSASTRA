 public class InterfaceContract : IFunplayCommand {
  public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){int passed=0;Action<bool,string> check=(ok,name)=>{if(!ok)throw new Exception("FAIL "+name);passed++;ctx.Log("PASS "+name);};var trace=new List<string>();var source=new InputStub{CurrentFrame=new CharacterInputFrame{move=Vector2.left}};var movement=new MovementStub(trace);var orientation=new OrientationStub();var hit=new HitStub(trace);var presentation=new PresentationStub(trace);var runtime=new BasicActionRuntime("interface-actor");var gate=new ControlAuthority();var bus=new Ossastra.Characters.CharacterDataBus("interface-actor",source,movement,orientation,runtime,gate);
  var initialize=typeof(Ossastra.Characters.CharacterRoot).GetMethod("InitializeSubsystems",BindingFlags.NonPublic|BindingFlags.Static);check(initialize!=null&&(bool)initialize.Invoke(null,new object[]{bus,new Ossastra.Characters.ICharacterSubsystem[]{movement,presentation}})&&string.Join(",",trace)=="movement,presentation","actual Root initialization protocol preserves declared order");
  check(!((object)source is Ossastra.Characters.ICharacterSubsystem) && !((object)hit is Ossastra.Characters.ICharacterSubsystem),"input source has no fake initialization obligation");source.Read();check(source.reads==1&&bus.Input.move==Vector2.left,"replaceable input source is consumed through same bus contract");movement.Tick(true,.1f);check(movement.observed==Vector2.left,"replaceable movement reads bus input");check(bus.AttackDirection==Vector3.left&&bus.VisualFacingX==1,"presentation facing and attack direction are separate interface data");orientation.VisualFacingX=-1;presentation.Present();check(presentation.observed==-1,"presentation replacement sees live orientation owner without cache");
  runtime.Request(new BasicAttackDefinition(.3,.5,.8,1.1,12),"interface",gate);check(bus.ActionActive,"interface substitution retains actual domain runtime authority");var facts=runtime.Advance(.3,false);hit.Resolve(facts[0].Release,Vector3.right);check(hit.observed==Vector3.right&&bus.AttackDirection==Vector3.left,"hit resolver receives explicit release-frame snapshot");check(hit.health.Current==88,"substituted target uses original sole Health settlement");bool duplicateRejected=false;try{hit.Resolve(facts[0].Release,Vector3.left);}catch(InvalidOperationException){duplicateRejected=true;}check(duplicateRejected&&hit.health.Current==88,"substitution cannot resolve a release twice");
  trace.Clear();var ok=(bool)initialize.Invoke(null,new object[]{bus,new Ossastra.Characters.ICharacterSubsystem[]{new FailingSubsystem(trace),movement}});check(!ok&&string.Join(",",trace)=="failure","failed initialization does not run later subsystem");
  var other=new Ossastra.Characters.CharacterDataBus("other",new InputStub{CurrentFrame=new CharacterInputFrame{move=Vector2.right}},new MovementStub(new List<string>()),new OrientationStub(),new BasicActionRuntime("other"),new ControlAuthority());check(other.Input.move==Vector2.right&&bus.Input.move==Vector2.left&&!other.ActionActive,"two interface-backed actors keep state isolated");ctx.Log("RESULT "+passed+"/12 passed");}
 }
