 var roots=scene.GetRootGameObjects();var transforms=roots.SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();var all=roots.SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(x=>x!=null).ToArray();var root=all.OfType<EnvironmentRoot>().Single();var sky=all.OfType<EnvironmentSkyBinding>().Single();var panel=all.OfType<EnvironmentSimulationPanel>().Single();var marker=all.OfType<EnvironmentObservationAuthoring>().Single();var map=all.OfType<Ossastra.MapResearch.MapResearchRoot>().Single();var commerce=all.OfType<Ossastra.Commerce.Runtime.CommerceService>().Single();var visibility=all.OfType<Ossastra.SceneVisibility.SceneVisibilityRoot>().Single();

}}
