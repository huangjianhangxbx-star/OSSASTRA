/* 功能：用正式Native Lit与WorldTile片元验证室外湿润温和响应，以及室内/雨棚/角色/信息排除。
 * 边界：真实主灯/SH有效才认fixture；不改源材质、灯、场景或环境档，不用静态源码匹配冒充受光。
 * 生命周期：仅私有材质、网格、纹理和RT；借用全局参数完整恢复，自有GPU对象finally释放。

using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public class EnvironmentWetnessContract:IFunplayCommand
{
 struct WetPair{public Color Dry,Wet;public bool Alpha=>Mathf.Abs(Dry.a-Wet.a)<.001f;public bool Same=>Mathf.Max(Mathf.Abs(Dry.r-Wet.r),Mathf.Abs(Dry.g-Wet.g),Mathf.Abs(Dry.b-Wet.b))<.001f;public bool GentleDrop=>Dry.r-Wet.r>.01f&&Wet.r/Mathf.Max(Dry.r,.001f)>.90f&&Wet.r/Mathf.Max(Dry.r,.001f)<.995f;}
 static Color Draw(Material material,Mesh mesh,RenderTexture target,Texture2D read)
 {
  var cmd=new CommandBuffer{name="Owned actual native/unlit wetness GPU contract"};
  try
  {
   cmd.SetRenderTarget(target);cmd.SetViewport(new Rect(0,0,target.width,target.height));cmd.ClearRenderTarget(false,true,Color.clear);cmd.SetViewProjectionMatrices(Matrix4x4.identity,Matrix4x4.identity);
   var block=new MaterialPropertyBlock();block.SetVector("unity_LightData",new Vector4(0,0,1,0));var sh=new SphericalHarmonicsL2();sh.AddAmbientLight(new Color(.2f,.2f,.2f));block.CopySHCoefficientArraysFrom(new[]{sh});cmd.DrawMesh(mesh,Matrix4x4.identity,material,0,0,block);Graphics.ExecuteCommandBuffer(cmd);
  }
  finally{cmd.Release();}
  RenderTexture.active=target;read.ReadPixels(new Rect(0,0,target.width,target.height),0,0);read.Apply();return read.GetPixel(target.width/2,target.height/2);
 }
 static WetPair Pair(Material material,Mesh mesh,RenderTexture target,Texture2D read)
 {
  Shader.SetGlobalFloat("_EnvironmentWetness",0);var dry=Draw(material,mesh,target,read);Shader.SetGlobalFloat("_EnvironmentWetness",1);return new WetPair{Dry=dry,Wet=Draw(material,mesh,target,read)};
 }
