/* 功能：把本轮自有新昼夜档收口为可辨认的冷色夜景；不改冻结旧档或业务存档。 */
using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;using Ossastra.EnvironmentSimulation;
public class EnvironmentNightBaseline:IFunplayCommand{public void Execute(ExecutionContext c){
if(Application.isPlaying)throw new Exception("Edit required");const string path="Assets/OSSASTRA/Environment/Profiles/MainDayCycle.asset";var day=AssetDatabase.LoadAssetAtPath<DayCycleProfileSO>(path);if(day==null)throw new Exception("Owned cycle missing");
Undo.RecordObject(day,"Set readable owned moonlit baseline");day.nightSunIntensity=.12f;day.nightAmbientSky=new Color(.15f,.20f,.30f);day.nightAmbientEquator=new Color(.10f,.135f,.20f);day.nightAmbientGround=new Color(.05f,.07f,.10f);day.nightAmbientIntensity=1;EditorUtility.SetDirty(day);AssetDatabase.SaveAssetIfDirty(day);
var root=UnityEngine.Object.FindObjectsOfType<EnvironmentRoot>().Single();if(root.dayCycle!=day)throw new Exception("Main owns unexpected cycle");Undo.RecordObject(root,"Set coherent decoration environment response");root.decorationLightResponse=1;EditorUtility.SetDirty(root);EditorSceneManager.MarkSceneDirty(root.gameObject.scene);c.ReturnValue=JsonConvert.SerializeObject(new{pass=true,path,day.nightSunIntensity,day.nightAmbientIntensity,root.decorationLightResponse});}}
