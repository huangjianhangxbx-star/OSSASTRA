 * Root Native only. Read-only source snapshot -> isolated PreviewScene, no source Renderer toggles.

        static MaterialPropertyBlock WithoutDepth(MaterialPropertyBlock source,Shader shader)
        {
            var result=new MaterialPropertyBlock();
            for(int i=0;i<shader.GetPropertyCount();i++)
            {
                string name=shader.GetPropertyName(i);if(name=="_DepthUnits"||name=="_DepthFactor"||!source.HasProperty(Shader.PropertyToID(name)))continue;
                switch(shader.GetPropertyType(i))
                {
                    case UnityEngine.Rendering.ShaderPropertyType.Color:result.SetColor(name,source.GetColor(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Vector:result.SetVector(name,source.GetVector(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Texture:result.SetTexture(name,source.GetTexture(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Int:result.SetInt(name,source.GetInt(name));break;
                    default:result.SetFloat(name,source.GetFloat(name));break;
                }
            }
            foreach(string name in new[]{"_MainTex_ST","_SecondTex_ST"})if(source.HasVector(name))result.SetVector(name,source.GetVector(name));
            return result;
        }
