        void BuildViews()
        {
            Action<float,float,float,bool> add=(p,y,d,o)=>{if(!views.Any(v=>v.pitch==p&&v.yaw==y&&v.distance==d&&v.ortho==o))views.Add(new View{pitch=p,yaw=y,distance=d,ortho=o,name=(o?"ortho":"persp")+"_p"+p+"_y"+y+"_d"+d});};
            foreach(bool o in new[]{false,true})foreach(float p in new[]{12f,35f,70f})foreach(float d in new[]{5f,25f,32f})add(p,315,d,o);
            if(matrixPreset=="Expanded")foreach(bool o in new[]{false,true})foreach(float p in new[]{12f,35f})foreach(float y in new[]{0f,45f,90f,135f,180f,225f,270f,315f})add(p,y,25,o);
            else if(matrixPreset!="Core")throw new ArgumentException("MatrixPreset must be Core or Expanded.");

        }
        void Prepare(View view)
        {
            float s=view.distance/25f;Vector3 focus=new Vector3(0,planeY,0);Quaternion r=Quaternion.Euler(view.pitch,view.yaw,0),yaw=Quaternion.Euler(0,view.yaw,0);

            ground.transform.SetPositionAndRotation(focus,Quaternion.identity);ground.transform.localScale=new Vector3(14*s,1,14*s);
            front.transform.SetPositionAndRotation(focus+toward*(2*s)-right*(1.5f*s)+Vector3.up*((gap+1)*s),yaw);front.transform.localScale=new Vector3(1.8f*s,2*s,1);
            behind.transform.SetPositionAndRotation(focus-toward*(2*s)+right*(1.5f*s)-Vector3.up*((gap+.85f)*s),yaw);behind.transform.localScale=new Vector3(1.8f*s,1.7f*s,1);
            EditorApplication.QueuePlayerLoopUpdate();

        void Tick()
        {
            try
            {
                double now=EditorApplication.timeSinceStartup;if(now<next)return;if(now-started>120)throw new TimeoutException("Bounded absolute-depth run exceeded 120 seconds.");

                if(stage==0){Prepare(view);stage=1;next=now+.05;return;}

                var standard=Capture(true,true,true,standardGround);var candidateFrame=Capture(true,true,true,candidateGround);
                Analyze(groundOnly,cardsOnly,standard,candidateFrame,view,viewIndex);stage=0;viewIndex++;next=now+.01;Publish("RUNNING",null);
            }
            catch(Exception error){End(error.ToString());}
        }

        }
        static int Kind(Color32 p)
        {if(p.r==255&&p.g==0&&p.b==0)return 1;if(p.r==0&&p.g==255&&p.b==0)return 2;if(p.r==0&&p.g==0&&p.b==255)return 3;return 0;}
        static bool Region(Color32[] a,int x,int y,int kind)
        {for(int dy=-Radius;dy<=Radius;dy++)for(int dx=-Radius;dx<=Radius;dx++)if(Kind(a[(y+dy)*W+x+dx])!=kind)return false;return true;}
        static bool Same(Color32 a,Color32 b)=>a.r==b.r&&a.g==b.g&&a.b==b.b;
        void Analyze(Color32[] onlyGround,Color32[] onlyCards,Color32[] standard,Color32[] actual,View view,int index)
        {
            var mask=new Color32[W*H];int fronts=0,backs=0,frontErrors=0,backErrors=0,interiors=0,errors=0,allErrors=0;var examples=new List<object>();
            for(int p=0;p<standard.Length;p++)if(!Same(standard[p],actual[p]))allErrors++;
            for(int y=Radius;y<H-Radius;y++)for(int x=Radius;x<W-Radius;x++)
            {
                int p=y*W+x,kind=Kind(standard[p]);if(kind==0||!Region(standard,x,y,kind))continue;interiors++;bool wrong=!Same(standard[p],actual[p]);
                if(wrong){errors++;mask[p]=new Color32(255,40,255,255);if(examples.Count<10)examples.Add(new{x,y,expected=new[]{(int)standard[p].r,standard[p].g,standard[p].b},actual=new[]{(int)actual[p].r,actual[p].g,actual[p].b}});}
                bool onPlane=Region(onlyGround,x,y,3);
                if(kind==1&&onPlane&&Region(onlyCards,x,y,1)){fronts++;if(wrong){frontErrors++;mask[p]=new Color32(255,30,30,255);}}
                if(kind==3&&onPlane&&Region(onlyCards,x,y,2)){backs++;if(wrong){backErrors++;mask[p]=new Color32(20,140,255,255);}}
            }
            bool valid=fronts>=Minimum&&backs>=Minimum;everyPoseValid&=valid;
            if(valid){interiorWrong+=errors;frontWrong+=frontErrors;behindWrong+=backErrors;}
            if(saveImages){string prefix=Path.Combine(output,"pose"+index);Save(onlyGround,prefix+"_GroundOnly.png");Save(onlyCards,prefix+"_CardsOnly.png");Save(standard,prefix+"_Standard.png");Save(actual,prefix+"_Candidate.png");Save(mask,prefix+"_ErrorMask.png");}
            rows.Add(new{pose=index,view.name,view.pitch,view.yaw,view.distance,orthographic=view.ortho,frontOverlapSamples=fronts,behindOverlapSamples=backs,minimum=Minimum,valid,
                interiorSamples=interiors,interiorMismatch=errors,frontMismatch=frontErrors,behindMismatch=backErrors,strictAllPixelsMismatch=allErrors,edgeOrBackgroundMismatch=allErrors-errors,examples,
                result=!valid?"UNRESOLVED":errors>0?"RED":"GREEN"});
        }

        Shader CreateReference()
        {
            var methods=typeof(ShaderUtil).GetMethods(BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Static).Where(m=>m.Name=="CreateShaderAsset"&&m.ReturnType==typeof(Shader)).ToArray();
            var method=methods.FirstOrDefault(m=>{var p=m.GetParameters();return p.Length==2&&p[0].ParameterType==typeof(string)&&p[1].ParameterType==typeof(bool);});object[] args={ReferenceSource,true};
            if(method==null){method=methods.FirstOrDefault(m=>{var p=m.GetParameters();return p.Length==1&&p[0].ParameterType==typeof(string);});args=new object[]{ReferenceSource};}
            if(method==null)throw new MissingMethodException("No in-memory CreateShaderAsset entry point.");
            var shader=method.Invoke(null,args) as Shader;if(shader==null||EditorUtility.IsPersistent(shader))throw new InvalidOperationException("Reference shader was not transient.");
            shader.hideFlags=HideFlags.HideAndDontSave;if(!shader.isSupported||ShaderUtil.ShaderHasError(shader)){Object.DestroyImmediate(shader);throw new InvalidOperationException("Opaque reference ShaderHasError/unsupported.");}return shader;
        }
