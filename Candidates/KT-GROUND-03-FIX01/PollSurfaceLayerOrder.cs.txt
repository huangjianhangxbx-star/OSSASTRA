using System;using System.Linq;using System.Reflection;using Newtonsoft.Json.Linq;using Funplay.Editor.Tools.Scripting;
public sealed class PollKT03Fix01LayerOrder:IFunplayCommand
{
    public void Execute(ExecutionContext ctx)
    {
        ctx.ReturnValue=AppDomain.CurrentDomain.GetAssemblies().Select(a=>a.GetType("KT03Fix01LayerOrderRedContracts",false)).Where(t=>t!=null).Select(t=>{
            var value=(string)t.GetField("ResultJson",BindingFlags.Public|BindingFlags.Static).GetValue(null);
            return new{assembly=t.Assembly.FullName,module=t.Module.ModuleVersionId.ToString("N"),result=value=="NOT_STARTED"?new JObject(new JProperty("status",value)):JObject.Parse(value)};
        }).ToArray();
    }
}
