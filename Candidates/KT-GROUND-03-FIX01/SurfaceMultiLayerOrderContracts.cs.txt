 * Root Native only. Read-only source snapshot -> isolated PreviewScene, no source Renderer toggles.
 * A pixel is tested only inside both 5x5 opaque masks and with RGB layer contrast >=20.
 * Same-pose Prefix[k] must show Single[k]; removing it must reveal the independently covered lower layer.

        void BuildViews()
        {
            Action<float,float,float,float> add=(pitch,yaw,distance,dx)=>{
                if(views.Any(v=>v.pitch==pitch&&v.yaw==yaw&&v.distance==distance&&v.dx==dx))return;
                views.Add(new View{pitch=pitch,yaw=yaw,distance=distance,dx=dx,name="p"+pitch+"_y"+yaw+"_d"+distance+"_x"+dx});};
            Action<float,float,float> micro=(pitch,yaw,distance)=>{foreach(float dx in new[]{-.001f,0,.001f})add(pitch,yaw,distance,dx);};
            if(matrixPreset=="TwoViews"){micro(35,315,25);micro(12,315,32);}
            else if(matrixPreset=="Octants"){foreach(float p in new[]{12f,35f})foreach(float y in new[]{0f,45f,90f,135f,180f,225f,270f,315f})add(p,y,25,0);}
            else if(matrixPreset=="Neighborhood"){foreach(float p in new[]{12f,35f})foreach(float y in new[]{314.9f,315f,315.1f})micro(p,y,25);}
            else if(matrixPreset=="PitchDistance"){foreach(float p in new[]{12f,35f})foreach(float d in new[]{5f,25f,32f})micro(p,315,d);}
            else if(matrixPreset=="Bounded")
            {
                foreach(float p in new[]{12f,35f})
                {
                    foreach(float d in new[]{5f,25f,32f})add(p,315,d,0);
                    foreach(float y in new[]{0f,45f,90f,135f,180f,225f,270f,315f,314.9f,315.1f})add(p,y,25,0);
                    micro(p,315,25);
                }
            }
            else throw new ArgumentException("Unknown bounded MatrixPreset.");
            if(views.Count>32)throw new InvalidOperationException("View matrix exceeds its 32-pose budget per surface.");
        }
        static bool[] Strong(Color32[] image)
        {
            var integral=new int[(Width+1)*(Height+1)];int stride=Width+1;
            for(int y=0;y<Height;y++){int sum=0;for(int x=0;x<Width;x++){if(image[y*Width+x].a>=250)sum++;integral[(y+1)*stride+x+1]=integral[y*stride+x+1]+sum;}}
            var strong=new bool[Width*Height];int expected=(Radius*2+1)*(Radius*2+1);
            for(int y=Radius;y<Height-Radius;y++)for(int x=Radius;x<Width-Radius;x++)
            {int x0=x-Radius,x1=x+Radius+1,y0=y-Radius,y1=y+Radius+1;strong[y*Width+x]=integral[y1*stride+x1]-integral[y0*stride+x1]-integral[y1*stride+x0]+integral[y0*stride+x0]==expected;}
            return strong;
        }
        void AnalyzeMulti(Color32[] empty,Color32[][] singles,Color32[][] prefixes,int[] keys,View view,int index)
        {
            bool clearValid=empty.All(c=>c.a==0);var strong=singles.Select(Strong).ToArray();var mask=new Color32[Width*Height];
            var layerRows=new List<object>();int poseSamples=0,poseWrong=0,poseAmbiguous=0,testedLayers=0,visibleHighLayers=0,worst=1;long worstScore=-1;bool allVisibleLayersCovered=true;
            for(int high=1;high<keys.Length;high++)
            {
                int samples=0,badHigh=0,badRemoval=0,unknown=0,highOpaque=strong[high].Count(v=>v);var examples=new List<object>();var lowerWins=new Dictionary<int,int>();
                if(highOpaque>=MinSamples)visibleHighLayers++;
                if(clearValid)for(int p=0;p<Width*Height;p++)
                {
                    if(!strong[high][p])continue;int lower=high-1;while(lower>=0&&singles[lower][p].a<250)lower--;
                    if(lower<0||!strong[lower][p]||Distance(singles[high][p],singles[lower][p])<Contrast)continue;samples++;
                    int highError=Distance(prefixes[high][p],singles[high][p]);int lowerError=Distance(prefixes[high-1][p],singles[lower][p]);
                    bool wrongHigh=false,wrongRemoved=false;int observed=-1,removedObserved=-1;
                    if(highError>Tolerance)
                    {
                        int best=int.MaxValue;for(int j=0;j<high;j++)if(singles[j][p].a>=250){int d=Distance(prefixes[high][p],singles[j][p]);if(d<best){best=d;observed=j;}}
                        if(best<=Tolerance){badHigh++;wrongHigh=true;int layer=keys[observed];lowerWins[layer]=lowerWins.TryGetValue(layer,out int n)?n+1:1;mask[p]=new Color32(255,30,30,255);}else{unknown++;mask[p]=new Color32(30,100,255,255);}
                    }
                    if(lowerError>Tolerance)
                    {
                        int best=int.MaxValue;for(int j=0;j<lower;j++)if(singles[j][p].a>=250){int d=Distance(prefixes[high-1][p],singles[j][p]);if(d<best){best=d;removedObserved=j;}}
                        if(best<=Tolerance){badRemoval++;wrongRemoved=true;mask[p]=new Color32(255,30,255,255);}else{unknown++;if(!wrongHigh)mask[p]=new Color32(30,100,255,255);}
                    }
                    if((wrongHigh||wrongRemoved)&&examples.Count<8)examples.Add(new{x=p%Width,y=p/Width,expectedHigh=keys[high],expectedAfterRemoval=keys[lower],observedHigh=observed<0?(int?)null:keys[observed],observedAfterRemoval=removedObserved<0?(int?)null:keys[removedObserved],
                        singleHigh=Rgba(singles[high][p]),singleLower=Rgba(singles[lower][p]),prefixHigh=Rgba(prefixes[high][p]),afterRemoval=Rgba(prefixes[high-1][p])});
                }
                bool valid=clearValid&&samples>=MinSamples;if(valid){testedLayers++;coveredLayers.Add(keys[high]);}if(highOpaque>=MinSamples&&!valid)allVisibleLayersCovered=false;
                poseSamples+=samples;poseWrong+=valid?badHigh+badRemoval:0;poseAmbiguous+=unknown;
                long score=(long)(badHigh+badRemoval)*100000+samples;if(score>worstScore){worstScore=score;worst=high;}
                layerRows.Add(new{highLayer=keys[high],highStrongOpaque=highOpaque,samples,valid,wrongPrefixWinner=badHigh,wrongAfterRemovalWinner=badRemoval,unresolvedColorDifference=unknown,lowerWins,examples,
                    coverage=valid?"TESTED":highOpaque<MinSamples?"INSUFFICIENT_VISIBLE_INTERIOR":"NO_SUFFICIENT_CONTRAST_OR_INTERSECTION"});
            }
            bool poseValid=clearValid&&poseSamples>=MinSamples&&testedLayers>0&&allVisibleLayersCovered;everyPoseValid&=poseValid;
            // A proven wrong winner in a sufficiently sampled layer remains RED even when another layer has no usable coverage.
            tested+=poseSamples;wrong+=poseWrong;validatedWrong+=poseWrong;ambiguous+=poseAmbiguous;
            if(saveImages)
            {
                string prefix=Path.Combine(output,"pose"+index);Save(prefixes.Last(),prefix+"_Full.png");Save(mask,prefix+"_ErrorMask.png");
                Save(singles[worst],prefix+"_SingleHigh"+keys[worst]+".png");Save(prefixes[worst],prefix+"_PrefixHigh"+keys[worst]+".png");Save(prefixes[worst-1],prefix+"_RemovedHigh"+keys[worst]+".png");
                var lowerOracle=new Color32[Width*Height];for(int p=0;p<lowerOracle.Length;p++){int j=worst-1;while(j>=0&&singles[j][p].a<250)j--;if(j>=0)lowerOracle[p]=singles[j][p];}Save(lowerOracle,prefix+"_ExpectedAfterRemoval.png");
            }
            rows.Add(new{pose=index,view.name,view.pitch,view.yaw,view.distance,view.dx,sourceSurface=surfaceChoice,activeLayers=keys,position=new[]{camera.transform.position.x,camera.transform.position.y,camera.transform.position.z},
                clearAlphaVerified=clearValid,poseSamples,poseWrong,poseAmbiguous,testedLayers,visibleHighLayers,allVisibleLayersCovered,poseValid,layerRows,
                result=poseWrong>0?"RED":!poseValid||poseAmbiguous>0?"UNRESOLVED":"GREEN"});
        }

        static MaterialPropertyBlock WithoutDepth(MaterialPropertyBlock source,Shader shader)
        {
            var result=new MaterialPropertyBlock();
            for(int i=0;i<shader.GetPropertyCount();i++)
            {
                string name=shader.GetPropertyName(i);if(name=="_DepthUnits"||name=="_DepthFactor"||!source.HasProperty(Shader.PropertyToID(name)))continue;
                switch(shader.GetPropertyType(i))
                {
                    case UnityEngine.Rendering.ShaderPropertyType.Color:result.SetColor(name,source.GetColor(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Vector:result.SetVector(name,source.GetVector(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Texture:result.SetTexture(name,source.GetTexture(name));break;
                    case UnityEngine.Rendering.ShaderPropertyType.Int:result.SetInt(name,source.GetInt(name));break;
                    default:result.SetFloat(name,source.GetFloat(name));break;
                }
            }
            foreach(string name in new[]{"_MainTex_ST","_SecondTex_ST"})if(source.HasVector(name))result.SetVector(name,source.GetVector(name));
            return result;
        }
