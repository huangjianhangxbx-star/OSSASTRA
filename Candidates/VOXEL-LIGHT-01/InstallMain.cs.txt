using System;
using System.Linq;
using System.IO;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
using Ossastra.WorldRendering.VoxelSurface;
using Ossastra.SceneVisibility.Rooms.Rendering;
using Ossastra.EnvironmentSimulation;
using Ossastra.Development.Environment;
using Funplay.Editor.Tools.Scripting;

public sealed class VoxelInstallMain : IFunplayCommand
{
 const string Root="Assets/OSSASTRA/WorldRendering/VoxelSurface";
 public void Execute(ExecutionContext ctx)
 {
  var scene=SceneManager.GetActiveScene();
  if(EditorApplication.isPlaying||scene.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new Exception("wrong edit scene");
  var author=EditorUtility.InstanceIDToObject(123452) as EnvironmentObservationAuthoring;
  var visibility=EditorUtility.InstanceIDToObject(123140) as Ossastra.SceneVisibility.SceneVisibilityRoot;
  var renderer=AssetDatabase.LoadAssetAtPath<UniversalRendererData>("Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/Rendering/RoomVisibilityRenderer.asset");
  var feature=renderer.rendererFeatures.OfType<RoomMaskRendererFeature>().Single();
  var shader=Shader.Find("Ossastra/WorldRendering/VoxelLit");var legal=Shader.Find("Ossastra/WorldRendering/SightVoxelLit");
  if(author==null||visibility==null||author.observationRoot==null||shader==null||legal==null)throw new Exception("missing audited source");
  if(scene.GetRootGameObjects().Any(g=>g.GetComponentsInChildren<VoxelSurfaceComparison>(true).Length>0))throw new Exception("already installed: inspect instead of replay");
  Folder(Root+"/Profiles");Folder(Root+"/Materials");Folder(Root+"/Geometry");
  var selected=new[]{120324,121644,120836,120600,121654,123236,120854,122570,122088,120174,121580,120454,120268,122988,122616,120334,121472,122718}.Select(id=>EditorUtility.InstanceIDToObject(id) as Renderer).ToList();
  if(selected.Any(r=>r==null||r.gameObject.scene!=scene||r.sharedMaterials.Any(m=>m==null||m.shader.name!="Universal Render Pipeline/Lit")))throw new Exception("audited targets changed");
  var snapshot=string.Join("\n",selected.Select(r=>PathOf(r.transform)+" "+r.GetInstanceID()+" "+string.Join(";",r.sharedMaterials.Select(m=>AssetDatabase.GetAssetPath(m)+" "+EditorJsonUtility.ToJson(m)))));
  File.WriteAllText("E:/WORLDCREATOR/XingHaiHuiLang/研究/VOXEL-LIGHT-01/original-selected-materials.txt",snapshot);
  ctx.RegisterObjectModification(feature);feature.positiveVoxelLitShader=legal;EditorUtility.SetDirty(feature);AssetDatabase.SaveAssetIfDirty(feature);
  var b=ScriptableObject.CreateInstance<VoxelSurfaceProfileSO>();b.name="Voxel_CalibratedPBR";b.settings.indirectDiffuse=.65f;b.settings.environmentSpecular=.25f;
  var c=ScriptableObject.CreateInstance<VoxelSurfaceProfileSO>();c.name="Voxel_FacetedCandidate";c.settings=b.settings.Clone();c.settings.mainShape=.65f;
  AssetDatabase.CreateAsset(b,Root+"/Profiles/CalibratedPBR.asset");AssetDatabase.CreateAsset(c,Root+"/Profiles/FacetedCandidate.asset");
  var group=new GameObject("VOXEL-LIGHT-01_显式材质对照");ctx.RegisterObjectCreation(group);group.transform.SetParent(author.observationRoot,false);
  // Four combined hard-edge voxel meshes, same geometry and palette in all comparison modes.
  var primitive=GameObject.CreatePrimitive(PrimitiveType.Cube);var cube=primitive.GetComponent<MeshFilter>().sharedMesh;
  try
  {
   for(int role=0;role<4;role++)
   {
    var blocks=new List<CombineInstance>();
    if(role==0)for(int x=0;x<4;x++)for(int y=0;y<3;y++)blocks.Add(Block(cube,new Vector3(x*.52f,y*.52f,0),Vector3.one*.5f));
    if(role==1)for(int x=0;x<4;x++)for(int y=0;y<=x;y++)blocks.Add(Block(cube,new Vector3(x*.52f,y*.34f,0),new Vector3(.5f,.32f,1.1f)));
    if(role==2)for(int y=0;y<5;y++)blocks.Add(Block(cube,new Vector3(0,y*.45f,0),new Vector3(.55f,.43f,.55f)));
    if(role==3)for(int y=0;y<3;y++)for(int x=-1;x<=1;x++)for(int z=-1;z<=1;z++)if(y<2||x*x+z*z<2)blocks.Add(Block(cube,new Vector3(x*.48f,y*.4f,z*.48f),new Vector3(.46f,.38f,.46f)));
    var mesh=new Mesh{name="VoxelHardEdges_"+role};mesh.CombineMeshes(blocks.ToArray(),true,true);mesh.RecalculateBounds();AssetDatabase.CreateAsset(mesh,Root+"/Geometry/Fixture_"+role+".asset");
    var source=new Material(Shader.Find("Universal Render Pipeline/Lit")){name=new[]{"VoxelGrayWall","VoxelLightStoneSteps","VoxelDarkWood","VoxelGreenCanopy"}[role]};source.SetColor("_BaseColor",new[]{new Color(.48f,.5f,.52f),new Color(.68f,.62f,.48f),new Color(.25f,.17f,.1f),new Color(.17f,.44f,.25f)}[role]);source.SetFloat("_Metallic",0);source.SetFloat("_Smoothness",.12f);AssetDatabase.CreateAsset(source,Root+"/Materials/"+source.name+"_Original.mat");
    var go=new GameObject(source.name,typeof(MeshFilter),typeof(MeshRenderer));ctx.RegisterObjectCreation(go);go.transform.SetParent(group.transform,false);go.transform.localPosition=new Vector3(-4+role*2.7f,.3f,6.3f);go.GetComponent<MeshFilter>().sharedMesh=mesh;var mr=go.GetComponent<MeshRenderer>();mr.sharedMaterial=source;selected.Add(mr);
   }
  }
  finally{UnityEngine.Object.DestroyImmediate(primitive);}
  var module=visibility.roomModule;ctx.RegisterObjectModification(module);var added=selected.Where(r=>r.transform.IsChildOf(group.transform)).ToArray();module.sharedSurfaces=module.sharedSurfaces.Concat(added).Distinct().ToArray();module.staticEnvironmentSurfaces=module.staticEnvironmentSurfaces.Concat(added).Distinct().ToArray();EditorUtility.SetDirty(module);
  var materialMap=new Dictionary<Material,Material>();
  foreach(var r in selected)foreach(var src in r.sharedMaterials)if(!materialMap.ContainsKey(src))
  {
   var dest=new Material(src){name="Voxel_"+src.name,shader=shader}; // Explicit source properties/keywords retained.
   new VoxelSurfaceSettings().ApplyTo(dest);AssetDatabase.CreateAsset(dest,Root+"/Materials/"+dest.name+".mat");materialMap[src]=dest;
  }
  var comparison=group.AddComponent<VoxelSurfaceComparison>();comparison.calibratedProfile=b;comparison.shapedProfile=c;comparison.mode=VoxelSurfaceComparisonMode.Calibrated;
  comparison.bindings=selected.Select(r=>new VoxelSurfaceComparison.Binding{renderer=r,original=r.sharedMaterials,calibrated=r.sharedMaterials.Select(m=>materialMap[m]).ToArray(),shaped=r.sharedMaterials.Select(m=>materialMap[m]).ToArray()}).ToArray();EditorUtility.SetDirty(comparison);
  EditorSceneManager.MarkSceneDirty(scene);ctx.Log("Installed explicit comparison: "+selected.Count+" renderers, "+materialMap.Count+" author materials, four combined voxel fixtures. Edit source remains Original; Play starts Calibrated. Camera/lights/2D unchanged. Main not saved yet.");
 }
 static string PathOf(Transform t)=>t.parent==null?t.name:PathOf(t.parent)+"/"+t.name;
 static CombineInstance Block(Mesh m,Vector3 p,Vector3 s)=>new CombineInstance{mesh=m,transform=Matrix4x4.TRS(p,Quaternion.identity,s)};
 static void Folder(string p){if(AssetDatabase.IsValidFolder(p))return;Folder(p.Substring(0,p.LastIndexOf('/')));AssetDatabase.CreateFolder(p.Substring(0,p.LastIndexOf('/')),p.Substring(p.LastIndexOf('/')+1));}
}
