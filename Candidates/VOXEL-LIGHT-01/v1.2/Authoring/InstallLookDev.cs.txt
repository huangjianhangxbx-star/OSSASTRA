/* 功能：当前主场增量创建可复现的三槽体素校准小样与独立材质/Look资产。
 * 边界：只新增本轮登记对象和资产；原22绑定/灯/地面/2D/全场Look不改。Root唯一Native执行。
 * 对照：同一几何/法线/三槽空间分配用于M0/M1，不按面朝向给固有色上色。
 */
using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine.SceneManagement;
using Ossastra.WorldRendering.VoxelSurface;
using Ossastra.SceneVisibility.Rooms;
using Funplay.Editor.Tools.Scripting;

public sealed class InstallVoxelLookDev : IFunplayCommand
{
    const string Root = "Assets/OSSASTRA/WorldRendering/VoxelSurface/LookDev";
    struct Block { public Vector3 position, scale; public int variant; }
    readonly Dictionary<Material,int> roles = new Dictionary<Material,int>();
    readonly Dictionary<Material,int> variants = new Dictionary<Material,int>();
    readonly List<VoxelSurfaceComparison.Binding> added = new List<VoxelSurfaceComparison.Binding>();
    VoxelSurfaceComparison comparison;
    Shader voxelShader, originalShader;
    public void Execute(ExecutionContext ctx)
    {
        var scene = SceneManager.GetActiveScene();
        if (EditorApplication.isPlaying || scene.path != "Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity" || scene.isDirty) throw new Exception("Requires protected clean main in Edit");
        comparison = UnityEngine.Object.FindObjectOfType<VoxelSurfaceComparison>();
        var rooms = UnityEngine.Object.FindObjectOfType<RoomVisibilityModule>();
        voxelShader = Shader.Find("Ossastra/WorldRendering/VoxelLit"); originalShader = Shader.Find("Universal Render Pipeline/Lit");
        if (comparison == null || comparison.bindings.Length != 22 || rooms == null || voxelShader == null || originalShader == null) throw new Exception("Expected original22 bindings and existing room backend");
        if (AssetDatabase.IsValidFolder(Root)) throw new Exception("LookDev exists; inspect, never replay installer");
        Folder(Root); Folder(Root+"/Materials"); Folder(Root+"/Profiles"); Folder(Root+"/Looks"); Folder(Root+"/Samples");
        ctx.RegisterObjectModification(comparison); ctx.RegisterObjectModification(rooms);
        var live = Profile("OriginalCLive", comparison.SessionSettings);
        comparison.originalCLiveProfile = live;
        var response = live.settings.Clone(); response.mainShape=.45f; response.litFaceBoost=.15f; response.backFaceDarkening=0;
        var r1 = Profile("CalibratedResponse", response);
        var photo = live.settings.Clone(); photo.litFaceBoost=.5f; photo.backFaceDarkening=0;
        Profile("Screenshot50Reference_UnsavedTrial",photo);
        // 角色与大底板不在这个显式source表；前22槽顺序与作者引用完整保留。
        int[] stone = {0,1,2,3,4,5,6,7,8,18,19};
        foreach(int index in stone) foreach(var source in comparison.bindings[index].shaped) Tag(source,0,1);
        foreach(int index in new[]{13,20}) foreach(var source in comparison.bindings[index].shaped) Tag(source,1,1);
        foreach(int index in new[]{12,21}) foreach(var source in comparison.bindings[index].shaped) Tag(source,2,1);
        var group = new GameObject("VoxelMaterialLookDev_SameGeometry"); ctx.RegisterObjectCreation(group);
        group.transform.SetParent(comparison.transform,false); group.transform.localPosition=new Vector3(0,.08f,10.4f);
        var primitive=GameObject.CreatePrimitive(PrimitiveType.Cube);
        try
        {
            Mesh cube=primitive.GetComponent<MeshFilter>().sharedMesh;
            Sample(ctx,group.transform,"StoneSteps",new Vector3(-4,0,0),0,new Color(.43f,.47f,.43f),BuildSteps(),cube);
            Sample(ctx,group.transform,"CorniceWall",new Vector3(-.8f,0,0),0,new Color(.48f,.50f,.52f),BuildWall(),cube);
            Sample(ctx,group.transform,"DeepWood",new Vector3(2.4f,0,0),1,new Color(.25f,.17f,.1f),BuildWood(),cube);
            Sample(ctx,group.transform,"LeafCluster",new Vector3(4,0,0),2,new Color(.17f,.44f,.25f),BuildLeaves(),cube);
            Sample(ctx,group.transform,"NeutralWitness",new Vector3(.2f,0,-2.1f),-1,new Color(.5f,.5f,.5f),new List<Block>{B(Vector3.up*.28f,new Vector3(1.3f,.5f,1),0)},cube);
        }
        finally { UnityEngine.Object.DestroyImmediate(primitive); }
        comparison.bindings=comparison.bindings.Concat(added).ToArray();
        var surfaces=added.Select(b=>b.renderer).ToArray();
        rooms.sharedSurfaces=(rooms.sharedSurfaces??Array.Empty<Renderer>()).Concat(surfaces).Distinct().ToArray();
        rooms.staticEnvironmentSurfaces=(rooms.staticEnvironmentSurfaces??Array.Empty<Renderer>()).Concat(surfaces).Distinct().ToArray();
        var clearMap=Palette("Clear",new Color(.74f,.70f,.60f),new Color(.27f,.19f,.12f),new Color(.24f,.38f,.15f));
        var neutralMap=Palette("Neutral",new Color(.70f,.71f,.67f),new Color(.27f,.19f,.12f),new Color(.24f,.38f,.15f));
        var sandMap=Palette("Sand",new Color(.75f,.66f,.50f),new Color(.27f,.19f,.12f),new Color(.24f,.38f,.15f));
        var richMap=Palette("Rich",new Color(.68f,.61f,.47f),new Color(.28f,.18f,.10f),new Color(.21f,.36f,.13f));
        var clear=Look("ClearRecommended","清透推荐",r1,clearMap,false);
        var rich=Look("RichCandidate","浓厚候选",r1,richMap,false);
        var colorOnly=Look("M1R0_ColorOnly","开发·仅配色",live,clearMap,true);
        var responseOnly=Look("M0R1_ResponseOnly","开发·仅响应",r1,roles.Keys.ToDictionary(m=>m,m=>m),true);
        var neutral=Look("NeutralColorTrial","开发·中性浅灰",live,neutralMap,true);
        var sand=Look("SandColorTrial","开发·暖砂",live,sandMap,true);
        var savedResponse=Profile("SavedCandidateResponse",response);
        var target=Look("SavedCandidate","已存独立候选",savedResponse,clearMap,false);
        comparison.lookPresets=new[]{clear,rich,colorOnly,responseOnly,neutral,sand}; comparison.saveLookTarget=target; comparison.defaultLook=null;
        EditorUtility.SetDirty(comparison); EditorUtility.SetDirty(rooms);
        EditorSceneManager.MarkSceneDirty(scene);
        if(!EditorSceneManager.SaveScene(scene))throw new Exception("Could not save explicit lookdev author scene");
        ctx.ReturnValue=new{scene=scene.path,bindings=comparison.bindings.Length,added=added.Count,sourceMaterials=roles.Count,originalMode=comparison.mode.ToString(),originalCLive=JsonUtility.ToJson(live.settings),r1=JsonUtility.ToJson(response),assets=Root,groupPosition=group.transform.position};
    }
    void Tag(Material source,int role,int variant){if(source==null)throw new Exception("missing role source");if(roles.TryGetValue(source,out int old)&&old!=role)throw new Exception("conflicting role");roles[source]=role;variants[source]=variant;}
    static VoxelSurfaceProfileSO Profile(string name,VoxelSurfaceSettings value){var p=ScriptableObject.CreateInstance<VoxelSurfaceProfileSO>();p.name=name;p.settings=value.Clone();p.settings.Sanitize();p.settings.diagnostic=0;AssetDatabase.CreateAsset(p,Root+"/Profiles/"+name+".asset");AssetDatabase.SaveAssetIfDirty(p);return p;}
    static VoxelSurfaceLookSO Look(string name,string label,VoxelSurfaceProfileSO response,Dictionary<Material,Material> map,bool development)
    {var look=ScriptableObject.CreateInstance<VoxelSurfaceLookSO>();look.name=name;look.label=label;look.developmentOnly=development;look.responseProfile=response;look.materials=map.Select(p=>new VoxelSurfaceLookSO.MaterialOverride{source=p.Key,replacement=p.Value}).ToArray();AssetDatabase.CreateAsset(look,Root+"/Looks/"+name+".asset");AssetDatabase.SaveAssetIfDirty(look);return look;}
    Dictionary<Material,Material> Palette(string name,Color stone,Color wood,Color leaf)
    {
        Folder(Root+"/Materials/"+name);var result=new Dictionary<Material,Material>();
        foreach(var p in roles){var source=p.Key;Color color=p.Value==0?stone:p.Value==1?wood:leaf;float shift=(variants[source]-1)*(p.Value==0?.025f:.015f);color.r+=shift;color.g+=shift;color.b+=shift;
            var mat=new Material(source){name=name+"_"+source.name};mat.SetColor("_BaseColor",color);if(mat.HasProperty("_Color"))mat.SetColor("_Color",color);AssetDatabase.CreateAsset(mat,Root+"/Materials/"+name+"/"+mat.name+".mat");AssetDatabase.SaveAssetIfDirty(mat);result.Add(source,mat);}
        return result;
    }
    void Sample(ExecutionContext ctx,Transform parent,string name,Vector3 position,int role,Color color,List<Block> blocks,Mesh cube)
    {
        int count=role<0?1:3;var parts=new List<Mesh>();var merged=new List<CombineInstance>();
        try
        {
            for(int variant=0;variant<count;variant++)
            {var temp=new Mesh();parts.Add(temp);temp.CombineMeshes(blocks.Where(b=>count==1||b.variant==variant).Select(b=>new CombineInstance{mesh=cube,transform=Matrix4x4.TRS(b.position,Quaternion.identity,b.scale)}).ToArray(),true,true);merged.Add(new CombineInstance{mesh=temp,transform=Matrix4x4.identity});}
            var mesh=new Mesh{name="LookDev_"+name};mesh.CombineMeshes(merged.ToArray(),false,true);mesh.RecalculateBounds();AssetDatabase.CreateAsset(mesh,Root+"/Samples/"+name+".asset");
            var originals=new Material[count];var shaped=new Material[count];
            for(int v=0;v<count;v++)
            {string matName=name+"_M0_"+v;var a=new Material(originalShader){name=matName};a.SetColor("_BaseColor",color);a.SetFloat("_Metallic",0);a.SetFloat("_Smoothness",.12f);AssetDatabase.CreateAsset(a,Root+"/Materials/"+matName+".mat");var c=new Material(a){name=name+"_CSource_"+v,shader=voxelShader};new VoxelSurfaceSettings().ApplyTo(c);AssetDatabase.CreateAsset(c,Root+"/Materials/"+c.name+".mat");originals[v]=a;shaped[v]=c;if(role>=0)Tag(c,role,v);}
            var go=new GameObject("LookDev_"+name,typeof(MeshFilter),typeof(MeshRenderer));ctx.RegisterObjectCreation(go);go.transform.SetParent(parent,false);go.transform.localPosition=position;go.GetComponent<MeshFilter>().sharedMesh=mesh;var renderer=go.GetComponent<MeshRenderer>();renderer.sharedMaterials=originals;
            added.Add(new VoxelSurfaceComparison.Binding{renderer=renderer,original=originals,calibrated=shaped,shaped=shaped});
        }
        finally{foreach(var m in parts)UnityEngine.Object.DestroyImmediate(m);}
    }
    static List<Block> BuildSteps(){var a=new List<Block>();for(int x=0;x<4;x++)for(int y=0;y<=x;y++)a.Add(B(new Vector3(x*.52f,y*.32f+.18f,0),new Vector3(.50f,.30f,1.60f),(x+y*2)%3));return a;}
    static List<Block> BuildWall(){var a=new List<Block>();for(int x=0;x<4;x++)for(int y=0;y<3;y++)if(!(y==1&&(x==1||x==2)))a.Add(B(new Vector3(x*.52f,y*.52f+.28f,0),new Vector3(.5f,.5f,.5f),(x+2*y)%3));a.Add(B(new Vector3(.78f,.80f,.17f),new Vector3(1.03f,.46f,.15f),1));for(int x=0;x<5;x++)a.Add(B(new Vector3(-.13f+x*.46f,1.70f,0),new Vector3(.44f,.19f,.72f),x%3));return a;}
    static List<Block> BuildWood(){var a=new List<Block>();for(int y=0;y<5;y++)a.Add(B(new Vector3(0,y*.45f+.25f,0),new Vector3(.55f,.43f,.55f),y%3));return a;}
    static List<Block> BuildLeaves(){var a=new List<Block>();for(int y=0;y<3;y++)for(int x=-1;x<=1;x++)for(int z=-1;z<=1;z++)if(y<2||x*x+z*z<2)a.Add(B(new Vector3(x*.48f,y*.40f+.22f,z*.48f),new Vector3(.46f,.38f,.46f),Math.Abs(x*5+z*7+y*11)%3));return a;}
    static Block B(Vector3 p,Vector3 s,int v)=>new Block{position=p,scale=s,variant=v};
    static void Folder(string path){if(AssetDatabase.IsValidFolder(path))return;string parent=path.Substring(0,path.LastIndexOf('/'));Folder(parent);AssetDatabase.CreateFolder(parent,path.Substring(path.LastIndexOf('/')+1));}
}
