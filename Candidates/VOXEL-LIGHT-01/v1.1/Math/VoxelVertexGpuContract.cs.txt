/* 功能：验证逐顶点额外灯聚合无法拆分时，独立背光压暗整体回退原C。
 * 边界：Root唯一Native执行；真实DX11浮点像素，不改场景/灯/作者资产；临时资源全部归还。
 */
using System;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using Funplay.Editor.Tools.Scripting;
public sealed class VoxelVertexGpuContract : IFunplayCommand
{
    public void Execute(ExecutionContext ctx)
    {
        if(Application.dataPath!="E:/WORLDCREATOR/new/Project_ANKALUTE/Assets")throw new InvalidOperationException("wrong project");
        bool oldAsync=ShaderUtil.allowAsyncCompilation;var previous=RenderTexture.active;
        Shader shader=null;Material material=null;RenderTexture output=null;Texture2D read=null;CommandBuffer command=null;
        try
        {
            ShaderUtil.allowAsyncCompilation=false;
            var factory=typeof(ShaderUtil).GetMethods(BindingFlags.Public|BindingFlags.NonPublic|BindingFlags.Static)
                .FirstOrDefault(m=>m.Name=="CreateShaderAsset"&&m.GetParameters().Length==1&&m.GetParameters()[0].ParameterType==typeof(string));
            if(factory==null)throw new InvalidOperationException("no transient shader factory");
            shader=(Shader)factory.Invoke(null,new object[]{Source});
            material=new Material(shader){hideFlags=HideFlags.HideAndDontSave};
            material.SetFloat("_VoxelKeyLightValid",1);material.SetFloat("_VoxelBackFaceDarkening",.5f);
            output=new RenderTexture(2,2,0,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear){hideFlags=HideFlags.HideAndDontSave};output.Create();
            read=new Texture2D(2,2,TextureFormat.RGBAFloat,false,true){hideFlags=HideFlags.HideAndDontSave};
            command=new CommandBuffer{name="Voxel v11 vertex aggregate back-control contract"};
            command.SetRenderTarget(output);command.SetViewport(new Rect(0,0,2,2));command.ClearRenderTarget(false,true,Color.magenta);
            command.DrawProcedural(Matrix4x4.identity,material,0,MeshTopology.Triangles,3);Graphics.ExecuteCommandBuffer(command);
            RenderTexture.active=output;read.ReadPixels(new Rect(0,0,2,2),0,0,false);read.Apply(false,false);
            Color actual=read.GetPixel(0,0);
            var errors=ShaderUtil.GetShaderMessages(shader).Where(m=>m.severity.ToString()=="Error").Select(m=>m.message).ToArray();
            if(errors.Length!=0||!shader.isSupported||float.IsNaN(actual.r)||actual.g<.999f)
                throw new InvalidOperationException("invalid GPU harness: "+string.Join(" | ",errors));
            bool pass=Mathf.Abs(actual.r-1)<.0001f;
            ctx.ReturnValue=new{pass,stage=pass?"GREEN":"RED",assertions=1,actualBackFillKeep=actual.r,expectedBackFillKeep=1,
                failure=pass?null:"per-vertex strong local lights are inseparable; back adjustment must keep C0",shaderSupported=shader.isSupported,graphics=SystemInfo.graphicsDeviceType.ToString(),errors};
        }
        finally
        {
            RenderTexture.active=previous;if(command!=null)command.Release();
            if(output!=null){output.Release();UnityEngine.Object.DestroyImmediate(output);}
            if(read!=null)UnityEngine.Object.DestroyImmediate(read);if(material!=null)UnityEngine.Object.DestroyImmediate(material);
            if(shader!=null)UnityEngine.Object.DestroyImmediate(shader);ShaderUtil.allowAsyncCompilation=oldAsync;
        }
    }
    const string Source=@"Shader ""Hidden/VOXEL-LIGHT-01/v11VertexAggregateContract"" {SubShader{Pass{
        Cull Off ZWrite Off ZTest Always
        HLSLPROGRAM
        #pragma target 4.5
        #pragma vertex Vert
        #pragma fragment Frag
        #define _ADDITIONAL_LIGHTS_VERTEX 1
        #include ""Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl""
        #include ""Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl""
        float _VoxelMainShape,_VoxelIndirectDiffuse,_VoxelEnvironmentSpecular,_VoxelDirectSpecular;
        float _VoxelHighlightStrength,_VoxelHighlightKnee,_VoxelHighlightCeiling,_VoxelDiagnostic,_VoxelVertexColors;
        float _VoxelLitFaceBoost,_VoxelBackFaceDarkening;
        #define ENVIRONMENT_NATIVE_LIT 1
        #include ""Assets/OSSASTRA/Scripts/WorldRendering/VoxelSurface/Rendering/VoxelSurfaceLighting.hlsl""
        float4 Vert(uint id:SV_VertexID):SV_POSITION{float2 uv=float2((id<<1)&2,id&2);return float4(uv*2-1,0,1);}
        float4 Frag():SV_Target{return float4(VoxelBackFillKeep(-1,1,0,1,1),1,0,1);}
        ENDHLSL
    }}}";
}
