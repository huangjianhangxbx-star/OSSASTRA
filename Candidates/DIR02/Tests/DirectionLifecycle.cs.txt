using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class DirectionLifecycle:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var c=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var a=c.Characters.Single(r=>r.ActorId=="temporary-a");var body=a.GetComponentInChildren<Spine.Unity.SkeletonAnimation>();var direction=body.GetComponent<C.CharacterDirectionPresentation>();var gear=a.GetComponent<C.CharacterEquipmentSpineView>();var rows=new List<object>();var steps=new List<(double,Action)>();int failed=0,index=0;double since=EditorApplication.timeSinceStartup;int life=a.LifeGeneration;Spine.Skeleton previous=null;bool paused=false;
void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,face=direction.FaceId,hidden=direction.HiddenSlotCount,life=a.LifeGeneration,clip=body.AnimationState.GetTrack(0)?.Animation?.Name});}
foreach(var r in c.Characters){r.GetComponent<C.CompanionCombatBrain>().enabled=false;r.GetComponent<C.CharacterFollow>().enabled=false;r.CancelAction("DIR02-lifecycle-fixture");r.SendMessage("OnApplicationFocus",true);}foreach(var e in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){e.AttacksEnabled=false;e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;}var engage=UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>();if(engage!=null)engage.enabled=false;
steps.Add((.2,()=>{previous=body.Skeleton;body.Initialize(true);Check("Skeleton rebuild changes instance",!ReferenceEquals(previous,body.Skeleton));}));
steps.Add((.5,()=>{Check("rebuilt first rendered pose has structural track and mask",body.AnimationState.GetTrack(9)!=null&&direction.HiddenSlotCount>300&&gear.LeftSkeleton!=null&&gear.LeftGripError<.001f);direction.enabled=false;Check("disable removes structural intent without clearing main",body.AnimationState.GetTrack(0)!=null);direction.enabled=true;}));
steps.Add((.8,()=>{Check("reenable restores profile and mask",body.AnimationState.GetTrack(9)!=null&&direction.HiddenSlotCount>300);a.SendMessage("OnApplicationPause",true);paused=true;}));
steps.Add((1.1,()=>{Check("OS suspend retains one valid body face",direction.HiddenSlotCount>300&&body.AnimationState.GetTrack(9)!=null);a.SendMessage("OnApplicationPause",false);paused=false;a.SendMessage("OnApplicationFocus",false);a.SendMessage("OnApplicationFocus",true);}));
steps.Add((1.4,()=>{Check("focus recovery has no stale branch",direction.HiddenSlotCount>300&&a.CanSimulateIntent);a.enabled=false;a.enabled=true;}));
steps.Add((1.7,()=>{Check("root reenable retains one direction component",a.GetComponentsInChildren<C.CharacterDirectionPresentation>(true).Length==1&&direction.HiddenSlotCount>300);var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();Ossastra.Combat.CombatContactGateway.Resolve(Ossastra.CharacterCore.AttackRelease.ForEmission(Guid.NewGuid(),0,enemy.ActorId,9999,0),enemy.Vitals,new Ossastra.Combat.IHitTarget[]{a.Vitals},enemy.transform.position);Check("actual damage pipeline death hands control away",!a.IsAlive&&c.Current!=a);}));
steps.Add((2,()=>Check("dead body equipment renderers hidden",gear.LeftSkeleton==null||!gear.LeftSkeleton.GetComponent<Renderer>().enabled)));
steps.Add((13,()=>Check("normal respawn service restores generation and face",a.IsAlive&&a.LifeGeneration>life&&body.AnimationState.GetTrack(9)!=null&&direction.HiddenSlotCount>300&&gear.LeftGripError<.001f)));
void Flush(string state)=>SessionState.SetString("DIR02_LIFECYCLE",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,boundaryInjection=true,rows}));EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(index==steps.Count){EditorApplication.update-=tick;Flush("completed");return;}if(EditorApplication.timeSinceStartup-since>=steps[index].Item1){steps[index++].Item2();Flush("running");}}catch(Exception ex){failed++;rows.Add(new{error=ex.ToString()});if(paused)a.SendMessage("OnApplicationPause",false);EditorApplication.update-=tick;Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="DIR02 lifecycle scheduled";}}
