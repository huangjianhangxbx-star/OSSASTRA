using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class Dir02CombatStory:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var c=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var a=c.Characters.Single(r=>r.ActorId=="temporary-a");var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(!a.IsAlive)throw new Exception("Living blue required");foreach(var r in c.Characters){r.GetComponent<C.CompanionCombatBrain>().enabled=true;r.GetComponent<C.CharacterFollow>().enabled=true;r.SendMessage("OnApplicationFocus",true);}enemy.AttacksEnabled=true;enemy.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=true;UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>().enabled=true;c.SendMessage("OnApplicationFocus",true);
var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;var keyboard=InputSystem.AddDevice<Keyboard>("DIR02CombatKeyboard");var mouse=InputSystem.AddDevice<Mouse>("DIR02CombatMouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in c.Characters.SelectMany(r=>{var i=r.GetComponent<C.CharacterInputHandler>();return new[]{i.movementAction,i.mouseAction,i.rightHandAction,i.leftHandAction,i.dodgeAction,i.skillOneAction,i.skillTwoAction,i.skillThreeAction};}).Concat(new[]{c.switchAction}).Where(x=>x!=null).Distinct()){overrides[action]=action.SaveBindingOverridesAsJson();for(int n=0;n<action.bindings.Count;n++){var b=action.bindings[n];if(!b.isComposite&&b.path!=null)action.ApplyBindingOverride(n,b.path.Replace("<Keyboard>",keyboard.path).Replace("<Mouse>",mouse.path));}}
EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();var phases=new List<(string,Action,Func<bool>,double)>();var rows=new List<object>();int index=0,failed=0,lastFrame=-1,hp=enemy.Vitals.Health.Current,accepted=0,releases=0;bool entered=false;double since=0;ushort buttons=0;var abilities=a.GetComponent<C.CharacterAbilities>();
void Keys(params Key[] keys)=>InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));void Aim(){var p=Camera.main.WorldToScreenPoint(enemy.transform.position);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(p.x,p.y),buttons=buttons});}void Mouse(ushort value){buttons=value;Aim();}void Add(string name,Action entry,Func<bool> check,double timeout=7)=>phases.Add((name,entry,check,timeout));void Delay(string name,Action entry,double seconds)=>Add(name,entry,()=>EditorApplication.timeSinceStartup-since>=seconds,seconds+3);
bool Approach(){var delta=enemy.transform.position-a.transform.position;delta.y=0;if(delta.magnitude<1.25f){Keys();return true;}var keys=new List<Key>();if(delta.x>.2)keys.Add(Key.D);if(delta.x<-.2)keys.Add(Key.A);if(delta.z>.2)keys.Add(Key.W);if(delta.z<-.2)keys.Add(Key.S);Keys(keys.ToArray());return false;}
Delay("neutral",()=>{Keys();Mouse(0);},.3);Add("normal Z selects blue",()=>{if(c.Current!=a)Keys(Key.Z);},()=>c.Current==a);Delay("release Z",()=>Keys(),.2);Add("walk into combat through normal movement",()=>{},Approach,15);Delay("guard approaches actual enemy",()=>Mouse(2),.5);Add("guard contact or enemy attack is real",()=>{},()=>abilities.GuardedCount>0||a.Vitals.Health.Current<100||enemy.Vitals.Health.Current<hp,12);Delay("release guard",()=>Mouse(0),.4);
Add("normal mouse attack accepts",()=>{releases=a.ReleaseCount;Mouse(1);},()=>a.ActionActive);Add("actual melee hit reduces enemy health",()=>Mouse(0),()=>a.ReleaseCount>releases&&enemy.Vitals.Health.Current<hp,4);Delay("recover",()=>{},.8);Delay("normal Q key",()=>{accepted=abilities.AcceptedCount;Keys(Key.Q);},.12);Add("body skill accepted by gameplay",()=>Keys(),()=>abilities.AcceptedCount>accepted,3);Add("skill returns to locomotion",()=>{},()=>!abilities.IsBusy&&!a.ActionActive,6);Delay("normal R key",()=>{accepted=abilities.AcceptedCount;Keys(Key.R);},.12);Add("second equipped skill accepts mapped body",()=>Keys(),()=>abilities.AcceptedCount>accepted,3);Add("second skill finishes",()=>{},()=>!abilities.IsBusy&&!a.ActionActive,6);
void Flush(string state)=>SessionState.SetString("DIR02_COMBAT",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,naturalInputOnly=true,virtualDevices=true,rows}));void Cleanup(){foreach(var pair in overrides)pair.Key.LoadBindingOverridesFromJson(pair.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}
EditorApplication.CallbackFunction tick=null;tick=()=>{if(Time.frameCount==lastFrame)return;lastFrame=Time.frameCount;try{if(index==phases.Count){EditorApplication.update-=tick;Cleanup();Flush("completed");return;}var p=phases[index];if(!entered){entered=true;since=EditorApplication.timeSinceStartup;p.Item2();}Aim();if(p.Item3()){rows.Add(new{name=p.Item1,ok=true,hpA=a.Vitals.Health.Current,hpEnemy=enemy.Vitals.Health.Current,face=a.GetComponentInChildren<C.CharacterDirectionPresentation>().FaceId,clip=a.GetComponentInChildren<Spine.Unity.SkeletonAnimation>().AnimationState.GetTrack(0)?.Animation?.Name,accepted=abilities.AcceptedCount});index++;entered=false;Flush("running");}else if(EditorApplication.timeSinceStartup-since>p.Item4)throw new Exception("Timeout "+p.Item1+" / "+abilities.LastResult+" / "+a.LastFact);}catch(Exception e){failed++;rows.Add(new{error=e.ToString()});EditorApplication.update-=tick;Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="DIR02 real combat scheduled";}}
