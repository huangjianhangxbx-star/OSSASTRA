/* 离屏姿态对照媒体：无玩法Root，不冒充输入实战；同一a4进度、八向。 */
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Spine;using Spine.Unity;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class RenderDirectionFrames:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var model=AssetDatabase.LoadAssetAtPath<SkeletonDataAsset>("Assets/OSSASTRA/LocalMedia/DIR02Blue/Blue43_SkeletonData.asset");var profile=AssetDatabase.LoadAssetAtPath<C.CharacterDirectionProfileSO>("Assets/OSSASTRA/Configs/Characters/BlueFiveFace/BlueFiveFace_Direction.asset");var scene=EditorSceneManager.NewPreviewScene();var oldActive=RenderTexture.active;var target=new RenderTexture(384,384,24);Texture2D pixels=null;
try{var go=new GameObject("DIR02-pose-only");SceneManager.MoveGameObjectToScene(go,scene);go.layer=31;SkeletonAnimation.AddToGameObject(go,model);var animation=go.GetComponent<SkeletonAnimation>();animation.UpdateTiming=UpdateTiming.ManualUpdate;animation.Initialize(false);var renderer=go.GetComponent<SkeletonRenderer>();var cameraGo=new GameObject("DIR02-evidence-camera");SceneManager.MoveGameObjectToScene(cameraGo,scene);var camera=cameraGo.AddComponent<Camera>();camera.scene=scene;camera.overrideSceneCullingMask=EditorSceneManager.GetSceneCullingMask(scene);camera.orthographic=true;camera.orthographicSize=1.6f;camera.transform.position=new Vector3(0,.85f,-10);camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=new Color(.055f,.07f,.09f);camera.cullingMask=1<<31;var lightGo=new GameObject("DIR02-preview-light");SceneManager.MoveGameObjectToScene(lightGo,scene);var light=lightGo.AddComponent<Light>();light.type=LightType.Directional;light.intensity=1.1f;light.transform.rotation=Quaternion.Euler(30,0,0);light.cullingMask=1<<31;camera.targetTexture=target;target.Create();pixels=new Texture2D(384,384,TextureFormat.RGBA32,false);var results=new List<object>();var directions=new[]{Vector3.forward,new Vector3(1,0,1),Vector3.right,new Vector3(1,0,-1),Vector3.back,new Vector3(-1,0,-1),Vector3.left,new Vector3(-1,0,1)};
for(int i=0;i<directions.Length;i++){animation.Initialize(true);var selection=profile.Select(directions[i],default);var face=profile.faces.Single(f=>f.faceId==selection.FaceId);animation.Skeleton.ScaleX=selection.Mirrored?-1:1;var track=animation.AnimationState.SetAnimation(0,"a4",false);track.TrackTime=track.Animation.Duration*.42f;animation.AnimationState.SetAnimation(9,"方向_"+face.rootBone,true);animation.Update(0);var mask=new C.CharacterSlotVisibility(animation.Skeleton,profile.boundarySlot,profile.builtInWeaponSlots);mask.Apply(false,false);renderer.LateUpdate();camera.Render();RenderTexture.active=target;pixels.ReadPixels(new Rect(0,0,384,384),0,0);pixels.Apply();string path="Candidates/DIR02/LocalEvidence/a4-42-"+i+"-"+selection.FaceId+(selection.Mirrored?"-mirror":"")+".png";System.IO.File.WriteAllBytes(path,pixels.EncodeToPNG());mask.Restore();results.Add(new{path,face=selection.FaceId,mirror=selection.Mirrored,clip="a4",normalized=.42f});}ctx.ReturnValue=new{poseEvidenceOnly=true,frames=results};}
finally{RenderTexture.active=oldActive;target.Release();UnityEngine.Object.DestroyImmediate(target);if(pixels!=null)UnityEngine.Object.DestroyImmediate(pixels);EditorSceneManager.ClosePreviewScene(scene);}}}
