using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Spine;using Spine.Unity;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class EquipmentPoseContracts:IFunplayCommand {
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var rig=AssetDatabase.LoadAssetAtPath<C.CharacterRigProfileSO>("Assets/OSSASTRA/Configs/Characters/BlueFiveFace/BlueFiveFace_Rig.asset");var data=rig.model.GetSkeletonData(false);var profile=AssetDatabase.LoadAssetAtPath<C.CharacterDirectionProfileSO>("Assets/OSSASTRA/Configs/Characters/BlueFiveFace/BlueFiveFace_Direction.asset");var rows=new List<object>();int tested=0;var left=rig.CompileRoute("equipment.left",data);var right=rig.CompileRoute("equipment.right",data);
 foreach(var face in profile.faces)foreach(var clip in new[]{"_stand","_move","a1","a2","a3","a4","_dash","_damaged","_stun","skill_架盾2","skill_盾冲！","skill_大戳"})foreach(float time in new[]{0f,.1f,.13333334f,.15f,.25f,.4f,.59f,.6f,.7f,.82f})foreach(float mirror in new[]{1f,-1f}){
 var skeleton=new Skeleton(data);skeleton.ScaleX=mirror;var state=new Spine.AnimationState(new AnimationStateData(data));var track=state.SetAnimation(0,clip,false);state.SetAnimation(9,"方向_"+face.rootBone,true);track.TrackTime=Mathf.Min(time,track.Animation.Duration);state.Apply(skeleton);skeleton.UpdateWorldTransform(Spine.Physics.Pose);var mask=new C.CharacterSlotVisibility(skeleton,profile.boundarySlot,profile.builtInWeaponSlots);mask.Apply(true,false);float fraction=track.Animation.Duration<=0?0:Mathf.Clamp01(track.TrackTime/track.Animation.Duration);
 var l=left.Select(face.faceId,clip,fraction);var r=right.Select(face.faceId,clip,fraction);bool ok=mask.Contains(skeleton.Bones.Items[l.BoneIndex])&&mask.Contains(skeleton.Bones.Items[r.BoneIndex]);tested++;if(!ok)rows.Add(new{face=face.faceId,clip,time,mirror,left=l.BoneName,right=r.BoneName});mask.Restore();}
 ctx.ReturnValue=new{samples=tested,failed=rows.Count,rows};}
}
