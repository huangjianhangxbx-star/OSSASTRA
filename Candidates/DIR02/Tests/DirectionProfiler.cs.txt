using System;using System.Linq;using System.Collections.Generic;using UnityEditor;using UnityEngine;using Unity.Profiling;using Funplay.Editor.Tools.Scripting;
public class DirectionProfiler:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var recorder=ProfilerRecorder.StartNew(ProfilerCategory.Scripts,"OSSASTRA.Direction.Visibility",64);var gc=ProfilerRecorder.StartNew(ProfilerCategory.Memory,"GC Allocated In Frame",64);var values=new List<long>();var allocations=new List<long>();int last=-1;EditorApplication.CallbackFunction tick=null;tick=()=>{if(last==Time.frameCount)return;last=Time.frameCount;if(recorder.Valid&&recorder.LastValue>0)values.Add(recorder.LastValue);if(gc.Valid)allocations.Add(gc.LastValue);if(values.Count>=120||allocations.Count>=180){EditorApplication.update-=tick;var result=new{state="completed",markerValid=recorder.Valid,frames=values.Count,meanNanoseconds=values.Count==0?0:values.Average(),maxNanoseconds=values.Count==0?0:values.Max(),wholeFrameGcBytesMean=allocations.Count==0?0:allocations.Average(),note="Editor Play有限采样；GC为整帧而非方向模块归因，不证明零GC或目标机帧率"};recorder.Dispose();gc.Dispose();SessionState.SetString("DIR02_PROFILER",Newtonsoft.Json.JsonConvert.SerializeObject(result));}};EditorApplication.update+=tick;ctx.ReturnValue="Profiler recorder scheduled";}}
