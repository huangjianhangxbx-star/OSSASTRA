void Pad(Vector2 aim,bool cast=false)=>InputSystem.QueueStateEvent(pad,cast?new GamepadState{leftStick=aim,rightStick=-aim}.WithButton(GamepadButton.LeftShoulder).WithButton(GamepadButton.North):new GamepadState{leftStick=aim,rightStick=-aim});void Add(string name,Action enter,Func<bool> check,double timeout=5)=>steps.Add((name,enter,check,timeout));
Add("neutral rearm",()=>Pad(Vector2.zero),()=>EditorApplication.timeSinceStartup-since>.35);Add("native E accepted once",()=>Pad(Vector2.right,true),()=>{if(ability.AcceptedCount!=accepted+1||!((C.ICharacterActionDirectionSource)ability).TryGetBodyActionDirection(out frozen))return false;id=a.DataBus.ActionId;if(direction.ReadGameplayDirection().x<.99f)throw new Exception("Body facing must retain accepted direction during backward windup");return frozen.x>.99f;});Add("new movement does not alter named rush accepted direction or body face",()=>Pad(Vector2.up),()=>input.CurrentFrame.aimDirection.y>.9f&&direction.FaceId=="left"&&direction.Mirrored&&a.DataBus.ActionId==id&&direction.ReadGameplayDirection().x>.99f);Add("accepted E finishes through original timeline",()=>{},()=>!ability.IsBusy&&!a.ActionActive,6);Add("after named rush direction returns to current movement",()=>{},()=>direction.FaceId=="back"&&ability.AcceptedCount==accepted+1);
