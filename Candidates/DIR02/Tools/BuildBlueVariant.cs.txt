 Rect VisibleBounds(Skeleton skeleton)
 {
  float minX=float.PositiveInfinity,minY=float.PositiveInfinity,maxX=float.NegativeInfinity,maxY=float.NegativeInfinity;
  foreach(var slot in skeleton.Slots){if(slot.AppliedPose.GetColor().a<=0)continue;float[] vertices;var attachment=slot.AppliedPose.Attachment;if(attachment is RegionAttachment region){vertices=new float[8];region.ComputeWorldVertices(slot,region.GetOffsets(slot.AppliedPose),vertices,0,2);}else if(attachment is MeshAttachment mesh){vertices=new float[mesh.WorldVerticesLength];mesh.ComputeWorldVertices(skeleton,slot,vertices);}else continue;for(int i=0;i<vertices.Length;i+=2){minX=Mathf.Min(minX,vertices[i]);maxX=Mathf.Max(maxX,vertices[i]);minY=Mathf.Min(minY,vertices[i+1]);maxY=Mathf.Max(maxY,vertices[i+1]);}}
  if(float.IsInfinity(minX)||maxY<=minY)throw new Exception("No visible body bounds");return Rect.MinMaxRect(minX,minY,maxX,maxY);
 }
