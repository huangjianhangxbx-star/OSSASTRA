using System;using System.Linq;using System.Collections;using System.Collections.Generic;using System.Reflection;using UnityEditor;using UnityEngine;using A=Ossastra.CharacterAnimation;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class DirectionComposerResearch:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var active=UnityEngine.SceneManagement.SceneManager.GetActiveScene();var dirty=active.isDirty;var prefab=PrefabUtility.LoadPrefabContents("Assets/OSSASTRA/Prefabs/Characters/PlayerCharacter.prefab");try{var root=prefab.GetComponent<C.CharacterRoot>();var body=root.GetComponentInChildren<A.CharacterAnimationController>(true);if(root==null||body==null)throw new Exception("Missing researched root/body");var skeleton=body.GetComponent<Spine.Unity.SkeletonAnimation>();var config=body.GetComponent<A.CharacterAnimationConfigManager>();var composer=body.GetComponent<A.CharacterAnimationComposer>();var handle=body.GetComponent<A.AnimationHandleSystem>();if(skeleton==null||config==null||composer==null||handle==null)throw new Exception("Missing researched animation assembly");handle.Bind(root,skeleton,"body");composer.Initialize(root,config,handle);var nested=typeof(A.CharacterAnimationComposer).GetNestedType("ResolvedAnimationPlan",BindingFlags.NonPublic);var listType=typeof(List<>).MakeGenericType(nested);var merge=typeof(A.CharacterAnimationComposer).GetMethod("MergePlans",BindingFlags.Instance|BindingFlags.NonPublic);var rows=new List<object>();
object Owner(A.AnimationIntentSourceStateMachine source,bool exclusive,int[] tracks){var resolved=Activator.CreateInstance(nested,true);var intent=new A.AnimationIntent{root=root,actorId=root.ActorId,target="body",intentSourceStateMatine=source,priority=exclusive?20:5,exclusiveFullBody=exclusive,actionId=Guid.NewGuid()};var plan=new A.CharacterAnimationPlan();foreach(var track in tracks)plan.clips.Add(new A.CharacterAnimationClip{animaName=track==0?"atk1":"idle1_1",track=track,playbackRate=1});nested.GetField("intent").SetValue(resolved,intent);nested.GetField("plan").SetValue(resolved,plan);return resolved;}
A.CharacterAnimationPlan Merge(params object[] owners){var list=(IList)Activator.CreateInstance(listType);foreach(var owner in owners)list.Add(owner);return (A.CharacterAnimationPlan)merge.Invoke(composer,new object[]{list});}
var direction=Owner(A.AnimationIntentSourceStateMachine.Movement,false,new[]{9});var normal=Merge(Owner(A.AnimationIntentSourceStateMachine.MainHand,false,new[]{0}),direction);rows.Add(new{name="different owners without exclusivity",tracks=normal.clips.Select(c=>c.track).ToArray(),clears=normal.clearTracks.ToArray()});var exclusive=Merge(Owner(A.AnimationIntentSourceStateMachine.MainHand,true,new[]{0}),direction);rows.Add(new{name="exclusive main clears separate direction owner",tracks=exclusive.clips.Select(c=>c.track).ToArray(),clears=exclusive.clearTracks.ToArray()});var same=Merge(Owner(A.AnimationIntentSourceStateMachine.MainHand,true,new[]{0,9}));rows.Add(new{name="same full plan owner can retain track9",tracks=same.clips.Select(c=>c.track).ToArray(),clears=same.clearTracks.ToArray()});ctx.ReturnValue=new{researchOnly=true,reproduced=normal.clips.Any(c=>c.track==9)&&exclusive.clearTracks.Contains(9)&&!exclusive.clips.Any(c=>c.track==9)&&same.clips.Any(c=>c.track==9),rows,sourcePrefabUnsaved=true,activeScenePreserved=active==UnityEngine.SceneManagement.SceneManager.GetActiveScene()&&active.isDirty==dirty};}finally{PrefabUtility.UnloadPrefabContents(prefab);}}}
