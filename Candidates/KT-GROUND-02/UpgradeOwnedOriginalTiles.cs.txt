/* 本轮生成的尚未交付OriginalLab升级到保留Native作者锁语义的Tile；不迁移保护旧场。
 * 定义和材质通过已验证正式API生成新版本；不改模板、源纹理和旧Tile文件。
 */
using System;using System.Linq;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
using G=Ossastra.GroundResearch;using E=Ossastra.GroundResearch.Editor;
public sealed class UpgradeOwnedOriginalTiles:IFunplayCommand
{
 public void Execute(ExecutionContext ctx)
 {
  var scene=SceneManager.GetActiveScene();if(scene.path!=E.GroundWorkshopOwnership.OriginalScene||scene.isDirty)throw new InvalidOperationException("只升级本轮干净OriginalLab。");
  var map=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<G.GroundWorkshopMap>(true)).Single();
  for(int i=0;i<map.layerBindings.Length;i++)
  {
   var tile=map.layerBindings[i].ownedTile;if(tile is G.GroundAuthorTile||tile is G.GroundAuthorRuleTile)continue;
   bool locked=E.GroundWorkshopAuthoring.IsLayerLocked(map,i);
   if(locked)E.GroundWorkshopAuthoring.SetLayerLocked(map,i,false);
   var change=E.GroundWorkshopAuthoring.PrepareReplacement(map,i,map.theme.layers[i].recipe,G.GroundReplacementScope.ExistingAndFuture);
   E.GroundWorkshopAuthoring.Commit(change);
   if(locked)E.GroundWorkshopAuthoring.SetLayerLocked(map,i,true);
  }
  E.GroundWorkshopMaps.Save(map);
  ctx.ReturnValue=new{path=scene.path,dirty=scene.isDirty,tiles=map.layerBindings.Select(b=>b.ownedTile.GetType().Name).ToArray()};
 }
}
