/* 本轮参考图的显式镜头适配：固定单面草在90度成薄片，克隆地图私有定义改Yaw，不改导入原定义、不重撒。 */
using System;using System.Linq;using UnityEditor;using UnityEngine;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
using G=Ossastra.GroundResearch;using E=Ossastra.GroundResearch.Editor;
public sealed class AdaptReferenceFloraYaw:IFunplayCommand
{
 public void Execute(ExecutionContext ctx)
 {
  var scene=SceneManager.GetActiveScene();if(scene.path!=E.GroundWorkshopOwnership.ReferenceScene||scene.isDirty)throw new InvalidOperationException("仅接受本轮干净参考Lab");
  var map=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<G.GroundWorkshopMap>(true)).Single();E.GroundWorkshopOwnership.Require(map);
  var instances=map.decorationRoot.GetComponentsInChildren<G.GroundDecorationInstance>(true);
  string before=string.Join("|",instances.Select(i=>i.instanceId+i.transform.localPosition.ToString("F6")+i.transform.localRotation.ToString("F6")+i.transform.localScale.ToString("F6")+i.isLocked).OrderBy(x=>x,StringComparer.Ordinal));
  var oldDefs=map.theme.decorations;var oldJson=oldDefs.Select(EditorJsonUtility.ToJson).ToArray();E.GroundWorkshopMaps.CopyTheme(map);
  var definitions=new G.GroundDecorationDefinition[oldDefs.Length];
  for(int n=0;n<oldDefs.Length;n++)
  {
   var source=oldDefs[n];var value=UnityEngine.Object.Instantiate(source);value.mode=G.GroundDecorationMode.UprightYaw;value.revision=source.revision+"-map-yaw-v1";
   value.displayName=source.displayName+" · 本图镜头适配";value.evidence=source.evidence+"\nUNITY_ADAPTATION：本图可旋转镜头，使用仅yaw面向相机；不声称原作运行时Billboard已核实。";
   string path=map.registration.assetFolder+"/DecorationYaw_"+Guid.NewGuid().ToString("N")+".asset";AssetDatabase.CreateAsset(value,path);AssetDatabase.SaveAssetIfDirty(value);definitions[n]=value;
   foreach(var instance in instances.Where(i=>i.definition==source)){Undo.RecordObject(instance,"本图装饰镜头适配");instance.definition=value;EditorUtility.SetDirty(instance);}
  }
  map.theme.decorations=definitions;EditorUtility.SetDirty(map.theme);map.GetComponent<G.GroundDecorationOrientation>().Refresh();E.GroundWorkshopMaps.Save(map);
  string after=string.Join("|",instances.Select(i=>i.instanceId+i.transform.localPosition.ToString("F6")+i.transform.localRotation.ToString("F6")+i.transform.localScale.ToString("F6")+i.isLocked).OrderBy(x=>x,StringComparer.Ordinal));
  ctx.ReturnValue=new{path=scene.path,scene.isDirty,count=instances.Length,rootsAndLocksUnchanged=before==after,sourceDefinitionsUnchanged=oldDefs.Select(EditorJsonUtility.ToJson).SequenceEqual(oldJson),
   allInstancesRetainSourceIds=instances.All(i=>i.assetId==i.definition.assetId),modes=instances.Select(i=>i.definition.mode.ToString()).Distinct().ToArray(),definitionPaths=definitions.Select(AssetDatabase.GetAssetPath).ToArray()};
 }
}
