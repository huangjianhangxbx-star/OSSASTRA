/* KT02有界旋转证据：原InputAction接收虚拟Mouse滚轮，8稳定角及至少1中间角。
 * 输出为原GameplayCamera原生Render的本机PNG，不是Editor/GameView截图，不宣称物理硬件输入或源图可公开。
 * root先SessionState.SetString("KT02.RotationArtifacts.Output",完整03-tutorial/images/ReferenceRotate目录)。
 * 不写模型/绑定/HP/时钟/角色Transform；临时RT和camera设置逐次finally恢复，45秒硬上限。
 */
using System;
using System.IO;
using System.Linq;
using System.Text;
using System.Collections.Generic;
using System.Security.Cryptography;
using UnityEditor;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.LowLevel;
using Funplay.Editor.Tools.Scripting;
using G=Ossastra.GroundResearch;
using GE=Ossastra.GroundResearch.Editor;
using M=Ossastra.MapResearch;

public sealed class KT02RotationArtifactStory:IFunplayCommand
{
    public static bool Running;
    public static long RunId;
    public static object Report;
    static M.MapResearchRoot map;static G.GroundWorkshopMap ground;static Mouse mouse;
    static string scenePath,output,error,initialRoots;static float wheel;static int step,expectedSector,lastFrame,midCount;
    static double started,phaseStarted;static string phase;
    static List<object> frames,captures;static List<Filter> filters;
    static bool rootsStayedFixed,cameraSettingsRestored,visualConstraintsPassed;static int[] modeCoverage;
    sealed class Filter{public InputActionMap map;public bool had;public InputDevice[] devices;}
    static bool Allowed(Scene scene)
    {
        string preview=GE.GroundWorkshopPlaygroundSession.PreviewScenePath;
        return scene.path==GE.GroundWorkshopOwnership.ReferenceScene||(!string.IsNullOrEmpty(preview)&&GE.GroundWorkshopPlaygroundSession.HasSession&&scene.path==preview);
    }
    static T[] InScene<T>(Scene scene)where T:Component=>scene.GetRootGameObjects().SelectMany(root=>root.GetComponentsInChildren<T>(true)).ToArray();
    static string RootChecksum()
    {
        var text=new StringBuilder();text.Append(ground.GetInstanceID()).Append(ground.registration==null?"none":ground.registration.mapId);
        text.Append(ground.transform.position.ToString("F6")).Append(ground.transform.rotation.ToString("F6")).Append(ground.transform.lossyScale.ToString("F6"));
        text.Append(ground.decorationRoot.position.ToString("F6")).Append(ground.decorationRoot.rotation.ToString("F6")).Append(ground.decorationRoot.lossyScale.ToString("F6"));
        foreach(var instance in ground.decorationRoot.GetComponentsInChildren<G.GroundDecorationInstance>(true).OrderBy(value=>value.instanceId,StringComparer.Ordinal))
            text.Append(instance.instanceId).Append(instance.GetInstanceID()).Append(instance.transform.localPosition.ToString("F6")).Append(instance.transform.localRotation.ToString("F6")).Append(instance.transform.localScale.ToString("F6")).Append(instance.transform.position.ToString("F6")).Append(instance.transform.rotation.ToString("F6")).Append(instance.transform.lossyScale.ToString("F6"));
        using(var sha=SHA256.Create())return Convert.ToBase64String(sha.ComputeHash(Encoding.UTF8.GetBytes(text.ToString())));
    }
    public void Execute(ExecutionContext ctx)
    {
        if(Running)throw new InvalidOperationException("Rotation story already running");
        RunId=DateTime.UtcNow.Ticks;frames=new List<object>();captures=new List<object>();filters=new List<Filter>();error="";step=midCount=0;lastFrame=-1;
        rootsStayedFixed=cameraSettingsRestored=visualConstraintsPassed=true;modeCoverage=new int[3];started=phaseStarted=EditorApplication.timeSinceStartup;phase="warmup";Report=null;
        try
        {
            foreach(var assembly in AppDomain.CurrentDomain.GetAssemblies())foreach(string name in new[]{"KT02M6InputStory","KT02M6CombatOnlyStory","KT02M6Performance"})
            {var type=assembly.GetType(name,false);if(type!=null&&(bool)type.GetField("Running").GetValue(null))throw new InvalidOperationException("Finish current M6 input/performance story before artifact captures");}
            var scene=SceneManager.GetActiveScene();scenePath=scene.path;
            if(!Application.isPlaying||EditorApplication.isPaused||Time.timeScale!=1f||!Allowed(scene))throw new InvalidOperationException("Normal-time ReferenceScene or exact recorded PreviewSession Play required");
            map=InScene<M.MapResearchRoot>(scene).Single();ground=InScene<G.GroundWorkshopMap>(scene).Single();
            if(map.IsStructureMode||!map.IsValid||map.GameplayCamera==null||ground.decorationRoot==null)throw new InvalidOperationException("Initialized normal gameplay camera and author decoration root required");
            if(Mathf.Abs(ground.surface.WorldArea-3072f)>.01f)throw new InvalidOperationException("Actual full 3072m² scene required");
            string configured=SessionState.GetString("KT02.RotationArtifacts.Output","");
            if(string.IsNullOrWhiteSpace(configured)||!Path.IsPathRooted(configured))throw new InvalidOperationException("root must explicitly configure absolute output path");
            output=Path.GetFullPath(configured).TrimEnd('/','\\');
            if(!output.Replace('\\','/').EndsWith("/03-tutorial/images/ReferenceRotate",StringComparison.OrdinalIgnoreCase)||
                output.StartsWith(Path.GetFullPath(Application.dataPath),StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Output must be research/03-tutorial/images/ReferenceRotate outside Assets");
            Directory.CreateDirectory(output);
            if(map.cameraInput==null||map.cameraInput.rotateWheel==null||!map.cameraInput.rotateWheel.enabled||!map.cameraInput.rotateWheel.bindings.Any(binding=>binding.effectivePath=="<Mouse>/scroll/y"))throw new InvalidOperationException("Existing enabled mouse wheel camera InputAction required; no rebinding");
            mouse=InputSystem.AddDevice<Mouse>("KT02RotationMouse");wheel=0;
            var actionMap=map.cameraInput.rotateWheel.actionMap;
            if(actionMap!=null){var old=actionMap.devices;filters.Add(new Filter{map=actionMap,had=old.HasValue,devices=old.HasValue?old.Value.ToArray():null});actionMap.devices=new InputDevice[]{mouse};}
            initialRoots=RootChecksum();Running=true;InputSystem.onBeforeUpdate+=BeforeInput;EditorApplication.update+=Tick;
            EditorApplication.playModeStateChanged+=OnPlayState;AssemblyReloadEvents.beforeAssemblyReload+=OnReload;
            var gameView=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView",false);if(gameView!=null)EditorWindow.GetWindow(gameView).Focus();
            Publish(false);ctx.ReturnValue=new{started=true,RunId,scene=scenePath,output,limitSeconds=45,pollType="KT02RotationArtifactStory",pollField="Report",nativeCameraRenderOnly=true,physicalDevicesMustBeIdle=true};
        }
        catch(Exception exception){error=exception.ToString();Cleanup();ctx.ReturnValue=Report;}
    }
    static void BeforeInput()
    {
        if(!Running||mouse==null||InputState.currentUpdateType!=InputUpdateType.Dynamic)return;
        InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(Screen.width*.5f,Screen.height*.55f),scroll=new Vector2(0,wheel)});wheel=0;
    }
    static void Tick()
    {
        if(!Running)return;
        try
        {
            double now=EditorApplication.timeSinceStartup;
            if(now-started>45)throw new TimeoutException("Rotation artifacts exceeded 45 seconds; partial output retained");
            if(!Application.isPlaying||EditorApplication.isPaused||Time.timeScale!=1f||SceneManager.GetActiveScene().path!=scenePath||!Allowed(SceneManager.GetActiveScene())||map==null||ground==null)throw new InvalidOperationException("Recorded normal gameplay lifecycle changed");
            if(Time.frameCount==lastFrame)return;lastFrame=Time.frameCount;
            string checksum=RootChecksum();rootsStayedFixed&=checksum==initialRoots;
            if(frames.Count<1800)frames.Add(new{frame=Time.frameCount,seconds=now-started,phase,actualYaw=map.ActualYaw,targetYaw=map.TargetYaw,sector=map.TargetSector,mapObject=ground.GetInstanceID(),mapId=ground.registration==null?null:ground.registration.mapId,rootChecksum=checksum,rootsFixed=checksum==initialRoots});
            if(!rootsStayedFixed)throw new InvalidOperationException("Decoration author root TRS changed during camera rotation");
            if(phase=="warmup")
            {
                if(now-phaseStarted>.6&&Mathf.Abs(Mathf.DeltaAngle(map.ActualYaw,map.TargetYaw))<.1f){expectedSector=(map.TargetSector+1)%8;wheel=120;phase="orbit";phaseStarted=now;}
                return;
            }
            float difference=Mathf.Abs(Mathf.DeltaAngle(map.ActualYaw,map.TargetYaw));
            if(map.TargetSector==expectedSector&&difference>=5&&difference<=40&&midCount==0){Capture("mid_"+step);midCount++;}
            if(map.TargetSector==expectedSector&&difference<.1f&&now-phaseStarted>.15)
            {
                Capture("stable_"+step);step++;
                if(step==8){if(midCount<1)throw new InvalidOperationException("Eight settled angles observed but no real 5–40° transition frame; no mid-angle proof inferred");Cleanup();}
                else{expectedSector=(map.TargetSector+1)%8;wheel=120;phaseStarted=now;}
            }
            else if(now-phaseStarted>4)throw new TimeoutException("Wheel did not settle to expected sector "+expectedSector);
            if(Running)Publish(false);
        }
        catch(Exception exception){error=exception.ToString();Cleanup();}
    }
    static object VisualReadback(Camera camera)
    {
        var instances=ground.decorationRoot.GetComponentsInChildren<G.GroundDecorationInstance>(true);
        var modes=new List<object>();
        foreach(G.GroundDecorationMode mode in Enum.GetValues(typeof(G.GroundDecorationMode)))
        {
            int count=0,failures=0;float maxAngle=0,maxFoot=0;
            foreach(var instance in instances.Where(value=>value.definition!=null&&value.definition.mode==mode))
            {
                count++;if(instance.visual==null){failures++;continue;}var visual=instance.visual;var renderer=visual.GetComponent<SpriteRenderer>();
                if(renderer==null||renderer.sprite==null){failures++;continue;}
                float angle=mode==G.GroundDecorationMode.UprightYaw?Quaternion.Angle(visual.rotation,Quaternion.Euler(0,camera.transform.eulerAngles.y,0)):
                    Quaternion.Angle(visual.localRotation,mode==G.GroundDecorationMode.Flat?Quaternion.Euler(90,0,0):Quaternion.identity);
                var sprite=renderer.sprite;var definition=instance.definition;
                var foot=new Vector3((definition.footpointNormalized.x*sprite.rect.width-sprite.pivot.x)/sprite.pixelsPerUnit,(definition.footpointNormalized.y*sprite.rect.height-sprite.pivot.y)/sprite.pixelsPerUnit,0);
                float footError=Vector3.Distance(visual.TransformPoint(foot),instance.transform.position+Vector3.up*definition.groundOffset);
                maxAngle=Mathf.Max(maxAngle,angle);maxFoot=Mathf.Max(maxFoot,footError);if(angle>.15f||footError>.0005f)failures++;
            }
            modeCoverage[(int)mode]=Math.Max(modeCoverage[(int)mode],count);visualConstraintsPassed&=failures==0;
            modes.Add(new{mode=mode.ToString(),count,observed=count>0,failures,maximumAngleErrorDegrees=maxAngle,maximumFootpointDriftMeters=maxFoot});
        }
        return modes.ToArray();
    }
    static void Capture(string label)
    {
        Camera camera=map.GameplayCamera;var oldTarget=camera.targetTexture;var oldActive=RenderTexture.active;
        float oldFar=camera.farClipPlane;var oldClear=camera.clearFlags;var oldBackground=camera.backgroundColor;var oldRect=camera.rect;
        RenderTexture target=null;Texture2D pixels=null;string file=null;object visual=null;var renderErrors=new List<string>();
        Application.LogCallback receive=(condition,stack,type)=>{if(type==LogType.Error||type==LogType.Exception||type==LogType.Assert)renderErrors.Add(condition);};
        try
        {
            int width=Math.Min(1600,Math.Max(640,Screen.width)),height=Math.Max(360,Mathf.RoundToInt(width*(float)Math.Max(1,Screen.height)/Math.Max(1,Screen.width)));
            target=new RenderTexture(width,height,24,RenderTextureFormat.ARGB32){name="KT02 scoped native rotation render",hideFlags=HideFlags.HideAndDontSave};target.Create();camera.targetTexture=target;
            Application.logMessageReceived+=receive;camera.Render();Application.logMessageReceived-=receive;
            if(renderErrors.Count>0)throw new InvalidOperationException("Native camera render logged errors: "+string.Join("; ",renderErrors));
            RenderTexture.active=target;pixels=new Texture2D(width,height,TextureFormat.RGB24,false);pixels.ReadPixels(new Rect(0,0,width,height),0,0);pixels.Apply(false,false);
            var colors=pixels.GetPixels32();int minimum=765,maximum=0;
            for(int index=0;index<colors.Length;index+=Math.Max(1,colors.Length/4096)){int sum=colors[index].r+colors[index].g+colors[index].b;minimum=Math.Min(minimum,sum);maximum=Math.Max(maximum,sum);}
            if(maximum-minimum<8)throw new InvalidOperationException("Native camera output is effectively uniform; no map-image proof inferred");
            visual=VisualReadback(camera);rootsStayedFixed&=RootChecksum()==initialRoots;if(!rootsStayedFixed)throw new InvalidOperationException("Native camera render changed decoration author roots");file=Path.Combine(output,RunId+"_"+label+"_sector"+map.TargetSector+".png");
            using(var stream=new FileStream(file,FileMode.CreateNew,FileAccess.Write,FileShare.None)){byte[] bytes=pixels.EncodeToPNG();stream.Write(bytes,0,bytes.Length);}
            captures.Add(new{label,file,frame=Time.frameCount,actualYaw=map.ActualYaw,targetYaw=map.TargetYaw,sector=map.TargetSector,width,height,camera= camera.GetInstanceID(),groundMap=ground.GetInstanceID(),rootChecksum=RootChecksum(),visualModes=visual,pixelSampleRgbSumRange=maximum-minimum,renderErrors=renderErrors.ToArray(),renderMethod="original GameplayCamera.Render to scoped RenderTexture; not Editor/GameView screenshot",publicPackEligible=false});
        }
        finally
        {
            Application.logMessageReceived-=receive;
            camera.targetTexture=oldTarget;RenderTexture.active=oldActive;camera.farClipPlane=oldFar;camera.clearFlags=oldClear;camera.backgroundColor=oldBackground;camera.rect=oldRect;
            cameraSettingsRestored&=camera.targetTexture==oldTarget&&RenderTexture.active==oldActive&&camera.farClipPlane==oldFar&&camera.clearFlags==oldClear&&camera.backgroundColor==oldBackground&&camera.rect==oldRect;
            if(pixels!=null)UnityEngine.Object.DestroyImmediate(pixels);if(target!=null){target.Release();UnityEngine.Object.DestroyImmediate(target);}
        }
    }
    static void Publish(bool done){Report=new{done,status=!done?"RUNNING":string.IsNullOrEmpty(error)&&step==8&&midCount>0&&rootsStayedFixed&&cameraSettingsRestored&&visualConstraintsPassed?"PASS":"FAILED",RunId,Running,scene=scenePath,seconds=EditorApplication.timeSinceStartup-started,stableAngles=step,midAngles=midCount,rootsStayedFixed,cameraSettingsRestored,visualConstraintsPassed,modeCoverage,threeModesCovered=modeCoverage!=null&&modeCoverage.All(count=>count>0),unobservedModesAreNotPassed=true,error,output,captures=captures==null?null:captures.ToArray(),frames=frames==null?null:frames.ToArray(),nativeGameplayCameraRender=true,editorScreenshot=false,physicalHardwareVerified=false,publicPackEligible=false};}
    static void OnPlayState(PlayModeStateChange state){if(Running&&state==PlayModeStateChange.ExitingPlayMode)Cancel("Play exit interrupted artifact story");}
    static void OnReload(){if(Running)Cancel("Assembly reload interrupted artifact story");}
    public static void Cancel(string reason){error=reason;Cleanup();}
    static void Cleanup()
    {
        Running=false;wheel=0;InputSystem.onBeforeUpdate-=BeforeInput;EditorApplication.update-=Tick;EditorApplication.playModeStateChanged-=OnPlayState;AssemblyReloadEvents.beforeAssemblyReload-=OnReload;
        try{if(mouse!=null)InputSystem.ResetDevice(mouse);}catch(Exception exception){error+="\nReset mouse: "+exception.Message;}
        if(filters!=null)foreach(var filter in filters)try{filter.map.devices=filter.had?filter.devices:null;}catch(ObjectDisposedException){}
        if(mouse!=null)InputSystem.RemoveDevice(mouse);mouse=null;Publish(true);
    }
}
