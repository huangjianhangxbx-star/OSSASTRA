using System;
using System.Linq;
using Funplay.Editor.Tools.Scripting;
public sealed class KT02PollRotationArtifactStory:IFunplayCommand
{
    public void Execute(ExecutionContext ctx)
    {
        var type=AppDomain.CurrentDomain.GetAssemblies().Select(assembly=>assembly.GetType("KT02RotationArtifactStory",false)).Where(value=>value!=null).OrderByDescending(value=>(long)value.GetField("RunId").GetValue(null)).FirstOrDefault();
        ctx.ReturnValue=type==null?(object)new{error="No KT02 rotation artifact story loaded"}:type.GetField("Report").GetValue(null);
    }
}
