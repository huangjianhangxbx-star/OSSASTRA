/* 为本轮纯原创样板显式制作一次地形/道路/花草；仅新OriginalLab，无每帧重建。 */
using System;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine;using UnityEngine.SceneManagement;using UnityEngine.Tilemaps;using Funplay.Editor.Tools.Scripting;
using G=Ossastra.GroundResearch;using E=Ossastra.GroundResearch.Editor;
public sealed class BuildOriginalAuthorSample:IFunplayCommand
{
 public void Execute(ExecutionContext ctx)
 {
  for(int i=0;i<SceneManager.sceneCount;i++)if(SceneManager.GetSceneAt(i).isDirty)throw new InvalidOperationException("有未保存场，样板不切换。");
  var scene=EditorSceneManager.OpenScene(E.GroundWorkshopOwnership.OriginalScene,OpenSceneMode.Single);
  var map=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<G.GroundWorkshopMap>(true)).Single();E.GroundWorkshopOwnership.Require(map);
  var area=map.surface.EditableCells;int moss=Array.FindIndex(map.theme.layers,l=>l.purpose==G.GroundLayerPurpose.Overlay),road=Array.FindIndex(map.theme.layers,l=>l.purpose==G.GroundLayerPurpose.Road);
  bool empty=map.surface.layers[moss].tilemap.GetTilesBlock(area).All(t=>t==null)&&map.surface.layers[road].tilemap.GetTilesBlock(area).All(t=>t==null)&&map.decorationRoot.childCount==0;
  if(!empty)throw new InvalidOperationException("原创样板已有作者内容，保留原值，不重复造样。");
  var brush=ScriptableObject.CreateInstance<E.FrontierGroundBrush>();brush.hideFlags=HideFlags.HideAndDontSave;
  try
  {
   if(!brush.Configure(map.surface,moss,1))throw new InvalidOperationException("覆盖层不允许写入");
   var target=map.surface.layers[moss].tilemap;
   brush.BoxFill(map.surface.grid,target.gameObject,new BoundsInt(-6,-3,0,5,4,1));brush.FinishStroke(false);
   brush.BoxFill(map.surface.grid,target.gameObject,new BoundsInt(1,1,0,4,3,1));brush.FinishStroke(false);
   if(!brush.Configure(map.surface,road,1))throw new InvalidOperationException("道路层不允许写入");
   target=map.surface.layers[road].tilemap;
   for(int x=area.xMin+1;x<area.xMax-1;x++){int y=x< -2?-1:x<2?0:1;brush.Paint(map.surface.grid,target.gameObject,new Vector3Int(x,y,0));}brush.FinishStroke(false);
  }
  finally{brush.FinishStroke(false);UnityEngine.Object.DestroyImmediate(brush);}
  map.theme.densityPer100=1.4f;map.theme.minimumSpacing=2.7f;map.theme.seed=123;EditorUtility.SetDirty(map.theme);
  int generated=E.GroundWorkshopDecorations.Scatter(map,map.theme.decorations,area,123,1.4f,2.7f,true);
  var p=map.surface.grid.GetCellCenterWorld(new Vector3Int(-3,2,0));p.y=map.surface.grid.transform.position.y;
  var manual=E.GroundWorkshopDecorations.Place(map,map.theme.decorations.First(d=>d.mode!=G.GroundDecorationMode.Flat),p,true);E.GroundWorkshopDecorations.SetProtected(manual,true);
  E.GroundWorkshopMaps.Save(map);
  var deps=AssetDatabase.GetDependencies(scene.path,true);bool independent=deps.All(p=>p==scene.path||p.StartsWith(E.GroundWorkshopOwnership.Root+"/",StringComparison.Ordinal)||p.StartsWith("Assets/OSSASTRA/Scripts/GroundResearch/",StringComparison.Ordinal)||p.StartsWith("Packages/",StringComparison.Ordinal));
  ctx.ReturnValue=new{path=scene.path,dirty=scene.isDirty,generated,manualProtected=manual.isLocked,decorationCount=map.decorationRoot.childCount,
   cells=map.surface.layers.Select(l=>new{l.displayName,count=l.tilemap.GetTilesBlock(area).Count(t=>t!=null)}).ToArray(),independent,foreignDependencies=deps.Where(p=>!(p==scene.path||p.StartsWith(E.GroundWorkshopOwnership.Root+"/",StringComparison.Ordinal)||p.StartsWith("Assets/OSSASTRA/Scripts/GroundResearch/",StringComparison.Ordinal)||p.StartsWith("Packages/",StringComparison.Ordinal))).ToArray()};
 }
}
