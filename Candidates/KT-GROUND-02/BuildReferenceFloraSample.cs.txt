/* 本机七层参考样板接三份已验证导入花草；只保存新参考场及其独有Theme。 */
using System;using System.Linq;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
using G=Ossastra.GroundResearch;using E=Ossastra.GroundResearch.Editor;
public sealed class BuildReferenceFloraSample:IFunplayCommand
{
 public void Execute(ExecutionContext ctx)
 {
  for(int i=0;i<SceneManager.sceneCount;i++)if(SceneManager.GetSceneAt(i).isDirty)throw new InvalidOperationException("保留未保存场。");
  var scene=EditorSceneManager.OpenScene(E.GroundWorkshopOwnership.ReferenceScene,OpenSceneMode.Single);
  var map=scene.GetRootGameObjects().SelectMany(r=>r.GetComponentsInChildren<G.GroundWorkshopMap>(true)).Single();E.GroundWorkshopOwnership.Require(map);
  if(map.decorationRoot.childCount!=0)throw new InvalidOperationException("新参考场已有花草，不重复生成。");
  var document=E.GroundWorkshopCatalog.Load(SessionState.GetString("GroundCatalog.TestManifest",""));
  string[] ids={"kt-c93658d1b0fcb1d590583217","kt-3504e2a3461a11c3287ff8e2","kt-15fbe741ca5ed239c61c3b9f"};
  var definitions=ids.Select(id=>(G.GroundDecorationDefinition)E.GroundWorkshopCatalog.ImportEntry(document,document.entries.Single(e=>e.assetId==id))).ToArray();
  map.theme.decorations=definitions;map.theme.densityPer100=1.4f;map.theme.minimumSpacing=2.7f;map.theme.seed=123;EditorUtility.SetDirty(map.theme);
  int count=E.GroundWorkshopDecorations.Scatter(map,definitions,map.surface.EditableCells,123,1.4f,2.7f,true);E.GroundWorkshopMaps.Save(map);
  ctx.ReturnValue=new{path=scene.path,dirty=scene.isDirty,generated=count,definitions=definitions.Select(d=>new{d.assetId,d.displayName,mode=d.mode.ToString()}).ToArray()};
 }
}
