using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Newtonsoft.Json;using Funplay.Editor.Tools.Scripting;
public sealed class GrassMainInspect:IFunplayCommand {
 public void Execute(ExecutionContext ctx){var s=SceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new InvalidOperationException("Expected main Edit");
 Func<Transform,string> path=t=>{string p=t.name;while(t.parent!=null){t=t.parent;p=t.name+"/"+p;}return p;};
 var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();
 var result=new{scene=s.path,dirty=s.isDirty,roots=s.GetRootGameObjects().Select(g=>new{id=g.GetInstanceID(),g.name,active=g.activeSelf}).ToArray(),components=all.SelectMany(t=>t.GetComponents<MonoBehaviour>()).Where(m=>m!=null).Select(m=>new{type=m.GetType().FullName,id=m.GetInstanceID(),path=path(m.transform),json=EditorJsonUtility.ToJson(m)}).ToArray(),surfaces=all.Where(t=>t.GetComponent<MeshCollider>()!=null||t.name.Contains("Plateau")).Select(t=>new{id=t.GetInstanceID(),path=path(t),pos=t.position.ToString("F3"),scale=t.lossyScale.ToString("F3"),active=t.gameObject.activeInHierarchy,collider=t.GetComponent<Collider>()==null?null:new{id=t.GetComponent<Collider>().GetInstanceID(),center=t.GetComponent<Collider>().bounds.center.ToString("F4"),size=t.GetComponent<Collider>().bounds.size.ToString("F4")},materials=t.GetComponent<Renderer>()==null?null:t.GetComponent<Renderer>().sharedMaterials.Select(m=>new{name=m==null?null:m.name,shader=m==null||m.shader==null?null:m.shader.name,path=AssetDatabase.GetAssetPath(m)}).ToArray()}).ToArray()};
 string file="artifacts/VEGETATION-01A/main-integration/initial.json";if(File.Exists(file))throw new InvalidOperationException("Snapshot already exists");File.WriteAllText(file,JsonConvert.SerializeObject(result,Formatting.Indented));ctx.ReturnValue=new{file,scene=s.path,dirty=s.isDirty,roots=all.Length,surfaceCount=result.surfaces.Length};}
}
