 string ExclusionReason(Vector3 p)
 {
  foreach(var building in buildings)
  {
   var local=building.transform.InverseTransformPoint(p);var scale=building.transform.lossyScale;
   foreach(var box in building.occupancyVolumes){var d=local-box.center;var h=box.size*.5f;if(Mathf.Abs(d.x)<=h.x+BuildingPadding/Mathf.Abs(scale.x)&&Mathf.Abs(d.z)<=h.z+BuildingPadding/Mathf.Abs(scale.z))return "building-footprint";}
  }
  foreach(var road in roads)
  {
   float y=road.surface.WorldBounds.center.y;if(Mathf.Abs(p.y-y)>.55f)continue;
   foreach(var offset in new[]{Vector2.zero,Vector2.left*.26f,Vector2.right*.26f,Vector2.up*.26f,Vector2.down*.26f})
    if(road.map.HasTile(road.map.WorldToCell(new Vector3(p.x+offset.x,y,p.z+offset.y))))return "registered-road";
  }
  foreach(var c in overhangs)
  {
   var bounds=c.bounds;if(p.x<bounds.min.x||p.x>bounds.max.x||p.z<bounds.min.z||p.z>bounds.max.z||bounds.max.y<p.y+.04f)continue;
   float top=bounds.max.y+1;RaycastHit hit;if(c.Raycast(new Ray(new Vector3(p.x,top,p.z),Vector3.down),out hit,top-p.y+.1f)&&hit.point.y>p.y+.04f)return "higher-explicit-surface";
  }
  return null;
 }
