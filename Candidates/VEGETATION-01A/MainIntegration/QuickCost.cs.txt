using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Ossastra.MapResearch;using Ossastra.WorldRendering.Vegetation;using Funplay.Editor.Tools.Scripting;
public sealed class GrassMainQuickCost:IFunplayCommand{
 static EditorApplication.CallbackFunction update;static GrassLayer grass;static MapResearchRoot map;static bool follow,enabled;static Vector3 target;static int stage,lastFrame,warm;static double deadline;static List<double>[] frame,gpu;static FrameTiming[] timings=new FrameTiming[1];
 static double? Q(List<double> a,double p){if(a.Count==0)return null;var s=a.OrderBy(x=>x).ToArray();return s[(int)Math.Floor((s.Length-1)*p)];}
 static void Finish(string error){EditorApplication.update-=update;update=null;if(grass!=null)grass.enabled=enabled;if(map!=null){map.followParticipant=follow;map.fixedTarget=target;}var rows=Enumerable.Range(0,3).Select(i=>new{state=i==1?"grass-on":"grass-off",samples=frame[i].Count,p50=Q(frame[i],.5),p95=Q(frame[i],.95),gpuSamples=gpu[i].Count,gpuP50=Q(gpu[i],.5),gpuP95=Q(gpu[i],.95)}).ToArray();File.WriteAllText("artifacts/VEGETATION-01A/main-integration/main-cost.json",Newtonsoft.Json.JsonConvert.SerializeObject(new{error,scope="Editor fixed camera off/on/off; whole-frame elapsed and optional FrameTiming GPU, not Player budget or CPU-effective work",rows,restored=true},Newtonsoft.Json.Formatting.Indented));}
 public void Execute(ExecutionContext ctx){if(!Application.isPlaying||update!=null)throw new InvalidOperationException("Expected idle Main Play");var roots=SceneManager.GetActiveScene().GetRootGameObjects();grass=roots.Single(g=>g.name=="GrassMain_原生地台半幅_二维草浪").GetComponent<GrassLayer>();map=roots.Single(g=>g.name=="MapResearchRoot").GetComponent<MapResearchRoot>();enabled=grass.enabled;follow=map.followParticipant;target=map.fixedTarget;map.followParticipant=false;map.fixedTarget=new Vector3(10,.5f,-12);stage=0;warm=60;lastFrame=-1;deadline=EditorApplication.timeSinceStartup+90;frame=Enumerable.Range(0,3).Select(i=>new List<double>()).ToArray();gpu=Enumerable.Range(0,3).Select(i=>new List<double>()).ToArray();grass.enabled=false;
 update=()=>{try{if(!Application.isPlaying||EditorApplication.timeSinceStartup>deadline){Finish("interrupted-or-deadline");return;}if(Time.frameCount==lastFrame)return;lastFrame=Time.frameCount;if(warm-->0)return;frame[stage].Add(Time.unscaledDeltaTime*1000);FrameTimingManager.CaptureFrameTimings();if(FrameTimingManager.GetLatestTimings(1,timings)>0&&timings[0].gpuFrameTime>0)gpu[stage].Add(timings[0].gpuFrameTime);if(frame[stage].Count>=180){stage++;if(stage==3){Finish(null);return;}grass.enabled=stage==1;warm=60;}}catch(Exception e){Finish(e.ToString());}};EditorApplication.update+=update;ctx.ReturnValue=new{started=true,file="artifacts/VEGETATION-01A/main-integration/main-cost.json",target="fixed45 grass area",states=3,samplesPerState=180};}
}
