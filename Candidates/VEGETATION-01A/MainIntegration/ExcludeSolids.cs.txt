using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Vegetation;using Funplay.Editor.Tools.Scripting;
public sealed class GrassExcludeSolids:IFunplayCommand{public void Execute(ExecutionContext ctx){if(Application.isPlaying)throw new InvalidOperationException("Edit required");var s=SceneManager.GetActiveScene();if(s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new InvalidOperationException("Wrong scene");var l=s.GetRootGameObjects().Single(g=>g.name=="GrassMain_原生地台半幅_二维草浪").GetComponent<GrassLayer>();if(l==null||l.data==null)throw new InvalidOperationException("Missing own grass");var boxes=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<BoxCollider>(true)).Where(b=>b.enabled&&!b.isTrigger&&b.gameObject.activeInHierarchy).ToArray();var previous=l.data.placements;float pad=l.profile.maximumBend+l.profile.widthRange.y*.5f;Func<Vector3,bool> inside=p=>boxes.Any(b=>{var q=b.transform.InverseTransformPoint(p)-b.center;var half=b.size*.5f;var scale=b.transform.lossyScale;return Mathf.Abs(q.x)<=half.x+pad/Mathf.Abs(scale.x)&&Mathf.Abs(q.z)<=half.z+pad/Mathf.Abs(scale.z)&&q.y>=-half.y-.15f/Mathf.Abs(scale.y)&&q.y<=half.y+.01f;});var keep=previous.Where(p=>!inside(p.worldRoot)).ToArray();ctx.RegisterObjectModification(l.data);ctx.RegisterObjectModification(l);l.data.placements=keep;l.exclusions=boxes;EditorUtility.SetDirty(l.data);AssetDatabase.SaveAssetIfDirty(l.data);l.RebuildFromSaved();EditorSceneManager.MarkSceneDirty(s);var result=new{before=previous.Length,after=keep.Length,excluded=previous.Length-keep.Length,boxes=boxes.Length,pad,l.BlockCount,acceptedInside=keep.Count(p=>inside(p.worldRoot)),mainSaved=false};File.WriteAllText("artifacts/VEGETATION-01A/main-integration/solid-exclusions.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=result;}}
