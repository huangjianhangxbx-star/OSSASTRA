using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Ossastra.WorldRendering.Vegetation;using Ossastra.SceneVisibility.Rooms;using Funplay.Editor.Tools.Scripting;

approved B profile");mesh=new Mesh();mesh.vertices=new[]{new Vector3(-1,0,-1),new Vector3(-1,0,1),new Vector3(1,0,1),new
