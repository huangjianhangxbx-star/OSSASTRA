 var result=new{status="captured-not-borrowed",utc=DateTime.UtcNow.ToString("o"),scene=scene.path,sceneGuid=AssetDatabase.AssetPathToGUID(scene.path),sceneSha=Hash(scene.path),dirty=scene.isDirty,roots=roots.Select(g=>new{id=GlobalObjectId.GetGlobalObjectIdSlow(g).ToString(),instance=g.GetInstanceID(),name=g.name,active=g.activeSelf}).ToArray(),objects,protectedFiles=profilePaths.Select(p=>new{path=p,sha=Hash(p)}).ToArray(),sceneSaved=false};File.WriteAllText(path,JObject.FromObject(result).ToString());ctx.ReturnValue=new{path,components=objects.Length,roots=roots.Length,scene.isDirty,sceneSaved=false};}}
