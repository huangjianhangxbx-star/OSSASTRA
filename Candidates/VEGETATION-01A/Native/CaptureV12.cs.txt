using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Vegetation;using Ossastra.WorldRendering.Vegetation.Development;using Funplay.Editor.Tools.Scripting;
public sealed class CaptureGrassV12:IFunplayCommand{public void Execute(ExecutionContext ctx){const string dir="Assets/OSSASTRA/WorldRendering/Vegetation/GrassLab/";var scene=SceneManager.GetSceneByPath(dir+"SC_GrassWindLab.unity");if(!scene.isLoaded)throw new InvalidOperationException("Owned lab required");var controls=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<GrassLabControls>(true)).Single();var p=AssetDatabase.LoadAssetAtPath<GrassLayerProfileSO>(dir+"GrassLayerProfile.asset");if(p==null)throw new InvalidOperationException("Owned profile missing");const string output="artifacts/VEGETATION-01A/v12-r0.json";if(File.Exists(output))throw new InvalidOperationException("Snapshot exists, inspect");var result=new{utc=DateTime.UtcNow.ToString("o"),play=Application.isPlaying,scene=scene.path,profile=EditorJsonUtility.ToJson(p,true),controls=EditorJsonUtility.ToJson(controls,true),layers=controls.layers.Select(l=>new{l.name,json=EditorJsonUtility.ToJson(l,true),l.ClumpCount,l.BlockCount}).ToArray(),mainSaved=false};File.WriteAllText(output,Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=new{output,clumps=controls.layers.Sum(l=>l.ClumpCount),p.density,p.maximumBend};}}
