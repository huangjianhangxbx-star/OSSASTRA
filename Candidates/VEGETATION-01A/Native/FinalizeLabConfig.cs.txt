using System;using System.IO;using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Vegetation;using Ossastra.WorldRendering.Vegetation.Development;using Ossastra.WorldRendering.Vegetation.Editor;using Funplay.Editor.Tools.Scripting;
public sealed class FinalizeGrassLabConfig:IFunplayCommand{public void Execute(ExecutionContext ctx){const string scenePath="Assets/OSSASTRA/WorldRendering/Vegetation/GrassLab/SC_GrassWindLab.unity";var scene=SceneManager.GetSceneByPath(scenePath);if(Application.isPlaying||SceneManager.sceneCount!=1||!scene.isLoaded)throw new InvalidOperationException("Only owned Lab Edit");var c=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<GrassLabControls>(true)).Single();var map=c.layers[0].terrainColor;var guid=AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(map.texture));int version=map.version;var clock=System.Diagnostics.Stopwatch.StartNew();string bake=GrassLayerEditor.BakeTerrainColor(c.layers[0],256);clock.Stop();if(map.version!=version+1||AssetDatabase.AssetPathToGUID(AssetDatabase.GetAssetPath(map.texture))!=guid)throw new InvalidOperationException("Map update lost identity/version");
ctx.RegisterObjectModification(c);var run=c.GetComponent<GrassLabPlayerRun>();if(run==null)run=c.gameObject.AddComponent<GrassLabPlayerRun>();run.sampler=c.GetComponent<GrassLabCostSampler>();if(run.sampler==null)throw new InvalidOperationException("Existing cost sampler missing");run.sampler.controls=c;run.sampler.layers=c.layers;
c.candidateA.material.DisableKeyword("_GRASS_WIND");c.candidateB.material.EnableKeyword("_GRASS_WIND");foreach(var m in new[]{c.candidateA.material,c.candidateB.material}){EditorUtility.SetDirty(m);AssetDatabase.SaveAssetIfDirty(m);}c.environment.SetTimeHours(12);c.environment.SetClockRunning(false);c.environment.RefreshNow();c.cameraPitch=45;c.showPanel=true;c.expanded=false;c.acceptKeyboardInput=true;c.RefreshView();EditorSceneManager.MarkSceneDirty(scene);if(!EditorSceneManager.SaveScene(scene,scenePath,false))throw new InvalidOperationException("Owned scene save failed");var result=new{bakeMilliseconds=clock.Elapsed.TotalMilliseconds,bakeFrequency="manual/dirty only; no camera or runtime repaint",bakeTextureGuidPreserved=true,version=map.version,memoryRGBA32Bytes=c.layers.Sum(l=>l.terrainColor.texture.width*l.terrainColor.texture.height*4),layers=c.layers.Select(l=>new{l.name,l.ClumpCount,l.BlockCount,l.VertexCount,l.TriangleCount}).ToArray(),shaderVariants="Asset A static + B _GRASS_WIND; both included by explicit dependencies",sceneSaved=scene.path,mainTouched=false};File.WriteAllText("artifacts/VEGETATION-01A/v12/final-lab-config.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=result;}}
