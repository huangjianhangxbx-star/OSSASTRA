using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEditor;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public sealed class ImportGrassCandidates:IFunplayCommand{
 static void Folder(string path){if(AssetDatabase.IsValidFolder(path))return;int i=path.LastIndexOf('/');Folder(path.Substring(0,i));AssetDatabase.CreateFolder(path.Substring(0,i),path.Substring(i+1));}
 public void Execute(ExecutionContext ctx){if(Application.isPlaying||EditorApplication.isCompiling||BuildPipeline.isBuildingPlayer)throw new InvalidOperationException("Stable Editor required");const string root="Assets/OSSASTRA/Scripts/WorldRendering/Vegetation/";var pairs=new List<KeyValuePair<string,string>>();foreach(var p in Directory.GetFiles("Candidates/VEGETATION-01A/Runtime","*.cs.txt"))pairs.Add(new KeyValuePair<string,string>(p,root+"Runtime/"+Path.GetFileNameWithoutExtension(p)));pairs.Add(new KeyValuePair<string,string>("Candidates/VEGETATION-01A/Scatter/GrassScatter.cs.txt",root+"Runtime/GrassScatter.cs"));foreach(var p in Directory.GetFiles("Candidates/VEGETATION-01A/Rendering"))pairs.Add(new KeyValuePair<string,string>(p,root+"Rendering/"+Path.GetFileName(p)));pairs.Add(new KeyValuePair<string,string>("Candidates/VEGETATION-01A/Authoring/GrassLayerEditor.cs.txt",root+"Editor/GrassLayerEditor.cs"));pairs.Add(new KeyValuePair<string,string>("Candidates/VEGETATION-01A/Authoring/GrassLabControls.cs.txt",root+"Development/GrassLabControls.cs"));foreach(var p in pairs){if(!File.Exists(p.Key)||File.Exists(p.Value))throw new InvalidOperationException("Candidate missing or destination already exists: "+p.Key+" -> "+p.Value);}
 AssetDatabase.StartAssetEditing();try{foreach(var p in pairs){Folder(Path.GetDirectoryName(p.Value).Replace('\\','/'));File.WriteAllBytes(p.Value,File.ReadAllBytes(p.Key));AssetDatabase.ImportAsset(p.Value,ImportAssetOptions.ForceUpdate);}}finally{AssetDatabase.StopAssetEditing();}string output=Path.GetFullPath("artifacts/VEGETATION-01A/import-manifest.json");File.WriteAllText(output,Newtonsoft.Json.JsonConvert.SerializeObject(pairs,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=new{output,files=pairs.Count,sceneSaved=false,sharedBackendsModified=false};}}
