using System;using System.IO;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using Newtonsoft.Json.Linq;using Ossastra.WorldRendering.Vegetation;using Ossastra.WorldRendering.Vegetation.Development;using Funplay.Editor.Tools.Scripting;
public sealed class SetGrassVideoView:IFunplayCommand{public void Execute(ExecutionContext ctx){const string scenePath="Assets/OSSASTRA/WorldRendering/Vegetation/GrassLab/SC_GrassWindLab.unity";var scene=SceneManager.GetSceneByPath(scenePath);if(!Application.isPlaying||!scene.isLoaded)throw new InvalidOperationException("Owned Lab Play required");var c=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<GrassLabControls>(true)).Single();var q=JObject.Parse(File.ReadAllText("artifacts/VEGETATION-01A/v12/view-request.json"));if(q["candidateB"]!=null)c.SwitchCandidate((bool)q["candidateB"]);if(q["appearance"]!=null)c.SetAppearance((GrassAppearance)(int)q["appearance"]);if(q["motion"]!=null)c.SetMotion((GrassMotion)(int)q["motion"]);if(q["debug"]!=null)c.SetArtDebug((int)q["debug"]);if(q["rgb"]!=null)c.SetRgbDetail((float)q["rgb"]);if(q["hour"]!=null)c.SetTimeHours((double)q["hour"]);if(q["pitch"]!=null)c.cameraPitch=Mathf.Clamp((float)q["pitch"],25,60);if(q["step"]!=null)c.RequestRotation((int)q["step"]);if(q["panel"]!=null)c.showPanel=(bool)q["panel"];if(q["expanded"]!=null)c.expanded=(bool)q["expanded"];if(q["windMode"]!=null)c.SetMode((GrassWindMode)(int)q["windMode"],(bool?)q["shadows"]??false);if(q["directionX"]!=null)c.SetWindDirection(new Vector2((float)q["directionX"],(float)q["directionZ"]));
foreach(var layer in c.layers){var p=layer.profile;if(p==null||p.hideFlags!=HideFlags.DontSave)throw new InvalidOperationException("Only owned runtime profile can tune");if(q["depthStrength"]!=null)p.depthColorStrength=Mathf.Clamp01((float)q["depthStrength"]);if(q["windColorStrength"]!=null)p.windColorStrength=Mathf.Clamp01((float)q["windColorStrength"]);if(q["tint"]!=null)p.tint=Color.white*Mathf.Clamp((float)q["tint"],.1f,1);layer.RefreshAppearance();foreach(var m in layer.Renderers.Where(r=>r!=null).Select(r=>r.sharedMaterial).Where(m=>m!=null).Distinct()){m.SetColor("_Color",p.tint);if(q["displacement"]!=null)m.SetFloat("_WindDisplacement",Mathf.Clamp01((float)q["displacement"]));}}
if(q["localLight"]!=null){var lamp=scene.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Light>(true)).Single(l=>l.name=="LocalWarmLight");lamp.enabled=(bool)q["localLight"];c.environment.RefreshNow();}if(q["targetX"]!=null)c.target=new Vector3((float)q["targetX"],(float?)q["targetY"]??.7f,(float)q["targetZ"]);if(q["distance"]!=null)c.distance=Mathf.Clamp((float)q["distance"],5,32);c.RefreshView();ctx.ReturnValue=new{c.UsingCandidateB,pitch=c.cameraPitch,hour=c.environment.timeHours,appearance=c.layers[0].profile.appearance.ToString(),motion=c.layers[0].profile.motion.ToString(),debug=c.layers[0].profile.artDebug,clumps=c.layers.Where(l=>l.isActiveAndEnabled).Sum(l=>l.ClumpCount)};}}
