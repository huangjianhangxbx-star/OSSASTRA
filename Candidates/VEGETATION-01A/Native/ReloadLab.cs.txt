using System;using System.IO;using System.Linq;using System.Security.Cryptography;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.SceneManagement;using Ossastra.WorldRendering.Vegetation;using Ossastra.WorldRendering.Vegetation.Development;using Funplay.Editor.Tools.Scripting;
public sealed class ReloadGrassLab:IFunplayCommand{
static string Hash(string p){using(var h=SHA256.Create())using(var s=File.OpenRead(p))return BitConverter.ToString(h.ComputeHash(s)).Replace("-","");}
static GrassLabControls Controls(Scene s)=>s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<GrassLabControls>(true)).Single();
public void Execute(ExecutionContext ctx){const string path="Assets/OSSASTRA/WorldRendering/Vegetation/GrassLab/SC_GrassWindLab.unity";var scene=SceneManager.GetSceneByPath(path);if(Application.isPlaying||SceneManager.sceneCount!=1||!scene.isLoaded||scene.isDirty)throw new InvalidOperationException("Exact clean owned Lab Edit required");var c=Controls(scene);var paths=c.layers.SelectMany(l=>new[]{AssetDatabase.GetAssetPath(l.profile),AssetDatabase.GetAssetPath(l.data),AssetDatabase.GetAssetPath(l.terrainColor),AssetDatabase.GetAssetPath(l.terrainColor.texture)}).Distinct().ToArray();var hashes=paths.Select(Hash).ToArray();var meshes=c.layers.SelectMany(l=>l.Renderers).Select(r=>r.GetComponent<MeshFilter>().sharedMesh).ToArray();var mats=c.layers.SelectMany(l=>l.Renderers).Select(r=>r.sharedMaterial).Distinct().ToArray();foreach(var l in c.layers)l.enabled=false;bool released=c.layers.All(l=>l.BlockCount==0&&l.OwnedMaterialCount==0)&&meshes.All(m=>m==null)&&mats.All(m=>m==null);foreach(var l in c.layers)l.enabled=true;bool rebuilt=c.layers.Sum(l=>l.ClumpCount)==4099&&c.layers.Sum(l=>l.BlockCount)==27;if(!released||!rebuilt)throw new InvalidOperationException("Disable/rebuild resource contract failed");EditorSceneManager.CloseScene(scene,true);scene=EditorSceneManager.OpenScene(path,OpenSceneMode.Single);c=Controls(scene);bool reload=c.layers.Sum(l=>l.ClumpCount)==4099&&c.layers.All(l=>l.profile==c.candidateB&&l.terrainColor.surfaceId==l.surfaces.Single().id&&l.terrainColor.texture!=null)&&Mathf.Abs(c.candidateB.depthColorStrength-.04f)<.00001f&&Mathf.Abs(c.cameraPitch-45)<.001f&&paths.Select(Hash).SequenceEqual(hashes);var result=new{released,rebuilt,coldReopened=reload,clumps=c.layers.Sum(l=>l.ClumpCount),blocks=c.layers.Sum(l=>l.BlockCount),dirty=scene.isDirty,authorDataPreserved=paths.Select(Hash).SequenceEqual(hashes),maps=c.layers.Select(l=>new{l.name,l.terrainColor.surfaceId,l.terrainColor.version}).ToArray()};File.WriteAllText("artifacts/VEGETATION-01A/v12/reload-lifecycle.json",Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));ctx.ReturnValue=result;if(!reload)throw new InvalidOperationException("Cold reopen differs");}}
