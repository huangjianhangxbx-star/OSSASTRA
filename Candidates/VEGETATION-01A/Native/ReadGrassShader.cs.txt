using System;using System.Linq;using UnityEditor;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public sealed class ReadGrassShader:IFunplayCommand{public void Execute(ExecutionContext ctx){var paths=new[]{"Assets/OSSASTRA/Scripts/WorldRendering/Vegetation/Rendering/GrassLayer.shader","Assets/OSSASTRA/Scripts/WorldRendering/Vegetation/Rendering/TerrainBaseColorBake.shader"};foreach(var m in typeof(ShaderUtil).GetMethods(System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.Static).Where(m=>m.Name=="CompilePass"))Debug.Log(m.ToString());ctx.ReturnValue=paths.Select(p=>{var s=AssetDatabase.LoadAssetAtPath<Shader>(p);return new{path=p,present=s!=null,supported=s!=null&&s.isSupported,messages=s==null?null:ShaderUtil.GetShaderMessages(s).Select(m=>new{m.message,m.line,severity=m.severity.ToString()}).ToArray()};}).ToArray();}}
