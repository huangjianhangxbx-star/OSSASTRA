 static Mesh PatchMesh(Mesh source,Transform sourceTransform,Transform owner,int[] selected,int number)
 {
  var positions=source.vertices;var normals=source.normals;var uv=source.uv;
  var indices=new Dictionary<int,int>();var v=new List<Vector3>();var n=new List<Vector3>();var tex=new List<Vector2>();var triangles=new List<int>();
  foreach(int original in selected)
  {
   if(!indices.TryGetValue(original,out var index))
   {
    index=v.Count;indices.Add(original,index);
    var world=sourceTransform.TransformPoint(positions[original])+Vector3.up*.003f;
    v.Add(owner.InverseTransformPoint(world));
    var worldNormal=sourceTransform.localToWorldMatrix.inverse.transpose.MultiplyVector(normals[original]).normalized;
    n.Add(owner.localToWorldMatrix.transpose.MultiplyVector(worldNormal).normalized);
    tex.Add(uv.Length==positions.Length?uv[original]:Vector2.zero);
   }
   triangles.Add(index);
  }
  var mesh=new Mesh{name=ZoneNames[number]+"_Mesh",indexFormat=v.Count>65000?IndexFormat.UInt32:IndexFormat.UInt16};
  mesh.SetVertices(v);mesh.SetNormals(n);mesh.SetUVs(0,tex);mesh.SetTriangles(triangles,0);mesh.RecalculateBounds();return mesh;
 }
 static Texture2D WindData()
 {
  const int size=64;var texture=new Texture2D(size,size,TextureFormat.RGBA32,true,true)
   {name="R1_WindData64",wrapMode=TextureWrapMode.Repeat,filterMode=FilterMode.Bilinear};
  var pixels=new Color[size*size];
  for(int y=0;y<size;y++)for(int x=0;x<size;x++)
  {
   double a=2*Math.PI*(x+.5)/size,b=2*Math.PI*(y+.5)/size;
   // Integer spatial frequencies are periodic on both texture axes; diagnostic wind data, not new painted grass.
   float value=(float)(.5+.18*Math.Sin(a+.7)*Math.Cos(b+.3)+.14*Math.Sin(2*a+b+.5)+.08*Math.Cos(a-3*b+.2));
   value=Mathf.Clamp01(value);pixels[y*size+x]=new Color(value,value,value,1);
  }
  texture.SetPixels(pixels);texture.Apply(true,false);return texture;
 }
