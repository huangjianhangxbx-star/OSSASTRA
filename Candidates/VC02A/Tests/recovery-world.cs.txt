using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class RecoveryWorldProbe:IFunplayCommand
{
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
 {
  if(!Application.isPlaying)throw new Exception("Play required");var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var a=c.Current;var b=c.Characters.First(r=>r!=a);var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();var recovery=a.Vitals.Recovery;if(recovery==null)throw new Exception("Missing recovery component");
  foreach(var r in c.Characters){r.GetComponent<CompanionCombatBrain>().enabled=false;r.SendMessage("OnApplicationFocus",true);r.CancelAction("recovery-probe");r.Vitals.RestoreForDevelopment();}
  var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();if(engagement!=null)engagement.enabled=false;e.AttacksEnabled=false;var brain=e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();if(brain!=null)brain.enabled=false;e.RestoreForDevelopment();
  a.GetComponent<CharacterMovement>().Teleport(new Vector3(0,.03f,-2));b.GetComponent<CharacterMovement>().Teleport(new Vector3(-4,.03f,-2));var body=e.GetComponent<CharacterController>();body.enabled=false;e.transform.position=new Vector3(1.1f,.03f,-2);body.enabled=true;Physics.SyncTransforms();
  var rows=new List<object>();int failed=0,next=0,last=-1;var start=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();double hold=0;var parent=Guid.NewGuid();var queue=new CombatDeliveryQueue();
  void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,hp=a.Vitals.Health.Current,gray=recovery.Gray,parents=recovery.ActiveParents});}
  void Flush(string state){SessionState.SetString("VC02_RECOVERY_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,rows}));}
  void Hurt(int damage){var runtime=new BasicActionRuntime(e.ActorId);runtime.Request(new BasicAttackDefinition(0,0,0,.1,damage,0),"actual-contact",new ControlAuthority());e.GetComponent<CombatMeleeHit>().Resolve(runtime.Advance(0,false).Single(f=>f.Release!=null).Release,Vector3.left);}
  var delivery=new DeliveryParameters{mode=DeliveryMode.Sector,range=2,radius=.6f,arc=120};
  bool Emit(int index,double rate=.5)=>queue.Emit(a.Vitals,parent,index,delivery,20,0,a.transform.position,Vector3.right,e.transform.position,()=>a.IsAlive,null,null,.25f,.05,rate);
  steps.Add((.2,()=>{Hurt(40);Check("actual HP loss creates equal gray",a.Vitals.Health.Current==60&&recovery.Gray==40);a.SetPaused(true);hold=recovery.HoldRemaining;}));
  steps.Add((1,()=>{Check("pause freezes recovery hold",Math.Abs(recovery.HoldRemaining-hold)<.000001);a.SetPaused(false);}));
  steps.Add((1.2,()=>{var snapshot=UnityEngine.Object.FindObjectOfType<Ossastra.UserInterface.CharacterHudAdapter>().Read(false);Check("HUD projects actual gray HP",snapshot.grayHp==40&&snapshot.hp==60);recovery.RetainParent(parent);queue.RegisterParent(parent);Check("physical first pulse hits",Emit(0));Check("actual damage reclaims weapon budget",a.Vitals.Health.Current==65&&recovery.Gray==35);var hp=e.Vitals.Health.Current;Check("duplicate emission rejected",!Emit(0)&&e.Vitals.Health.Current==hp);Check("second pulse valid without new recovery budget",Emit(1)&&a.Vitals.Health.Current==65);queue.CompleteParent(parent);recovery.ReleaseParent(parent);Check("parent budget lifetime released",recovery.ActiveParents==0);}));
  steps.Add((1.5,()=>{var bottle=UnityEngine.Object.FindObjectOfType<HealingBottleService>();Check("shared Y heals through original service",bottle.TryUse()&&a.Vitals.Health.Current==100);Check("normal heal clamps gray",recovery.Gray==0);Hurt(30);parent=Guid.NewGuid();recovery.RetainParent(parent);queue.RegisterParent(parent);var before=a.Vitals.Health.Current;Check("periodic contact can damage without reclaim",Emit(0,0)&&a.Vitals.Health.Current==before);queue.CompleteParent(parent);recovery.ReleaseParent(parent);}));
  steps.Add((1.9,()=>{Hurt(999);Check("real death clears gray and cannot reclaim",!a.IsAlive&&recovery.Gray==0);Check("real death hands control to surviving partner",c.Current==b);Check("dead source cannot emit",!Emit(2));Check("death clears parent records",recovery.ActiveParents==0);}));
  EditorApplication.CallbackFunction tick=null;tick=()=>{if(Time.frameCount==last)return;last=Time.frameCount;try{if(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].Item1)steps[next++].Item2();if(next==steps.Count){EditorApplication.update-=tick;queue.Clear();Flush("completed");}}catch(Exception ex){EditorApplication.update-=tick;queue.Clear();failed++;rows.Add(new{error=ex.ToString()});Flush("failed");}};EditorApplication.update+=tick;Flush("running");
 }
}
