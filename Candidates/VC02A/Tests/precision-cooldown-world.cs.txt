using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.CharacterCore.Abilities;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class PrecisionProbe:IFunplayCommand
{
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
 {
  if(!Application.isPlaying)throw new Exception("Play required");
  var coordinator=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var actor=coordinator.Characters.Single(r=>r.ActorId=="temporary-b");if(coordinator.Current!=actor)coordinator.SwitchNext();
  var module=actor.GetComponent<CharacterAbilities>();var gear=actor.GetComponent<CharacterEquipment>();var state=gear.State.RightRuntime;var input=actor.GetComponent<CharacterInputHandler>();
  foreach(var root in coordinator.Characters){root.GetComponent<CompanionCombatBrain>().enabled=false;root.CancelAction("precision-probe");root.Vitals.RestoreForDevelopment();root.SendMessage("OnApplicationFocus",true);}
  foreach(var candidateEnemy in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){candidateEnemy.AttacksEnabled=false;var brain=candidateEnemy.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();if(brain!=null)brain.enabled=false;}
  var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();if(engagement!=null)engagement.enabled=false;
  var settingsBefore=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(settingsBefore);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
  var keyboard=InputSystem.AddDevice<Keyboard>("VC02PrecisionKeyboard");var old=input.skillOneAction.SaveBindingOverridesAsJson();for(int i=0;i<input.skillOneAction.bindings.Count;i++)input.skillOneAction.ApplyBindingOverride(i,input.skillOneAction.bindings[i].path.Replace("<Keyboard>",keyboard.path));
  EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();coordinator.SendMessage("OnApplicationFocus",true);
  var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();var enemyHp=enemy.Vitals.Health.Current;var collider=enemy.GetComponent<CharacterController>();collider.enabled=false;enemy.transform.position=actor.transform.position+actor.DataBus.AttackDirection*.5f;collider.enabled=true;Physics.SyncTransforms();var rows=new List<object>();int failed=0,next=0,last=-1,accepted=module.AcceptedCount;var start=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();
  void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,result=module.LastResult,ammo=state.Magazine.Count,reloading=state.Magazine.IsReloading});}
  void KeyDown(bool value){InputSystem.QueueStateEvent(keyboard,value?new KeyboardState(Key.Q):new KeyboardState());}
  void Flush(string status){SessionState.SetString("VC02_PRECISION",Newtonsoft.Json.JsonConvert.SerializeObject(new{state=status,failed,rows}));}
  steps.Add((.2,()=>{Check("Q config exposes precision in own profession",module.ConfigureSkill(0,"precision"));state.Magazine.Consume(Guid.NewGuid());Check("cooldown source grant",module.ToggleExperimentalSource("precision-cooldown"));Check("precondition actual magazine three",state.Magazine.Count==3);KeyDown(false);}));
  steps.Add((.5,()=>KeyDown(true)));steps.Add((.6,()=>{KeyDown(false);Check("ordinary Q accepted exactly once",module.AcceptedCount==accepted+1);Check("pending reload preserves existing rounds",state.Magazine.Count==3&&state.Magazine.IsReloading);}));
  steps.Add((1.5,()=>{Check("completion commits four actual mortar rounds",state.Magazine.Count==4&&state.Magazine.Next==AmmoKind.Mortar&&!state.Magazine.IsReloading);Check("body and run return ready",!actor.ActionActive&&!module.IsBusy);Check("canonical precision CD consumes minus4 modifier",module.SkillCooldown(0)>6&&module.SkillCooldown(0)<8);state.Tick(20,true);state.Magazine.Consume(Guid.NewGuid());KeyDown(true);}));
  steps.Add((1.65,()=>{KeyDown(false);Check("second request pending",state.Magazine.IsReloading);actor.CancelAction("probe interrupt");Check("interruption clears only pending reload",!state.Magazine.IsReloading&&state.Magazine.Count==3);}));
  steps.Add((2.7,()=>{Check("cancel has no late refill",state.Magazine.Count==3);state.Tick(20,true);Check("more source grants through normal configuration",module.ToggleExperimentalSource("precision-more"));Check("instant source grants through normal configuration",module.ToggleExperimentalSource("precision-instant"));KeyDown(true);}));
  steps.Add((2.85,()=>{KeyDown(false);Check("instant branch commits six actual mortar rounds",state.Magazine.Count==6&&state.Magazine.Next==AmmoKind.Mortar&&!state.Magazine.IsReloading);}));
  steps.Add((3.8,()=>{Check("instant has no second delayed refill",state.Magazine.Count==6);Check("instant reload body never falls back to template melee",enemy.Vitals.Health.Current==enemyHp);Check("sources can be revoked",module.ToggleExperimentalSource("precision-more")&&module.ToggleExperimentalSource("precision-instant"));Check("revoke does not rewrite committed magazine",state.Magazine.Count==6);Check("rocket alias available Q",module.ConfigureSkill(0,"rocket"));Check("alias shares E cooldown identity",module.SkillDefinition(0).cooldownIdentity==module.SkillDefinition(1).cooldownIdentity);Check("fixed E cannot be replaced",!module.ConfigureSkill(1,"precision"));}));
  EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;KeyDown(false);input.skillOneAction.LoadBindingOverridesFromJson(old);InputSystem.RemoveDevice(keyboard);InputSystem.settings=settingsBefore;UnityEngine.Object.DestroyImmediate(settings);}
  tick=()=>{if(Time.frameCount==last)return;last=Time.frameCount;try{if(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].Item1){var late=EditorApplication.timeSinceStartup-start-steps[next].Item1;if(late>.02)start+=late;steps[next++].Item2();}if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;rows.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");
 }
}
