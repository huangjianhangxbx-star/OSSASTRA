using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class SixSkillsProbe:IFunplayCommand
{
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
 {
  if(!Application.isPlaying)throw new Exception("Play required");
  var coordinator=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();
  var settingsBefore=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(settingsBefore);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
  var keyboard=InputSystem.AddDevice<Keyboard>("VC02SkillsKeyboard");var mouse=InputSystem.AddDevice<Mouse>("VC02SkillsMouse");var overrides=new Dictionary<InputAction,string>();
  foreach(var input in coordinator.Characters.Select(r=>r.GetComponent<CharacterInputHandler>()))foreach(var action in new[]{input.skillOneAction,input.skillTwoAction,input.skillThreeAction,input.mouseAction})
  {overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var p=action.bindings[i].path;action.ApplyBindingOverride(i,p.Replace("<Keyboard>",keyboard.path).Replace("<Mouse>",mouse.path));}}
  EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();coordinator.SendMessage("OnApplicationFocus",true);
  foreach(var root in coordinator.Characters){root.SendMessage("OnApplicationFocus",true);root.GetComponent<CompanionCombatBrain>().enabled=false;root.CancelAction("six-skills-probe");root.Vitals.RestoreForDevelopment();}
  var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();if(engagement!=null)engagement.enabled=false;enemy.AttacksEnabled=false;var brain=enemy.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();if(brain!=null)brain.enabled=false;
  var rows=new List<object>();int failed=0,next=0,lastFrame=-1;double start=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();
  Ossastra.Characters.CharacterRoot actor=null;CharacterAbilities module=null;var facts=new List<ActionFact>();int accepted=0,emitted=0,hp=0;float paid=0;Vector3 position=Vector3.zero;Guid acceptedId=Guid.Empty;
  void Flush(string state){SessionState.SetString("VC02_SIX_SKILLS",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,rows}));}
  void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,last=module?.LastResult,accepted=module?.AcceptedCount,emitted=module?.EmittedCount,hp=enemy.Vitals.Health.Current,position=actor?.transform.position.ToString(),facts=facts.Select(f=>f.Kind.ToString()).ToArray()});Flush("running");}
  void SendKey(Key? key){InputSystem.QueueStateEvent(keyboard,key.HasValue?new KeyboardState(key.Value):new KeyboardState());}
  void Setup(string actorId,float distance)
  {
   SendKey(null);actor=coordinator.Characters.Single(r=>r.ActorId==actorId);if(coordinator.Current!=actor)coordinator.SwitchNext();module=actor.GetComponent<CharacterAbilities>();
   foreach(var r in coordinator.Characters){r.CancelAction("reset-probe");r.GetComponent<CharacterStamina>().Restore();r.GetComponent<CharacterMovement>().Teleport(new Vector3(r==actor?0:-4,.03f,-2));}
   enemy.RestoreForDevelopment();var controller=enemy.GetComponent<CharacterController>();controller.enabled=false;enemy.transform.position=new Vector3(distance,.03f,-2);controller.enabled=true;Physics.SyncTransforms();
   var point=Camera.main.WorldToScreenPoint(enemy.transform.position);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(point.x,point.y)});
   accepted=module.AcceptedCount;emitted=module.EmittedCount;hp=enemy.Vitals.Health.Current;paid=(float)actor.GetComponent<CharacterStamina>().Current;position=actor.transform.position;facts.Clear();actor.ActionObserved+=facts.Add;
  }
  void Finish(string id,int damage)
  {
   Check(id+" accepts once through ordinary InputSystem",module.AcceptedCount==accepted+1);
   Check(id+" actual release and HP receipt",module.EmittedCount==emitted+1&&enemy.Vitals.Health.Current==hp-damage);
   Check(id+" returns body to ready",!actor.ActionActive&&!module.IsBusy);
   actor.ActionObserved-=facts.Add;
  }
  steps.Add((.2,()=>Setup("temporary-a",1.1f)));steps.Add((.5,()=>SendKey(Key.Q)));steps.Add((.6,()=>SendKey(null)));steps.Add((1.1,()=>Finish("hunter-break",20)));
  steps.Add((1.3,()=>Setup("temporary-a",1.1f)));steps.Add((1.6,()=>SendKey(Key.R)));steps.Add((1.7,()=>SendKey(null)));steps.Add((2.3,()=>Finish("hunter-sweep",18)));
  steps.Add((2.5,()=>Setup("temporary-a",1.1f)));steps.Add((2.8,()=>SendKey(Key.E)));
  steps.Add((3.05,()=>{acceptedId=actor.DataBus.ActionId;Check("rush retreat is actual movement before release",actor.transform.position.x<position.x-.5f&&module.EmittedCount==emitted);Check("rush pays only initial28",Math.Abs(actor.GetComponent<CharacterStamina>().Current-(paid-28))<.1);SendKey(null);}));
  steps.Add((4.1,()=>{Finish("hunter-native-E",75);Check("rush phases retain single accepted identity",facts.Count(f=>f.Kind==ActionFactKind.Accepted)==1&&facts.Count(f=>f.Kind==ActionFactKind.PhaseChanged)==2&&facts.All(f=>f.ActionId==acceptedId));Check("rush ordinary contact stops charge while target is displaced",actor.transform.position.x<1.1f&&enemy.transform.position.x>2.5f);}));
  steps.Add((4.3,()=>Setup("temporary-b",1.1f)));steps.Add((4.6,()=>SendKey(Key.Q)));steps.Add((4.7,()=>SendKey(null)));steps.Add((5.2,()=>Finish("al-stab",20)));
  steps.Add((5.4,()=>Setup("temporary-b",1.1f)));steps.Add((5.7,()=>SendKey(Key.R)));steps.Add((5.8,()=>SendKey(null)));steps.Add((6.4,()=>Finish("al-spin",18)));
  steps.Add((6.6,()=>Setup("temporary-b",3)));steps.Add((6.9,()=>SendKey(Key.E)));steps.Add((7,()=>SendKey(null)));
  steps.Add((8.1,()=>{Finish("al-native-E",85);Check("rocket uses actual movement before landing damage",Vector3.Distance(actor.transform.position,position)>1);Check("native E retains canonical cooldown",module.SkillCooldown(1)>4);Check("E cannot be reconfigured",!module.ConfigureSkill(1,"al-stab"));}));
  EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;SendKey(null);foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=settingsBefore;UnityEngine.Object.DestroyImmediate(settings);}
  tick=()=>{if(Time.frameCount==lastFrame)return;lastFrame=Time.frameCount;try{if(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].Item1){var late=EditorApplication.timeSinceStartup-start-steps[next].Item1;if(late>.02)start+=late;steps[next++].Item2();}if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;rows.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");
 }
}
