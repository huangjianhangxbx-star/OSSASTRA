using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using UnityEditor;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.CharacterCore.Abilities;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class EnhancementProbe:IFunplayCommand
{
 public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx)
 {
  if(!Application.isPlaying)throw new Exception("Play required");var coordinator=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var actor=coordinator.Characters.Single(r=>r.ActorId=="temporary-b");if(coordinator.Current!=actor)coordinator.SwitchNext();var module=actor.GetComponent<CharacterAbilities>();var gear=actor.GetComponent<CharacterEquipment>();var state=gear.State.RightRuntime;var input=actor.GetComponent<CharacterInputHandler>();var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();
  foreach(var root in coordinator.Characters){root.GetComponent<CompanionCombatBrain>().enabled=false;root.CancelAction("magazine-probe");root.Vitals.RestoreForDevelopment();root.SendMessage("OnApplicationFocus",true);root.GetComponent<CharacterMovement>().Teleport(new Vector3(root==actor?0:-4,.03f,-2));}
  enemy.AttacksEnabled=false;var brain=enemy.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();if(brain!=null)brain.enabled=false;var engagement=UnityEngine.Object.FindObjectOfType<CombatEngagementSession>();if(engagement!=null)engagement.enabled=false;
  var controller=enemy.GetComponent<CharacterController>();controller.enabled=false;enemy.transform.position=new Vector3(3,.03f,-2);controller.enabled=true;Physics.SyncTransforms();enemy.RestoreForDevelopment();
  var settingsBefore=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(settingsBefore);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
  var mouse=InputSystem.AddDevice<Mouse>("VC02MagazineMouse");var keyboard=InputSystem.AddDevice<Keyboard>("VC02MagazineKeys");var overrides=new Dictionary<InputAction,string>();
  foreach(var a in new[]{input.offhandAction,input.mouseAction,input.skillOneAction}){overrides[a]=a.SaveBindingOverridesAsJson();for(int i=0;i<a.bindings.Count;i++)a.ApplyBindingOverride(i,a.bindings[i].path.Replace("<Mouse>",mouse.path).Replace("<Keyboard>",keyboard.path));}
  EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();coordinator.SendMessage("OnApplicationFocus",true);var aim=Camera.main.WorldToScreenPoint(enemy.transform.position);
  void Mouse(bool held){InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(aim.x,aim.y),buttons=(ushort)(held?2:0)});}
  void Key(bool held){InputSystem.QueueStateEvent(keyboard,held?new KeyboardState(UnityEngine.InputSystem.Key.Q):new KeyboardState());}
  var rows=new List<object>();int failed=0,next=0,last=-1,accepted=module.AcceptedCount,hp=enemy.Vitals.Health.Current;var start=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();
  void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,result=module.LastResult,ammo=state.Magazine.Count,hp=enemy.Vitals.Health.Current,accepted=module.AcceptedCount});}
  void Flush(string status){SessionState.SetString("VC02_SHOT_ENHANCEMENT",Newtonsoft.Json.JsonConvert.SerializeObject(new{state=status,failed,rows}));}
  steps.Add((.2,()=>{Check("equip own magazine delivery through normal equipment API",gear.TryEquip(gear.OwnedOffhands.Single(d=>d.definitionId=="vc02.magazine-shooter")));Check("precision remains Q configuration",module.ConfigureSkill(0,"precision"));Mouse(false);Key(false);}));
  steps.Add((.3,()=>{foreach(var id in new[]{"precision-critical","precision-speed","precision-damage"})Check("ordinary source grant "+id,module.ToggleExperimentalSource(id));}));steps.Add((.5,()=>Mouse(true)));steps.Add((.6,()=>Mouse(false)));
  steps.Add((1.2,()=>{Check("right mouse ordinary shot accepted once",module.AcceptedCount==accepted+1);Check("shot consumes main payload not left probe magazine",state.Magazine.Count==3&&gear.State.LeftRuntime.Magazine.Count==4);Check("actual projectile causes source32 damage",enemy.Vitals.Health.Current==hp-32);Key(true);}));
  steps.Add((1.3,()=>Key(false)));steps.Add((2.3,()=>{Check("precision fills same consumed magazine with mortar",state.Magazine.Count==4&&state.Magazine.Next==AmmoKind.Mortar);enemy.RestoreForDevelopment();hp=enemy.Vitals.Health.Current;Mouse(true);}));
  steps.Add((2.45,()=>Mouse(false)));steps.Add((3.4,()=>{Check("mortar consumes only one special round",state.Magazine.Count==3);Check("mortar actual landing applies frozen135percent and225percent critical",enemy.Vitals.Health.Current==hp-243);Check("same release uses shared queue cleanup",module.InFlightCount==0);Check("critical is consumed only after Applied contact",state.ShotEnhancements.Snapshot().CriticalSource==null);Check("reload shot rate remains active",Math.Abs(state.ShotEnhancements.Snapshot().Rate-1.4)<.01);Mouse(false);}));
steps.Add((3.8,()=>{Check("last round source grant",module.ToggleExperimentalSource("last-round"));state.Magazine.Consume(Guid.NewGuid());state.Magazine.Consume(Guid.NewGuid());hp=enemy.Vitals.Health.Current;Check("last round is actual remaining mortar",state.Magazine.Count==1);}));
steps.Add((4.1,()=>Mouse(true)));steps.Add((4.2,()=>Mouse(false)));
steps.Add((4.95,()=>{Check("reload damage plus last round adds before multiplier208",enemy.Vitals.Health.Current==hp-208);enemy.RestoreForDevelopment();hp=enemy.Vitals.Health.Current;Check("enhanced last round grant",module.ToggleExperimentalSource("last-round-plus"));state.Magazine.CancelAutomaticReload();state.Magazine.BeginReload(Guid.NewGuid(),AmmoKind.Mortar,1,true,actor.SimulationTime,0);}));
steps.Add((5.2,()=>Mouse(true)));steps.Add((5.3,()=>Mouse(false)));
steps.Add((7.5,()=>{Check("enhanced source supersedes basic and yields256",enemy.Vitals.Health.Current==hp-256);foreach(var id in new[]{"precision-critical","precision-speed","precision-damage","last-round","last-round-plus"})Check("ordinary source revoke "+id,module.ToggleExperimentalSource(id));}));
steps.Add((7.9,()=>Check("withdraw removes future shot buffs",state.ShotEnhancements.Snapshot().DamageMultiplier==1&&state.ShotEnhancements.Snapshot().Rate==1)));
  EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;Mouse(false);Key(false);foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(mouse);InputSystem.RemoveDevice(keyboard);InputSystem.settings=settingsBefore;UnityEngine.Object.DestroyImmediate(settings);}
  tick=()=>{if(Time.frameCount==last)return;last=Time.frameCount;try{if(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].Item1){var late=EditorApplication.timeSinceStartup-start-steps[next].Item1;if(late>.02)start+=late;steps[next++].Item2();}if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;rows.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");
 }
}
