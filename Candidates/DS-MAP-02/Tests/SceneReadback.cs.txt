using System;using System.Linq;using UnityEngine;using UnityEditor;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
using C=Ossastra.Characters;using E=Ossastra.Enemies;using Ossastra.MapResearch;using Ossastra.MapResearchIntegration;using Ossastra.MapResearchIntegration.Rendering;using Spine.Unity;
public sealed class Ds02SceneReadback:IFunplayCommand
{
 public void Execute(ExecutionContext ctx){var scene=SceneManager.GetActiveScene();var roots=scene.GetRootGameObjects();var actors=roots.SelectMany(r=>r.GetComponentsInChildren<C.CharacterRoot>(true)).ToArray();var enemies=roots.SelectMany(r=>r.GetComponentsInChildren<E.EnemyRoot>(true)).ToArray();var bridge=roots.SelectMany(r=>r.GetComponentsInChildren<MapResearchPartyBridge>(true)).SingleOrDefault();
 ctx.ReturnValue=new{scene=scene.path,scene.isDirty,play=Application.isPlaying,actors=actors.Select(a=>new{a.ActorId,hp=a.Vitals?.Health?.Current,alive=a.IsAlive,ready=a.DataBus!=null}).ToArray(),enemies=enemies.Select(e=>new{e.ActorId,hp=e.Vitals?.Health?.Current,e.AttacksEnabled}).ToArray(),coordinators=roots.Sum(r=>r.GetComponentsInChildren<C.CharacterControlCoordinator>(true).Length),current=bridge?.coordinator?.Current?.ActorId,registered=bridge?.RegisteredCount,missingScripts=roots.SelectMany(r=>r.GetComponentsInChildren<Transform>(true)).Sum(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)),cameras=UnityEngine.Object.FindObjectsOfType<Camera>(true).Select(c=>new{c.name,c.enabled,c.gameObject.activeInHierarchy}).ToArray(),enabledListeners=UnityEngine.Object.FindObjectsOfType<AudioListener>(true).Count(l=>l.isActiveAndEnabled),depth=roots.SelectMany(r=>r.GetComponentsInChildren<MapResearchSpineDepth>(true)).Select(p=>new{p.name,p.enabled,p.IsBound,p.Status,spacing=p.sourceRenderer==null?0:p.sourceRenderer.zSpacing,mesh=p.GetComponent<MeshFilter>()?.sharedMesh?.GetInstanceID(),bounds=p.GetComponent<MeshFilter>()?.sharedMesh?.bounds.ToString()}).ToArray()};}
}
