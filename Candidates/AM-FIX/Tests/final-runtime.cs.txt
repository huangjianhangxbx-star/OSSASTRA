using System.Linq;using UnityEngine;using C=Ossastra.Characters;using Spine.Unity;using Funplay.Editor.Tools.Scripting;
public class FinalFixRuntime:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){ctx.ReturnValue=Object.FindObjectsOfType<C.CharacterRoot>().Select(r=>new{r.ActorId,r.IsAlive,r.CanSimulateIntent,missing=r.GetComponentsInChildren<Transform>(true).Sum(t=>UnityEditor.GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)),machines=r.StateArbitrator.Snapshots.Select(s=>new{s.machine,s.initialized}).ToArray(),skeletons=r.GetComponentsInChildren<SkeletonAnimation>().Select(s=>new{s.name,valid=s.Skeleton!=null&&s.AnimationState!=null,clip=s.AnimationState?.GetTrack(0)?.Animation?.Name}).ToArray()}).ToArray();}}
