var settingsBefore=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(settingsBefore);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;var pad=InputSystem.AddDevice<Gamepad>("MainGroupPad");var overrides=new Dictionary<InputAction,string>();foreach(var action in new[]{input.rightHandAction,input.leftHandAction,input.weaponModeAction,input.gamepadAimAction}){overrides[action]=action.SaveBindingOverridesAsJson();for(int n=0;n<action.bindings.Count;n++){var b=action.bindings[n];if(!b.isComposite&&b.path!=null)action.ApplyBindingOverride(n,b.path.Replace("<Gamepad>",pad.path));}}EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();

Add("first group stage reaches attack-ready",()=>Pad(),()=>actor.DataBus.AttackReady);

}}
