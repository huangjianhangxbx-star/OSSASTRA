using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using C=Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class PulseConsumers:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var actor=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>().Current;var input=actor.GetComponent<C.CharacterInputHandler>();var session=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();var interaction=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventoryInteraction>();var view=UnityEngine.Object.FindObjectOfType<Ossastra.UserInterface.InventoryView>();var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;
var keyboard=InputSystem.AddDevice<Keyboard>("PulseKeyboard");var mouse=InputSystem.AddDevice<Mouse>("PulseMouse");var skillBefore=input.skillOneAction.SaveBindingOverridesAsJson();var dodgeBefore=input.dodgeAction.SaveBindingOverridesAsJson();var interactBefore=interaction.interactAction.SaveBindingOverridesAsJson();var point=new Vector2(Screen.width*.6f,Screen.height*.4f);var blocker=new object();var position=actor.transform.position;var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}void UpdateInput(){InputSystem.Update();input.Read();}void Mouse(ushort buttons)=>InputSystem.QueueStateEvent(mouse,new MouseState{position=point,buttons=buttons});
try{
actor.CancelAction("pulse-fixture");actor.SendMessage("OnApplicationFocus",true);UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>().SendMessage("OnApplicationFocus",true);interaction.SendMessage("OnApplicationFocus",true);view.SetOpen(false);
for(int n=0;n<input.dodgeAction.bindings.Count;n++)if(!input.dodgeAction.bindings[n].isComposite)input.dodgeAction.ApplyBindingOverride(n,mouse.leftButton.path);
Mouse(0);UpdateInput();Mouse(0);UpdateInput();Mouse(1);Mouse(0);UpdateInput();var frame=input.CurrentFrame;Check("world down/up preserves dodge pressed when final held is false",frame.dodgePressed);actor.GetComponent<C.CharacterStamina>().Restore();Check("world pulse reaches actual dodge acceptance",actor.GetComponent<C.CharacterDodge>().Request(frame.move,frame.dodgePressed));actor.CancelAction("pulse-fixture");
Mouse(0);UpdateInput();input.RegisterPointerRegion(blocker,new Rect(point.x-5,Screen.height-point.y-5,10,10));Mouse(1);Mouse(0);UpdateInput();Check("same-update UI down/up cannot become world dodge",!input.CurrentFrame.dodgePressed);input.RemovePointerRegion(blocker);
Mouse(0);UpdateInput();Mouse(1);UpdateInput();input.ResetPlayerInput();UpdateInput();Check("reset rejects old hold without synthesizing edge",!input.CurrentFrame.dodgePressed);Mouse(0);UpdateInput();Mouse(1);Mouse(0);UpdateInput();Check("real release then quick press rearms",input.CurrentFrame.dodgePressed);
for(int n=0;n<interaction.interactAction.bindings.Count;n++)if(!interaction.interactAction.bindings[n].isComposite)interaction.interactAction.ApplyBindingOverride(n,keyboard.fKey.path);
actor.GetComponent<C.CharacterMovement>().Teleport(session.ContainerPosition("external").Value+Vector3.right);Physics.SyncTransforms();view.SetOpen(false);InputSystem.QueueStateEvent(keyboard,new KeyboardState());InputSystem.Update();interaction.SendMessage("Update");InputSystem.QueueStateEvent(keyboard,new KeyboardState());InputSystem.Update();interaction.SendMessage("Update");
InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.F));InputSystem.QueueStateEvent(keyboard,new KeyboardState());InputSystem.Update();interaction.SendMessage("Update");Check("world F pulse reaches actual InventoryInteraction and opens chest",view.IsOpen&&view.OpenedExternal=="external");
var module=actor.GetComponent<C.CharacterAbilities>();var gear=actor.GetComponent<C.CharacterEquipment>();var skill=module.SkillDefinition(0);var skillKey=gear.State.RightInstanceId+":"+(string.IsNullOrWhiteSpace(skill.cooldownIdentity)?skill.definitionId:skill.cooldownIdentity);gear.State.RightRuntime.Cooldowns.MakeReady(skillKey,gear.State.RightRuntime.Clock(skill.cooldownPolicy));
view.SetOpen(false);actor.CancelAction("pulse-skill");actor.GetComponent<C.CharacterStamina>().Restore();for(int n=0;n<input.skillOneAction.bindings.Count;n++)if(!input.skillOneAction.bindings[n].isComposite)input.skillOneAction.ApplyBindingOverride(n,keyboard.qKey.path);
InputSystem.QueueStateEvent(keyboard,new KeyboardState());UpdateInput();module.TickInput(input.CurrentFrame);InputSystem.QueueStateEvent(keyboard,new KeyboardState());UpdateInput();module.TickInput(input.CurrentFrame);var accepted=module.AcceptedCount;InputSystem.QueueStateEvent(keyboard,new KeyboardState(Key.Q));InputSystem.QueueStateEvent(keyboard,new KeyboardState());UpdateInput();module.TickInput(input.CurrentFrame);Check("quick skill pulse reaches formal ability consumer once",input.CurrentFrame.skillOnePressed&&!input.CurrentFrame.skillOneHeld&&module.AcceptedCount==accepted+1&&actor.ActionActive);actor.CancelAction("pulse-skill");module.TickEffects(0);
foreach(var mode in new[]{Ossastra.CharacterCore.Abilities.GestureMode.Press,Ossastra.CharacterCore.Abilities.GestureMode.Hold,Ossastra.CharacterCore.Abilities.GestureMode.Release,Ossastra.CharacterCore.Abilities.GestureMode.TapOrLongPress}){var gesture=new Ossastra.CharacterCore.Abilities.GestureRecognizer(mode,.3);gesture.Sample(false,0,1);var signal=gesture.Sample(false,true,true,.1,1);Check("pulse semantic "+mode,signal==(mode==Ossastra.CharacterCore.Abilities.GestureMode.Press?Ossastra.CharacterCore.Abilities.GestureSignal.Trigger:mode==Ossastra.CharacterCore.Abilities.GestureMode.Hold?Ossastra.CharacterCore.Abilities.GestureSignal.Begin:mode==Ossastra.CharacterCore.Abilities.GestureMode.Release?Ossastra.CharacterCore.Abilities.GestureSignal.Trigger:Ossastra.CharacterCore.Abilities.GestureSignal.Tap));}
}finally{view.SetOpen(false);input.RemovePointerRegion(blocker);input.skillOneAction.LoadBindingOverridesFromJson(skillBefore);input.dodgeAction.LoadBindingOverridesFromJson(dodgeBefore);interaction.interactAction.LoadBindingOverridesFromJson(interactBefore);actor.GetComponent<C.CharacterMovement>().Teleport(position);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}
ctx.ReturnValue=new{failed,rows};
}}
