using System;using System.Linq;using UnityEngine;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class F01Repro:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("B Play required");var coordinator=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var actor=coordinator?.Current;if(actor==null)throw new Exception("Current missing");var engage=UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>();if(engage!=null)engage.enabled=false;foreach(var e in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){e.AttacksEnabled=false;e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;}
var main=actor.GetComponent<C.CharacterMainAction>();var stamina=actor.GetComponent<C.CharacterStamina>();stamina.Restore();if(main==null||!main.Configure(main.Choices.First().definitionId))throw new Exception(main?.LastResult??"main missing");actor.StateArbitrator.Combat.RefreshMainProfile();var prepared=main.TryPrepare(0,out var phase);var copied=prepared?phase.Phase.CopyPresentationPlan() as A.CharacterAnimationPlan:null;var before=stamina.Current;int accepted=0;Action<Ossastra.CharacterCore.ActionFact> observe=f=>{if(f.Kind==Ossastra.CharacterCore.ActionFactKind.Accepted)accepted++;};actor.ActionObserved+=observe;string error=null;
try{actor.StateArbitrator.Combat.RequestInput(default);actor.StateArbitrator.Combat.RequestInput(new C.CharacterInputFrame{attackHeld=true});}catch(Exception ex){error=ex.ToString();}finally{actor.ActionObserved-=observe;}
ctx.ReturnValue=new{prepared,completePlan=copied!=null&&copied.HasOperations,accepted,paid=before-stamina.Current,error,active=actor.ActionActive,passed=copied!=null&&error==null&&accepted==1};actor.CancelAction("f01-fixture-cleanup");}}
