using System;using System.Linq;using UnityEngine;using Spine.Unity;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using I=Ossastra.Inventory;using Funplay.Editor.Tools.Scripting;
public class F03Equipment:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var actor=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>()?.Current;var session=UnityEngine.Object.FindObjectOfType<I.InventorySession>();if(actor==null||session==null)throw new Exception("B session required");actor.CancelAction("fixture");var equipment=actor.GetComponent<C.CharacterEquipment>();if(equipment.RightDefinition?.bodyAnimationOverride!=null)throw new Exception("fixture requires no body override");var hand=I.InventorySession.HandId(actor.ActorId,I.InventoryHand.Right);var item=session.Ledger.Items.Single(i=>i.Location.ContainerId==hand);var go=new GameObject("F03Fixture");SkeletonAnimation.AddToGameObject(go,actor.GetComponentInChildren<SkeletonAnimation>().skeletonDataAsset);var skeleton=go.GetComponent<SkeletonAnimation>();skeleton.UpdateTiming=UpdateTiming.ManualUpdate;var handle=go.AddComponent<A.AnimationHandleSystem>();handle.Bind(actor,skeleton,"body");var config=go.AddComponent<A.CharacterAnimationConfigManager>();var composer=go.AddComponent<A.CharacterAnimationComposer>();var map=UnityEngine.Object.Instantiate(actor.GetComponentInChildren<A.CharacterAnimationConfigManager>().baseMap);var router=ScriptableObject.CreateInstance<A.AnimationLogicRouterSO>();router.rulesMovement.Add(new A.AnimationLogicRouterSO.RouteRule{inputSemantic=A.EAnimSemantic.Idle,outputID=A.EAnimationLogicState.Jog,conditions=new System.Collections.Generic.List<A.AnimationLogicRouterSO.Condition>{new A.AnimationLogicRouterSO.Condition{type=A.AnimationLogicRouterSO.ConditionType.NoWeapon}},priority=100});config.baseMap=map;config.logicRouter=router;config.Initialize(actor);composer.Initialize(actor,config,handle,"body",intent=>equipment.RightDefinition);var intent=new A.AnimationIntent{root=actor,actorId=actor.ActorId,target="body",lifeGeneration=actor.LifeGeneration,intentSourceStateMatine=A.AnimationIntentSourceStateMachine.Movement,semantic=A.EAnimSemantic.Idle};
try{composer.SubmitIntent(intent);composer.Commit(0);var before=skeleton.AnimationState.GetTrack(0).Animation.Name;var unequip=session.Move(session.Ledger.Plan(item.Id,"team"));if(!unequip.Ok)throw new Exception(unequip.Reason);composer.SubmitIntent(intent);composer.Commit(0);var empty=skeleton.AnimationState.GetTrack(0).Animation.Name;var equip=session.Move(session.Ledger.Plan(item.Id,hand));if(!equip.Ok)throw new Exception(equip.Reason);composer.SubmitIntent(intent);composer.Commit(0);var after=skeleton.AnimationState.GetTrack(0).Animation.Name;ctx.ReturnValue=new{before,empty,after,passed=before=="idle1_1"&&empty=="run"&&after=="idle1_1"};}finally{if(equipment.RightDefinition==null)session.Move(session.Ledger.Plan(item.Id,hand));UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(map);UnityEngine.Object.DestroyImmediate(router);}}}
