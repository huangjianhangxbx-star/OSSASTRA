using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using C=Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class FixMoveExit:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var control=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var actor=control.Current;var input=actor.GetComponent<C.CharacterInputHandler>();var movement=actor.GetComponent<C.CharacterMovement>();var state=actor.StateArbitrator.Movement;var pad=InputSystem.AddDevice<Gamepad>("MoveExitPad");var original=input.movementAction.SaveBindingOverridesAsJson();for(int n=0;n<input.movementAction.bindings.Count;n++){var b=input.movementAction.bindings[n];if(!b.isComposite&&b.path!=null)input.movementAction.ApplyBindingOverride(n,b.path.Replace("<Gamepad>",pad.path));}
var oldSettings=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(oldSettings);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;actor.SendMessage("OnApplicationFocus",true);control.SendMessage("OnApplicationFocus",true);EditorWindow.GetWindow(typeof(Editor).Assembly.GetType("UnityEditor.GameView")).Focus();foreach(var r in control.Characters){r.GetComponent<C.CompanionCombatBrain>().enabled=false;r.GetComponent<C.CharacterFollow>().enabled=false;}var enemy=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();foreach(var e in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){e.AttacksEnabled=false;e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;}UnityEngine.Object.FindObjectOfType<CombatEngagementSession>().enabled=false;
var rows=new List<object>();int failed=0,index=0;double since=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,moving=movement.IsMoving,active=state.IsActive,enters=state.EnterCount,exits=state.ExitCount});}void Pad(Vector2 move)=>InputSystem.QueueStateEvent(pad,new GamepadState{leftStick=move});
steps.Add((.1,()=>Pad(Vector2.zero)));steps.Add((.4,()=>Pad(Vector2.up)));steps.Add((.7,()=>{Check("moving entered",state.IsActive&&state.EnterCount>state.ExitCount);actor.enabled=false;Check("disable clears velocity and matches Exit immediately",!movement.IsMoving&&movement.Velocity==Vector3.zero&&!state.IsActive&&state.EnterCount==state.ExitCount);Pad(Vector2.zero);actor.enabled=true;}));steps.Add((1.0,()=>Pad(Vector2.up)));steps.Add((1.3,()=>{Check("reenabled movement enters again",state.IsActive&&state.EnterCount>state.ExitCount);CombatContactGateway.Resolve(AttackRelease.ForEmission(Guid.NewGuid(),0,enemy.ActorId,9999,0),enemy.Vitals,new IHitTarget[]{actor.Vitals},enemy.transform.position);Check("death clears velocity and matches Exit immediately",!actor.IsAlive&&!movement.IsMoving&&movement.Velocity==Vector3.zero&&!state.IsActive&&state.EnterCount==state.ExitCount);}));
void Flush(string status)=>SessionState.SetString("AM_FIX_MOVEMENT",Newtonsoft.Json.JsonConvert.SerializeObject(new{state=status,failed,rows}));EditorApplication.CallbackFunction tick=null;tick=()=>{try{if(index==steps.Count){EditorApplication.update-=tick;input.movementAction.LoadBindingOverridesFromJson(original);InputSystem.RemoveDevice(pad);InputSystem.settings=oldSettings;UnityEngine.Object.DestroyImmediate(settings);Flush("completed");return;}if(EditorApplication.timeSinceStartup-since>=steps[index].Item1){steps[index++].Item2();Flush("running");}}catch(Exception e){failed++;rows.Add(new{error=e.ToString()});EditorApplication.update-=tick;Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="movement death/disable story scheduled";
}}
