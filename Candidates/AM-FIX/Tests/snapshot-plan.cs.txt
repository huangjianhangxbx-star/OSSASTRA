using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Ossastra.CharacterCore;using Funplay.Editor.Tools.Scripting;
public class SnapshotPlan:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var actor=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>().Current;var view=actor.GetComponentInChildren<A.CharacterAnimationController>();var config=view.GetComponent<A.CharacterAnimationConfigManager>();var prior=config.baseMap;var priorRouter=config.logicRouter;var map=UnityEngine.Object.Instantiate(prior);config.baseMap=map;config.logicRouter=null;var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok});}
try{var entry=map.plansMelee.First(e=>e.logicID==actor.Definition.basicAttacks[0].animationLogicId);var original=entry.plan;original.clips[0].scalesWithAction=true;original.clips[0].sourceDuration=2;original.clips.Add(new A.CharacterAnimationClip{animaName="idle1_1",track=2,loop=true,playbackRate=1.7f,mixDuration=.13f});original.clearTracks.Add(5);original.queuedClips.Add(new A.CharacterAnimationQueuedClip{clip=new A.CharacterAnimationClip{animaName="run",track=2,loop=true,playbackRate=1.3f,mixDuration=.2f},delay=.27f,runtimeAfterPreviousEnd=true});
var frozen=view.FreezeAction(actor.Definition.basicAttacks[0]);var timeline=new BasicAttackDefinition(.1,.2,.3,.5,10);var retimed=frozen.WithTimeline(timeline);var plan=(A.CharacterAnimationPlan)retimed.CopyPresentationPlan();Check("WithTimeline keeps full tracks queue clear and runtime logic",plan.clips.Count==2&&plan.clips[1].track==2&&plan.queuedClips.Count==1&&plan.clearTracks.Contains(5)&&plan.runtimeLogicId==actor.Definition.basicAttacks[0].animationLogicId);Check("only action-scaled rate follows final duration",Math.Abs(plan.clips[0].playbackRate-4)<.0001&&Math.Abs(plan.clips[1].playbackRate-1.7f)<.0001);Check("queue playback and timing parameters survive",plan.queuedClips[0].clip.loop&&plan.queuedClips[0].runtimeAfterPreviousEnd&&Math.Abs(plan.queuedClips[0].delay-.27f)<.0001&&Math.Abs(plan.queuedClips[0].clip.playbackRate-1.3f)<.0001);plan.clips[0].animaName="mutated-copy";plan.queuedClips.Clear();plan.clearTracks.Clear();var independent=(A.CharacterAnimationPlan)retimed.CopyPresentationPlan();Check("snapshot consumers receive independent deep plans",independent.clips[0].animaName!="mutated-copy"&&independent.queuedClips.Count==1&&independent.clearTracks.Contains(5));original.clips[0].animaName="mutated-config";Check("timeline replacement never rereads edited SO",((A.CharacterAnimationPlan)frozen.WithTimeline(timeline).CopyPresentationPlan()).clips[0].animaName!="mutated-config");}finally{config.baseMap=prior;config.logicRouter=priorRouter;UnityEngine.Object.DestroyImmediate(map);}ctx.ReturnValue=new{failed,rows};}}
