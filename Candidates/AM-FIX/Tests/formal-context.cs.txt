using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using C=Ossastra.Characters;using A=Ossastra.CharacterAnimation;using Funplay.Editor.Tools.Scripting;
public class FormalContext:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var control=UnityEngine.Object.FindObjectOfType<C.CharacterControlCoordinator>();var actor=control.Current;var view=actor.GetComponentInChildren<A.CharacterAnimationController>();var config=view.GetComponent<A.CharacterAnimationConfigManager>();var composer=view.GetComponent<A.CharacterAnimationComposer>();var gear=actor.GetComponent<C.CharacterEquipment>();var module=actor.GetComponent<C.CharacterAbilities>();var stamina=actor.GetComponent<C.CharacterStamina>();var session=UnityEngine.Object.FindObjectOfType<Ossastra.Inventory.InventorySession>();var ui=UnityEngine.Object.FindObjectOfType<Ossastra.UserInterface.InventoryView>();
foreach(var r in control.Characters){r.CancelAction("route-fixture");r.SendMessage("OnApplicationFocus",true);r.GetComponent<C.CompanionCombatBrain>().enabled=false;r.GetComponent<C.CharacterFollow>().enabled=false;}control.SendMessage("OnApplicationFocus",true);foreach(var e in UnityEngine.Object.FindObjectsOfType<Ossastra.Enemies.EnemyRoot>()){e.AttacksEnabled=false;e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().enabled=false;}UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>().enabled=false;
var prior=config.logicRouter;var router=ScriptableObject.CreateInstance<A.AnimationLogicRouterSO>();config.logicRouter=router;var rows=new List<object>();int failed=0;void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,clip=view.CurrentClip,source=view.CapturedSource.ToString(),last=module.LastResult});}void Reset(){actor.CancelAction("route-fixture");module.TickEffects(0);stamina.Restore();ui.SetOpen(false);}
void Rules(A.AnimationIntentSourceStateMachine source){router.rulesMelee.Clear();foreach(var semantic in new[]{A.EAnimSemantic.Attack_Normal_1,A.EAnimSemantic.Attack_Normal_2,A.EAnimSemantic.Attack_Normal_3,A.EAnimSemantic.Attack_Normal_4,A.EAnimSemantic.Ability})router.rulesMelee.Add(new A.AnimationLogicRouterSO.RouteRule{matchSource=true,source=source,inputSemantic=semantic,outputID=A.EAnimationLogicState.Attack_Normal_2,priority=100,conditions=new List<A.AnimationLogicRouterSO.Condition>{new A.AnimationLogicRouterSO.Condition{type=A.AnimationLogicRouterSO.ConditionType.HasWeapon},new A.AnimationLogicRouterSO.Condition{type=A.AnimationLogicRouterSO.ConditionType.HasIntent,intentSource=source}}});router.Revision++;}
A.AnimationPlanOverrideRuleSetSO temporaryOverride=null;A.AnimationPlanOverrideRuleSetSO previousOverride=null;C.EquipmentDefinitionSO changedDefinition=null;var oldPosition=actor.transform.position;var oldLeft=gear.LeftDefinition;var oldLeftId=gear.State.LeftInstanceId;
try{
Reset();Rules(A.AnimationIntentSourceStateMachine.MainHand);var skill=module.SkillDefinition(0);var accepted=module.RequestSkill(0,C.CharacterActionSource.Player);view.Present();var frozen=composer.CurrentBodyIntent?.frozenPlan;Check("formal body skill routes with candidate HasIntent and equipment",accepted&&frozen?.runtimeLogicId==A.EAnimationLogicState.Attack_Normal_2&&view.CurrentClip==frozen.clips.First(c=>c.track==0).animaName);
var id=actor.DataBus.ActionId;foreach(var rule in router.rulesMelee)rule.outputID=A.EAnimationLogicState.Attack_Normal_3;router.Revision++;view.Present();Check("editing Router preserves current accepted phase",actor.DataBus.ActionId==id&&composer.CurrentBodyIntent?.frozenPlan?.runtimeLogicId==A.EAnimationLogicState.Attack_Normal_2);
Reset();var key=gear.State.RightInstanceId+":"+(string.IsNullOrWhiteSpace(skill.cooldownIdentity)?skill.definitionId:skill.cooldownIdentity);gear.State.RightRuntime.Cooldowns.MakeReady(key,gear.State.RightRuntime.Clock(skill.cooldownPolicy));accepted=module.RequestSkill(0,C.CharacterActionSource.Player);view.Present();Check("next actual body skill sees revised Router",accepted&&composer.CurrentBodyIntent?.frozenPlan?.runtimeLogicId==A.EAnimationLogicState.Attack_Normal_3);Reset();
actor.GetComponent<C.CharacterMovement>().Teleport(session.ContainerPosition("external").Value+Vector3.right);Physics.SyncTransforms();var shield=session.Ledger.Items.Single(i=>i.DefinitionId=="vc02.shield-b");var transfer=session.Move(session.Ledger.Plan(shield.Id,Ossastra.Inventory.InventorySession.HandId(actor.ActorId,Ossastra.Inventory.InventoryHand.Left)));rows.Add(new {transfer=transfer.Ok,reason=transfer.Reason,access=session.CanAccessContainer("external")});var definition=gear.OwnedOffhands.SingleOrDefault(d=>d.definitionId=="vc02.shield-b");Check("actual chest shield B equips via inventory swap",transfer.Ok&&definition!=null&&gear.LeftDefinition==definition);
temporaryOverride=ScriptableObject.CreateInstance<A.AnimationPlanOverrideRuleSetSO>();previousOverride=definition.bodyAnimationOverride;changedDefinition=definition;temporaryOverride.rulesMelee.Add(new A.AnimationPlanOverrideRuleSetSO.Rule{targetID=A.EAnimationLogicState.Attack_Normal_2,replacementPlan=config.baseMap.GetDefault(A.EAnimationLogicState.Attack_Normal_3).CloneFor(actor)});definition.bodyAnimationOverride=temporaryOverride;Rules(A.AnimationIntentSourceStateMachine.OffHand);var binding=gear.LeftDefinition.abilityBindings.First(b=>b.ability.executor.UsesBodyAction||b.longPressAbility!=null&&b.longPressAbility.executor.UsesBodyAction);var ability=binding.ability.executor.UsesBodyAction?binding.ability:binding.longPressAbility;gear.State.LeftRuntime.Cooldowns.MakeReady(gear.State.LeftInstanceId+":"+(string.IsNullOrWhiteSpace(ability.cooldownIdentity)?ability.definitionId:ability.cooldownIdentity),gear.State.LeftRuntime.Clock(ability.cooldownPolicy));accepted=module.TryRequest(ability,false,binding.operation,C.CharacterActionSource.Player);view.Present();Check("actual offhand body action uses left equipment override and routes",accepted&&view.CapturedSource==A.AnimationIntentSourceStateMachine.OffHand&&composer.CurrentBodyIntent?.frozenPlan?.runtimeLogicId==A.EAnimationLogicState.Attack_Normal_2&&view.CurrentClip=="atk3");definition.bodyAnimationOverride=previousOverride;Reset();
router.rulesMelee.Clear();router.rulesOther.Add(new A.AnimationLogicRouterSO.RouteRule{matchSource=true,source=A.AnimationIntentSourceStateMachine.Dodge,inputSemantic=A.EAnimSemantic.Dodge,outputID=A.EAnimationLogicState.Attack_Normal_2});router.Revision++;var dodge=actor.GetComponent<C.CharacterDodge>();accepted=dodge.Request(Vector2.right,true);view.Present();var dodgeId=dodge.ActionId;var prepared=dodge.PreparedAnimation;router.rulesOther[0].outputID=A.EAnimationLogicState.Attack_Normal_3;router.Revision++;view.Present();Check("one dodge identity retains its prepared plan",accepted&&dodge.ActionId==dodgeId&&ReferenceEquals(prepared,dodge.PreparedAnimation)&&composer.GetIntent(A.AnimationIntentSourceStateMachine.Dodge)?.frozenPlan?.runtimeLogicId==A.EAnimationLogicState.Attack_Normal_2);
}catch(Exception error){failed++;rows.Add(new {error=error.ToString()});}finally{if(changedDefinition!=null)changedDefinition.bodyAnimationOverride=previousOverride;if(temporaryOverride!=null)UnityEngine.Object.DestroyImmediate(temporaryOverride);Reset();if(oldLeft!=null&&gear.LeftDefinition!=oldLeft)session.Move(session.Ledger.Plan(oldLeftId,Ossastra.Inventory.InventorySession.HandId(actor.ActorId,Ossastra.Inventory.InventoryHand.Left)));actor.GetComponent<C.CharacterMovement>().Teleport(oldPosition);config.logicRouter=prior;UnityEngine.Object.DestroyImmediate(router);}
ctx.ReturnValue=new{failed,rows};
}}
