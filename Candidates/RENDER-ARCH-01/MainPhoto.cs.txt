using System;using System.Linq;using UnityEngine;using UnityEditor;using Newtonsoft.Json;using Ossastra.CharacterLighting;using Ossastra.Commerce.Runtime;
public static class FusionPhoto{
 const string Action="setup";
 public static string Run(){var sys=UnityEngine.Object.FindFirstObjectByType<CharacterLightingSystem>();var service=UnityEngine.Object.FindFirstObjectByType<CommerceService>();if(!Application.isPlaying||sys==null||service==null||service.profileId!="render-arch01-isolated")return "ISOLATED PLAY REQUIRED";var panel=UnityEngine.Object.FindFirstObjectByType<Ossastra.Development.CharacterLighting.CharacterLightingPanel>();var env=sys.environment.environment;var sky=sys.environment;var map=panel.map;
 if(Action=="setup"){panel.SetExpanded(false);panel.environmentPanel.SetExpanded(false);panel.environmentPanel.legacyDiagnostics.showPanel=false;panel.coordinator.SetInputCapture(sys,true);map.pitch=45;map.followSmoothTime=0;map.followParticipant=false;map.fixedTarget=new Vector3(26,.1f,-12);panel.environmentPanel.legacyDiagnostics.PlaceCurrent(map.fixedTarget);foreach(var r in sys.receivers){var f=r.GetComponent<Ossastra.Characters.CharacterFollow>();if(f!=null)f.enabled=false;foreach(var b in r.GetComponents<MonoBehaviour>())if(b!=null&&b.GetType().Name=="CharacterCombatBrain")b.enabled=false;var movement=r.GetComponent<Ossastra.Characters.CharacterMovement>();if(movement!=null)movement.Teleport(map.fixedTarget+(r==sys.receivers[0]?Vector3.zero:new Vector3(1.4f,0,0)));}env.SetClockRunning(false);env.SetTimeHours(18);sky.overrideSkyColors=true;sky.zenithColor=new Color(.28f,.13f,.15f);sky.horizonColor=new Color(.9f,.45f,.19f);}
 if(Action=="freeze")foreach(var r in sys.receivers)foreach(var rig in r.GetComponentsInChildren<Spine.Unity.SkeletonAnimation>(true)){rig.timeScale=0;if(rig.AnimationState!=null)rig.AnimationState.TimeScale=0;}
 if(Action=="A"||Action=="B")foreach(var r in sys.receivers)r.Settings.enhancedEnvironmentTint=Action=="B";
 if(Action=="noon"){env.SetTimeHours(12);sky.overrideSkyColors=false;}
 if(Action=="natural"){env.SetTimeHours(17);sky.overrideSkyColors=false;}
 if(Action=="night"){env.SetTimeHours(0);sky.overrideSkyColors=false;}
 if(Action=="room"){map.fixedTarget=panel.environmentPanel.legacyDiagnostics.knownRoomPosition;panel.environmentPanel.legacyDiagnostics.PlaceCurrent(map.fixedTarget);env.SetTimeHours(0);sky.overrideSkyColors=false;}
 env.RefreshNow();return JsonConvert.SerializeObject(new{Action,env.timeHours,sky.overrideSkyColors,target=sky.ResolveCharacterEnvironmentTint(env.Current).ToString("F4"),sys.CoverageBytes,sys.OuterGlowBytes,sys.OuterGlowDraws,sys.OuterGlowStatus,camera=map.gameplayCamera.transform.position.ToString(),actors=sys.receivers.Select(r=>new{r.name,pos=r.transform.position.ToString(),settings=JsonUtility.ToJson(r.Settings),diagnostic=sys.Describe(r)})});}
}
