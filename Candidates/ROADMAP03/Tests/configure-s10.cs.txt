using System;using System.Linq;using UnityEngine;using UnityEngine.InputSystem;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class ConfigureS10:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(scene.isDirty||Application.isPlaying||!scene.path.Contains("ROADMAP03"))throw new Exception("Clean own scene required");foreach(var root in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>()){ctx.RegisterObjectModification(root.gameObject);if(root.GetComponent<Ossastra.Characters.CharacterStamina>()!=null)throw new Exception("Already configured");root.gameObject.AddComponent<Ossastra.Characters.CharacterStamina>();root.gameObject.AddComponent<Ossastra.Characters.CharacterDodge>();root.gameObject.AddComponent<Ossastra.Combat.CombatDefense>();var input=root.GetComponent<Ossastra.Characters.CharacterInputHandler>();input.dodgeAction.AddBinding("<Keyboard>/leftShift");EditorUtility.SetDirty(input);}
var go=new GameObject("DefenseResourceDebug_SAMPLE");ctx.RegisterObjectCreation(go);go.AddComponent<Ossastra.Development.DefenseLoadoutDebugView>();EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=scene.path;}}
