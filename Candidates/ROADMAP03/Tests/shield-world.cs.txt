using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Funplay.Editor.Tools.Scripting;
public class ShieldWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var a=c.Current;var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(a==null||e==null)throw new Exception("world missing");e.RestoreForDevelopment();e.AttacksEnabled=false;foreach(var r in c.Characters){r.GetComponent<CompanionCombatBrain>().enabled=false;r.CancelAction("fixture");r.SendMessage("OnApplicationFocus",true);}c.SendMessage("OnApplicationFocus",true);
var guard=a.GetComponent<ShieldGuard>();var gear=a.GetComponent<CharacterEquipment>();var stamina=a.GetComponent<CharacterStamina>();var posture=a.GetComponent<CombatPosture>();a.Vitals.RestoreForDevelopment();posture.ResetState();stamina.Restore();a.GetComponent<CharacterMovement>().Teleport(new Vector3(0,.03f,-2));
var original=InputSystem.settings;var settings=UnityEngine.Object.Instantiate(original);settings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;settings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=settings;var mouse=InputSystem.AddDevice<Mouse>("ShieldProbe");var overrides=new Dictionary<InputAction,string>();foreach(var action in c.Characters.SelectMany(r=>{var i=r.GetComponent<CharacterInputHandler>();return new[]{i.mouseAction,i.offhandAction,i.attackAction};})){overrides[action]=action.SaveBindingOverridesAsJson();for(int j=0;j<action.bindings.Count;j++){var path=action.bindings[j].path;if(path!=null&&path.Contains("<Mouse>"))action.ApplyBindingOverride(j,path.Replace("<Mouse>",mouse.path));}}
var type=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");EditorWindow.GetWindow(type).Focus();
var rows=new List<object>();int failed=0,index=0;double start=EditorApplication.timeSinceStartup;var steps=new List<(double,Action)>();
void Flush(string state){SessionState.SetString("ROADMAP_S11_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,rows}));}
void Check(string name,bool ok){if(!ok)failed++;rows.Add(new{name,ok,guard=guard.IsGuarding,stamina=stamina.Current,hp=a.Vitals.Health.Current,durability=gear.State.Durability});Flush("running");}
void Mouse(int buttons){var p=Camera.main.WorldToScreenPoint(a.transform.position+Vector3.right*2);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(p.x,p.y),buttons=(ushort)buttons});}
AttackRelease Release(){var rt=new BasicActionRuntime(e.ActorId);rt.Request(new BasicAttackDefinition(0,0,0,.1,10,20),"shield-fixture",new ControlAuthority());return rt.Advance(0,false).Single(x=>x.Release!=null).Release;}
void Add(double t,Action f){steps.Add((t,f));}Mouse(0);
Add(.2,()=>Mouse(2));Add(.5,()=>{Check("RMB raises shield and pays",guard.IsGuarding&&stamina.Current<92&&stamina.Current>85);var before=posture.Current;var release=Release();var result=CombatContactGateway.Resolve(release,e.Vitals,new IHitTarget[]{a.Vitals,a.Vitals},a.transform.position+guard.GuardDirection);Check("front guard zero HP",result.Count==1&&result[0].Reason==HitReason.Guarded&&a.Vitals.Health.Current==a.Vitals.Health.Maximum);Check("half posture and one durability",Math.Abs(posture.Current-(before-10))<.01&&gear.State.Durability==99);bool duplicate=false;try{CombatContactGateway.Resolve(release,e.Vitals,new IHitTarget[]{a.Vitals},a.transform.position+guard.GuardDirection);}catch(InvalidOperationException){duplicate=true;}Check("release cannot repeat",duplicate&&gear.State.Durability==99);var back=CombatContactGateway.Resolve(Release(),e.Vitals,new IHitTarget[]{a.Vitals},a.transform.position-guard.GuardDirection);Check("back bypasses shield",back[0].Reason==HitReason.Applied&&a.Vitals.Health.Current==a.Vitals.Health.Maximum-10);});
Add(.7,()=>stamina.Spend(Guid.NewGuid(),stamina.Current,a.SimulationTime,"fixture-empty"));Add(1,()=>{Check("exhaust lowers shield",!guard.IsGuarding);stamina.Restore();});Add(1.2,()=>{Check("held cannot restart",!guard.IsGuarding);Mouse(0);});Add(1.4,()=>Mouse(2));Add(1.6,()=>{Check("release repress rearms",guard.IsGuarding);c.SwitchNext();});Add(1.9,()=>{Check("switch clears old guard",!guard.IsGuarding);Check("held not inherited",!c.Current.GetComponent<ShieldGuard>().IsGuarding);Mouse(0);});
EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(mouse);InputSystem.settings=original;UnityEngine.Object.DestroyImmediate(settings);}
tick=()=>{try{while(index<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[index].Item1)steps[index++].Item2();if(index==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;rows.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.ReturnValue="scheduled";
}}
