using UnityEngine.InputSystem;using System;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class ConfigureS13:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var s=EditorSceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/ROADMAP03/SC_ROADMAP03_DefenseLoadout.unity")throw new Exception("own scene required");s=EditorSceneManager.OpenScene(s.path,OpenSceneMode.Single);if(UnityEngine.Object.FindObjectOfType<HealingBottleService>()!=null)throw new Exception("already configured");var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();var engagement=UnityEngine.Object.FindObjectOfType<Ossastra.Combat.CombatEngagementSession>();if(c==null||engagement==null)throw new Exception("missing dependencies");var go=new GameObject("SharedHealing_SAMPLE");ctx.RegisterObjectCreation(go);var service=go.AddComponent<HealingBottleService>();var input=go.AddComponent<PartyConsumableInput>();void Set(UnityEngine.Object o,string name,UnityEngine.Object value){var so=new SerializedObject(o);so.FindProperty(name).objectReferenceValue=value;so.ApplyModifiedProperties();}Set(service,"coordinator",c);Set(input,"service",service);input.healingAction.AddBinding("<Keyboard>/y");var camp=GameObject.CreatePrimitive(PrimitiveType.Cylinder);camp.name="PrototypeCampfire_SAMPLE_G";ctx.RegisterObjectCreation(camp);camp.transform.position=new Vector3(-4,.05f,-3);camp.transform.localScale=new Vector3(.65f,.05f,.65f);var fire=camp.AddComponent<PrototypeCampfire>();Set(fire,"service",service);Set(fire,"engagement",engagement);fire.interactAction.AddBinding("<Keyboard>/g");EditorUtility.SetDirty(input);EditorUtility.SetDirty(fire);EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=new{scene=s.path,camp=camp.transform.position};}}
