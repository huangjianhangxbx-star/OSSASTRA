using System;using UnityEngine;using UnityEngine.InputSystem;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class ConfigureS11:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var scene=EditorSceneManager.GetActiveScene();if(scene.isDirty||Application.isPlaying||!scene.path.Contains("ROADMAP03"))throw new Exception("Clean own scene required");foreach(var root in UnityEngine.Object.FindObjectsOfType<Ossastra.Characters.CharacterRoot>()){if(root.GetComponent<Ossastra.Characters.CharacterEquipment>()!=null)throw new Exception("Already configured");ctx.RegisterObjectModification(root.gameObject);root.gameObject.AddComponent<Ossastra.Characters.CharacterEquipment>();root.gameObject.AddComponent<Ossastra.Characters.ShieldGuard>();var input=root.GetComponent<Ossastra.Characters.CharacterInputHandler>();ctx.RegisterObjectModification(input);input.offhandAction.AddBinding("<Mouse>/rightButton");EditorUtility.SetDirty(input);}EditorSceneManager.MarkSceneDirty(scene);EditorSceneManager.SaveScene(scene);ctx.ReturnValue=scene.path;}}
