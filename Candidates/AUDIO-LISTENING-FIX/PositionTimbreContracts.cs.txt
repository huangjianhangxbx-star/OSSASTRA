using System;using System.Linq;using System.Reflection;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class PositionTimbreContracts:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){var root=Ossastra.Audio.GameAudioRoot.Instance;var actor=UnityEngine.Object.FindObjectOfType<Ossastra.Characters.CharacterControlCoordinator>()?.Current;if(!Application.isPlaying||root==null||actor==null)throw new Exception("Owned Play required");var pose=actor.transform.position;var tick=root.GetType().GetMethod("LateUpdate",BindingFlags.Instance|BindingFlags.NonPublic);var rows=new System.Collections.Generic.List<object>();try{foreach(var z in new[]{3f,-4.6f}){Place(actor.transform,new Vector3(0,pose.y,z));tick.Invoke(root,null);using(var lease=root.RegisterSfxSource("DEVELOPMENT_ONLY:position:"+z)){var d=lease.Submit("player.blade.swing.1","one",actor.transform.position);root.SfxPlayer.Tick(AudioSettings.dspTime);var v=root.SfxPlayer.Voices.Single(v=>v.VoiceId==d.VoiceId);rows.Add(new{z,accepted=d.Accepted,gain=v.Source.volume,cutoff=v.LowPass.enabled?v.LowPass.cutoffFrequency:22000,reverb=v.Reverb.enabled,listener=root.SfxPlayer.AcousticOrigin.y});}}}finally{Place(actor.transform,pose);tick.Invoke(root,null);}bool clear=rows.All(r=>(float)r.GetType().GetProperty("cutoff").GetValue(r)>=20000&&!(bool)r.GetType().GetProperty("reverb").GetValue(r));float g0=(float)rows[0].GetType().GetProperty("gain").GetValue(rows[0]),g1=(float)rows[1].GetType().GetProperty("gain").GetValue(rows[1]);c.ReturnValue=new{checks=new[]{new{name="upper and lower self sounds stay dry and clear",pass=clear},new{name="self source gain is invariant across screen position",pass=Mathf.Abs(g0-g1)<.0001f}},rows};}static void Place(Transform t,Vector3 p){var b=t.GetComponent<CharacterController>();bool enabled=b!=null&&b.enabled;if(enabled)b.enabled=false;t.position=p;if(enabled)b.enabled=true;Physics.SyncTransforms();}}
