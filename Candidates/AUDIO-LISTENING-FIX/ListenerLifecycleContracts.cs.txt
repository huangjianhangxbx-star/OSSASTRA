// 实际根禁用/恢复、切人及全部角色暂不可用的听点边界；退出Play重置本场事务。
using System;using System.Linq;using System.Collections.Generic;using System.Reflection;using UnityEngine;using Funplay.Editor.Tools.Scripting;
public class ListenerLifecycleContracts:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){var root=Ossastra.Audio.GameAudioRoot.Instance;var cc=UnityEngine.Object.FindObjectOfType<Ossastra.Characters.CharacterControlCoordinator>();if(!Application.isPlaying||root==null||cc?.Current==null)throw new Exception("Owned Play required");var checks=new List<object>();Action<string,bool> check=(n,v)=>checks.Add(new{name=n,pass=v});var tick=root.GetType().GetMethod("LateUpdate",BindingFlags.Instance|BindingFlags.NonPublic);Func<AudioListener> current=()=>UnityEngine.Object.FindObjectsOfType<AudioListener>().SingleOrDefault(l=>l.isActiveAndEnabled);var native=UnityEngine.Object.FindObjectsOfType<AudioListener>(true).Where(l=>!l.name.StartsWith("Gameplay Audio Listener")).ToArray();root.enabled=false;check("disabled root restores enabled native listening",native.Any(l=>l.isActiveAndEnabled));root.enabled=true;root.ResumeMusic();tick.Invoke(root,null);check("reenabled root has exactly one active listener",current()!=null&&current().name.StartsWith("Gameplay Audio Listener"));var actor=cc.Current;var focusField=cc.GetType().GetField("focused",BindingFlags.Instance|BindingFlags.NonPublic);bool wasFocused=(bool)focusField.GetValue(cc);cc.SendMessage("OnApplicationFocus",true);bool switched=cc.SwitchNext();cc.SendMessage("OnApplicationFocus",wasFocused);tick.Invoke(root,null);check("real control handoff moves hearing to current actor",switched&&Vector3.Distance(current().transform.position,cc.Current.transform.position)<1.5f);var before=current().transform.position;var actors=cc.Characters.Select(a=>new{a,active=a.gameObject.activeSelf}).ToArray();try{foreach(var a in actors)a.a.gameObject.SetActive(false);tick.Invoke(root,null);check("no available character retains last hearing point for tails",Vector3.Distance(current().transform.position,before)<.01f);}finally{foreach(var a in actors)a.a.gameObject.SetActive(a.active);}c.ReturnValue=new{checks};}}
