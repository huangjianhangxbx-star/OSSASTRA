// 真实uGUI指针几何/导航/按钮与统一Mixer、Prefs边界；推荐音量作为本轮修正保留，未暂停世界。
using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.EventSystems;using UnityEngine.UI;using Funplay.Editor.Tools.Scripting;
public class SettingsInteractionStory:IFunplayCommand{
public static object Report;static Ossastra.UserInterface.InventoryView view;static Ossastra.Audio.GameAudioRoot root;static Ossastra.Characters.CharacterControlCoordinator cc;static Slider environment;static List<object> rows;static double start;static int stage;static float oldEnemy,worldTime;static string error;
public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext c){try{view=UnityEngine.Object.FindObjectOfType<Ossastra.UserInterface.InventoryView>();root=Ossastra.Audio.GameAudioRoot.Instance;cc=UnityEngine.Object.FindObjectOfType<Ossastra.Characters.CharacterControlCoordinator>();if(!Application.isPlaying||view==null||root==null)throw new Exception("Owned Play required");rows=new List<object>();Report=null;error="";stage=0;worldTime=Time.time;view.ToggleMenu();var entry=view.GetComponentsInChildren<Button>(true).Single(b=>b.name=="AudioSettings");entry.onClick.Invoke();Canvas.ForceUpdateCanvases();Check("menu opens actual audio controls",view.AudioSettingsOpen);Check("menu captures character input without pausing world",!cc.IsPlayerInputAllowed(cc.Current.GetComponent<Ossastra.Characters.CharacterInputHandler>())&&Time.timeScale==1);var master=view.GetComponentsInChildren<Slider>(true).Single(s=>s.name=="MasterVolume");var corners=new Vector3[4];((RectTransform)master.handleRect.parent).GetWorldCorners(corners);var point=(corners[0]+corners[2])*.5f;var data=new PointerEventData(EventSystem.current){position=point,button=PointerEventData.InputButton.Left};master.OnPointerDown(data);Check("real pointer geometry sets master to midpoint",Mathf.Abs(root.Mix.MasterVolume-.5f)<.005f);root.UnifiedMix.Tick();root.UnifiedMix.Mixer.GetFloat("MasterGain",out var db);Check("slider reaches sole Mixer master gain",Mathf.Abs(db-(-6.0206f))<.02f);master.OnMove(new AxisEventData(EventSystem.current){moveDir=MoveDirection.Right,moveVector=Vector2.right});Check("navigation changes selected slider",Mathf.Abs(root.Mix.MasterVolume-.6f)<.005f);master.OnPointerUp(data);Check("interaction completion persists actual gain",Mathf.Abs(PlayerPrefs.GetFloat("ossastra.audio.master")-.6f)<.005f);var mute=view.GetComponentsInChildren<Button>(true).Single(b=>b.name=="MuteToggle");bool before=root.Mix.Muted;mute.onClick.Invoke();Check("mute button changes real user mute",root.Mix.Muted!=before);mute.onClick.Invoke();oldEnemy=root.UnifiedMix.GetBusVolume(Ossastra.Audio.SfxBus.Enemy);root.UnifiedMix.SetBusVolume(Ossastra.Audio.SfxBus.Enemy,.73f);view.GetComponentsInChildren<Button>(true).Single(b=>b.name=="RecommendedAudio").onClick.Invoke();Check("recommended levels are usable and preserve unrelated buses",root.Mix.MasterVolume==1&&Mathf.Abs(root.Mix.UserBgmVolume-.65f)<.001f&&root.UnifiedMix.SfxVolume==1&&root.UnifiedMix.GetBusVolume(Ossastra.Audio.SfxBus.Ambience)==0&&Mathf.Abs(root.UnifiedMix.GetBusVolume(Ossastra.Audio.SfxBus.Enemy)-.73f)<.001f);root.UnifiedMix.SetBusVolume(Ossastra.Audio.SfxBus.Enemy,oldEnemy);root.SaveUserSettings();environment=view.GetComponentsInChildren<Slider>(true).Single(s=>s.name=="AmbienceVolume");environment.value=.2f;start=EditorApplication.timeSinceStartup;EditorApplication.update+=Tick;c.ReturnValue=new{started=true,seconds=3};}catch(Exception e){error=e.ToString();Cleanup();c.ReturnValue=Report;}}
static void Check(string name,bool pass)=>rows.Add(new{name,pass});
static void Tick(){try{var t=EditorApplication.timeSinceStartup-start;if(!Application.isPlaying)throw new Exception("Play ended");if(stage==0&&t>.8){Check("optional ambience can actually start a budgeted loop",root.Sfx.ActiveVoices.Any(v=>v.Definition.loop));environment.value=0;stage++;}if(stage==1&&t>1.5){Check("turning ambience off releases actual loop permit",!root.Sfx.ActiveVoices.Any(v=>v.Definition.loop));view.CloseAudioSettings();view.OpenAudioSettings();Check("reopening reuses one audio panel",view.AudioSettingsOpen&&view.GetComponentsInChildren<RectTransform>(true).Count(r=>r.name=="AudioSettingsPanel")==1);view.ToggleMenu();Check("menu exit closes settings and restores Tab action",!view.AudioSettingsOpen&&view.toggleAction.enabled);stage++;}if(stage==2&&t>2.2){Check("world time advances while settings are open",Time.time>worldTime+1);Check("closing menu releases input capture",cc.IsPlayerInputAllowed(cc.Current.GetComponent<Ossastra.Characters.CharacterInputHandler>()));Cleanup();}if(t>6)throw new Exception("UI story deadline");}catch(Exception e){error=e.ToString();Cleanup();}}
static void Cleanup(){EditorApplication.update-=Tick;if(view!=null&&view.AudioSettingsOpen)view.CloseAudioSettings();if(root!=null){root.UnifiedMix.SetBusVolume(Ossastra.Audio.SfxBus.Enemy,oldEnemy);root.UnifiedMix.SetBusVolume(Ossastra.Audio.SfxBus.Ambience,0);root.SaveUserSettings();}Report=new{done=true,error,checks=rows?.ToArray(),recommendedSettingsRetained=true};}}
