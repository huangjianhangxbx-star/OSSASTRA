using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Enemies;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class ChaseWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");
var service=UnityEngine.Object.FindObjectOfType<PartyRespawnCoordinator>();var debug=UnityEngine.Object.FindObjectOfType<CombatDeathDebugView>();var d=UnityEngine.Object.FindObjectOfType<CompanionFollowDirector>();var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(service==null||debug==null||d==null||e==null)throw new Exception("Missing world");
var c=d.Coordinator;var a=c.Characters[0];var b=c.Characters[1];var roots=new[]{a,b};var inputs=roots.Select(x=>x.GetComponent<CharacterInputHandler>()).ToArray();var healths=roots.Select(x=>x.Vitals.Health).ToArray();var initialGeneration=a.LifeGeneration;
var originalSettings=InputSystem.settings;var testSettings=UnityEngine.Object.Instantiate(originalSettings);testSettings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;testSettings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=testSettings;var keyboard=InputSystem.AddDevice<Keyboard>("M02B1Keyboard");var mouse=InputSystem.AddDevice<Mouse>("M02B1Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in inputs.SelectMany(i=>new[]{i.movementAction,i.mouseAction,i.attackAction,i.cancelAction,i.pauseAction}).Concat(new[]{c.switchAction})){overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var path=action.bindings[i].path;if(path==null||action.bindings[i].isComposite)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(i,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(i,path.Replace("<Mouse>",mouse.path));}}
var game=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");if(game!=null)EditorWindow.GetWindow(game).Focus();
var records=new List<object>();var steps=new List<(double at,Action run)>();int next=0,failed=0;var start=EditorApplication.timeSinceStartup;GameObject wall=null;double pausedAt=0;Vector3 before=default;int generation=0;

void Check(string label,bool pass){if(!pass)failed++;records.Add(new{label,pass,a=a.IsAlive,b=b.IsAlive,hpA=a.Vitals.Health.Current,hpB=b.Vitals.Health.Current,hpE=e.Vitals.Health.Current,stage=a.AttackStage,buffer=a.HasBufferedAttack,releasesA=a.ReleaseCount,last=e.LastFact,phase=e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().Phase.ToString(),enemyPosition=e.transform.position.ToString(),target=e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>().TargetId,generationA=a.LifeGeneration,generationB=b.LifeGeneration,current=c.Current==null?"none":c.Current.ActorId,time=service.SimulationTime,status=service.Status});Flush("running");}

debug.RestartForDevelopment();d.enabled=false;e.AttacksEnabled=false;foreach(var r in roots){r.SendMessage("OnApplicationFocus",true);r.SetPaused(false);}c.SendMessage("OnApplicationFocus",true);e.SendMessage("OnApplicationFocus",true);service.SendMessage("OnApplicationFocus",true);Keys();Mouse(false);
void Add(double at,Action run){steps.Add((at,run));}
var brain=e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();var move=e.GetComponent<Ossastra.Enemies.EnemyMovement>();var body=e.GetComponent<CharacterController>();var home=brain.Home;Vector3 saved=default;int enemyHp=0;

Add(.1,()=>{Reset();Check("S4-independent-enemy-root",e.GetComponent<Ossastra.Characters.CharacterRoot>()==null&&e.GetComponent<CharacterInputHandler>()==null&&e.GetComponent<CharacterFollow>()==null);});

Add(3.5,()=>{Keys();Check("S4-target-not-current-mirror",c.Current==b&&brain.TargetId==a.ActorId);Place(a,home+Vector3.left*8);Place(b,home+Vector3.back*8);enemyHp=e.Vitals.Health.Current;});
Add(5.8,()=>Check("S4-returns-physically-no-heal",Vector3.Distance(home,e.transform.position)<.25f&&e.Vitals.Health.Current==enemyHp));
Add(6,()=>{Reset();Place(a,home+Vector3.right*3);wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.transform.position=home+Vector3.right*1.5f+Vector3.up*.8f;wall.transform.localScale=new Vector3(.2f,1.6f,2);Physics.SyncTransforms();saved=e.transform.position;});

Add(7,()=>{wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.transform.position=home+Vector3.left*1.2f+Vector3.up*.25f;wall.transform.localScale=new Vector3(.2f,.5f,2);Physics.SyncTransforms();});
Add(8.6,()=>Check("S4-physical-low-wall-holds",e.transform.position.x>home.x-1.0f&&(brain.Phase==Ossastra.EnemyCore.ChasePhase.Blocked||brain.Phase==Ossastra.EnemyCore.ChasePhase.ReturnHome)));

EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;e.AttacksEnabled=false;d.enabled=true;if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=originalSettings;UnityEngine.Object.DestroyImmediate(testSettings);}
tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");while(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].at)steps[next++].run();if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;records.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.Log("Scheduled S1 actual-world 7 seconds");
}}
