using System.Linq;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public class ConfigureChase:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
var s=EditorSceneManager.OpenScene("Assets/OSSASTRA/ROADMAP01/SC_ROADMAP01_CombatLoop.unity");var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();
var old=e.GetComponent<Ossastra.Enemies.StationaryEnemyBrain>();var source=new SerializedObject(old).FindProperty("targets");
var brain=e.GetComponent<Ossastra.Enemies.ChaseEnemyBrain>();if(brain==null)brain=e.gameObject.AddComponent<Ossastra.Enemies.ChaseEnemyBrain>();var so=new SerializedObject(brain);var dest=so.FindProperty("targets");dest.arraySize=source.arraySize;for(int i=0;i<source.arraySize;i++)dest.GetArrayElementAtIndex(i).objectReferenceValue=source.GetArrayElementAtIndex(i).objectReferenceValue;so.ApplyModifiedPropertiesWithoutUndo();old.enabled=false;
foreach(var collider in e.GetComponentsInChildren<Collider>())if(!(collider is CharacterController))UnityEngine.Object.DestroyImmediate(collider);
var body=e.GetComponent<CharacterController>();if(body==null)body=e.gameObject.AddComponent<CharacterController>();body.height=1.5f;body.radius=.33f;body.center=new Vector3(0,.75f,0);body.stepOffset=.2f;body.skinWidth=.02f;
if(e.GetComponent<Ossastra.Enemies.EnemyMovement>()==null)e.gameObject.AddComponent<Ossastra.Enemies.EnemyMovement>();
if(GameObject.Find("Roadmap_LOS_Wall")==null){var wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.name="Roadmap_LOS_Wall";wall.transform.position=new Vector3(2,.8f,0);wall.transform.localScale=new Vector3(.25f,1.6f,3);}
var debug=e.GetComponent<Ossastra.Development.EnemyChaseDebugView>();if(debug==null)debug=e.gameObject.AddComponent<Ossastra.Development.EnemyChaseDebugView>();var ds=new SerializedObject(debug);ds.FindProperty("enemy").objectReferenceValue=e;ds.ApplyModifiedPropertiesWithoutUndo();EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);ctx.ReturnValue=new{scene=s.path,targets=dest.arraySize,npcs=UnityEngine.Object.FindObjectsOfType<Ossastra.Npcs.NpcRoot>().Length};}}
