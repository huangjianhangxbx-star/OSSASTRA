using System;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEngine.InputSystem;using UnityEngine.InputSystem.LowLevel;using Ossastra.Characters;using Ossastra.CharacterCore;using Ossastra.Combat;using Ossastra.Enemies;using Ossastra.Development;using Funplay.Editor.Tools.Scripting;
public class LifeWorld:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){
if(!Application.isPlaying)throw new Exception("Play required");
var service=UnityEngine.Object.FindObjectOfType<PartyRespawnCoordinator>();var debug=UnityEngine.Object.FindObjectOfType<CombatDeathDebugView>();var d=UnityEngine.Object.FindObjectOfType<CompanionFollowDirector>();var e=UnityEngine.Object.FindObjectOfType<Ossastra.Enemies.EnemyRoot>();if(service==null||debug==null||d==null||e==null)throw new Exception("Missing world");
var c=d.Coordinator;var a=c.Characters[0];var b=c.Characters[1];var roots=new[]{a,b};var inputs=roots.Select(x=>x.GetComponent<CharacterInputHandler>()).ToArray();var healths=roots.Select(x=>x.Vitals.Health).ToArray();var initialGeneration=a.LifeGeneration;
var originalSettings=InputSystem.settings;var testSettings=UnityEngine.Object.Instantiate(originalSettings);testSettings.backgroundBehavior=InputSettings.BackgroundBehavior.IgnoreFocus;testSettings.editorInputBehaviorInPlayMode=InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;InputSystem.settings=testSettings;var keyboard=InputSystem.AddDevice<Keyboard>("M02B1Keyboard");var mouse=InputSystem.AddDevice<Mouse>("M02B1Mouse");var overrides=new Dictionary<InputAction,string>();foreach(var action in inputs.SelectMany(i=>new[]{i.movementAction,i.mouseAction,i.attackAction,i.cancelAction,i.pauseAction}).Concat(new[]{c.switchAction})){overrides[action]=action.SaveBindingOverridesAsJson();for(int i=0;i<action.bindings.Count;i++){var path=action.bindings[i].path;if(path==null||action.bindings[i].isComposite)continue;if(path.Contains("<Keyboard>"))action.ApplyBindingOverride(i,path.Replace("<Keyboard>",keyboard.path));if(path.Contains("<Mouse>"))action.ApplyBindingOverride(i,path.Replace("<Mouse>",mouse.path));}}
var game=typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView");if(game!=null)EditorWindow.GetWindow(game).Focus();
var records=new List<object>();var steps=new List<(double at,Action run)>();int next=0,failed=0;var start=EditorApplication.timeSinceStartup;GameObject wall=null;double pausedAt=0;Vector3 before=default;int generation=0;
void Flush(string state){SessionState.SetString("ROADMAP_S1_WORLD",Newtonsoft.Json.JsonConvert.SerializeObject(new{state,failed,records},Newtonsoft.Json.Formatting.Indented));}
void Check(string label,bool pass){if(!pass)failed++;records.Add(new{label,pass,a=a.IsAlive,b=b.IsAlive,hpA=a.Vitals.Health.Current,hpB=b.Vitals.Health.Current,generationA=a.LifeGeneration,generationB=b.LifeGeneration,current=c.Current==null?"none":c.Current.ActorId,time=service.SimulationTime,status=service.Status});Flush("running");}
void Keys(params Key[] keys){InputSystem.QueueStateEvent(keyboard,new KeyboardState(keys));}
void Mouse(bool held){var p=Camera.main.WorldToScreenPoint(e.transform.position);InputSystem.QueueStateEvent(mouse,new MouseState{position=new Vector2(p.x,p.y),buttons=(ushort)(held?1:0)});}
void Kill(Ossastra.Characters.CharacterRoot r){CombatDeathDebugView.ApplyDevelopmentHit(r.Vitals,1000);}
debug.RestartForDevelopment();d.enabled=false;e.AttacksEnabled=false;foreach(var r in roots){r.SendMessage("OnApplicationFocus",true);r.SetPaused(false);}c.SendMessage("OnApplicationFocus",true);e.SendMessage("OnApplicationFocus",true);service.SendMessage("OnApplicationFocus",true);Keys();Mouse(false);
steps.Add((.2,()=>{a.GetComponent<CharacterMovement>().Teleport(e.transform.position+Vector3.left*1.15f);b.GetComponent<CharacterMovement>().Teleport(new Vector3(-4,.03f,-3));CombatDeathDebugView.ApplyDevelopmentHit(a.Vitals,80);Physics.SyncTransforms();e.AttacksEnabled=true;}));
steps.Add((2,()=>{Check("S1-real-enemy-kills-A-and-B-controls",!a.IsAlive&&c.Current==b&&a.Vitals.DeathCount>0);Check("S1-dead-no-collider",a.GetComponentsInChildren<Collider>().All(x=>!x.enabled));e.AttacksEnabled=false;service.SendMessage("OnApplicationFocus",false);pausedAt=service.SimulationTime;}));
steps.Add((4,()=>{Check("S1-focus-stops-clock",Math.Abs(service.SimulationTime-pausedAt)<.001);service.SendMessage("OnApplicationFocus",true);}));
steps.Add((14,()=>{Check("S1-single-same-identity-full",a.IsAlive&&ReferenceEquals(healths[0],a.Vitals.Health)&&a.Vitals.Health.Current==100&&a.GetComponent<CombatPosture>().Current==100&&a.LifeGeneration==initialGeneration+1);Check("S1-does-not-steal-current",c.Current==b);}));
steps.Add((14.2,()=>{wall=GameObject.CreatePrimitive(PrimitiveType.Cube);wall.name="RoadmapLifeProbe";wall.transform.position=new Vector3(-2,.8f,-3);wall.transform.localScale=new Vector3(6,2,4);Physics.SyncTransforms();Kill(a);Kill(b);Check("S1-wipe-clears-current",c.Current==null&&!a.IsAlive&&!b.IsAlive);}));
steps.Add((15.5,()=>{Check("S1-blocked-spawn-stays-pending",!a.IsAlive&&!b.IsAlive&&service.Status=="等待合法落点");UnityEngine.Object.DestroyImmediate(wall);wall=null;Physics.SyncTransforms();}));
steps.Add((16.2,()=>{Check("S1-wipe-batch-safe-restored",a.IsAlive&&b.IsAlive&&c.Current!=null&&Vector3.Distance(a.transform.position,b.transform.position)>1);Check("S1-same-healths-no-reconstruction",ReferenceEquals(healths[0],a.Vitals.Health)&&ReferenceEquals(healths[1],b.Vitals.Health));Keys(Key.D);Mouse(true);}));
steps.Add((16.35,()=>{Kill(a);Kill(b);generation=a.LifeGeneration;}));
steps.Add((17.2,()=>{before=c.Current.transform.position;Check("S1-held-after-respawn-no-action",a.LifeGeneration==generation+1&&!a.ActionActive&&!b.ActionActive&&c.Current.DataBus.Input.move==Vector2.zero);}));
steps.Add((17.5,()=>{Check("S1-held-movement-blocked",Vector3.Distance(before,c.Current.transform.position)<.03f);Keys();Mouse(false);}));
steps.Add((17.8,()=>Keys(Key.D)));
steps.Add((18.1,()=>{Check("S1-new-press-after-release-moves",Vector3.Distance(before,c.Current.transform.position)>.1f);Keys();Mouse(false);}));
EditorApplication.CallbackFunction tick=null;void Cleanup(){EditorApplication.update-=tick;e.AttacksEnabled=false;d.enabled=true;if(wall!=null)UnityEngine.Object.DestroyImmediate(wall);foreach(var kv in overrides)kv.Key.LoadBindingOverridesFromJson(kv.Value);InputSystem.RemoveDevice(keyboard);InputSystem.RemoveDevice(mouse);InputSystem.settings=originalSettings;UnityEngine.Object.DestroyImmediate(testSettings);}
tick=()=>{try{if(!Application.isPlaying)throw new Exception("Play ended");while(next<steps.Count&&EditorApplication.timeSinceStartup-start>=steps[next].at)steps[next++].run();if(next==steps.Count){Cleanup();Flush("completed");}}catch(Exception ex){failed++;records.Add(new{error=ex.ToString()});Cleanup();Flush("failed");}};EditorApplication.update+=tick;Flush("running");ctx.Log("Scheduled S1 actual-world 18.1 seconds");
}}
