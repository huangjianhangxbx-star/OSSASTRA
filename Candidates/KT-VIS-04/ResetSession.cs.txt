using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
public sealed class ResetRoomSession:IFunplayCommand {public void Execute(ExecutionContext ctx){var s=SceneManager.GetActiveScene();if(!Application.isPlaying||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new Exception("Wrong lab");var m=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Ossastra.SceneVisibility.Rooms.RoomVisibilityModule>(true)).Single();m.ResetSessionDiscovery();ctx.ReturnValue=new{resetSessionOnly=true,discovered=m.DiscoveredRoomCount};}}
