    static void Need(bool value, string error) { if (!value) throw new InvalidOperationException(error); }
    static string Full(string path) { return Path.GetFullPath(path).TrimEnd(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar); }
    static string Sha(string path) { using (var stream = File.OpenRead(path)) using (var hash = SHA256.Create()) return BitConverter.ToString(hash.ComputeHash(stream)).Replace("-", ""); }
    static string Hierarchy(Transform node) { string result = node.name; while (node.parent != null) { node = node.parent; result = node.name + "/" + result; } return result; }
    T[] Components<T>() where T : Component { return scene.GetRootGameObjects().SelectMany(go => go.GetComponentsInChildren<T>(true)).ToArray(); }
    T One<T>() where T : Component { var items = Components<T>(); Need(items.Length == 1, "要求唯一" + typeof(T).FullName + "，实际" + items.Length); return items[0]; }
    GameObject Empty(string name, Transform parent, Vector3 position)
    {
        var go = new GameObject(name); ctx.RegisterObjectCreation(go); createdObjects.Add(go);
        if (parent != null) go.transform.SetParent(parent, false);
        go.transform.position = position; return go;
    }
    Renderer Box(string name, Transform parent, Vector3 center, Vector3 size, Material material, bool solid = true)
    {
        Need(size.x > 0 && size.y > 0 && size.z > 0, "Box尺寸必须为正：" + name);
        var go = GameObject.CreatePrimitive(PrimitiveType.Cube); ctx.RegisterObjectCreation(go); createdObjects.Add(go); go.name = name;
        if (parent != null) go.transform.SetParent(parent, false);
        go.transform.position = center; go.transform.localScale = size;
        GameObjectUtility.SetStaticEditorFlags(go, 0);
        var renderer = go.GetComponent<MeshRenderer>(); renderer.sharedMaterial = material;
        renderer.shadowCastingMode = ShadowCastingMode.On; renderer.receiveShadows = true;
        if (!solid) ctx.DestroyObject(go.GetComponent<BoxCollider>());
        return renderer;
    }
    void Folder(string path)
    {
        if (AssetDatabase.IsValidFolder(path)) return;
        Need(!Directory.Exists(path) && !File.Exists(path + ".meta"), "存在未知文件系统目录，拒绝覆盖：" + path);
        string parent = Path.GetDirectoryName(path).Replace('\\', '/'); Folder(parent);
        Need(!string.IsNullOrEmpty(AssetDatabase.CreateFolder(parent, Path.GetFileName(path))), "创建自有目录失败：" + path); createdFolders.Add(path);
    }
    T Asset<T>(T value, string name) where T : Object
    {
        string path = AssetFolder + "/" + name;
        Need(!File.Exists(path) && !File.Exists(path + ".meta") && AssetDatabase.LoadMainAssetAtPath(path) == null, "拒绝覆盖资产：" + path);
        ctx.RegisterObjectCreation(value); AssetDatabase.CreateAsset(value, path); createdAssets.Add(path); AssetDatabase.SaveAssetIfDirty(value); return value;
    }
    Material Material(string name, Shader shader, Color color)
    {
        var material = new Material(shader) { name = name };
        if (material.HasProperty("_BaseColor")) material.SetColor("_BaseColor", color);
        if (material.HasProperty("_Color")) material.SetColor("_Color", color);
        if (material.HasProperty("_Smoothness")) material.SetFloat("_Smoothness", .15f);
        if (material.HasProperty("_Metallic")) material.SetFloat("_Metallic", 0);
        if (material.HasProperty("_Cull")) material.SetFloat("_Cull", 2);
        return Asset(material, name + ".mat");
    }
    Mesh Aperture(string name, float width, float height)
    {
        var mesh = new Mesh { name = name };
        mesh.vertices = new[] { new Vector3(-width / 2, -height / 2, 0), new Vector3(width / 2, -height / 2, 0), new Vector3(width / 2, height / 2, 0), new Vector3(-width / 2, height / 2, 0) };
        mesh.uv = new[] { Vector2.zero, Vector2.right, Vector2.one, Vector2.up };
        mesh.triangles = new[] { 0, 1, 2, 0, 2, 3 }; mesh.RecalculateNormals(); mesh.RecalculateBounds();
        return Asset(mesh, name + ".asset");
    }
    void MakeAssets()
    {
        var lit = Shader.Find("Universal Render Pipeline/Lit"); var unlit = Shader.Find("Universal Render Pipeline/Unlit");
        Need(lit != null && unlit != null && lit.isSupported && unlit.isSupported, "缺可用URP Lit/Unlit，未创建场景几何");
        Folder(AssetFolder);
        wall = Material("Structure", lit, new Color(.53f, .56f, .6f));
        floor = Material("InteriorFloor", lit, new Color(.22f, .27f, .28f));
        roof = Material("Roof", lit, new Color(.25f, .34f, .42f));
        Need(roof.FindPass("ShadowCaster") >= 0, "Roof材质没有ShadowCaster；不能用无阴影顶盖假装保光");
        cabinet = Material("Furniture", lit, new Color(.43f, .29f, .18f));
        vegetation = Material("Canopy", lit, new Color(.17f, .44f, .25f));
        trunk = Material("Trunk", lit, new Color(.25f, .17f, .1f));
        stone = Material("Stone", lit, new Color(.43f, .45f, .49f));
        emissive = Material("BrightMarker", unlit, new Color(1f, .82f, .08f));
        doorMesh = Aperture("DoorOpening_2p6x2p5", 2.6f, 2.5f);
        windowMesh = Aperture("WindowOpening_2p4x1p6", 2.4f, 1.6f);
        smallDoorMesh = Aperture("DoorOpening_1p6x2p5", 1.6f, 2.5f);
        motionClip = new AnimationClip { name = "WindowMotion", legacy = true, wrapMode = WrapMode.Loop };
        motionClip.SetCurve("", typeof(Transform), "localPosition.x", new AnimationCurve(new Keyframe(0, -.9f), new Keyframe(1, .9f), new Keyframe(2, -.9f)));
        motionClip = Asset(motionClip, "WindowMotion.anim");
    }
    void WallPart(string name, Transform parent, bool alongX, float fixedPosition, float lo, float hi, float bottom, float top, List<Renderer> structure)
    {
        Need(hi > lo && top > bottom, "墙段范围无效：" + name);
        var center = alongX ? new Vector3((lo + hi) / 2, (bottom + top) / 2, fixedPosition) : new Vector3(fixedPosition, (bottom + top) / 2, (lo + hi) / 2);
        var size = alongX ? new Vector3(hi - lo, top - bottom, WallThickness) : new Vector3(WallThickness, top - bottom, hi - lo);
        var renderer = Box(name, parent, center, size, wall); structure.Add(renderer); wallSolids.Add(renderer.GetComponent<BoxCollider>());
    }
    void SolidWall(string name, Transform parent, bool alongX, float at, float lo, float hi, List<Renderer> structure)
    { WallPart(name, parent, alongX, at, lo, hi, 0, WallHeight, structure); }
    // 四段均为完整有顶面/厚度的实体Box，开口没有隐藏的大整墙或ZTest Always替代几何。
    void OpeningWall(string name, Transform parent, bool alongX, float at, float lo, float hi, float center, float width, float bottom, float height, List<Renderer> structure)
    {
        float left = center - width / 2, right = center + width / 2, top = bottom + height;
        Need(left > lo && right < hi && bottom > 0 && top < WallHeight, "开口必须严格落在四段墙内：" + name);
        WallPart(name + "_Side0", parent, alongX, at, lo, left, 0, WallHeight, structure);
        WallPart(name + "_Side1", parent, alongX, at, right, hi, 0, WallHeight, structure);
        WallPart(name + "_Sill", parent, alongX, at, left, right, 0, bottom, structure);
        WallPart(name + "_Lintel", parent, alongX, at, left, right, top, WallHeight, structure);
    }
    RoomRegion Room(string name, string id, Transform building, Vector3 center, Vector2 size, List<Renderer> contents)
    {
        var go = Empty(name, building, center); var room = go.AddComponent<RoomRegion>(); room.roomId = id; room.floorId = "Ground";
        room.volumes = new[] { new RoomLocalBox { center = new Vector3(0, WallHeight / 2, 0), size = new Vector3(size.x, WallHeight, size.y) } };
        contents.Add(Box("Floor", go.transform, new Vector3(center.x, FloorTop / 2, center.z), new Vector3(size.x, FloorTop, size.y), floor));
        return room;
    }
    RoofVisibilityGroup Roof(string name, string id, Transform building, Vector3 center, Vector2 size)
    {
        var go = Empty(name, building, center); var group = go.AddComponent<RoofVisibilityGroup>(); group.roofGroupId = id;
        group.members = new[] { Box("RoofVisual_AndPhysicalSolid", go.transform, new Vector3(center.x, WallHeight + .1f, center.z), new Vector3(size.x, .2f, size.y), roof) };
        return group;
    }
    ViewPortal Portal(string name, string id, Transform parent, RoomRegion inside, RoomRegion outside, Vector3 position, Vector3 outward, Mesh aperture, bool traversable, bool open = true)
    {
        var go = Empty(name, parent, position); go.transform.rotation = Quaternion.LookRotation(outward, Vector3.up);
        var portal = go.AddComponent<ViewPortal>(); portal.portalId = id; portal.inside = inside; portal.outside = outside;
        portal.apertureMesh = aperture; portal.open = open; portal.canSeeThrough = true; portal.canTraverse = traversable; allPortals.Add(portal); return portal;
    }
    SceneOccluderGroup Occluder(GameObject go, string label, Renderer[] members, Collider[] volumes, bool reveal)
    {
        var group = go.AddComponent<SceneOccluderGroup>(); group.groupLabel = label; group.members = members; group.volumes = volumes;
        group.occlusionPolicy = reveal ? SceneOcclusionPolicy.CharacterReveal : SceneOcclusionPolicy.KeepNaturalOcclusion;
        group.allowWholeFade = reveal; group.minimumOpacity = .15f; newGroups.Add(group); return group;
    }
    Transform MovingMarker(string name, Vector3 position, Vector3 outward)
    {
        var anchor = Empty(name, sample.transform, position); anchor.transform.rotation = Quaternion.LookRotation(outward, Vector3.up);
        var visual = Box("AnimatedCurrentFrameMarker", anchor.transform, position, new Vector3(.55f, .85f, .4f), emissive, false);
        visual.transform.localPosition = Vector3.zero; visual.shadowCastingMode = ShadowCastingMode.Off;
        var animation = visual.gameObject.AddComponent<Animation>(); animation.AddClip(motionClip, "WindowMotion"); animation.clip = motionClip;
        animation.playAutomatically = true; animation.wrapMode = WrapMode.Loop; animation.cullingType = AnimationCullingType.AlwaysAnimate;
        return visual.transform;
    }
    void LocalLight(string name, Transform parent, Vector3 position, Color color)

    void Tree(string name, Vector3 position, bool allowCanopy)
    {
        var go = Empty(name, sample.transform, position);
        var stem = Box("NaturalSolidTrunk", go.transform, position + Vector3.up * 2.1f, new Vector3(.5f, 4.2f, .5f), trunk);
        Occluder(stem.gameObject, name + " / 实体树干", new[] { stem }, new Collider[] { stem.GetComponent<BoxCollider>() }, false);

        Occluder(canopy.gameObject, name + (allowCanopy ? " / 许可树冠" : " / 自然树冠"), new[] { canopy }, new Collider[] { canopy.GetComponent<BoxCollider>() }, allowCanopy);
    }

    static string CacheStamp(GroundSurfaceBinding binding)
    {
        var cache = binding.cacheRoot == null ? null : binding.cacheRoot.GetComponent<GroundSurfaceMeshCache>();

    }
