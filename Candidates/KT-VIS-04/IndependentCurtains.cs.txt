/* 功能：真实可见B地板最近Collider采样，独立切未知黑幕，不拿被墙挡住的投影当室内内容。 */
using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.SceneManagement;using Funplay.Editor.Tools.Scripting;
public sealed class IndependentRoomCurtains:IFunplayCommand {public void Execute(ExecutionContext ctx){
 var s=SceneManager.GetActiveScene();if(!Application.isPlaying||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new Exception("Wrong lab");var root=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Ossastra.SceneVisibility.SceneVisibilityRoot>(true)).Single();var m=root.roomModule;var b=m.buildings.SelectMany(x=>x.regions).Single(r=>r.roomId=="room-b");if(m.ActiveRoom==null||m.ActiveRoom.roomId!="room-a"||m.IsDiscovered(b))throw new Exception("Need A known B unknown");var c=root.gameplayCamera;var floor=b.members.Single(r=>r.name=="Floor");var ignores=m.buildings.SelectMany(x=>x.roofGroups).SelectMany(r=>r.labelRaycastExclusions).ToArray();var points=new List<Vector3>();
 for(float x=-9;x<=-3;x+=1)for(float z=-15;z<=-9;z+=1){var p=new Vector3(x,.105f,z);var v=c.WorldToViewportPoint(p);if(v.z<=0||v.x<=0||v.x>=1||v.y<=0||v.y>=1)continue;var ray=c.ViewportPointToRay(v);var hits=Physics.RaycastAll(ray.origin,ray.direction,Vector3.Distance(ray.origin,p)+.5f,~0,QueryTriggerInteraction.Ignore).Where(h=>!ignores.Contains(h.collider)).OrderBy(h=>h.distance).ToArray();if(hits.Length>0&&hits[0].collider.GetComponent<Renderer>()==floor)points.Add(p);}if(points.Count<2)throw new Exception("No actual visible B floor samples");
 var old=c.targetTexture;var active=RenderTexture.active;float aspect=c.aspect;bool ex=m.exteriorBlackoutEnabled,inter=m.interiorUnexploredEnabled;var rt=new RenderTexture(960,540,24,RenderTextureFormat.ARGB32){antiAliasing=4};var reader=new Texture2D(960,540,TextureFormat.RGB24,false);var rows=new List<object>();int darkCount=0,litCount=0;try{rt.Create();c.targetTexture=rt;c.aspect=16f/9f;m.exteriorBlackoutEnabled=false;
 for(int k=0;k<2;k++){m.interiorUnexploredEnabled=k==1;c.Render();RenderTexture.active=rt;reader.ReadPixels(new Rect(0,0,960,540),0,0);reader.Apply();var values=points.Select(p=>{var v=c.WorldToViewportPoint(p);return reader.GetPixel((int)(v.x*960),(int)(v.y*540));}).ToArray();if(k==0)litCount=values.Count(v=>v.maxColorComponent>.05f);else darkCount=values.Count(v=>v.maxColorComponent<.01f);string file="E:/WORLDCREATOR/XingHaiHuiLang/研究/可见性修复/KT-VIS-04/independent-interior-"+k+".png";File.WriteAllBytes(file,reader.EncodeToPNG());rows.Add(new{exterior=false,interior=k==1,visibleFloorSamples=points.Count,colors=values.Select(v=>v.ToString("F3")).ToArray(),file});}
 ctx.ReturnValue=new{passed=litCount>=2&&darkCount>=2,litCount,darkCount,points=points.Select(p=>p.ToString("R")).ToArray(),rows,scope="未知B actual floor采样/独立内部黑幕；非外部乘黑假阳性"};
 }finally{m.exteriorBlackoutEnabled=ex;m.interiorUnexploredEnabled=inter;c.targetTexture=old;c.aspect=aspect;RenderTexture.active=active;UnityEngine.Object.DestroyImmediate(reader);rt.Release();UnityEngine.Object.DestroyImmediate(rt);c.Render();}
}}
