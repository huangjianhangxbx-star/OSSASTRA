    static void CapturePhysicalBaseline()
    {
        physicalDevices = InputSystem.devices.ToArray();

    static bool PhysicalIdle()
    {
        var devices = InputSystem.devices.Where(d => d != keyboard && d != mouse).ToArray();
        if (devices.Length != physicalDevices.Length || devices.Any(d => !physicalDevices.Contains(d))) return false;
        if (physicalMice.Any(m => m.Device == null || !devices.Contains(m.Device) || Buttons(m.Device) != "00000" || Buttons(m.Device) != m.Buttons || (m.Device.position.ReadValue() - m.Position).sqrMagnitude > 1 || m.Device.scroll.ReadValue().sqrMagnitude > .01f)) return false;
        return devices.All(d => !(d is Keyboard k && k.allKeys.Any(b => b != null && b.isPressed)) && !(d is Gamepad g && (g.leftStick.ReadValue().sqrMagnitude > .04f || g.rightStick.ReadValue().sqrMagnitude > .04f || g.allControls.OfType<ButtonControl>().Any(b => b.isPressed))));
    }
    static void PhysicalEvent(InputEventPtr eventPtr, InputDevice device)
    {
        if (!Running || error != null || device == keyboard || device == mouse || device == null || !physicalDevices.Contains(device)) return;
        if (!eventPtr.IsA<StateEvent>() && !eventPtr.IsA<DeltaStateEvent>()) return;
        bool active = false;
        if (device is Keyboard k) active = k.allKeys.Any(b => b != null && b.ReadValueFromEvent(eventPtr, out float value) && value > .5f);
        else if (device is Mouse m)
        {
            active = new[] { m.leftButton, m.rightButton, m.middleButton, m.forwardButton, m.backButton }.Any(b => b.ReadValueFromEvent(eventPtr, out float value) && value > .5f);
            var baseline = physicalMice.SingleOrDefault(p => p.Device == m);
            if (baseline != null && m.position.ReadValueFromEvent(eventPtr, out Vector2 position) && (position - baseline.Position).sqrMagnitude > 1) active = true;
            if (m.scroll.ReadValueFromEvent(eventPtr, out Vector2 scroll) && scroll.sqrMagnitude > .01f) active = true;
        }
        else if (device is Gamepad g)
        {
            active = g.allControls.OfType<ButtonControl>().Any(b => b.ReadValueFromEvent(eventPtr, out float value) && value > .5f);
            if (g.leftStick.ReadValueFromEvent(eventPtr, out Vector2 left) && left.sqrMagnitude > .04f) active = true;
            if (g.rightStick.ReadValueFromEvent(eventPtr, out Vector2 right) && right.sqrMagnitude > .04f) active = true;
        }
        if (active) { keys = Array.Empty<Key>(); error = "检测到真实设备输入，立即撤回虚拟键；下一Editor更新按所有权恢复：" + device.name + "/" + device.deviceId; }
    }
    static void BeforeInput()
    {
        if (!Running || !Application.isPlaying || InputState.currentUpdateType != InputUpdateType.Dynamic || keyboard == null || mouse == null) return;
        if (!keyboard.enabled) InputSystem.EnableDevice(keyboard); if (!mouse.enabled) InputSystem.EnableDevice(mouse);
        InputSystem.QueueStateEvent(keyboard, new KeyboardState(error == null ? keys : Array.Empty<Key>()));
        InputSystem.QueueStateEvent(mouse, new MouseState { position = pointer });
    }

        Need(mapView != null && mapView.IsValid && !mapView.IsStructureMode && ReferenceEquals(orbitField.GetValue(mapView), fixtureOrbit) && fixtureOrbit.TargetSector == 0 && Mathf.Abs(Mathf.DeltaAngle(fixtureOrbit.ActualYaw, 0)) < .02f, "本候选orbit已被更换/旋转，保留外部状态。");
        Need(Mathf.Abs((float)zoomField.GetValue(mapView) - originalZoom) < .00001f && Mathf.Abs(mapView.pitch - originalPitch) < .00001f, "zoom/pitch外部更改，候选不覆盖。");
        Need(visibility.effectEnabled && !visibility.detectionOnly && rooms.IsReady && rooms.effectEnabled && rooms.revealRoofs && rooms.exteriorBlackoutEnabled && rooms.interiorUnexploredEnabled && rooms.maskDebugMode == 0 && string.IsNullOrEmpty(visibility.RenderError), "房间/局部效果或渲染错误状态变化。");
    }
    static void Tick()
    {
        if (!Running) return;
        try
        {
            Guard(); double now = EditorApplication.timeSinceStartup;
            Need(now - began < HardDeadlineSeconds, "50秒步行硬上限；不二次摆位或绕过阻挡。");
            Need(now - lastRenderAt < 3, "GameplayCamera三秒没有真实render，不能代签屋顶/黑幕。");
            EditorApplication.QueuePlayerLoopUpdate(); if (lastTickFrame == Time.frameCount) return; lastTickFrame = Time.frameCount;
            Vector3 position = actor.transform.position; float delta = Horizontal(position - previousPosition); previousPosition = position; actualWalkDistance += delta;
            if (delta > .0001f) movedFrames++; if (input.CurrentFrame.move.sqrMagnitude > .01f) inputFrames++;
            keys = Array.Empty<Key>();
            if (stage == "opening-neutral")
            {
                if (now - phaseAt >= 1.2 && Time.frameCount - neutralStartFrame >= 3) { Need(Mathf.Abs(Mathf.DeltaAngle(mapView.ActualYaw, 0)) < .02f, "原MapResearchRoot未消费临时0度观察。"); StartLeg(); }
            }
            else if (stage.StartsWith("walk-", StringComparison.Ordinal))
            {
                Need(now - phaseAt < 9, "目标步行9秒超时：" + Route[waypoint].Name);
                var difference = Route[waypoint].Point - position;
                if (Horizontal(difference) <= ArrivalTolerance)
                {
                    reachedCount++; stage = "observe-" + Route[waypoint].Name; phaseAt = now; arrivalChecked = false; noProgressSince = 0;
                    Add("reached-by-original-WASD", new { waypoint, target = Route[waypoint].Name, requested = Route[waypoint].Point.ToString("R"), actual = position.ToString("R"), distance = Horizontal(difference), cumulativeActualDistance = actualWalkDistance });
                }
                else
                {
                    if (Mathf.Abs(difference.x) >= Mathf.Abs(difference.z)) keys = new[] { difference.x > 0 ? Key.D : Key.A }; else keys = new[] { difference.z > 0 ? Key.W : Key.S };
                    if (delta > .001f) noProgressSince = now; else if (noProgressSince <= 0) noProgressSince = now;
                    Need(now - noProgressSince < 3, "原Motor连续三秒无位置增长：" + Route[waypoint].Name + "；保留真实碰撞失败。");
                }
            }
            else if (stage.StartsWith("observe-", StringComparison.Ordinal))
            {
                Need(Horizontal(Route[waypoint].Point - position) <= .35f, "到达后被真实运动移出容差，不伪造停留。");
                if (!arrivalChecked && now - phaseAt >= ArrivalObservationSeconds)
                {
                    var expected = Route[waypoint].Room;
                    Need(expected == null ? rooms.ActiveBuilding == null : rooms.ActiveBuilding == abc && rooms.ActiveRoom != null && rooms.ActiveRoom.roomId == expected && rooms.IsDiscovered(rooms.ActiveRoom), "真实房间解析/到达状态与停留点不一致：" + Route[waypoint].Name);
                    if (waypoint < 2) Need(!rooms.IsDiscovered(abc.regions.Single(r => r.roomId == "room-b")), "主控未进B却提前发现B。");
                    arrivalChecked = true; Add("arrival-observed-0p7-seconds", new { waypoint, target = Route[waypoint].Name, building = rooms.ActiveBuilding == null ? "Exterior" : rooms.ActiveBuilding.buildingId, room = rooms.ActiveRoom == null ? null : rooms.ActiveRoom.roomId, discovered = Discovered() });
                }
                if (arrivalChecked && now - phaseAt >= ArrivalObservationSeconds + Route[waypoint].FilmHold)
                {
                    waypoint++;
                    if (waypoint < Route.Length) StartLeg(); else { stage = "final-exterior-hold"; phaseAt = now; Add("route-completed", new { reachedCount, actualWalkDistance }); }
                }
            }
            else if (stage == "final-exterior-hold" && now - began >= MinimumFilmSeconds && now - phaseAt >= .7)
            {
                Need(reachedCount == Route.Length && actualWalkDistance >= MinimumWalkDistance() && movedFrames > 20 && inputFrames > 20, "缺原WASD/Motor实际位移证据；实际" + actualWalkDistance + "，路线容差下界" + MinimumWalkDistance());
                Need(sawInside && sawExteriorBefore && sawExteriorAfter && rooms.ActiveBuilding == null && renderedFrames > 30 && renderedEnds == renderedFrames, "缺进栋/离栋/完整相机scope证据。");
                Need(abc.regions.All(rooms.IsDiscovered) && rooms.IsDiscovered(independent.regions.Single()) == dWasDiscovered, "本次到达集合未覆盖ABC或错误影响D。");
                Finish(true); return;
            }
            if (now - lastPublish >= 1) { lastPublish = now; Publish(false); }
        }
        catch (Exception ex) { error = ex.ToString(); Finish(false); }
    }
    static void StartLeg() { stage = "walk-" + Route[waypoint].Name; phaseAt = noProgressSince = EditorApplication.timeSinceStartup; keys = Array.Empty<Key>(); Add("leg-start", new { waypoint, target = Route[waypoint].Name, worldTarget = Route[waypoint].Point.ToString("R") }); }
    static string Discovered() => string.Join(",", regions.Where(rooms.IsDiscovered).Select(r => r.roomId).OrderBy(id => id, StringComparer.Ordinal));
    static string Modes(Renderer[] values) => string.Join(",", values.Select(r => r == null ? "Missing" : r.shadowCastingMode.ToString()));
    static void CameraBegin(ScriptableRenderContext context, Camera value)
    {
        if (!Running || error != null || value != camera) return;
        try
        {
            Need(pendingRender == null, "GameplayCamera出现未闭合/递归scope。");
            bool inside = rooms.ActiveBuilding == abc; Need(rooms.ActiveBuilding == null || inside, "路线错误进入独立建筑D。");
            var expected = inside ? ShadowCastingMode.ShadowsOnly : ShadowCastingMode.On;
            Need(abcRoofs.All(r => r != null && r.enabled && r.shadowCastingMode == expected), "ABC三屋顶没有同帧统一" + expected + "，或被禁用。");
            Need(dRoofs.All(r => r != null && r.enabled && r.shadowCastingMode == ShadowCastingMode.On), "独立D屋顶被错误联动。");
            if (inside) sawInside = true; else if (!sawInside) sawExteriorBefore = true; else sawExteriorAfter = true;
            lastRenderAt = EditorApplication.timeSinceStartup; renderedFrames++;
            Vector3 position = actor.transform.position; var move = input.CurrentFrame.move;
            pendingRender = new FrameFact { render = renderedFrames, frame = Time.frameCount, waypoint = waypoint, seconds = lastRenderAt - began, stage = stage, building = inside ? abc.buildingId : "Exterior", room = rooms.ActiveRoom == null ? null : rooms.ActiveRoom.roomId, discovered = Discovered(), x = position.x, y = position.y, z = position.z, inputX = move.x, inputY = move.y, velocityX = motor.Velocity.x, velocityZ = motor.Velocity.z, requestedX = motor.RequestedVelocity.x, requestedZ = motor.RequestedVelocity.z, motorAllowed = motor.LastAllowed, motorMoving = motor.IsMoving, environmentBlocked = motor.LastEnvironmentSideBlocked, abcRoofModes = Modes(abcRoofs), dRoofModes = Modes(dRoofs) };
            trace.Add(pendingRender);
        }
        catch (Exception ex) { keys = Array.Empty<Key>(); error = "GameplayCamera begin: " + ex; }
    }
    static void CameraEnd(ScriptableRenderContext context, Camera value)
    {
        if (!Running || value != camera || pendingRender == null) return;
        try
        {
            bool found = RoomMaskCameraRegistry.TryGetStatistics(camera, out var stats); pendingRender.maskActive = found && stats.Active; pendingRender.maskError = found ? stats.Error : null;
            Need(!found || string.IsNullOrEmpty(stats.Error), "BlackoutStats.Error：" + (found ? stats.Error : ""));
            Need(pendingRender.building == "Exterior" || found && stats.Active, "在栋内没有实际mask后端执行。");
            pendingRender.cameraScopeRestored = abcRoofs.Concat(dRoofs).All(r => r != null && r.shadowCastingMode == ShadowCastingMode.On && r.enabled);
            Need(pendingRender.cameraScopeRestored, "相机End未按所有权恢复屋顶原状态。"); renderedEnds++;
        }
        catch (Exception ex) { keys = Array.Empty<Key>(); error = (error ?? "") + "\nGameplayCamera end: " + ex; }
        finally { pendingRender = null; }
    }
    static void Add(string kind, object data) { facts.Add(new { seconds = EditorApplication.timeSinceStartup - began, kind, data }); }
    public static string Cancel() { if (Running) { error = "Root取消WalkingProbe。"; Finish(false); } return ResultJson; }
    static void ModeChanged(PlayModeStateChange value) { if (Running && (value == PlayModeStateChange.ExitingPlayMode || value == PlayModeStateChange.EnteredEditMode)) { error = "Play退出，按所有权停止。"; Finish(false); } }
    static void Reload() { if (Running) { error = "域重载，按所有权停止。"; Finish(false); } }
    static void CleanupError(string label, Exception ex) { error = (error ?? "") + "\n" + label + ": " + ex.Message; passed = false; }
    static void Finish(bool ok)
    {
        if (finished) return; finished = true; passed = ok && error == null; Running = false; keys = Array.Empty<Key>();
        InputSystem.onBeforeUpdate -= BeforeInput; InputSystem.onEvent -= PhysicalEvent; EditorApplication.update -= Tick; RenderPipelineManager.beginCameraRendering -= CameraBegin; RenderPipelineManager.endCameraRendering -= CameraEnd;
        EditorApplication.playModeStateChanged -= ModeChanged; AssemblyReloadEvents.beforeAssemblyReload -= Reload;
        try { if (keyboard != null && InputSystem.devices.Contains(keyboard)) InputSystem.ResetDevice(keyboard); if (mouse != null && InputSystem.devices.Contains(mouse)) InputSystem.ResetDevice(mouse); } catch (Exception ex) { CleanupError("只复位自有设备", ex); }
        foreach (var filter in filters)
        {
            try { var current = filter.Map.devices; if (current.HasValue && current.Value.Count == 2 && current.Value.Contains(keyboard) && current.Value.Contains(mouse)) filter.Map.devices = filter.Had ? filter.Before : null; else { foreignFilters++; passed = false; error = (error ?? "") + "\nCurrent ActionMap出现外部filter替换，保留外部值。"; } }
            catch (Exception ex) { CleanupError("Current filter归还", ex); }
        }
        try { if (keyboard != null && InputSystem.devices.Contains(keyboard)) InputSystem.RemoveDevice(keyboard); } catch (Exception ex) { CleanupError("移除自有Keyboard", ex); }
        try { if (mouse != null && InputSystem.devices.Contains(mouse)) InputSystem.RemoveDevice(mouse); } catch (Exception ex) { CleanupError("移除自有Mouse", ex); }
        devicesRemoved = (keyboard == null || !InputSystem.devices.Contains(keyboard)) && (mouse == null || !InputSystem.devices.Contains(mouse));
        try
        {
            if (ownsOrbit && mapView != null && orbitField != null)
            {
                if (ReferenceEquals(orbitField.GetValue(mapView), fixtureOrbit)) { orbitField.SetValue(mapView, originalOrbit); ownsOrbit = false; }
                else { passed = false; error = (error ?? "") + "\n外部orbit替换已保留，未覆盖。"; }
            }
        }
        catch (Exception ex) { CleanupError("原orbit引用归还", ex); }
        try { if (ownsCamera && mapView != null) mapView.SetUiCameraCapture(CameraOwner, false); } catch (Exception ex) { CleanupError("自有camera capture归还", ex); } finally { ownsCamera = false; }
        foreach (var fixture in focus)
        {
            try
            {
                if (fixture.Owner == null) continue;
                bool current = (bool)fixture.Flag.GetValue(fixture.Owner);
                if (!fixture.Applied || current == fixture.Before) fixture.Restored = true;
                else if (current) { fixture.Callback.Invoke(fixture.Owner, new object[] { fixture.Before }); fixture.Restored = (bool)fixture.Flag.GetValue(fixture.Owner) == fixture.Before; }
                else { passed = false; error = (error ?? "") + "\n外部焦点值已保留：" + fixture.Label; }
            }
            catch (Exception ex) { CleanupError("原焦点夹具归还", ex); }
        }
        if (ownsLease) { SessionState.SetBool(Lease, false); ownsLease = false; }
        EditorApplication.QueuePlayerLoopUpdate(); Publish(true);
    }
    static void Publish(bool done)
    {
        bool orbitRestored = mapView == null || orbitField == null || ReferenceEquals(orbitField.GetValue(mapView), originalOrbit);
        bool zoomUnchanged = mapView == null || zoomField == null || Mathf.Abs((float)zoomField.GetValue(mapView) - originalZoom) < .00001f;
        bool pitchUnchanged = mapView == null || Mathf.Abs(mapView.pitch - originalPitch) < .00001f;
        ResultJson = JsonConvert.SerializeObject(new
        {
            done, Running, status = !done ? "RUNNING" : passed ? "PASS" : "FAILED_OR_CANCELLED", runId, stage, error, seconds = EditorApplication.timeSinceStartup - began, worldSeconds = Time.time - worldBegan,
            actorId, waypoint, reachedCount, waypointCount = Route.Length, renderedFrames, renderedEnds, movedFrames, inputFrames, actualWalkDistance, nominalRouteDistance = NominalRouteDistance(), minimumAcceptedWalkDistance = MinimumWalkDistance(),
            currentPosition = actor == null ? null : actor.transform.position.ToString("R"), initialPosition = initialPosition.ToString("R"),
            originalOrbitReferenceRestored = orbitRestored, zoomUnchanged, pitchUnchanged, ownCameraCaptureReleased = !ownsCamera, devicesRemoved, foreignFiltersPreserved = foreignFilters,
            focus = focus.Select(f => new { f.Label, f.Before, f.Applied, f.Restored }).ToArray(),
            sawInside, sawExteriorBefore, sawExteriorAfter, facts, trace = done ? trace.ToArray() : null, traceCount = trace.Count,
            scope = "原Current InputAction虚拟Keyboard/Mouse的真实WASD步行；一次初始Motor.Teleport位置夹具。临时原orbit对象yaw0；原MapResearchRoot.Update/CharacterRoot/Movement推进。未改绑定/共享SO/伙伴/UI actions、HP、AI、Collider、TimeScale或存档。正常共同保存writer可自行继续。每次GameplayCamera真实begin/end记录整栋顶盖与BlackoutStats；无RT/截图旁路。物理输入出现即取消；不重置实体设备。不代表实体硬件或用户审美验收。结束停在实际步行终点，不再Teleport恢复。"
        });
    }
