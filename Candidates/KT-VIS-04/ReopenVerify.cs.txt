/* 功能：只保存/重开本轮同一新Lab，验证持久登记、Shader引用及隔离档。 */
using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using UnityEditor;using UnityEditor.SceneManagement;using Funplay.Editor.Tools.Scripting;
public sealed class ReopenRoomVerify:IFunplayCommand {public void Execute(ExecutionContext ctx){const string path="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity";var s=SceneManager.GetActiveScene();if(Application.isPlaying||s.path!=path||SceneManager.sceneCount!=1)throw new Exception("Wrong edit lab");Ossastra.GroundSurfaces.Editor.GroundSurfaceAuthoring.SaveLab();if(s.isDirty)throw new Exception("Own lab save incomplete, do not discard");s=EditorSceneManager.OpenScene(path,OpenSceneMode.Single);var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(c=>c!=null).ToArray();var root=all.OfType<Ossastra.SceneVisibility.SceneVisibilityRoot>().Single();var map=all.OfType<Ossastra.MapResearch.MapResearchRoot>().Single();string reason;if(root.mapView!=map||root.roomModule==null||!root.ValidateConfiguration(out reason))throw new Exception("Reopened authoring invalid");var data=AssetDatabase.LoadAssetAtPath<UnityEngine.Rendering.Universal.UniversalRendererData>("Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/Rendering/RoomVisibilityRenderer.asset");var feature=data.rendererFeatures.OfType<Ossastra.SceneVisibility.Rooms.Rendering.RoomMaskRendererFeature>().Single();if(!feature.isActive||feature.worldLabelShader==null)throw new Exception("Feature persistence failed");ctx.ReturnValue=new{passed=true,scene=s.path,guid=AssetDatabase.AssetPathToGUID(path),pitch=map.pitch,buildings=root.roomModule.buildings.Length,roofGroups=root.roomModule.buildings.Sum(b=>b.roofGroups.Length),roofModes=root.roomModule.buildings.SelectMany(b=>b.roofGroups).SelectMany(r=>r.members).Select(r=>r.shadowCastingMode.ToString()).ToArray(),mapViewBound=true,feature=feature.isActive,labelShader=feature.worldLabelShader.name,profile=all.OfType<Ossastra.Commerce.Runtime.CommerceService>().Single().profileId};}}
