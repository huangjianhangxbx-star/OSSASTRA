using System;using System.Linq;using System.IO;using UnityEngine;using UnityEngine.SceneManagement;using UnityEditor;using Funplay.Editor.Tools.Scripting;
public sealed class CreateRoomTemplates:IFunplayCommand {public void Execute(ExecutionContext ctx){
 var s=SceneManager.GetActiveScene();if(Application.isPlaying||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity")throw new Exception("Exact Edit lab required");const string dir="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/Templates";if(!AssetDatabase.IsValidFolder(dir))AssetDatabase.CreateFolder("Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04","Templates");string[] names={"Building_ABC","Building_D_IndependentRoofControl","Tree_AllowCanopyReveal"};var rows=new System.Collections.Generic.List<object>();
 foreach(var name in names){string path=dir+"/"+name+".prefab";if(File.Exists(path))throw new Exception("Existing output, do not replay "+path);var original=GameObject.Find("RoomVisibilitySampleRoot/"+name);if(original==null)throw new Exception("Missing explicit sample "+name);GameObject copy=null;try{copy=UnityEngine.Object.Instantiate(original);copy.name=name;foreach(var room in copy.GetComponentsInChildren<Ossastra.SceneVisibility.Rooms.RoomRegion>(true))room.members=room.members.Where(r=>r!=null&&r.transform.IsChildOf(copy.transform)).ToArray();bool saved;var prefab=PrefabUtility.SaveAsPrefabAsset(copy,path,out saved);if(!saved||prefab==null)throw new Exception("Native prefab save failed");rows.Add(new{path,guid=AssetDatabase.AssetPathToGUID(path),note="原敌/工位等场景业务Renderer不随模板复制；新实例手动改唯一ID并接自己业务显示成员"});}finally{if(copy!=null)UnityEngine.Object.DestroyImmediate(copy);}}
 ctx.ReturnValue=new{ok=true,templates=rows,sceneInstancesReplaced=false,sourceChanged=false};
}}
