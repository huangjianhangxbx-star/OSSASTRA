/* 功能：给既有样板增量接入显式屋顶碰撞排除、开发面板及世界标签Shader；不重建或清场。 */
using System;using System.Linq;using UnityEngine;using UnityEngine.SceneManagement;using UnityEditor;using UnityEditor.SceneManagement;using Ossastra.SceneVisibility.Rooms;using Ossastra.SceneVisibility.Rooms.Rendering;using Ossastra.Development.SceneVisibility;using Funplay.Editor.Tools.Scripting;
public sealed class ConfigureRoomIncrement:IFunplayCommand {public void Execute(ExecutionContext ctx){
 var s=SceneManager.GetActiveScene();if(Application.isPlaying||EditorApplication.isCompiling||s.path!="Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/SC_RoomVisibilityLab.unity"||SceneManager.sceneCount!=1)throw new Exception("Exact owned Edit lab required");
 var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<MonoBehaviour>(true)).Where(c=>c!=null).ToArray();var root=all.OfType<Ossastra.SceneVisibility.SceneVisibilityRoot>().Single();var module=root.roomModule;if(module==null)throw new Exception("No module");
 foreach(var roof in module.buildings.SelectMany(b=>b.roofGroups)){ctx.RegisterObjectModification(roof);roof.labelRaycastExclusions=roof.members.Select(r=>r.GetComponent<Collider>()).Where(c=>c!=null).ToArray();EditorUtility.SetDirty(roof);}
 var panel=root.GetComponent<RoomVisibilityLabPanel>();if(panel==null){panel=root.gameObject.AddComponent<RoomVisibilityLabPanel>();ctx.RegisterObjectCreation(panel);}else ctx.RegisterObjectModification(panel);
 panel.visibility=root;panel.map=all.OfType<Ossastra.MapResearch.MapResearchRoot>().Single();panel.shutterPortal=module.buildings.SelectMany(b=>b.portals).Single(p=>p.portalId=="portal-a-west");panel.shutter=GameObject.Find("RoomVisibilitySampleRoot/Building_ABC/A_West_ActualOpaqueShutter_InitiallyClosed").transform;
 panel.shutterClosedPosition=new Vector3(-20.14f,1.7f,-11);panel.shutterOpenPosition=new Vector3(-20.14f,1.7f,-7.7f);panel.exteriorPosition=new Vector3(-16,.1f,-18.2f);panel.knownRoomPosition=new Vector3(-15.5f,.1f,-12.5f);panel.unknownRoomPosition=new Vector3(-6,.1f,-13.2f);EditorUtility.SetDirty(panel);
 var data=AssetDatabase.LoadAssetAtPath<UnityEngine.Rendering.Universal.UniversalRendererData>("Assets/OSSASTRA/GroundWorkshop/Samples/KT-VIS-04/Rendering/RoomVisibilityRenderer.asset");var feature=data.rendererFeatures.OfType<RoomMaskRendererFeature>().Single();ctx.RegisterObjectModification(feature);feature.worldLabelShader=Shader.Find("Ossastra/Rooms/WorldLabel");if(feature.worldLabelShader==null)throw new Exception("Missing world label shader");feature.SetActive(true);EditorUtility.SetDirty(feature);data.SetDirty();EditorUtility.SetDirty(data);AssetDatabase.SaveAssetIfDirty(data);
 string reason;if(!root.ValidateConfiguration(out reason))throw new Exception(reason);EditorSceneManager.MarkSceneDirty(s);Ossastra.GroundSurfaces.Editor.GroundSurfaceAuthoring.SaveLab();Ossastra.GroundSurfaces.Editor.GroundSurfaceUpdates.ClearPending();ctx.ReturnValue=new{ok=true,saved=!s.isDirty,roofExclusions=module.buildings.SelectMany(b=>b.roofGroups).Sum(r=>r.labelRaycastExclusions.Length),pitch=panel.map.pitch,shader=feature.worldLabelShader.name};
}}
