// 旧M02A真实RED：两角色相距4单位时非实控者不移动，证明未接入跟随。
using System;using UnityEngine;using UnityEditor;using Ossastra.Characters;using Funplay.Editor.Tools.Scripting;
public class NoFollowRed:IFunplayCommand{public void Execute(Funplay.Editor.Tools.Scripting.ExecutionContext ctx){var c=UnityEngine.Object.FindObjectOfType<CharacterControlCoordinator>();if(!Application.isPlaying||c==null||c.Characters.Count!=2)throw new Exception("Dual Play required");var a=c.Characters[0];var b=c.Characters[1];a.GetComponent<CharacterMovement>().Teleport(new Vector3(2,.03f,-3));b.GetComponent<CharacterMovement>().Teleport(new Vector3(-2,.03f,-3));var before=b.transform.position;var started=EditorApplication.timeSinceStartup;EditorApplication.CallbackFunction tick=null;tick=()=>{if(EditorApplication.timeSinceStartup-started<.7)return;EditorApplication.update-=tick;SessionState.SetString("OSSASTRA_M02C_RED",Newtonsoft.Json.JsonConvert.SerializeObject(new{distance=Vector3.Distance(a.transform.position,b.transform.position),bTravel=Vector3.Distance(before,b.transform.position),bInput=b.DataBus.Input.move.ToString(),pass=Vector3.Distance(before,b.transform.position)>.2f}));};EditorApplication.update+=tick;ctx.Log("Scheduled .7s old non-current follow RED");}}
