i=0;i<2;i++){var g=UnityEngine.Object.Instantiate(prefab,parent.transform);g.transform.position=new after=actors[0].DataBus.Input.move;ctx.Log(Newtonsoft.Json.JsonConvert.SerializeObject(new{before=before.ToString(),after=after.ToString(),rawAfter=input.CurrentFrame.move.ToString(),pass=before.x>.9f&&after==Vector2.zero}));}finally{InputSystem.RemoveDevice(keyboard);UnityEngine.Object.DestroyImmediate(parent);}}}
