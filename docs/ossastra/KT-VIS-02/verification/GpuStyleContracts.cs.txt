 * Reuses the previous backend contract's independent original-material images +

 * Requires stable Edit and SessionState[GroundSurface.Vis02Output] = existing evidence directory.

        ctx.ReturnValue = new { status = active == null ? "FINISHED" : "STARTED", pollType = "Vis02GpuStyleContracts", pollField = "ResultJson", result = ResultJson };

        void Tick()
        {
            if (EditorApplication.timeSinceStartup < next) return;
            try
            {
                Need(!Application.isPlaying && !EditorApplication.isPlayingOrWillChangePlaymode && !EditorApplication.isCompiling && !EditorApplication.isUpdating,
                    "源编辑器生命周期改变；不继续或伪造结果。");
                Need(SceneManager.GetActiveScene() == source && source.isDirty == sourceDirty && RootIds(source) == sourceRoots && Time.timeScale == sourceTimeScale,
                    "源场景/dirty/root/timeScale变化。");
                if (floorOnly)
                {
                    if (stage == 0) SetupResolution();
                    else if (stage == 1) MinimumOpacityFloor();
                    else if (resolution == 0) { resolution = 1; stage = -1; }
                    else { Finish(null); return; }
                }
                else if (stage == 0) SetupResolution();
                else if (stage == 1) Soft();
                else if (stage == 2) DitherStable();
                else if (stage == 3) SoftAgain();
                else if (stage == 4) WholeHalfAndDepth();
                else if (stage == 5) Endpoints();
                else if (stage == 6) UnionAndSupport();
                else
                {
                    if (resolution == 0) { resolution = 1; stage = -1; }
                    else { Finish(null); return; }
                }
                stage++; next = EditorApplication.timeSinceStartup + .05; Publish("RUNNING", null, false); EditorApplication.QueuePlayerLoopUpdate();
            }
            catch (Exception error) { Finish(error.ToString()); }
        }

        void WholeHalfAndDepth()
        {
            phase = "whole half opacity and original triangle depth order";
            for (int i = 0; i < 2; i++) { frames[i].CircleA = frames[i].CircleB = frames[i].PreserveSupport = default(CaptureCircle); frames[i].WholeFade = .5f; }
            backend.SetEdgeStyle(VisibilityEdgeStyle.Soft, 0, 1); Color[] soft = Render(true);
            backend.SetEdgeStyle(VisibilityEdgeStyle.SoftDitherEdge, .18f, 1); Color[] dotted = Render(true);
            Equal("wholepartial-no-dots-without-spatial-feather", soft, dotted, null, 0);
            int samples = 0, wrong = 0, aNear = 0, bNear = 0, front = 0; float maxError = 0;
            // Sparse 3px sampling is predeclared to keep the independent CPU oracle bounded at 1080.
            for (int y = 3; y < height - 3; y += 3) for (int x = 3; x < width - 3; x += 3)
            {
                Ray ray = camera.ViewportPointToRay(new Vector3((x + .5f) / width, (y + .5f) / height, 0));
                float da, db; if (!Interior(a, ray, out da) || !Interior(b, ray, out db)) continue;
                float depth = float.PositiveInfinity; foreach (var item in bases) { float d; if (Hit(item, ray, false, out d)) depth = Math.Min(depth, d); }
                int p = y * width + x; Color expected = baseImage[p];
                if (da < db) { aNear++; if (db < depth) expected = Mix(expected, singleB[p], .5f); if (da < depth) expected = Mix(expected, singleA[p], .5f); }
                else { bNear++; if (da < depth) expected = Mix(expected, singleA[p], .5f); if (db < depth) expected = Mix(expected, singleB[p], .5f); }
                if (depth < da && depth < db) front++;
                float e = Error(expected, dotted[p]); maxError = Math.Max(maxError, e); samples++; if (e > ColorTolerance) wrong++;
            }
            Add(new Row {Name = "per-pixel-crossing-depth-and-front-object", Samples = samples, Wrong = wrong, MaxError = maxError,
                Passed = samples > 1000 && aNear > 100 && bNear > 100 && front > 50 && wrong == 0,
                Reason = "independent original-material references plus original-triangle depth; aNear=" + aNear + " bNear=" + bNear + " front=" + front});
            Save(dotted, "wholehalf-no-exterior-dots.png");
        }

        void MinimumOpacityFloor()
        {
            phase = "nonzero floor GPU regression"; const float floor = .8f;
            SetCircles(1-floor); Show(true,false,true);
            // On the old backend the field is absent; still render its actual dither output and
            // fail on observed undershoot rather than treating an API-name check as the regression.

            if(field!=null){object value=frames[0];field.SetValue(value,floor);frames[0]=(RegisteredGroupFrame)value;}
            backend.SetDebugView(0); backend.SetEdgeStyle(VisibilityEdgeStyle.SoftDitherEdge,.18f,1);
            Color[] actual=Render(true); int samples=0,violations=0,innerSamples=0,innerViolations=0;float minimum=1,innerMinimum=1;
            for(int y=2;y<height-2;y++)for(int x=2;x<width-2;x++)
            {
                int p=y*width+x;float d=Distance(frames[0].CircleA,p);
                if(d<=frames[0].CircleA.InnerPixels+2||d>=frames[0].CircleA.OuterPixels-2)continue;
                Ray ray=camera.ViewportPointToRay(new Vector3((x+.5f)/width,(y+.5f)/height,0));float groupDepth;
                if(!Interior(a,ray,out groupDepth))continue;
                float depth=float.PositiveInfinity;foreach(var item in bases){if(!item.Renderer.enabled)continue;float z;if(Hit(item,ray,false,out z))depth=Math.Min(depth,z);}
                if(groupDepth>=depth)continue;
                Color delta=singleA[p]-baseImage[p];int channel=Math.Abs(delta.r)>Math.Abs(delta.g)?0:1;
                if(Math.Abs(delta.b)>Math.Abs(delta[channel]))channel=2;
                if(Math.Abs(delta[channel])<.25f)continue;

                samples++;minimum=Math.Min(minimum,inferred);if(float.IsNaN(inferred)||float.IsInfinity(inferred)||inferred<floor-.0015f)violations++;
                if(d<frames[0].CircleA.InnerPixels+.4f*(frames[0].CircleA.OuterPixels-frames[0].CircleA.InnerPixels))
                {
                    innerSamples++;innerMinimum=Math.Min(innerMinimum,inferred);
                    if(float.IsNaN(inferred)||float.IsInfinity(inferred)||inferred<floor-.0015f)innerViolations++;
                }
            }
            Add(new Row{Name="actual-feather-keeps-explicit-minimum-opacity",Samples=samples,Wrong=violations,MaxError=Math.Max(0,floor-minimum),
                Passed=samples>500&&innerSamples>500&&violations==0,
                Reason="floor="+floor+" measured minimum="+minimum+" inner-ring minimum="+innerMinimum+" inner violations="+innerViolations+"/"+innerSamples+
                    "; dark Base only, independent original-color mixture, opacity tolerance .0015"});
            Add("floor-plumbing-adds-no-capture-target",backend.OwnedRTCount==2&&captured==1,"one visible group remains exactly two RTs; second source is disabled");
            Save(actual,"minimum-opacity-floor.png");
        }
        void CompareDitherBand()
        {
            int core = 0, exterior = 0, edge = 0, changed = 0, wrongCore = 0, wrongExterior = 0; float maxBandError = 0;
            double redMean = 0, greenMean = 0, blueMean = 0;
            for (int y = 2; y < height - 2; y++) for (int x = 2; x < width - 2; x++)
            {
                int p = y * width + x; float distance = Distance(frames[0].CircleA, p); float delta = Error(softImage[p], ditherImage[p]);
                if (distance < frames[0].CircleA.InnerPixels - 2) {core++; if (delta != 0) wrongCore++;}
                else if (distance > frames[0].CircleA.OuterPixels + 2) {exterior++; if (delta != 0) wrongExterior++;}
                else
                {
                    edge++; if (delta > .0002f) changed++; maxBandError = Math.Max(maxBandError, delta);
                    redMean += ditherImage[p].r - softImage[p].r; greenMean += ditherImage[p].g - softImage[p].g; blueMean += ditherImage[p].b - softImage[p].b;
                }
            }
            Add(new Row {Name="spatial-core-and-exterior-have-no-dot-change", Samples=core+exterior, Wrong=wrongCore+wrongExterior,
                Passed=core>100 && exterior>100 && wrongCore==0 && wrongExterior==0, Reason="actual GPU fixed-pose style difference, not a CPU style mirror"});
            float mean = edge > 0 ? (float)Math.Max(Math.Abs(redMean / edge), Math.Max(Math.Abs(greenMean / edge), Math.Abs(blueMean / edge))) : 1;
            Add(new Row {Name="only-feather-has-restrained-mean-preserving-pattern", Samples=edge, Wrong=0, MaxError=maxBandError,
                Passed=changed>100 && maxBandError<.15f && mean<.012f, Reason="changed="+changed+" max RGB mean shift="+mean});
            // A 1px reference cell must exhibit neighbouring-pixel differences at both actual viewport sizes.
            int neighborPairs = 0, neighborChanges = 0;
            for (int y = height/2-12; y < height/2+12; y++) for (int x = width/2+(int)(height*.19f); x < width/2+(int)(height*.23f); x++)
            {
                int p=y*width+x; Color n=ditherImage[p+1]-softImage[p+1], c=ditherImage[p]-softImage[p];
                // Blue is monotone with coverage in this explicitly declared blue-near fixture.
                // Count residual sign reversals, not a smooth feather gradient masquerading as fine dots.
                neighborPairs++; if (Math.Abs(n.b)>.0005f&&Math.Abs(c.b)>.0005f&&Math.Sign(n.b)!=Math.Sign(c.b)) neighborChanges++;
            }
            Add("actual-pixel-small-pattern-at-"+height+"p", neighborPairs>100 && neighborChanges>neighborPairs/3,
                "neighbouring edge residual sign reversals="+neighborChanges+"/"+neighborPairs+"; smooth gradients cannot satisfy fine-dot gate");
        }

        float Distance(CaptureCircle circle,int p)
        { float dx=p%width+.5f-circle.CenterViewport.x*width,dy=p/width+.5f-circle.CenterViewport.y*height;return Mathf.Sqrt(dx*dx+dy*dy); }
        void Equal(string name,Color[] expected,Color[] actual,Func<int,bool> mask,float tolerance)
        {
            int samples=0,wrong=0;float max=0;
            for(int p=0;p<actual.Length;p++)if(mask==null||mask(p))
            {float e=Error(expected[p],actual[p]);samples++;max=Math.Max(max,e);if(e>tolerance)wrong++;}
            Add(new Row{Name=name,Samples=samples,Wrong=wrong,MaxError=max,Passed=samples>100&&wrong==0});
        }
        void Add(string name,bool passed,string reason) => Add(new Row{Name=name,Passed=passed,Reason=reason});

        static float Error(Color a,Color b)
        {
            float r=Math.Abs(a.r-b.r),g=Math.Abs(a.g-b.g),blue=Math.Abs(a.b-b.b);
            return float.IsNaN(r)||float.IsNaN(g)||float.IsNaN(blue)?float.PositiveInfinity:Math.Max(r,Math.Max(g,blue));
        }
        static Color Mix(Color behind,Color front,float alpha) => behind*(1-alpha)+front*alpha;

        {

            // FloorOnly uses a dark high-contrast background. The ordinary suite retains rear/front geometry.
            // Otherwise the inner ring could be rejected solely because the colored body is too similar to A.
            for(int i=0;i<bases.Count;i++)bases[i].Renderer.enabled=showBase&&(!floorOnly||i==0);
        }
        Color[] Render(bool enabled)

        static bool Interior(Surface surface,Ray ray,out float distance) => Hit(surface,ray,true,out distance);
        static bool Hit(Surface s,Ray ray,bool interior,out float distance)
        {
            distance=float.PositiveInfinity;bool hit=false;
            for(int i=0;i<s.Triangles.Length;i+=3)
            {
                Vector3 a=s.Vertices[s.Triangles[i]],e1=s.Vertices[s.Triangles[i+1]]-a,e2=s.Vertices[s.Triangles[i+2]]-a;
                Vector3 p=Vector3.Cross(ray.direction,e2);float det=Vector3.Dot(e1,p);if(Math.Abs(det)<1e-7f)continue;
                float inv=1/det;Vector3 t=ray.origin-a;float u=Vector3.Dot(t,p)*inv;Vector3 q=Vector3.Cross(t,e1);float v=Vector3.Dot(ray.direction,q)*inv;
                float margin=interior?.005f:0;if(u<margin||v<margin||u+v>1-margin)continue;
                float d=Vector3.Dot(e2,q)*inv;if(d>=0&&d<distance){hit=true;distance=d;}
            }
            return hit;
        }
        string ResourceIds()
        {
            var ids=new List<int>();
            Action<object> add=value=>{var o=value as Object;if(o!=null)ids.Add(o.GetInstanceID());};
            var flags=BindingFlags.Instance|BindingFlags.Public|BindingFlags.NonPublic;Type type=backend.GetType();
            foreach(string n in new[]{"quad","auxiliaryObject","proxyRoot"})add(type.GetField(n,flags).GetValue(backend));
            foreach(var value in (IEnumerable)type.GetField("displays",flags).GetValue(backend))add(value);
            foreach(var group in (IEnumerable)type.GetField("slots",flags).GetValue(backend))if(group!=null)
            {
                Type gt=group.GetType();foreach(string n in new[]{"Color","Depth","Root"})add(gt.GetField(n,flags).GetValue(group));
                foreach(var item in (IEnumerable)gt.GetField("Sources",flags).GetValue(group))if(item!=null)
                {
                    Type st=item.GetType();foreach(string n in new[]{"Copy","Mesh"})add(st.GetField(n,flags).GetValue(item));
                    foreach(string n in new[]{"Materials","DepthMaterials"})foreach(var m in (IEnumerable)st.GetField(n,flags).GetValue(item))add(m);
                }
            }
            ids.Sort();return string.Join(",",ids)+"|"+backend.OwnedRTCount+"|"+backend.OwnedMaterialCount+"|"+backend.OwnedMeshCount;
        }
